9 # Free cyclic submodules and spanning fractions
10 
11 Let `R` be a simple right Noetherian ring with classical right quotient ring
12 `Q`, and let `M` be a finitely generated right `R`-module with module of
13 fractions `M_Q` (`Goldie.FractionModule R M`) and Goldie rank
14 `rk(M) = length_Q(M_Q)`. Write `s = rk(R_R) = length_Q(Q_Q)`.
15 
16 * `Goldie.eq_top_of_forall_toFractionModule_smul_mem` (`R` simple, no
17   Noetherian hypothesis): a `Q`-submodule of `M_Q` that contains the fraction of
18   `m e` for every `m ∈ M` and one fixed `e ≠ 0` is all of `M_Q`.
19 * `Goldie.exists_injective_toSpanSingleton`: if `s ≤ rk(M)`, some `p ∈ M` has
20   zero annihilator, so `pR ≅ R_R`.
21 * `Goldie.exists_goldieRank_eq_add_quotient`: if `s ≤ rk(M)`, some `p ∈ M`
22   satisfies `rk(M) = s + rk(M/pR)`.
23 * `Goldie.exists_span_fraction_eq_top`: for nonzero `e₀, …, e_{n-1}` and
24   `rk(M) = l < n` there are `m_j ∈ M`, zero for `j ≥ l`, such that the fractions
25   of the `m_j e_j` span `M_Q`.
26

28

43 /-- **Simplicity argument.** Let `R` be simple, `e ≠ 0` and `U` a
44 `Q`-submodule of `M_Q` containing the fraction of `m e` for every `m ∈ M`. Then
45 `U = M_Q`: the `r` with `M r ⊆ U` form a two-sided ideal containing `e`, hence
46 `1`, and every fraction `m c⁻¹` lies in `U`. -/
47 theorem eq_top_of_forall_toFractionModule_smul_mem [IsSimpleRing R] {e : R} (he : e ≠ 0)
48     (U : Submodule (OppositeQuotientRing R) (FractionModule R M))
49     (hU : ∀ m : M, toFractionModule R M (op e • m) ∈ U) : U = ⊤ := by
50   -- the two-sided ideal of all `r` with `toFrac(m r) ∈ U` for every `m`
51   let J : TwoSidedIdeal R :=
52     TwoSidedIdeal.mk' {r | ∀ m : M, toFractionModule R M (op r • m) ∈ U}
53     (fun m => by simp)
54     (fun hx hy m => by rw [op_add, add_smul, map_add]; exact U.add_mem (hx m) (hy m))
55     (fun hx m => by rw [op_neg, neg_smul, map_neg]; exact U.neg_mem (hx m))
56     (fun {x y} hy m => by rw [op_mul, mul_smul]; exact hy _)
57     (fun {x y} hx m => by
58       rw [op_mul, mul_smul, map_smul,
59         ← OreModule.oreDiv_one_smul_eq (S := oppositeSubmonoid R⁰)]
60       exact U.smul_mem _ (hx m))
61   have h1 : (1 : R) ∈ J :=
62     IsSimpleRing.one_mem_of_ne_zero_mem J he ((TwoSidedIdeal.mem_mk' ..).mpr hU)
63   rw [eq_top_iff]
64   intro x _
65   induction x using OreLocalization.ind with | _ m c =>
66   have hx : m /ₒ c = ((1 : Rᵐᵒᵖ) /ₒ c) • toFractionModule R M m := by
67     rw [OreModule.toLocalization_apply, OreLocalization.one_div_smul, one_mul]
68   rw [hx]
69   refine U.smul_mem _ ?_
70   simpa using (TwoSidedIdeal.mem_mk' ..).mp h1 m

72 /-- **A free cyclic submodule.** If `R` is simple right Noetherian and
73 `s = rk(R_R) ≤ rk(M)`, then some `p ∈ M` has zero annihilator: `r ↦ p r` is
74 injective, so `pR ≅ R_R`. Embed `Q_Q` in `M_Q` and clear the denominator of the
75 image `m c⁻¹` of `1`. -/
76 theorem exists_injective_toSpanSingleton [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
77     [Module.Finite Rᵐᵒᵖ M] (h : goldieRank R R ≤ goldieRank R M) :
78     ∃ p : M, Function.Injective (LinearMap.toSpanSingleton Rᵐᵒᵖ M p) := by
79   have : Module.Finite (OppositeQuotientRing R) (FractionModule R M) :=
80     OreModule.finite (oppositeSubmonoid R⁰) M
81   rw [goldieRank_self] at h
82   obtain ⟨v, hv⟩ := exists_injective_smul_of_length_le h
83   induction v using OreLocalization.ind with | _ m c =>
84   refine ⟨m, ?_⟩
85   rw [← LinearMap.ker_eq_bot, eq_bot_iff]
86   intro a ha
87   rw [LinearMap.mem_ker, LinearMap.toSpanSingleton_apply] at ha
88   have h0 : ((a * c) /ₒ (1 : oppositeSubmonoid R⁰)) • (m /ₒ c) = 0 := by
89     rw [OreModule.oreDiv_one_smul_eq, mul_smul, OreModule.smul_oreDiv_self,
90       ← map_smul, ha, map_zero]
91   have h1 : ((a * c) /ₒ (1 : oppositeSubmonoid R⁰) : OppositeQuotientRing R) = 0 :=
92     hv (by
93       dsimp only
94       rw [h0]
95       exact (zero_smul (OppositeQuotientRing R) (m /ₒ c : FractionModule R M)).symm)
96   have h2 : a * c = 0 := numeratorRingHom_injective (h1.trans (map_zero _).symm)
97   exact (Submodule.mem_bot _).mpr
98     ((mem_oppositeSubmonoid_nonZeroDivisors.mp c.2).2 a h2)

100 /-- **Splitting off `R_R`.** If `rk(R_R) ≤ rk(M)`, there is `p ∈ M`
101 with `rk(M) = rk(R_R) + rk(M/pR)`. -/
102 theorem exists_goldieRank_eq_add_quotient [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
103     [Module.Finite Rᵐᵒᵖ M] (h : goldieRank R R ≤ goldieRank R M) :
104     ∃ p : M, goldieRank R M =
105       goldieRank R R + goldieRank R (M ⧸ Submodule.span Rᵐᵒᵖ {p}) := by
106   obtain ⟨p, hp⟩ := exists_injective_toSpanSingleton h
107   refine ⟨p, ?_⟩
108   have hexact : Function.Exact (LinearMap.toSpanSingleton Rᵐᵒᵖ M p)
109       (Submodule.span Rᵐᵒᵖ {p}).mkQ := by
110     rw [LinearMap.exact_iff, Submodule.ker_mkQ]
111     exact LinearMap.span_singleton_eq_range Rᵐᵒᵖ M p
112   rw [goldieRank_eq_add_of_exact hp (Submodule.mkQ_surjective _) hexact, goldieRank_op_self,
113     ← goldieRank_self]

115 /-- **Successive choice.** Let `e_0, …, e_{n-1}` be nonzero elements of the
116 simple right Noetherian ring `R` and `rk(M) = l < n`. There are `m_j ∈ M`, with
117 `m_j = 0` for `j ≥ l`, such that the fractions of the `m_j e_j` span `M_Q` over
118 `Q`. Each choice raises the length of the span by at least one until it
119 reaches `l`, by the simplicity argument
120 (`eq_top_of_forall_toFractionModule_smul_mem`). -/
121 theorem exists_span_fraction_eq_top [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
122     [Module.Finite Rᵐᵒᵖ M] {n l : ℕ} (e : Fin n → R)
123     (he : ∀ j, e j ≠ 0) (hln : l < n) (hM : goldieRank R M = l) :
124     ∃ m : Fin n → M, (∀ j : Fin n, l ≤ (j : ℕ) → m j = 0) ∧
125       Submodule.span (OppositeQuotientRing R)
126         (Set.range fun j => toFractionModule R M (op (e j) • m j)) = ⊤ := by
127   classical
128   have : Module.Finite (OppositeQuotientRing R) (FractionModule R M) :=
129     OreModule.finite (oppositeSubmonoid R⁰) M
130   let U : (Fin n → M) → Submodule (OppositeQuotientRing R) (FractionModule R M) := fun m =>
131     Submodule.span (OppositeQuotientRing R)
132       (Set.range fun j => toFractionModule R M (op (e j) • m j))
133   have hlenM : Module.length (OppositeQuotientRing R) (FractionModule R M) = l := hM
134   have key : ∀ k ≤ l, ∃ m : Fin n → M, (∀ j : Fin n, k ≤ (j : ℕ) → m j = 0) ∧
135       (k : ℕ∞) ≤ Module.length (OppositeQuotientRing R) (U m) := by
136     intro k
137     induction k with
138     | zero => exact fun _ => ⟨0, fun _ _ => rfl, by simp⟩
139     | succ k ih =>
140       intro hk
141       obtain ⟨m, hm0, hmk⟩ := ih (by omega)
142       by_cases hlen : ((k + 1 : ℕ) : ℕ∞) ≤ Module.length (OppositeQuotientRing R) (U m)
143       · exact ⟨m, fun j hj => hm0 j (by omega), hlen⟩
144       have hU : U m ≠ ⊤ := by
145         intro htop
146         apply hlen
147         rw [htop, Module.length_top, hlenM]
148         exact_mod_cast hk
149       let i : Fin n := ⟨k, by omega⟩
150       obtain ⟨m', hm'⟩ : ∃ m' : M, toFractionModule R M (op (e i) • m') ∉ U m := by
151         by_contra! h
152         exact hU (eq_top_of_forall_toFractionModule_smul_mem (he i) _ h)
153       refine ⟨Function.update m i m', fun j hj => ?_, ?_⟩
154       · have hji : j ≠ i := fun h => by rw [h] at hj; simp [i] at hj
155         rw [Function.update_of_ne hji]
156         exact hm0 j (by omega)
157       have hlt : U m < U (Function.update m i m') := by
158         refine lt_of_le_of_ne (Submodule.span_le.mpr ?_) ?_
159         · rintro _ ⟨j, rfl⟩
160           by_cases hj : j = i
161           · rw [hj]
162             dsimp only
163             rw [hm0 i le_rfl, smul_zero, map_zero]
164             exact zero_mem _
165           · exact Submodule.subset_span ⟨j, by simp [Function.update_of_ne hj]⟩
166         · intro heq
167           apply hm'
168           rw [heq]
169           exact Submodule.subset_span ⟨i, by simp⟩
170       have h1 := Submodule.height_strictMono hlt
171       rw [← Module.length_submodule, ← Module.length_submodule] at h1
172       push_cast
173       exact Order.add_one_le_of_lt (lt_of_le_of_lt hmk h1)
174   obtain ⟨m, hm0, hml⟩ := key l le_rfl
175   refine ⟨m, hm0, ?_⟩
176   by_contra hne
177   have := Submodule.length_lt hne
178   rw [hlenM] at this
179   exact (lt_of_le_of_lt hml this).false
