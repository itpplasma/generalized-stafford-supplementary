9 # Idempotents of the quotient ring with common denominators
10 
11 Let `R` be a simple right Noetherian ring with classical right quotient ring
12 `Q`, and let `s = length_Q(Q_Q) = goldieRank R R`. Since `Q` is semisimple,
13 `Q_Q` is a direct sum of `s` simple right ideals, and the decomposition
14 `1 = E₁ + ⋯ + E_s` gives complete orthogonal idempotents with `Eᵢ Q` simple.
15 Clearing denominators in `R` gives the device used for upper bounds on the
16 number of generators of a module:
17 
18 * a regular `c ∈ R` with `fᵢ = Eᵢ c ∈ R` for all `i`;
19 * regular `tᵢ ∈ R` with `rᵢ = c⁻¹ Eᵢ tᵢ ∈ R`, stated without inverses as
20   `c rᵢ = Eᵢ tᵢ` in `Q`.
21 
22 Then `fᵢ rⱼ = 0` for `i ≠ j`, and `eᵢ = fᵢ rᵢ = Eᵢ tᵢ ≠ 0`
23 (`Goldie.exists_idempotent_denominators`).
24 
25 As elsewhere, `Q = (Q°)ᵐᵒᵖ` with `Q° = OppositeQuotientRing R`; the right ideal
26 `Eᵢ Q` of `Q` is the left ideal `Q° · unop Eᵢ` of `Q°`, and its simplicity is
27 stated for that left ideal.
28 
29 ## References
30 
31 J. C. McConnell and J. C. Robson, *Noncommutative Noetherian Rings*, 2.1.3 and
32 2.3.6; T. Y. Lam, *A First Course in Noncommutative Rings*, §3 and §21
33 (idempotents attached to a direct sum decomposition). Nothing here is new.

48 /-- `Q°` is a direct sum of `n = length(Q°)` simple left ideals `Q° eᵢ`, given by
49 complete orthogonal idempotents `eᵢ`. -/
50 theorem exists_completeOrthogonalIdempotents_oppositeQuotientRing :
51     ∃ (n : ℕ) (e : Fin n → OppositeQuotientRing R),
52       (n : ℕ∞) = Module.length (OppositeQuotientRing R) (OppositeQuotientRing R) ∧
53       CompleteOrthogonalIdempotents e ∧
54       ∀ i, IsSimpleModule (OppositeQuotientRing R)
55         (Submodule.span (OppositeQuotientRing R) {e i}) := by
56   classical
57   set Q := OppositeQuotientRing R
58   obtain ⟨s, hind, hsup, hsimple⟩ := IsSemisimpleModule.exists_sSupIndep_sSup_simples_eq_top Q Q
59   have hind' : iSupIndep fun m : s => m.1 := (sSupIndep_iff s).mp hind
60   have hfin : Finite s := WellFoundedGT.finite_of_iSupIndep hind'
61     fun m => have := hsimple _ m.2
62       (Submodule.nontrivial_iff_ne_bot).mp (IsSimpleModule.nontrivial Q m.1)
63   let eq := Finite.equivFin s
64   let V : Fin (Nat.card s) → Ideal Q := fun i => (eq.symm i).1
65   have hV : iSupIndep V := hind'.comp eq.symm.injective
66   have hVsup : iSup V = ⊤ := by
67     rw [← hsup, sSup_eq_iSup']
68     exact eq.symm.iSup_comp (g := fun m : s => m.1)
69   have hint : DirectSum.IsInternal V :=
70     DirectSum.isInternal_submodule_of_iSupIndep_of_iSup_eq_top hV hVsup
71   let := hint.chooseDecomposition
72   have hsimpleV : ∀ i, IsSimpleModule Q (V i) := fun i => hsimple _ (eq.symm i).2
73   have hspan : ∀ i, Submodule.span Q {DirectSum.idempotent V i} = V i := by
74     intro i
75     apply le_antisymm
76     · rw [Submodule.span_le, Set.singleton_subset_iff]
77       exact (DirectSum.decompose V (1 : Q) i).2
78     · intro x hx
79       rw [← DirectSum.decompose_of_mem_same V hx, DirectSum.decompose_eq_mul_idempotent]
80       exact Submodule.smul_mem _ x (Submodule.subset_span rfl)
81   refine ⟨Nat.card s, DirectSum.idempotent V, ?_,
82     DirectSum.completeOrthogonalIdempotents_idempotent V, fun i => by rw [hspan]; exact hsimpleV i⟩
83   have e := LinearEquiv.ofBijective (DirectSum.coeLinearMap V) hint
84   rw [← e.length_eq, (DFinsupp.linearEquivFunOnFintype).length_eq, Module.length_pi_of_fintype]
85   rw [Finset.sum_congr rfl fun i _ => Module.length_eq_one_iff.mpr (hsimpleV i),
86     Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_one]
87 
88 end Decomposition

90 /-- **Idempotents with common denominators** (the device of McConnell–Robson 2.3.6
91 used for generator bounds). Let `R` be simple and right Noetherian with
92 classical right quotient ring `Q` and `ι : R → Q`. There are `n = rk(R_R) =
93 length_Q(Q_Q)` complete orthogonal idempotents `E₁, …, E_n` of `Q` with each
94 `Eᵢ Q` simple, a regular `c`, regular `tᵢ` and elements `fᵢ, rᵢ` of `R` with
95 
96 * `Eᵢ c = fᵢ` and `c rᵢ = Eᵢ tᵢ` in `Q` (so `rᵢ = c⁻¹ Eᵢ tᵢ`);
97 * `fᵢ rⱼ = 0` for `i ≠ j`;
98 * `fᵢ rᵢ = Eᵢ tᵢ` in `Q`, and `fᵢ rᵢ ≠ 0`. -/
99 theorem exists_idempotent_denominators [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R] :
100     ∃ (n : ℕ) (E : Fin n → QuotientRing R) (c : R) (t f r : Fin n → R),
101       (n : ℕ∞) = goldieRank R R ∧
102       CompleteOrthogonalIdempotents E ∧
103       (∀ i, IsSimpleModule (OppositeQuotientRing R)
104         (Submodule.span (OppositeQuotientRing R) {unop (E i)})) ∧
105       c ∈ R⁰ ∧ (∀ i, t i ∈ R⁰) ∧
106       (∀ i, E i * toQuotientRing R c = toQuotientRing R (f i)) ∧
107       (∀ i, toQuotientRing R c * toQuotientRing R (r i) = E i * toQuotientRing R (t i)) ∧
108       (∀ i j, i ≠ j → f i * r j = 0) ∧
109       (∀ i, toQuotientRing R (f i * r i) = E i * toQuotientRing R (t i)) ∧
110       (∀ i, f i * r i ≠ 0) := by
111   classical
112   let ι := toQuotientRing R
113   obtain ⟨n, e, hn, he, hsimple⟩ := exists_completeOrthogonalIdempotents_oppositeQuotientRing
114     (R := R)
115   let E : Fin n → QuotientRing R := fun i => op (e i)
116   have hE : CompleteOrthogonalIdempotents E := by
117     refine ⟨⟨fun i => ?_, fun i j hij => ?_⟩, ?_⟩
118     · change op (e i) * op (e i) = op (e i)
119       rw [← op_mul, he.idem i]
120     · change op (e i) * op (e j) = 0
121       rw [← op_mul, he.ortho (Ne.symm hij), op_zero]
122     · change ∑ i, opAddEquiv (e i) = 1
123       rw [← map_sum, he.complete]
124       rfl
125   have hE0 : ∀ i, E i ≠ 0 := by
126     intro i h0
127     have h2 : e i = 0 := op_injective h0
128     have h := (hsimple i).nontrivial
129     rw [h2] at h
130     exact (Submodule.nontrivial_iff_ne_bot.mp h) (by simp)
131   -- the right ideal of denominators clearing `E i`
132   let J : Fin n → Submodule Rᵐᵒᵖ R := fun i =>
133     { carrier := {x | ∃ a, E i * ι x = ι a}
134       add_mem' := by
135         rintro x y ⟨a, ha⟩ ⟨b, hb⟩
136         exact ⟨a + b, by rw [map_add, mul_add, ha, hb, map_add]⟩
137       zero_mem' := ⟨0, by rw [map_zero, mul_zero]⟩
138       smul_mem' := by
139         rintro c x ⟨a, ha⟩
140         refine ⟨a * unop c, ?_⟩
141         change E i * ι (x * unop c) = ι (a * unop c)
142         rw [map_mul, ← mul_assoc, ha, map_mul] }
143   obtain ⟨c, hc, hcJ⟩ := exists_mem_nonZeroDivisors_finset Finset.univ J fun i _ => by
144     obtain ⟨a, s, hs, -, hsa⟩ := rightOre_clear (S := R⁰) (E i)
145     exact ⟨s, ⟨a, hsa⟩, hs⟩
146   choose f hf using fun i => hcJ i (Finset.mem_univ i)
147   have hcu := isUnit_toQuotientRing hc
148   have hrt : ∀ i, ∃ r t : R, t ∈ R⁰ ∧ ι c * ι r = E i * ι t := by
149     intro i
150     obtain ⟨a, t, ht, -, hta⟩ := rightOre_clear (S := R⁰) (↑hcu.unit⁻¹ * E i)
151     refine ⟨a, t, ht, ?_⟩
152     rw [← hta, ← mul_assoc, ← mul_assoc]
153     change ↑hcu.unit * ↑hcu.unit⁻¹ * E i * ι t = E i * ι t
154     rw [Units.mul_inv, one_mul]
155   choose r t ht hrt using hrt
156   have hfr : ∀ i j, ι (f i * r j) = E i * E j * ι (t j) := by
157     intro i j
158     rw [map_mul, ← hf i, mul_assoc, hrt j, mul_assoc]
159   refine ⟨n, E, c, t, f, r, hn.trans (goldieRank_self R).symm, hE, hsimple, hc, ht, hf, hrt,
160     fun i j hij => ?_, fun i => ?_, fun i h0 => ?_⟩
161   · apply toQuotientRing_injective
162     change ι (f i * r j) = ι 0
163     rw [hfr, hE.ortho hij, zero_mul, map_zero]
164   · rw [hfr, (hE.idem i).eq]
165   · have h1 : E i * ι (t i) = 0 := by
166       rw [← (hE.idem i).eq, ← hfr]
167       rw [h0, map_zero]
168     exact hE0 i ((isUnit_toQuotientRing (ht i)).mul_left_eq_zero.mp h1)
