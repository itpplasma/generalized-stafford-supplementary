9 # Goldie's theorem: the quotient ring of a simple right Noetherian ring
10 
11 Let `R` be a simple right Noetherian ring. By `Goldie.exists_right_ore` its
12 regular elements form a right Ore set, so `R` has a classical right quotient
13 ring `Q`, whose elements are right fractions `r s⁻¹` with `s` regular. Goldie's
14 theorem says that `Q` is a simple Artinian ring; by the Wedderburn–Artin
15 theorem it is a matrix ring `Mₙ(D)` over a division ring `D`.
16 
17 Mathlib's `OreLocalization` builds left fractions `s⁻¹ r`. As in
18 `AlgebraicAnalysis.OreRightLocalization.RightOreLocalization`, we form the Ore
19 localization `Q°` of the opposite ring `Rᵐᵒᵖ` at the opposite copy of `R⁰`
20 (`Goldie.OppositeQuotientRing R`) and set `Q = (Q°)ᵐᵒᵖ`
21 (`Goldie.QuotientRing R`, definitionally `RightOreLocalization R R⁰`). Left
22 ideals of `Q°` are right ideals of `Q`; simplicity and semisimplicity pass
23 between a ring and its opposite (Mathlib instances).
24 
25 The Ore set structure is data, so the results below take an instance
26 `[OreLocalization.OreSet (oppositeSubmonoid R⁰)]`;
27 `Goldie.nonempty_oreSet_oppositeSubmonoid` shows that one exists. (Different
28 choices give isomorphic rings by the universal property; this is not used.)
29 
30 ## Main results
31 
32 * `Goldie.isSimpleRing_quotientRing`: `Q` is a simple ring;
33 * `Goldie.isSemisimpleRing_quotientRing`, `Goldie.isArtinianRing_quotientRing`,
34   `Goldie.isArtinianRing_oppositeQuotientRing`: `Q` is semisimple, and left and
35   right Artinian;
36 * `Goldie.exists_ringEquiv_matrix_divisionRing`: `Q ≃ Mₙ(D)` for some `n ≠ 0`
37   and division ring `D`;
38 * `Goldie.length_oppositeQuotientRing_ne_top`,
39   `Goldie.length_oppositeQuotientRing_pos`: `Q` has finite positive length as a
40   right module over itself;
41 * `Goldie.toQuotientRing`: the canonical ring map `R → Q`, injective and
42   sending regular elements to units.
43 
44 ## Proof
45 
46 Write `ι : R → Q` for the canonical map; every `q ∈ Q` has the form
47 `q = ι(r) ι(t)⁻¹` with `t` regular, `ι` is injective, and `ι(t)` is a unit for
48 regular `t`.
49 
50 *Every essential right ideal `E` of `Q` is `Q`.* The right ideal
51 `J = ι⁻¹(E)` of `R` is essential: for `y ≠ 0`, `ι(y) Q` meets `E` in some
52 `ι(y) q ≠ 0`; writing `q = ι(r) ι(t)⁻¹` gives `ι(y r) = ι(y) q ι(t) ∈ E`, and
53 `y r ≠ 0`. By Goldie (`exists_mem_nonZeroDivisors_of_isEssential`) `J`
54 contains a regular `s`, and `ι(s) ∈ E` is a unit.
55 
56 *`Q` is semisimple.* For a right ideal `L` of `Q` choose, by Zorn's lemma, a
57 right ideal `L'` maximal with `L ∩ L' = 0`. Then `L + L'` is essential: if
58 `N ∩ (L + L') = 0`, the modular law gives `L ∩ (L' + N) = 0`, so `N ≤ L'` by
59 maximality and `N = 0`. Hence `L + L' = Q` and `L'` is a complement of `L`.
60 
61 *`Q` is simple.* A nonzero two-sided ideal `T` of `Q` contains some
62 `q = ι(r) ι(t)⁻¹ ≠ 0`, hence `ι(r) = q ι(t) ≠ 0`; then `ι⁻¹(T)` is a nonzero
63 two-sided ideal of the simple ring `R`, so `1 ∈ T`.
64 
65 A simple semisimple ring is Artinian, and Wedderburn–Artin applies
66 (`IsSimpleRing.exists_ringEquiv_matrix_divisionRing`).
67

88 /-- **Goldie (right Ore condition).** The opposite copy of the regular
89 elements of a simple right Noetherian ring is an Ore set of `Rᵐᵒᵖ` in
90 Mathlib's convention. -/
91 theorem nonempty_oreSet_oppositeSubmonoid {R : Type u} [Ring R] [IsSimpleRing R]
92     [IsNoetherian Rᵐᵒᵖ R] : Nonempty (OreLocalization.OreSet (oppositeSubmonoid R⁰)) := by
93   refine OreLocalization.nonempty_oreSet_iff.mpr ⟨?_, ?_⟩
94   · rintro r₁ r₂ ⟨s, hs⟩ he
95     refine ⟨1, ?_⟩
96     have hs' : unop s ∈ R⁰ := hs
97     have h : unop s * (unop r₁ - unop r₂) = 0 := by
98       rw [mul_sub, sub_eq_zero]
99       exact congrArg unop he
100     rw [unop_injective (sub_eq_zero.mp (hs'.1 _ h))]
101   · rintro r ⟨s, hs⟩
102     obtain ⟨a', s', hs', he⟩ := exists_right_ore (unop r) (show unop s ∈ R⁰ from hs)
103     exact ⟨op a', ⟨op s', hs'⟩, unop_injective he⟩

126 /-- Every element of `Q°` becomes a numerator after left multiplication by a
127 regular numerator: `t q = r` for some regular `t`. -/
128 theorem exists_numerator_mul_eq (q : OppositeQuotientRing R) :
129     ∃ t ∈ oppositeSubmonoid R⁰, ∃ r : Rᵐᵒᵖ, numeratorRingHom (S := oppositeSubmonoid R⁰) t * q =
130       numeratorRingHom r := by
131   induction q using OreLocalization.ind with
132   | _ r s =>
133     refine ⟨s, s.2, r, ?_⟩
134     change ((s : Rᵐᵒᵖ) /ₒ (1 : oppositeSubmonoid R⁰)) * (r /ₒ s) = r /ₒ 1
135     exact OreLocalization.mul_cancel

157 /-- In `Q°`, a left ideal meeting every nonzero left ideal is everything (in `Q`:
158 every essential right ideal is `Q`). -/
159 theorem eq_top_of_forall_inf_eq_bot (E : Submodule (OppositeQuotientRing R)
160     (OppositeQuotientRing R)) (hE : ∀ N, N ⊓ E = ⊥ → N = ⊥) : E = ⊤ := by
161   let ι := numeratorRingHom (R := Rᵐᵒᵖ) (S := oppositeSubmonoid R⁰)
162   let J : Submodule Rᵐᵒᵖ R :=
163     { carrier := {r | ι (op r) ∈ E}
164       add_mem' := fun {x y} hx hy => by
165         change ι (op (x + y)) ∈ E
166         rw [op_add, map_add]
167         exact E.add_mem hx hy
168       zero_mem' := by
169         change ι (op 0) ∈ E
170         rw [op_zero, map_zero]
171         exact E.zero_mem
172       smul_mem' := fun c x hx => by
173         change ι (op (x * unop c)) ∈ E
174         rw [op_mul, op_unop, map_mul]
175         exact E.smul_mem (ι c) hx }
176   have hJ : IsEssential J := by
177     intro y hy
178     have hq0 : ι (op y) ≠ 0 := fun h =>
179       hy (op_injective (numeratorRingHom_injective (h.trans (map_zero ι).symm)))
180     have hspan : Submodule.span _ {ι (op y)} ⊓ E ≠ ⊥ := fun h =>
181       hq0 (Submodule.span_singleton_eq_bot.mp (hE _ h))
182     obtain ⟨z, ⟨hzspan, hzE⟩, hz0⟩ := Submodule.exists_mem_ne_zero_of_ne_bot hspan
183     obtain ⟨q, rfl⟩ := Submodule.mem_span_singleton.mp hzspan
184     obtain ⟨t, ht, r, hr⟩ := exists_numerator_mul_eq q
185     have key : ι (op (y * unop r)) = ι t * (q • ι (op y)) := by
186       rw [op_mul, op_unop, map_mul, ← hr, smul_eq_mul, mul_assoc]
187     refine ⟨unop r, fun h0 => hz0 ?_, ?_⟩
188     · have h1 : ι t * (q • ι (op y)) = 0 := by rw [← key, h0, op_zero, map_zero]
189       exact (isUnit_numeratorRingHom ht).mul_right_eq_zero.mp h1
190     · change ι (op (y * unop r)) ∈ E
191       rw [key]
192       exact E.smul_mem (ι t) hzE
193   obtain ⟨s, hsJ, hs⟩ := exists_mem_nonZeroDivisors_of_isEssential hJ
194   exact Ideal.eq_top_of_isUnit_mem E hsJ
195     (isUnit_numeratorRingHom (s := op s) hs)

197 /-- Every left ideal of `Q°` has a complement (in `Q`: every right ideal is a
198 direct summand). -/
199 theorem exists_isCompl (L : Submodule (OppositeQuotientRing R) (OppositeQuotientRing R)) :
200     ∃ L', IsCompl L L' := by
201   obtain ⟨L', hL', hmax⟩ := zorn_le₀ {N | Disjoint L N} fun c hc hchain =>
202     ⟨sSup c, hchain.directedOn.disjoint_sSup_right.mpr fun _ hb => hc hb,
203       fun _ hz => le_sSup hz⟩
204   refine ⟨L', hL', codisjoint_iff.mpr (eq_top_of_forall_inf_eq_bot _ fun N hN => ?_)⟩
205   have hdisj : Disjoint (L ⊔ L') N := disjoint_iff.mpr (by rw [inf_comm, hN])
206   have hle : L' ⊔ N ≤ L' :=
207     hmax (Disjoint.disjoint_sup_right_of_disjoint_sup_left hL' hdisj) le_sup_left
208   have hNL : N ≤ L ⊔ L' := (le_sup_right.trans hle).trans le_sup_right
209   rw [← hN, inf_eq_left.mpr hNL]

217 instance isSimpleRing_oppositeQuotientRing : IsSimpleRing (OppositeQuotientRing R) := by
218   have : Nontrivial (OppositeQuotientRing R) :=
219     OreLocalization.nontrivial_of_nonZeroDivisors fun _ hx =>
220       mem_oppositeSubmonoid_nonZeroDivisors.mp hx
221   refine IsSimpleRing.of_eq_bot_or_eq_top fun T => or_iff_not_imp_left.mpr fun hT => ?_
222   let ι := numeratorRingHom (R := Rᵐᵒᵖ) (S := oppositeSubmonoid R⁰)
223   obtain ⟨z, hzT, hz0⟩ : ∃ z ∈ T, z ≠ 0 := by
224     by_contra! h
225     exact hT (TwoSidedIdeal.ext fun x => ⟨fun hx => by simp [h x hx], fun hx => by
226       rw [(TwoSidedIdeal.mem_bot _).mp hx]; exact T.zero_mem⟩)
227   obtain ⟨t, ht, r, hr⟩ := exists_numerator_mul_eq z
228   have hrT : r ∈ T.comap ι := by
229     rw [TwoSidedIdeal.mem_comap, ← hr]
230     exact T.mul_mem_left _ _ hzT
231   have hr0 : r ≠ 0 := by
232     rintro rfl
233     rw [map_zero] at hr
234     exact hz0 ((isUnit_numeratorRingHom ht).mul_right_eq_zero.mp hr)
235   have h1 := IsSimpleRing.one_mem_of_ne_zero_mem _ hr0 hrT
236   rw [TwoSidedIdeal.mem_comap, map_one] at h1
237   exact (TwoSidedIdeal.one_mem_iff T).mp h1
