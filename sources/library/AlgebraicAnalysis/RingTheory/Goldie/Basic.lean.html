7 /-!
8 # Goldie theory for simple right Noetherian rings
9 
10 Let `R` be a simple ring which is right Noetherian (`IsNoetherian Rᵐᵒᵖ R`); zero
11 divisors are allowed. This file proves the ring-theoretic half of Goldie's
12 theorem:
13 
14 * the right singular ideal vanishes (`eq_zero_of_isEssential_rightAnn`);
15 * every nonzero right ideal contains a nonzero good element `b`, i.e. one with
16   `bR ∩ r(b) = 0` (`exists_good_mem`);
17 * an element with zero right annihilator generates an essential right ideal
18   and is regular (`isEssential_principal`, `mem_nonZeroDivisors_of_rightAnn`);
19 * every essential right ideal contains a regular element
20   (`exists_mem_nonZeroDivisors_of_isEssential`);
21 * the regular elements satisfy the right Ore condition (`exists_right_ore`),
22   so finitely many right ideals containing regular elements have a common
23   regular element (`exists_mem_nonZeroDivisors_finset`), and `(Rᵐᵒᵖ)⁰` is an
24   Ore set in Mathlib's convention (`nonempty_oreSet_op`);
25 * every right ideal `I` contains an element `a` such that `I / aR` is
26   Goldie-torsion: each `y ∈ I` has a regular `s` with `y s ∈ aR`
27   (`exists_mem_forall_exists_mul_mem`).
28 
29 Right ideals are `Submodule Rᵐᵒᵖ R`; `x * y` is always read in the displayed
30 order. "Regular" means two-sided regular, `s ∈ R⁰`. An essential right ideal
31 `E` is one meeting every nonzero right ideal; in elements, every `y ≠ 0` has a
32 right multiple `y r ≠ 0` in `E`.
33 
34 ## Proof of the last statement
35 
36 Let `C` be a right ideal maximal with `I ∩ C = 0` (it exists because `R` is
37 right Noetherian). Then `I + C` is essential: if `y ≠ 0` and `yR ∩ (I + C) = 0`,
38 then `I ∩ (C + yR) = 0` and `C + yR` strictly contains `C`. By Goldie,
39 `I + C` contains a regular `s = a + c` with `a ∈ I`, `c ∈ C`. For `y ∈ I` the
40 right ideal `{r | y r ∈ sR}` is essential, as the inverse image of the
41 essential right ideal `sR`, so it contains a regular `t`; write `y t = s r'`.
42 Then `c r' = y t - a r'` lies in `I ∩ C = 0`, so `y t = a r'`.
43 
44 ## References
45 
46 The argument follows the classical proof (A. W. Goldie, *The structure of
47 prime rings under ascending chain conditions*, Proc. London Math. Soc. (3) 8
48 (1958) 589–608, and *Semi-prime rings with maximum condition*, ibid. 10 (1960)
49 201–220; J. C. McConnell and J. C. Robson, *Noncommutative Noetherian Rings*,
50 §2.3, in particular 2.3.6 and 2.3.10; K. R. Goodearl and R. B. Warfield, *An
51 Introduction to Noncommutative Noetherian Rings*, Ch. 5–6), specialized to
52 simple rings: simplicity replaces semiprimeness and the nilpotency of the
53 singular ideal, and the right Noetherian hypothesis replaces finite uniform
54 dimension and the ascending chain condition on right annihilators. Nothing
55 here is new.
56 -/

73 /-- An essential right ideal: every nonzero `y` has a nonzero right multiple
74 `y r` in `E`, i.e. `E` meets every nonzero right ideal nontrivially. -/
75 def IsEssential (E : Submodule Rᵐᵒᵖ R) : Prop :=
76   ∀ y : R, y ≠ 0 → ∃ r : R, y * r ≠ 0 ∧ y * r ∈ E
77 
78 theorem IsEssential.mono {E F : Submodule Rᵐᵒᵖ R} (hE : IsEssential E) (h : E ≤ F) :
79     IsEssential F := fun y hy => by
80   obtain ⟨r, hr, hrE⟩ := hE y hy
81   exact ⟨r, hr, h hrE⟩
82 
83 theorem IsEssential.inf {E F : Submodule Rᵐᵒᵖ R} (hE : IsEssential E) (hF : IsEssential F) :
84     IsEssential (E ⊓ F) := fun y hy => by
85   obtain ⟨r, hr, hrE⟩ := hE y hy
86   obtain ⟨r', hr', hr'F⟩ := hF (y * r) hr
87   refine ⟨r * r', by rwa [← mul_assoc], ?_, by rwa [← mul_assoc]⟩
88   have := E.smul_mem (op r') hrE
89   rwa [op_smul_eq_mul, mul_assoc] at this

91 /-- The inverse image `{y | a y ∈ E}` of an essential right ideal is essential. -/
92 theorem IsEssential.comap {E : Submodule Rᵐᵒᵖ R} (hE : IsEssential E) (a : R) :
93     IsEssential (E.comap (leftMulLinear a)) := fun y hy => by
94   by_cases hay : a * y = 0
95   · refine ⟨1, by rwa [mul_one], ?_⟩
96     change a * (y * 1) ∈ E
97     rw [mul_one, hay]
98     exact E.zero_mem
99   · obtain ⟨r, hr, hrE⟩ := hE (a * y) hay
100     refine ⟨r, fun h => hr (by rw [mul_assoc, h, mul_zero]), ?_⟩
101     change a * (y * r) ∈ E
102     rwa [← mul_assoc]

104 theorem not_isEssential_bot [Nontrivial R] : ¬ IsEssential (⊥ : Submodule Rᵐᵒᵖ R) := by
105   intro h
106   obtain ⟨r, hr, hr0⟩ := h 1 one_ne_zero
107   exact hr ((Submodule.mem_bot _).mp hr0)

109 theorem IsEssential.exists_ne_zero_mem_inf {E K : Submodule Rᵐᵒᵖ R} (hE : IsEssential E)

119 /-- The right singular ideal `Z(R_R) = {y | r(y) essential}`. -/
120 def singularIdeal (R : Type u) [Ring R] : TwoSidedIdeal R :=
121   TwoSidedIdeal.mk' {y : R | IsEssential (rightAnn y)}
122     (fun y hy => ⟨1, by rwa [mul_one], by
123       change (0 : R) * (y * 1) = 0
124       rw [zero_mul]⟩)
125     (fun {x y} hx hy => (IsEssential.inf hx hy).mono fun z hz => by
126       change (x + y) * z = 0
127       rw [add_mul, mem_rightAnn.mp hz.1, mem_rightAnn.mp hz.2, add_zero])
128     (fun {x} hx => hx.mono fun z hz => by
129       change -x * z = 0
130       rw [neg_mul, mem_rightAnn.mp hz, neg_zero])
131     (fun {x y} hy => hy.mono fun z hz => by
132       change x * y * z = 0
133       rw [mul_assoc, mem_rightAnn.mp hz, mul_zero])
134     (fun {x y} hx => (hx.comap y).mono fun z hz => by
135       change x * (y * z) = 0 at hz
136       change x * y * z = 0
137       rwa [mul_assoc])

139 /-- **The right singular ideal of a simple ring is zero.** -/
140 theorem eq_zero_of_isEssential_rightAnn [IsSimpleRing R] {y : R}
141     (hy : IsEssential (rightAnn y)) : y = 0 := by
142   by_contra hne
143   have h1 : (1 : R) ∈ singularIdeal R :=
144     IsSimpleRing.one_mem_of_ne_zero_mem _ hne
145       ((TwoSidedIdeal.mem_mk' _ _ _ _ _ _ _).mpr hy)
146   have h1' : IsEssential (rightAnn (1 : R)) := (TwoSidedIdeal.mem_mk' _ _ _ _ _ _ _).mp h1
147   obtain ⟨r, hr, hr0⟩ := h1' 1 one_ne_zero
148   have : (1 : R) * (1 * r) = 0 := hr0
149   exact hr (by simpa using this)

156 /-- A simple right Noetherian ring has no nonzero nil right ideal. -/
157 theorem exists_not_nilpotent_mem [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
158     {K : Submodule Rᵐᵒᵖ R} (hK : K ≠ ⊥) : ∃ a ∈ K, ∀ n : ℕ, a ^ n ≠ 0 := by
159   classical
160   by_contra! hnil
161   obtain ⟨_, ⟨a, haK, ha0, rfl⟩, hmax⟩ :=
162     (set_has_maximal_iff_noetherian (R := Rᵐᵒᵖ) (M := R)).mpr inferInstance
163       {L | ∃ a ∈ K, a ≠ 0 ∧ rightAnn a = L}
164       (by
165         obtain ⟨a, haK, ha⟩ := Submodule.exists_mem_ne_zero_of_ne_bot hK
166         exact ⟨_, a, haK, ha, rfl⟩)
167   -- `a R a = 0`.
168   have haxa : ∀ x : R, a * x * a = 0 := by
169     intro x
170     by_contra hne
171     set y := x * a with hy
172     have hay : a * y ≠ 0 := by rwa [hy, ← mul_assoc]
173     have haxK : a * x ∈ K := by
174       have := K.smul_mem (op x) haK
175       rwa [op_smul_eq_mul] at this
176     obtain ⟨m, hm⟩ := hnil (a * x) haxK
177     have hex : ∃ n, a * y ^ (n + 1) = 0 := ⟨m, by
178       rw [hy, ← mul_pow_mul, pow_succ, hm, zero_mul, zero_mul]⟩
179     set t := Nat.find hex with ht
180     have hta : a * y ^ (t + 1) = 0 := Nat.find_spec hex
181     have hc0 : a * y ^ t ≠ 0 := by
182       rcases Nat.eq_zero_or_eq_succ_pred t with h0 | hs
183       · rw [h0, pow_zero, mul_one]; exact ha0
184       · rw [hs]
185         exact Nat.find_min hex (by omega)
186     have hcK : a * y ^ t ∈ K := by
187       have := K.smul_mem (op (y ^ t)) haK
188       rwa [op_smul_eq_mul] at this
189     have hle : rightAnn a ≤ rightAnn (a * y ^ t) := by
190       intro z hz
191       change a * y ^ t * z = 0
192       rw [hy, ← mul_pow_mul, mul_assoc, mem_rightAnn.mp hz, mul_zero]
193     have heq : rightAnn a = rightAnn (a * y ^ t) := by
194       by_contra hne'
195       exact hmax _ ⟨_, hcK, hc0, rfl⟩ (lt_of_le_of_ne hle hne')
196     have hyann : y ∈ rightAnn (a * y ^ t) := by
197       change a * y ^ t * y = 0
198       rw [mul_assoc, ← pow_succ, hta]
199     rw [← heq] at hyann
200     exact hay hyann
201   -- Simplicity: the two-sided ideal `{c | a R c = 0}` contains `a`.
202   let J : TwoSidedIdeal R := TwoSidedIdeal.mk' {c | ∀ r : R, a * r * c = 0}
203     (fun r => mul_zero _)
204     (fun {x y} hx hy r => by rw [mul_add, hx r, hy r, add_zero])
205     (fun {x} hx r => by rw [mul_neg, hx r, neg_zero])
206     (fun {x y} hy r => by rw [← mul_assoc, mul_assoc a r x, hy])
207     (fun {x y} hx r => by rw [← mul_assoc, hx r, zero_mul])
208   have h1 : (1 : R) ∈ J := IsSimpleRing.one_mem_of_ne_zero_mem J ha0
209     (by simpa [J, TwoSidedIdeal.mem_mk'] using haxa)
210   have h1' : ∀ r : R, a * r * 1 = 0 := by simpa [J, TwoSidedIdeal.mem_mk'] using h1
211   exact ha0 (by simpa using h1' 1)

213 /-- Every nonzero right ideal of a simple right Noetherian ring contains a
214 nonzero good element. -/
215 theorem exists_good_mem [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
216     {K : Submodule Rᵐᵒᵖ R} (hK : K ≠ ⊥) : ∃ b ∈ K, b ≠ 0 ∧ IsGood b := by
217   obtain ⟨a, haK, hnil⟩ := exists_not_nilpotent_mem hK
218   have hmono : Monotone fun n : ℕ => rightAnn (a ^ n) := by
219     refine monotone_nat_of_le_succ fun n z hz => ?_
220     change a ^ (n + 1) * z = 0
221     rw [pow_succ', mul_assoc, mem_rightAnn.mp hz, mul_zero]
222   obtain ⟨n, hn⟩ := (monotone_stabilizes_iff_noetherian (R := Rᵐᵒᵖ) (M := R)).mpr
223     inferInstance ⟨_, hmono⟩
224   have h1 : rightAnn (a ^ (n + 1)) = rightAnn (a ^ (n + 1 + (n + 1))) := by
225     have e1 := hn (n + 1) (by omega)
226     have e2 := hn (n + 1 + (n + 1)) (by omega)
227     exact e1.symm.trans e2
228   refine ⟨a ^ (n + 1), ?_, hnil _, fun z hz => ?_⟩
229   · have := K.smul_mem (op (a ^ n)) haK
230     rwa [op_smul_eq_mul, ← pow_succ'] at this
231   · have hz' : z ∈ rightAnn (a ^ (n + 1 + (n + 1))) := by
232       change a ^ (n + 1 + (n + 1)) * z = 0
233       rwa [pow_add, mul_assoc]
234     rw [← h1] at hz'
235     exact hz'

239 /-- Every essential right ideal contains an element with zero right

241 theorem exists_mem_rightAnn_eq_bot [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
242     {E : Submodule Rᵐᵒᵖ R} (hE : IsEssential E) :
243     ∃ x ∈ E, ∀ z : R, x * z = 0 → z = 0 := by
244   classical
245   obtain ⟨L, ⟨x, hxE, hxg, hLx⟩, hmax⟩ :=
246     (set_has_maximal_iff_noetherian (R := Rᵐᵒᵖ) (M := R)).mpr inferInstance
247       {L | ∃ x ∈ E, IsGood x ∧ L ⊓ rightAnn x = ⊥}
248       ⟨⊥, 0, E.zero_mem, fun z _ => by rw [zero_mul], bot_inf_eq _⟩
249   refine ⟨x, hxE, ?_⟩
250   by_contra! hex
251   have hann : rightAnn x ≠ ⊥ := by
252     obtain ⟨z, hz, hz0⟩ := hex
253     intro h
254     have : z ∈ rightAnn x := hz
255     rw [h] at this
256     exact hz0 ((Submodule.mem_bot _).mp this)
257   obtain ⟨y, hy, hy0⟩ := hE.exists_ne_zero_mem_inf hann
258   have hEK : E ⊓ rightAnn x ≠ ⊥ := fun h => hy0 ((Submodule.mem_bot _).mp (h ▸ hy))
259   obtain ⟨a, ⟨haE, hax⟩, ha0, hag⟩ := exists_good_mem hEK
260   have hxa : x * a = 0 := hax
261   -- `r(x + a) = r(x) ∩ r(a)`.
262   have hann' : ∀ w : R, (x + a) * w = 0 → x * w = 0 ∧ a * w = 0 := by
263     intro w hw
264     have hxw : x * (x * w) = 0 := by
265       have h2 : x * w = -(a * w) := eq_neg_of_add_eq_zero_left (by rwa [← add_mul])
266       rw [h2, mul_neg, ← mul_assoc, hxa, zero_mul, neg_zero]
267     have h1 := hxg w hxw
268     refine ⟨h1, ?_⟩
269     rwa [add_mul, h1, zero_add] at hw
270   have hgood : IsGood (x + a) := by
271     intro z hz
272     obtain ⟨h1, h2⟩ := hann' _ hz
273     have hxz : x * z = 0 := hxg z (by
274       rw [add_mul, mul_add, ← mul_assoc x a z, hxa, zero_mul, add_zero] at h1; exact h1)
275     have haz : a * z = 0 := hag z (by
276       rw [add_mul, mul_add, hxz, mul_zero, zero_add] at h2; exact h2)
277     rw [add_mul, hxz, haz, add_zero]
278   have hL' : L ⊔ principalRightIdeal a ∈ {L | ∃ x ∈ E, IsGood x ∧ L ⊓ rightAnn x = ⊥} := by
279     refine ⟨x + a, E.add_mem hxE haE, hgood, ?_⟩
280     rw [Submodule.eq_bot_iff]
281     rintro m ⟨hm, hmann⟩
282     obtain ⟨l, hl, _, ⟨s, rfl⟩, rfl⟩ := Submodule.mem_sup.mp hm
283     change l + a * s ∈ rightAnn (x + a) at hmann
284     obtain ⟨h1, h2⟩ := hann' _ hmann
285     have hxl : x * l = 0 := by
286       rwa [mul_add, ← mul_assoc, hxa, zero_mul, add_zero] at h1
287     have hl0 : l = 0 := by
288       have : l ∈ L ⊓ rightAnn x := ⟨hl, hxl⟩
289       rw [hLx] at this
290       exact (Submodule.mem_bot _).mp this
291     subst hl0
292     rw [zero_add] at h2 ⊢
293     exact hag s h2
294   have hlt : L < L ⊔ principalRightIdeal a := by
295     refine lt_of_le_of_ne le_sup_left fun h => ha0 ?_
296     have haL : a ∈ L := h ▸ Submodule.mem_sup_right ⟨1, mul_one a⟩
297     have : a ∈ L ⊓ rightAnn x := ⟨haL, hax⟩
298     rw [hLx] at this
299     exact (Submodule.mem_bot _).mp this
300   exact hmax _ hL' hlt

302 /-- An element with zero right annihilator generates an essential right ideal
303 (right Noetherian rings have no infinite direct sums `K ⊕ xK ⊕ x²K ⊕ ⋯`). -/
304 theorem isEssential_principal [IsNoetherian Rᵐᵒᵖ R] {x : R}
305     (hx : ∀ z : R, x * z = 0 → z = 0) : IsEssential (principalRightIdeal x) := by
306   intro y hy
307   by_contra! hdisj
308   set K := principalRightIdeal y with hK
309   set f := leftMulLinear x with hf
310   have hKx : ∀ k ∈ K, k ∈ LinearMap.range f → k = 0 := by
311     rintro _ ⟨r, rfl⟩ hk
312     by_contra h
313     exact hdisj r h hk
314   let B : ℕ → Submodule Rᵐᵒᵖ R := fun n => Nat.rec ⊥ (fun _ b => K ⊔ b.map f) n
315   have hB0 : B 0 = ⊥ := rfl
316   have hBs : ∀ n, B (n + 1) = K ⊔ (B n).map f := fun _ => rfl
317   have hmono : Monotone B := by
318     refine monotone_nat_of_le_succ fun n => ?_
319     induction n with
320     | zero => rw [hB0]; exact bot_le
321     | succ n ih => rw [hBs (n + 1), hBs n]; exact sup_le_sup_left (Submodule.map_mono ih) _
322   have hstep : ∀ m, B (m + 2) ≤ B (m + 1) → B (m + 1) ≤ B m := by
323     intro m hm u hu
324     have hxu' : x * u ∈ B (m + 2) := by
325       rw [hBs (m + 1)]
326       exact Submodule.mem_sup_right ⟨u, hu, rfl⟩
327     have hxu : x * u ∈ B (m + 1) := hm hxu'
328     rw [hBs m] at hxu
329     obtain ⟨k, hk, _, ⟨v, hv, rfl⟩, he⟩ := Submodule.mem_sup.mp hxu
330     have hk0 : k = 0 := hKx k hk ⟨u - v, by
331       change x * (u - v) = k
332       rw [mul_sub, ← he]
333       change k + x * v - x * v = k
334       abel⟩
335     subst hk0
336     have huv : u - v = 0 := hx _ (by
337       rw [mul_sub, ← he]
338       change 0 + x * v - x * v = 0
339       abel)
340     rwa [sub_eq_zero.mp huv]
341   have hdown : ∀ m, B (m + 1) ≤ B m → B 1 ≤ B 0 := by
342     intro m
343     induction m with
344     | zero => exact id
345     | succ m ih => exact fun h => ih (hstep m h)
346   obtain ⟨n, hn⟩ := (monotone_stabilizes_iff_noetherian (R := Rᵐᵒᵖ) (M := R)).mpr
347     inferInstance ⟨B, hmono⟩
348   have h1 : B 1 ≤ B 0 := hdown n (hn (n + 1) (by omega)).symm.le
349   have hyK : y ∈ B 1 := by rw [show (1 : ℕ) = 0 + 1 from rfl, hBs 0]; exact Submodule.mem_sup_left ⟨1, mul_one y⟩
350   exact hy ((Submodule.mem_bot _).mp (hB0 ▸ h1 hyK))

377 /-- **Goldie (right Ore condition).** In a simple right Noetherian ring, for
378 every `a` and regular `s` there are `a'` and regular `s'` with `a s' = s a'`. -/
379 theorem exists_right_ore [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R] (a : R) {s : R}
380     (hs : s ∈ R⁰) : ∃ a' s' : R, s' ∈ R⁰ ∧ a * s' = s * a' := by
381   obtain ⟨s', hs'F, hs'⟩ := exists_mem_nonZeroDivisors_of_isEssential
382     ((isEssential_principal hs.1).comap a)
383   obtain ⟨a', ha'⟩ := hs'F
384   exact ⟨a', s', hs', ha'.symm⟩

386 /-- Finitely many right ideals, each containing a regular element, have a
387 common regular element. -/
388 theorem exists_mem_nonZeroDivisors_finset [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
389     {ι : Type*} [DecidableEq ι] (s : Finset ι) (I : ι → Submodule Rᵐᵒᵖ R)
390     (hI : ∀ i ∈ s, ∃ x ∈ I i, x ∈ R⁰) : ∃ x ∈ R⁰, ∀ i ∈ s, x ∈ I i := by
391   induction s using Finset.induction_on with
392   | empty => exact ⟨1, Submonoid.one_mem _, by simp⟩
393   | @insert i s _ ih =>
394     obtain ⟨m, hm, hmI⟩ := ih fun j hj => hI j (Finset.mem_insert_of_mem hj)
395     obtain ⟨n, hnI, hn⟩ := hI i (Finset.mem_insert_self i s)
396     obtain ⟨a', s', hs', he⟩ := exists_right_ore n hm
397     refine ⟨n * s', Submonoid.mul_mem _ hn hs', fun j hj => ?_⟩
398     rcases Finset.mem_insert.mp hj with rfl | hj
399     · have := (I j).smul_mem (op s') hnI
400       rwa [op_smul_eq_mul] at this
401     · rw [he]
402       have := (I j).smul_mem (op a') (hmI j hj)
403       rwa [op_smul_eq_mul] at this

405 /-- A right ideal containing a regular element: every `f` has a regular `b`
406 with `f b ∈ I` (the quotient `R/I` is Goldie-torsion). -/
407 theorem exists_mul_mem_of_mem_nonZeroDivisors [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
408     {I : Submodule Rᵐᵒᵖ R} {s : R} (hsI : s ∈ I) (hs : s ∈ R⁰) (f : R) :
409     ∃ b ∈ R⁰, f * b ∈ I := by
410   obtain ⟨a', s', hs', he⟩ := exists_right_ore f hs
411   refine ⟨s', hs', ?_⟩
412   rw [he]
413   have := I.smul_mem (op a') hsI
414   rwa [op_smul_eq_mul] at this

416 /-- Regularity in `Rᵐᵒᵖ` is regularity in `R`. -/
417 theorem op_mem_nonZeroDivisors_iff {x : R} : op x ∈ (Rᵐᵒᵖ)⁰ ↔ x ∈ R⁰ := by
418   constructor
419   · rintro ⟨h1, h2⟩
420     exact ⟨fun y hy => op_injective (h2 (op y) (unop_injective hy)),
421       fun y hy => op_injective (h1 (op y) (unop_injective hy))⟩
422   · rintro ⟨h1, h2⟩
423     exact ⟨fun y hy => unop_injective (h2 (unop y) (op_injective hy)),
424       fun y hy => unop_injective (h1 (unop y) (op_injective hy))⟩

426 /-- **Goldie.** The regular elements of a simple right Noetherian ring form a
427 right Ore set; in Mathlib's left-handed convention, `(Rᵐᵒᵖ)⁰` is an Ore set of
428 `Rᵐᵒᵖ`, so the classical right quotient ring is the Ore localization of `Rᵐᵒᵖ`
429 at `(Rᵐᵒᵖ)⁰`. -/
430 theorem nonempty_oreSet_op [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R] :
431     Nonempty (OreLocalization.OreSet (Rᵐᵒᵖ)⁰) := by
432   refine OreLocalization.nonempty_oreSet_iff.mpr ⟨?_, ?_⟩
433   · rintro r₁ r₂ ⟨s, hs⟩ he
434     refine ⟨1, ?_⟩
435     have hs' : unop s ∈ R⁰ := op_mem_nonZeroDivisors_iff.mp hs
436     have h : unop s * (unop r₁ - unop r₂) = 0 := by
437       rw [mul_sub, sub_eq_zero]
438       exact congrArg unop he
439     rw [unop_injective (sub_eq_zero.mp (hs'.1 _ h))]
440   · rintro r ⟨s, hs⟩
441     obtain ⟨a', s', hs', he⟩ := exists_right_ore (unop r)
442       (op_mem_nonZeroDivisors_iff.mp hs)
443     exact ⟨op a', ⟨op s', op_mem_nonZeroDivisors_iff.mpr hs'⟩, unop_injective he⟩

447 /-- If `C` is maximal among right ideals with `I ⊓ C = ⊥`, then `I ⊔ C` is
448 essential. -/
449 theorem isEssential_sup_of_maximal {I C : Submodule Rᵐᵒᵖ R} (hIC : I ⊓ C = ⊥)
450     (hmax : ∀ C', C ≤ C' → I ⊓ C' = ⊥ → C' = C) : IsEssential (I ⊔ C) := by
451   intro y hy
452   by_contra! h
453   have hdisj : I ⊓ (C ⊔ principalRightIdeal y) = ⊥ := by
454     rw [Submodule.eq_bot_iff]
455     rintro z ⟨hzI, hz⟩
456     obtain ⟨c, hc, _, ⟨r, rfl⟩, rfl⟩ := Submodule.mem_sup.mp hz
457     have hyr : y * r = 0 := by
458       by_contra hne
459       refine h r hne ?_
460       have hsub : c + y * r - c ∈ I ⊔ C :=
461         Submodule.sub_mem _ (Submodule.mem_sup_left hzI) (Submodule.mem_sup_right hc)
462       rwa [add_sub_cancel_left] at hsub
463     change c + y * r ∈ I at hzI
464     change c + y * r = 0
465     rw [hyr, add_zero] at hzI ⊢
466     have : c ∈ I ⊓ C := ⟨hzI, hc⟩
467     rwa [hIC] at this
468   have heq := hmax _ le_sup_left hdisj
469   have hyC : y ∈ C := heq ▸ Submodule.mem_sup_right ⟨1, mul_one y⟩
470   exact h 1 (by rwa [mul_one]) (by rw [mul_one]; exact Submodule.mem_sup_right hyC)
471 
472 /-- **Every right ideal has a principal right ideal of full rank.** In a
473 simple right Noetherian ring, every right ideal `I` contains an element `a`
474 such that `I / aR` is Goldie-torsion: every `y ∈ I` has a regular `s` with
475 `y s ∈ aR`. -/
476 theorem exists_mem_forall_exists_mul_mem [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
477     (I : Submodule Rᵐᵒᵖ R) :
478     ∃ a ∈ I, ∀ y ∈ I, ∃ s ∈ R⁰, y * s ∈ principalRightIdeal a := by
479   obtain ⟨C, hC, hmax⟩ :=
480     (set_has_maximal_iff_noetherian (R := Rᵐᵒᵖ) (M := R)).mpr inferInstance
481       {C | I ⊓ C = ⊥} ⟨⊥, inf_bot_eq _⟩
482   have hE : IsEssential (I ⊔ C) := isEssential_sup_of_maximal hC fun C' hle hC' => by
483     by_contra hne
484     exact hmax C' hC' (lt_of_le_of_ne hle (Ne.symm hne))
485   obtain ⟨s, hsE, hs⟩ := exists_mem_nonZeroDivisors_of_isEssential hE
486   obtain ⟨a, haI, c, hcC, rfl⟩ := Submodule.mem_sup.mp hsE
487   refine ⟨a, haI, fun y hyI => ?_⟩
488   obtain ⟨t, ⟨r', hr'⟩, ht⟩ := exists_mem_nonZeroDivisors_of_isEssential
489     ((isEssential_principal hs.1).comap y)
490   change (a + c) * r' = y * t at hr'
491   have hytI : y * t ∈ I := by
492     have := I.smul_mem (op t) hyI
493     rwa [op_smul_eq_mul] at this
494   have harI : a * r' ∈ I := by
495     have := I.smul_mem (op r') haI
496     rwa [op_smul_eq_mul] at this
497   have hcr : c * r' ∈ I ⊓ C := by
498     refine ⟨?_, ?_⟩
499     · have : c * r' = y * t - a * r' := by rw [← hr', add_mul, add_sub_cancel_left]
500       rw [this]
501       exact I.sub_mem hytI harI
502     · have := C.smul_mem (op r') hcC
503       rwa [op_smul_eq_mul] at this
504   rw [hC, Submodule.mem_bot] at hcr
505   refine ⟨t, ht, r', ?_⟩
506   change a * r' = y * t
507   rw [← hr', add_mul, hcr, add_zero]
508 
509 end AlgebraicAnalysis.Goldie
