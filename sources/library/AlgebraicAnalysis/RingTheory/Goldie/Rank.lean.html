10 # Goldie rank of modules over a simple right Noetherian ring
11 
12 Let `R` be a ring whose regular elements form a right Ore set, with classical
13 right quotient ring `Q` (`Goldie.QuotientRing R`). For a right `R`-module `M`
14 the module of fractions `M_Q` consists of the fractions `m s⁻¹` with `s`
15 regular; it is a right `Q`-module. In Lean, right `R`-modules are
16 `Rᵐᵒᵖ`-modules, `M_Q` is Mathlib's Ore localization of `M` at the opposite copy
17 of `R⁰` (`Goldie.FractionModule R M`), and it is a left module over
18 `Q° = Goldie.OppositeQuotientRing R`, i.e. a right `Q`-module.
19 
20 * `Goldie.goldieTorsionPart R M`: the Goldie-torsion part `t(M)`, the span of
21   the elements killed by a regular element. It is the kernel of `M → M_Q`
22   (`Goldie.ker_toFractionModule`), so these elements already form a submodule
23   (`Goldie.mem_goldieTorsionPart_iff`).
24 * `Goldie.goldieRank R M = length_Q(M_Q)` (in `ℕ∞`). It is additive on short
25   exact sequences (`Goldie.goldieRank_eq_add_of_exact`), since localization is
26   exact; it is monotone under injective maps and decreases under quotients;
27   `goldieRank R (Fin m → Rᵐᵒᵖ) = m * goldieRank R R`; and it vanishes exactly on
28   Goldie-torsion modules (`Goldie.goldieRank_eq_zero_iff`).
29 
30 Now assume `R` is simple and right Noetherian, so that `Q` is simple Artinian
31 (Goldie's theorem, `AlgebraicAnalysis.RingTheory.Goldie.QuotientRing`).
32 
33 * `Goldie.goldieRank_ne_top`: a finitely generated `M` has finite Goldie rank,
34   and `Goldie.goldieRank_self = length_Q(Q_Q)` is finite and positive.
35 * **Uniform dimension.** `Goldie.uniformDimGE_quotient_iff`,
36   `Goldie.uniformDimLE_quotient_iff`: the uniform dimension of `M/t(M)` is the
37   Goldie rank of `M`; `Goldie.uniformDimGE_self_iff`,
38   `Goldie.uniformDimLE_self_iff`: the uniform dimension of `R_R` is
39   `goldieRank R R = length_Q(Q_Q)`.
40 * `Goldie.goldieRank_le_card_mul`: a module generated by `m` elements has
41   Goldie rank at most `m * goldieRank R R`; in uniform-dimension form
42   `Goldie.le_mul_of_uniformDim`.
43 
44 ## Proof of the uniform dimension bridge
45 
46 The map `M/t(M) → M_Q` is injective, and `(m s⁻¹) s = m` lies in its image for
47 every fraction. Hence `M/t(M)` and `M_Q` contain direct sums of the same numbers
48 of nonzero submodules (`OreModule.uniformDimGE_iff_of_injective`, by
49 saturating submodules). As `Q` is semisimple, so is `M_Q`, and its uniform
50 dimension is its length (`Module.uniformDimGE_iff_le_length`).
51

71 /-- The Goldie-torsion part `t(M)` of a right `R`-module `M`: the submodule
72 spanned by the elements `m` with `m s = 0` for some regular `s`. Under a right
73 Ore condition these elements already form a submodule
74 (`mem_goldieTorsionPart_iff`). -/
75 def goldieTorsionPart : Submodule Rᵐᵒᵖ M :=
76   Submodule.span Rᵐᵒᵖ {m : M | ∃ s ∈ R⁰, op s • m = 0}
77 
78 variable [OreLocalization.OreSet (oppositeSubmonoid R⁰)]
79 
80 /-- The module of fractions `M_Q` of a right `R`-module `M`: Mathlib's Ore
81 localization of `M` at the regular elements, a left module over
82 `Q° = OppositeQuotientRing R`, that is, a right module over the classical right
83 quotient ring `Q`. -/
84 abbrev FractionModule : Type (max u v) := OreLocalization (oppositeSubmonoid R⁰) M
85 
86 /-- The canonical map `M → M_Q`, `m ↦ m 1⁻¹`. -/
87 abbrev toFractionModule : M →ₗ[Rᵐᵒᵖ] FractionModule R M :=
88   toLocalization (oppositeSubmonoid R⁰) M
89 
90 /-- The **Goldie rank** `length_Q(M ⊗_R Q)` of a right `R`-module `M`, the length
91 of its module of fractions `M_Q` as a right `Q`-module. It is finite for
92 finitely generated `M` over a simple right Noetherian ring (`goldieRank_ne_top`).
93 The normalized rank of the paper is `goldieRank R M / goldieRank R R`. -/
94 noncomputable def goldieRank : ℕ∞ :=
95   Module.length (OppositeQuotientRing R) (FractionModule R M)
96 
97 end Definitions

106 /-- An element maps to zero in `M_Q` iff it is killed by a regular element. -/
107 theorem toFractionModule_eq_zero_iff {m : M} :
108     toFractionModule R M m = 0 ↔ ∃ s ∈ R⁰, op s • m = 0 := by
109   rw [toLocalization_eq_zero_iff]
110   exact ⟨fun ⟨s, hs, h⟩ => ⟨unop s, hs, h⟩, fun ⟨s, hs, h⟩ => ⟨op s, hs, h⟩⟩
111 
112 variable (R M) in
113 /-- The Goldie-torsion part is the kernel of `M → M_Q`. -/
114 theorem ker_toFractionModule : LinearMap.ker (toFractionModule R M) = goldieTorsionPart R M := by
115   apply le_antisymm
116   · intro m hm
117     exact Submodule.subset_span (toFractionModule_eq_zero_iff.mp hm)
118   · rw [goldieTorsionPart, Submodule.span_le]
119     intro m hm
120     exact toFractionModule_eq_zero_iff.mpr hm
121 
122 /-- **The Goldie-torsion elements form a submodule.** `m ∈ t(M)` iff `m s = 0`
123 for some regular `s`. -/
124 theorem mem_goldieTorsionPart_iff {m : M} :
125     m ∈ goldieTorsionPart R M ↔ ∃ s ∈ R⁰, op s • m = 0 := by
126   rw [← ker_toFractionModule, LinearMap.mem_ker, toFractionModule_eq_zero_iff]
127 
128 /-- `M` is Goldie-torsion iff `t(M) = M`. -/
129 theorem goldieTorsionPart_eq_top_iff : goldieTorsionPart R M = ⊤ ↔ IsGoldieTorsion R M := by
130   rw [eq_top_iff]
131   exact ⟨fun h m => mem_goldieTorsionPart_iff.mp (h Submodule.mem_top),
132     fun h m _ => mem_goldieTorsionPart_iff.mpr (h m)⟩
133 
134 variable (R) in
135 /-- `R_R` has no Goldie torsion: a regular element kills no nonzero element. -/
136 theorem goldieTorsionPart_self : goldieTorsionPart R R = ⊥ := by
137   rw [eq_bot_iff]
138   intro r hr
139   obtain ⟨s, hs, h⟩ := mem_goldieTorsionPart_iff.mp hr
140   exact (Submodule.mem_bot _).mpr (hs.2 r h)
141 
142 /-- Every element of `M_Q` is multiplied into the image of `M` by a regular
143 element: `(m s⁻¹) s = m`. -/
144 theorem exists_smul_mem_range (x : FractionModule R M) :
145     ∃ s ∈ oppositeSubmonoid R⁰, s • x ∈ LinearMap.range (toFractionModule R M) := by
146   induction x using OreLocalization.ind with | _ m s =>
147   exact ⟨s, s.2, m, (smul_oreDiv_self m s).symm⟩
148 
149 /-- The injective map `M/t(M) → M_Q` induced by `M → M_Q`. -/
150 noncomputable def quotientToFractionModule :
151     (M ⧸ goldieTorsionPart R M) →ₗ[Rᵐᵒᵖ] FractionModule R M :=
152   (goldieTorsionPart R M).liftQ (toFractionModule R M) (ker_toFractionModule R M).ge
153 
154 /-- The map `M/t(M) → M_Q` is injective. -/
155 theorem quotientToFractionModule_injective :
156     Function.Injective (quotientToFractionModule (R := R) (M := M)) := by
157   rw [← LinearMap.ker_eq_bot]
158   exact Submodule.ker_liftQ_eq_bot _ _ _ (ker_toFractionModule R M).le
159 
160 end Torsion

164 /-- **Additivity of Goldie rank.** For a short exact sequence
165 `0 → M → N → P → 0` of right `R`-modules, `rk N = rk M + rk P`; localization is
166 exact and length is additive. -/
167 theorem goldieRank_eq_add_of_exact {f : M →ₗ[Rᵐᵒᵖ] N} {g : N →ₗ[Rᵐᵒᵖ] P}
168     (hf : Function.Injective f) (hg : Function.Surjective g) (hfg : Function.Exact f g) :
169     goldieRank R N = goldieRank R M + goldieRank R P :=
170   Module.length_eq_add_of_exact _ _ (localizationMap_injective hf) (localizationMap_surjective hg)
171     (localizationMap_exact hfg)
172 
173 /-- Goldie rank is monotone under injective maps. -/
174 theorem goldieRank_le_of_injective {f : M →ₗ[Rᵐᵒᵖ] N} (hf : Function.Injective f) :
175     goldieRank R M ≤ goldieRank R N :=
176   Module.length_le_of_injective _ (localizationMap_injective (S := oppositeSubmonoid R⁰) hf)
177 
178 /-- Goldie rank does not increase under surjective maps; in particular a
179 quotient of `M` has Goldie rank at most that of `M`. -/
180 theorem goldieRank_le_of_surjective {g : N →ₗ[Rᵐᵒᵖ] P} (hg : Function.Surjective g) :
181     goldieRank R P ≤ goldieRank R N :=
182   Module.length_le_of_surjective _ (localizationMap_surjective (S := oppositeSubmonoid R⁰) hg)
183 
184 /-- A quotient of `M` has Goldie rank at most that of `M`. -/
185 theorem goldieRank_quotient_le (K : Submodule Rᵐᵒᵖ M) : goldieRank R (M ⧸ K) ≤ goldieRank R M :=
186   goldieRank_le_of_surjective K.mkQ_surjective
187 
188 /-- A submodule of `M` has Goldie rank at most that of `M`. -/
189 theorem goldieRank_submodule_le (K : Submodule Rᵐᵒᵖ M) : goldieRank R K ≤ goldieRank R M :=
190   goldieRank_le_of_injective K.injective_subtype
191 
192 /-- Goldie rank is invariant under linear equivalences. -/
193 theorem goldieRank_congr (e : M ≃ₗ[Rᵐᵒᵖ] N) : goldieRank R M = goldieRank R N :=
194   le_antisymm (goldieRank_le_of_injective e.injective) (goldieRank_le_of_surjective e.surjective)
195 
196 variable (R M N) in
197 /-- Goldie rank is additive on direct sums. -/
198 theorem goldieRank_prod : goldieRank R (M × N) = goldieRank R M + goldieRank R N :=
199   goldieRank_eq_add_of_exact LinearMap.inl_injective LinearMap.snd_surjective .inl_snd

201 variable (R M) in
202 /-- `rk(Mᵐ) = m · rk(M)`. -/
203 theorem goldieRank_pi_fin (m : ℕ) : goldieRank R (Fin m → M) = m * goldieRank R M := by
204   induction m with
205   | zero =>
206     have : Subsingleton (FractionModule R (Fin 0 → M)) := by
207       constructor
208       intro p q
209       induction p using OreLocalization.ind with | _ x s =>
210       induction q using OreLocalization.ind with | _ y t =>
211       rw [Subsingleton.elim x 0, Subsingleton.elim y 0, OreLocalization.zero_oreDiv,
212         OreLocalization.zero_oreDiv]
213     simp [goldieRank, Module.length_eq_zero]
214   | succ m ih =>
215     rw [← goldieRank_congr (Fin.consLinearEquiv Rᵐᵒᵖ fun _ : Fin (m + 1) => M), goldieRank_prod,
216       ih]
217     push_cast
218     ring
219 
220 /-- The Goldie rank vanishes exactly on Goldie-torsion modules. -/
221 theorem goldieRank_eq_zero_iff : goldieRank R M = 0 ↔ IsGoldieTorsion R M := by
222   rw [goldieRank, Module.length_eq_zero_iff]
223   constructor
224   · intro h m
225     exact toFractionModule_eq_zero_iff.mp (Subsingleton.elim _ _)
226   · intro h
227     refine ⟨fun p q => ?_⟩
228     suffices ∀ p : FractionModule R M, p = 0 by rw [this p, this q]
229     intro p
230     induction p using OreLocalization.ind with | _ m s =>
231     exact oreDiv_eq_zero_iff.mpr (toFractionModule_eq_zero_iff.mpr (h m))
232

236 theorem goldieRank_op_self :
237     goldieRank R Rᵐᵒᵖ = Module.length (OppositeQuotientRing R) (OppositeQuotientRing R) := rfl

239 variable (R) in
240 /-- `x ↦ op x` identifies `R`, as a right module over itself, with the left
241 regular module of `Rᵐᵒᵖ`. -/
242 def opRegularEquiv : R ≃ₗ[Rᵐᵒᵖ] Rᵐᵒᵖ where
243   toFun := op
244   invFun := unop
245   map_add' _ _ := rfl
246   map_smul' c x := by
247     change op (x * unop c) = c * op x
248     rw [op_mul, op_unop]
249   left_inv _ := rfl
250   right_inv _ := rfl

252 variable (R) in
253 /-- `s = rk(R_R) = length_Q(Q_Q)`. -/
254 theorem goldieRank_self :
255     goldieRank R R = Module.length (OppositeQuotientRing R) (OppositeQuotientRing R) :=
256   (goldieRank_congr (opRegularEquiv R)).trans (goldieRank_op_self R)

258 /-- A right module generated by the finite set `s` has Goldie rank at most
259 `|s| · rk(R_R)`. -/
260 theorem goldieRank_le_card_mul (s : Finset M) (hs : Submodule.span Rᵐᵒᵖ (s : Set M) = ⊤) :
261     goldieRank R M ≤ s.card * goldieRank R R := by
262   let v : Fin s.card → M := fun i => (s.equivFin.symm i : M)
263   have hv : Set.range v = (s : Set M) := by
264     ext m
265     constructor
266     · rintro ⟨i, rfl⟩
267       exact (s.equivFin.symm i).2
268     · intro hm
269       exact ⟨s.equivFin ⟨m, hm⟩, by simp [v]⟩
270   have hsurj : Function.Surjective (Fintype.linearCombination Rᵐᵒᵖ v) := by
271     rw [← LinearMap.range_eq_top, Fintype.range_linearCombination, hv, hs]
272   calc goldieRank R M ≤ goldieRank R (Fin s.card → Rᵐᵒᵖ) := goldieRank_le_of_surjective hsurj
273     _ = s.card * goldieRank R R := by
274       rw [goldieRank_pi_fin, goldieRank_op_self, goldieRank_self]
275 
276 end Rank

300 /-- **Uniform dimension equals Goldie rank.** `M/t(M)` contains a direct sum of
301 `n` nonzero submodules iff `n ≤ rk(M) = length_Q(M_Q)`. -/
302 theorem uniformDimGE_quotient_iff {n : ℕ} :
303     UniformDimGE Rᵐᵒᵖ (M ⧸ goldieTorsionPart R M) n ↔ (n : ℕ∞) ≤ goldieRank R M := by
304   rw [uniformDimGE_iff_of_injective (S := oppositeSubmonoid R⁰) quotientToFractionModule_injective
305     fun x => ?_]
306   · exact uniformDimGE_iff_le_length
307   · obtain ⟨s, hs, m, hm⟩ := exists_smul_mem_range x
308     exact ⟨s, hs, Submodule.Quotient.mk m, hm⟩
309 
310 /-- **Uniform dimension equals Goldie rank.** `M/t(M)` contains no direct sum of
311 `n + 1` nonzero submodules iff `rk(M) ≤ n`. -/
312 theorem uniformDimLE_quotient_iff {n : ℕ} :
313     UniformDimLE Rᵐᵒᵖ (M ⧸ goldieTorsionPart R M) n ↔ goldieRank R M ≤ n := by
314   rw [UniformDimLE, uniformDimGE_quotient_iff, not_le, Nat.cast_add, Nat.cast_one,
315     ENat.lt_add_one_iff (ENat.natCast_ne_top n)]
316 
317 /-- **Uniform dimension of `R_R`.** `R` contains a direct sum of `n` nonzero right
318 ideals iff `n ≤ rk(R_R) = length_Q(Q_Q)`. -/
319 theorem uniformDimGE_self_iff {n : ℕ} :
320     UniformDimGE Rᵐᵒᵖ R n ↔ (n : ℕ∞) ≤ goldieRank R R := by
321   rw [← uniformDimGE_quotient_iff,
322     uniformDimGE_congr (Submodule.quotEquivOfEqBot _ (goldieTorsionPart_self R))]
323 
324 /-- **Uniform dimension of `R_R`.** `R` contains no direct sum of `n + 1` nonzero
325 right ideals iff `rk(R_R) ≤ n`. -/
326 theorem uniformDimLE_self_iff {n : ℕ} :
327     UniformDimLE Rᵐᵒᵖ R n ↔ goldieRank R R ≤ n := by
328   rw [UniformDimLE, uniformDimGE_self_iff, not_le, Nat.cast_add, Nat.cast_one,
329     ENat.lt_add_one_iff (ENat.natCast_ne_top n)]

331 /-- **Generators bound the normalized rank from below** (`⌈ρ(M)⌉ ≤ μ(M)`). If `M`
332 is generated by a set of at most `m` elements, `M/t(M)` contains a direct sum of
333 `ℓ` nonzero submodules and `R_R` contains no direct sum of `s + 1` nonzero right
334 ideals, then `ℓ ≤ m s`. -/
335 theorem le_mul_of_uniformDim {m ℓ s : ℕ} (g : Finset M) (hg : g.card ≤ m)
336     (hspan : Submodule.span Rᵐᵒᵖ (g : Set M) = ⊤)
337     (hℓ : UniformDimGE Rᵐᵒᵖ (M ⧸ goldieTorsionPart R M) ℓ) (hs : UniformDimLE Rᵐᵒᵖ R s) :
338     ℓ ≤ m * s := by
339   have h1 := uniformDimGE_quotient_iff.mp hℓ
340   have h2 := goldieRank_le_card_mul g hspan
341   have h3 := uniformDimLE_self_iff.mp hs
342   have : (ℓ : ℕ∞) ≤ (m * s : ℕ) := by
343     calc (ℓ : ℕ∞) ≤ g.card * goldieRank R R := h1.trans h2
344       _ ≤ m * s := by gcongr
345       _ = (m * s : ℕ) := (Nat.cast_mul m s).symm
346   exact_mod_cast this
