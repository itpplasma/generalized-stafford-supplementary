9 /-!
10 # Goldie torsion
11 
12 A right module `M` over a ring `R` is *Goldie-torsion* (`IsGoldieTorsion R M`)
13 if every element of `M` is killed by a regular element of `R`. Over a domain
14 this is ordinary torsion. This file records the closure properties of the
15 class and, over a simple right Noetherian ring:
16 
17 * `R / I` is Goldie-torsion when `I` contains a regular element
18   (`isGoldieTorsion_quotient`);
19 * a finite-dimensional subspace of a Goldie-torsion bimodule has a common
20   regular right annihilator
21   (`IsGoldieTorsion.exists_common_regular_annihilator`);
22 * escape: for `g ≠ 0` and `b ≠ 0` (possibly singular), `g R b` is not contained
23   in any finite-dimensional left subspace (`IsGoldieTorsion.exists_smul_mul_not_mem`),
24   hence an injective finite-dimensional map stays injective after right
25   multiplication by some `r b` (`IsGoldieTorsion.exists_injective_right_mul`);
26 * Goldie torsion passes from a critical module to its section module
27   (`isGoldieTorsion_section`).
28 
29 The linear-algebra steps are those of `AlgebraicAnalysis.TorsionEscape`; only
30 the torsion input changes, from the right Ore condition of a domain to Goldie's
31 theorem (`AlgebraicAnalysis.Goldie.exists_mem_nonZeroDivisors_finset`).
32 -/

41 /-- A right `R`-module `M` is Goldie-torsion if every element is killed by a
42 regular element of `R`. -/
43 def IsGoldieTorsion (R : Type u) [Ring R] (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] :
44     Prop :=
45   ∀ m : M, ∃ s ∈ R⁰, op s • m = 0

52 theorem IsGoldieTorsion.of_injective (f : M →ₗ[Rᵐᵒᵖ] N) (hf : Function.Injective f)
53     (hN : IsGoldieTorsion R N) : IsGoldieTorsion R M := fun m => by
54   obtain ⟨s, hs, hsm⟩ := hN (f m)
55   exact ⟨s, hs, hf (by rw [map_smul, hsm, map_zero])⟩
56 
57 theorem IsGoldieTorsion.of_surjective (f : M →ₗ[Rᵐᵒᵖ] N) (hf : Function.Surjective f)
58     (hM : IsGoldieTorsion R M) : IsGoldieTorsion R N := fun n => by
59   obtain ⟨m, rfl⟩ := hf n
60   obtain ⟨s, hs, hsm⟩ := hM m
61   exact ⟨s, hs, by rw [← map_smul, hsm, map_zero]⟩

63 /-- Goldie torsion is ordinary torsion (regular elements are nonzero). -/
64 theorem IsGoldieTorsion.isTorsionRight [Nontrivial R] (hM : IsGoldieTorsion R M) :
65     IsTorsionRight (R := R) (M := M) := fun m => by
66   obtain ⟨s, hs, hsm⟩ := hM m
67   exact ⟨s, nonZeroDivisors.ne_zero hs, hsm⟩

69 /-- Over a domain, ordinary torsion is Goldie torsion. -/
70 theorem isGoldieTorsion_of_isTorsionRight [IsDomain R]
71     (hM : IsTorsionRight (R := R) (M := M)) : IsGoldieTorsion R M := fun m => by
72   obtain ⟨s, hs, hsm⟩ := hM m
73   exact ⟨s, mem_nonZeroDivisors_of_ne_zero hs, hsm⟩

75 /-- A quotient `R/I` with `I` containing a regular element is Goldie-torsion. -/
76 theorem isGoldieTorsion_quotient [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
77     {I : Submodule Rᵐᵒᵖ R} {s : R} (hsI : s ∈ I) (hs : s ∈ R⁰) :
78     IsGoldieTorsion R (R ⧸ I) := fun z => by
79   obtain ⟨f, rfl⟩ := Submodule.mkQ_surjective I z
80   obtain ⟨b, hb, hfb⟩ := exists_mul_mem_of_mem_nonZeroDivisors hsI hs f
81   refine ⟨b, hb, ?_⟩
82   rw [← map_smul]
83   exact (Submodule.Quotient.mk_eq_zero I).mpr hfb

85 /-- If every `y ∈ I` has a regular `s` with `y s ∈ aR`, then `I / aR` (the
86 quotient of `I` by `aR ∩ I`) is Goldie-torsion. -/
87 theorem isGoldieTorsion_quotient_of_forall_exists_mul_mem {I : Submodule Rᵐᵒᵖ R} {a : R}
88     (hIa : ∀ y ∈ I, ∃ s ∈ R⁰, y * s ∈ principalRightIdeal a) :
89     IsGoldieTorsion R (↥I ⧸ (principalRightIdeal a).comap I.subtype) := by
90   intro z
91   obtain ⟨⟨m, hm⟩, rfl⟩ := Submodule.mkQ_surjective _ z
92   obtain ⟨s, hs, hms⟩ := hIa m hm
93   refine ⟨s, hs, ?_⟩
94   rw [← map_smul]
95   exact (Submodule.Quotient.mk_eq_zero _).mpr hms

97 /-- In a simple right Noetherian ring every right ideal `I` contains an
98 element `a` with `I / aR` Goldie-torsion. -/
99 theorem exists_mem_isGoldieTorsion_quotient [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
100     (I : Submodule Rᵐᵒᵖ R) :
101     ∃ a ∈ I, IsGoldieTorsion R (↥I ⧸ (principalRightIdeal a).comap I.subtype) := by
102   obtain ⟨a, haI, hIa⟩ := exists_mem_forall_exists_mul_mem I
103   exact ⟨a, haI, isGoldieTorsion_quotient_of_forall_exists_mul_mem hIa⟩

105 end Closure

113 /-- A finite-dimensional subspace of a Goldie-torsion bimodule has a common
114 regular right annihilator. -/
115 theorem IsGoldieTorsion.exists_common_regular_annihilator (hW : IsGoldieTorsion R W)
116     (Y : Submodule E W) [Module.Finite E Y] :
117     ∃ c ∈ R⁰, ∀ y ∈ Y, op c • y = 0 := by
118   classical
119   let B := Module.finBasis E Y
120   obtain ⟨c, hc, hcB⟩ := exists_mem_nonZeroDivisors_finset Finset.univ
121     (fun i => rightAnnihilator (R := R) (B i : W))
122     (fun i _ => by
123       obtain ⟨r, hr, hri⟩ := hW (B i : W)
124       exact ⟨r, hri, hr⟩)
125   refine ⟨c, hc, ?_⟩
126   intro y hy
127   have heq := congrArg (fun z : Y => (z : W)) (B.sum_repr ⟨y, hy⟩)
128   simp only [Submodule.coe_sum, Submodule.coe_smul] at heq
129   rw [← heq, Finset.smul_sum]
130   apply Finset.sum_eq_zero
131   intro i _
132   rw [smul_comm, show op c • (B i : W) = 0 from hcB i (Finset.mem_univ i), smul_zero]

134 /-- **Escape.** In a Goldie-torsion bimodule over a simple right Noetherian
135 ring, for `g ≠ 0` and `b ≠ 0` (possibly singular), `g R b` is not contained in
136 any finite-dimensional left subspace. -/
137 theorem IsGoldieTorsion.exists_smul_mul_not_mem (hW : IsGoldieTorsion R W)
138     {g : W} (hg : g ≠ 0) {b : R} (hb : b ≠ 0)
139     (Y : Submodule E W) [Module.Finite E Y] :
140     ∃ r : R, op (r * b) • g ∉ Y := by
141   classical
142   by_contra! h
143   obtain ⟨c, hc, hcY⟩ := hW.exists_common_regular_annihilator Y
144   have hbc : b * c ∈ cyclicAnnihilator g := by
145     simp only [cyclicAnnihilator, TwoSidedIdeal.mem_mk', Set.mem_ofPred_eq]
146     intro r
147     rw [← mul_assoc, op_mul, mul_smul]
148     exact hcY _ (h r)
149   have hbc0 : b * c ≠ 0 := fun h0 => hb (hc.2 b h0)
150   have h1 := IsSimpleRing.one_mem_of_ne_zero_mem (cyclicAnnihilator g) hbc0 hbc
151   have hzero := h1 1
152   exact hg (by simpa using hzero)

157 /-- A finite-dimensional injective map into a Goldie-torsion bimodule stays
158 injective after right multiplication by `r b` for some `r`, for any `b ≠ 0`. -/
159 theorem IsGoldieTorsion.exists_injective_right_mul (hW : IsGoldieTorsion R W)
160     (σ : k →+* R) (κ : k →+* E) (hκ : ∀ c a, κ c * a = a * κ c)
161     (hscalar : ∀ c (w : W), op (σ c) • w = κ c • w)
162     (P : D →ₗ[E] W) (hP : Function.Injective P) {b : R} (hb : b ≠ 0) :
163     ∃ r : R, Function.Injective ((rightMulLinear (E := E) (r * b)).comp P) := by
164   apply exists_injective_of_escape σ κ hκ
165     (fun r => (rightMulLinear (E := E) (r * b)).comp P)
166   · intro r s c d
167     change op ((r + σ c * s) * b) • P d =
168       op (r * b) • P d + κ c • (op (s * b) • P d)
169     rw [add_mul, op_add, add_smul, mul_assoc]
170     congr 1
171     rw [op_mul, mul_smul, hscalar, smul_comm]
172   · intro d hd Y hY
173     let := hY
174     exact hW.exists_smul_mul_not_mem
175       (fun h => hd (hP (h.trans (map_zero P).symm))) hb Y
176 
177 end Escape

186 /-- Goldie torsion passes from a critical module to its section module: the
187 annihilator of a section is essential, hence contains a regular element
188 (Goldie). -/
189 theorem isGoldieTorsion_section (hC : Critical Rᵐᵒᵖ e C) (ht : IsGoldieTorsion R C) :
190     IsGoldieTorsion R (Section R e C) := by
191   intro w
192   have hess : IsEssential (rightAnnihilator (R := R) w) := by
193     intro y hy
194     by_cases hyw : op y • w = 0
195     · refine ⟨1, by rwa [mul_one], ?_⟩
196       change op (y * 1) • w = 0
197       rwa [mul_one]
198     obtain ⟨i, c, hc, he⟩ := Section.exists_smul_eq_ofElement hC (op y • w) hyw
199     rw [Section.right_action_assoc] at he
200     obtain ⟨s, hs, hsc⟩ := ht c
201     have hyi : y * i ≠ 0 := by
202       intro h0
203       rw [h0, op_zero, zero_smul] at he
204       exact hc (Section.ofElement_injective hC (he.symm.trans (map_zero _).symm))
205     refine ⟨i * s, fun h0 => hyi (hs.2 _ (by rwa [← mul_assoc] at h0)), ?_⟩
206     change op (y * (i * s)) • w = 0
207     rw [← mul_assoc, ← Section.right_action_assoc, he, ← map_smul, hsc, map_zero]
208   obtain ⟨s, hsw, hs⟩ := exists_mem_nonZeroDivisors_of_isEssential hess
209   exact ⟨s, hs, hsw⟩
210 
211 end Section
