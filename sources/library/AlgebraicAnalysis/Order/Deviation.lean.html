8 # Finite deviation of partially ordered sets
9 
10 `Order.DevLE n α` is the finite Gabriel–Rentschler deviation bound. At zero it
11 is the descending chain condition; at a successor, successive closed intervals
12 of every descending chain eventually have the preceding bound. It is not the
13 supremum of lengths of finite strict chains.
14 
15 The definition needs only a partial order. Strictly monotone maps pull bounds
16 back, and finite products preserve bounds. These two facts also apply to the
17 intersection/projection map for submodule lattices.
18 -/

24 /-- Finite deviation at most `n`. Zero means that descending chains stabilize.
25 The trivial order has every bound, including zero. -/
26 def DevLE : (n : ℕ) → (α : Type u) → [PartialOrder α] → Prop
27   | 0, α, _ => ∀ c : ℕ → α, Antitone c → ∃ N, ∀ i, N ≤ i → c i = c N
28   | n + 1, α, _ => ∀ c : ℕ → α, Antitone c →
29       ∃ N, ∀ i, N ≤ i → DevLE n (Set.Icc (c (i + 1)) (c i))
30 
31 variable {α : Type u} {β : Type v} [PartialOrder α] [PartialOrder β]

33 theorem devLE_zero_iff : DevLE 0 α ↔
34     ∀ c : ℕ → α, Antitone c → ∃ N, ∀ i, N ≤ i → c i = c N := Iff.rfl

36 theorem devLE_succ_iff (n : ℕ) : DevLE (n + 1) α ↔
37     ∀ c : ℕ → α, Antitone c →
38       ∃ N, ∀ i, N ≤ i → DevLE n (Set.Icc (c (i + 1)) (c i)) := Iff.rfl

40 /-- A strictly monotone map reflects finite deviation bounds. Injectivity on
41 incomparable pairs is unnecessary. -/
42 theorem DevLE.of_strictMono {n : ℕ} (h : DevLE n β) {f : α → β}
43     (hf : StrictMono f) : DevLE n α := by
44   induction n generalizing α β with
45   | zero =>
46     intro c hc
47     obtain ⟨N, hN⟩ := h (f ∘ c) (hf.monotone.comp_antitone hc)
48     refine ⟨N, fun i hi => ?_⟩
49     by_contra he
50     have hlt : c i < c N := lt_of_le_of_ne (hc hi) he
51     exact (ne_of_lt (hf hlt)) (hN i hi)
52   | succ n ih =>
53     intro c hc
54     obtain ⟨N, hN⟩ := h (f ∘ c) (hf.monotone.comp_antitone hc)
55     refine ⟨N, fun i hi => ?_⟩
56     let g : Set.Icc (c (i + 1)) (c i) → Set.Icc (f (c (i + 1))) (f (c i)) :=
57       fun x => ⟨f x, hf.monotone x.property.1, hf.monotone x.property.2⟩
58     exact ih (hN i hi) (f := g) (fun _ _ hxy => hf hxy)
59 
60 /-- Finite deviation is invariant under order isomorphisms. -/
61 theorem devLE_orderIso_iff (n : ℕ) (e : α ≃o β) : DevLE n α ↔ DevLE n β :=
62   ⟨fun h => h.of_strictMono e.symm.strictMono, fun h => h.of_strictMono e.strictMono⟩
63 
64 /-- Every interval inherits a deviation bound. -/
65 theorem DevLE.interval {n : ℕ} (h : DevLE n α) (a b : α) : DevLE n (Set.Icc a b) :=
66   h.of_strictMono (f := Subtype.val) (fun _ _ hxy => hxy)

68 theorem devLE_of_subsingleton (n : ℕ) [Subsingleton α] : DevLE n α := by
69   induction n generalizing α with
70   | zero => exact fun c _ => ⟨0, fun _ _ => Subsingleton.elim _ _⟩
71   | succ n ih => exact fun c _ => ⟨0, fun i _ => ih⟩
72 
73 /-- Bounds increase monotonically with the finite index. -/
74 theorem DevLE.succ {n : ℕ} (h : DevLE n α) : DevLE (n + 1) α := by
75   induction n generalizing α with
76   | zero =>
77     intro c hc
78     obtain ⟨N, hN⟩ := h c hc
79     refine ⟨N, fun i hi => ?_⟩
80     have he : c (i + 1) = c i := (hN (i + 1) (Nat.le_trans hi (Nat.le_succ i))).trans
81       (hN i hi).symm
82     have : Subsingleton (Set.Icc (c (i + 1)) (c i)) := by
83       rw [he]
84       infer_instance
85     exact devLE_of_subsingleton 0
86   | succ n ih =>
87     intro c hc
88     obtain ⟨N, hN⟩ := h c hc
89     exact ⟨N, fun i hi => ih (hN i hi)⟩
90 
91 theorem DevLE.mono {n m : ℕ} (h : DevLE n α) (hnm : n ≤ m) : DevLE m α := by
92   induction hnm with
93   | refl => exact h
94   | step _ ih => exact ih.succ

96 /-- Products preserve finite deviation bounds. -/
97 theorem DevLE.prod {n : ℕ} (hα : DevLE n α) (hβ : DevLE n β) :
98     DevLE n (α × β) := by
99   induction n generalizing α β with
100   | zero =>
101     intro c hc
102     obtain ⟨N, hN⟩ := hα (fun i => (c i).1) (fun _ _ hij => (hc hij).1)
103     obtain ⟨M, hM⟩ := hβ (fun i => (c i).2) (fun _ _ hij => (hc hij).2)
104     refine ⟨max N M, fun i hi => Prod.ext ?_ ?_⟩
105     · exact (hN i (le_trans (le_max_left _ _) hi)).trans
106         (hN _ (le_max_left _ _)).symm
107     · exact (hM i (le_trans (le_max_right _ _) hi)).trans
108         (hM _ (le_max_right _ _)).symm
109   | succ n ih =>
110     intro c hc
111     obtain ⟨N, hN⟩ := hα (fun i => (c i).1) (fun _ _ hij => (hc hij).1)
112     obtain ⟨M, hM⟩ := hβ (fun i => (c i).2) (fun _ _ hij => (hc hij).2)
113     refine ⟨max N M, fun i hi => ?_⟩
114     have hp := ih (hN i (le_trans (le_max_left _ _) hi))
115       (hM i (le_trans (le_max_right _ _) hi))
116     let f : Set.Icc (c (i + 1)) (c i) →
117         Set.Icc ((c (i + 1)).1) ((c i).1) × Set.Icc ((c (i + 1)).2) ((c i).2) :=
118       fun x => (⟨x.val.1, x.property.1.1, x.property.2.1⟩,
119         ⟨x.val.2, x.property.1.2, x.property.2.2⟩)
120     apply hp.of_strictMono (f := f)
121     intro x y hxy
122     exact hxy
123 
124 end Order
