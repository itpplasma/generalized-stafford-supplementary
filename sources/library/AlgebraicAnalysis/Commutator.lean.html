35/-- Iterated commutation with a Weyl-type relation. -/
36theorem ringCommutator_pow (z x : A) (h : ringCommutator z x = 1) :
37 ∀ n : ℕ, ringCommutator (z ^ n) x = n • z ^ (n - 1)
38 | 0 => by simp [ringCommutator]
39 | n + 1 => by
40 rw [pow_succ, ringCommutator_mul, h, ringCommutator_pow z x h n]
41 by_cases hn : n = 0
42 · subst n
43 simp
44 · rw [smul_mul_assoc, Nat.succ_sub_one, mul_one, add_nsmul]
45 have hpow : z ^ (n - 1) * z = z ^ n := by
46 rw [← pow_succ, Nat.sub_add_cancel (Nat.pos_of_ne_zero hn)]
