9 # PBW filtration of a quotient of a universal enveloping algebra
10 
11 Let `L` be a Lie algebra over a field `k`, and let `π : U(L) →ₐ[k] R` be a
12 surjective algebra map onto a `k`-algebra `R` (this covers `U(L)` itself and
13 every quotient `U(L) / J`). Let `gen x = π (ι x)` be the image of `x ∈ L`.
14 
15 The **PBW filtration** of `R` is `Fₙ = span {x₁ ⋯ xⱼ : j ≤ n, xᵢ ∈ L}`, the span of
16 products of at most `n` Lie elements. It is an ascending filtration in the sense
17 of `RingFiltration`.
18 
19 The only commutation fact used is the Lie bracket identity
20 `gen x * gen y - gen y * gen x = gen [x, y]`, which lies in `F₁`. Hence a
21 product of degree-one generators changes, under a permutation of its factors,
22 only by an element of lower filtration (`wordProd_sub_mem_pbwPiece`). No
23 Poincaré–Birkhoff–Witt theorem is used: only this spanning direction is needed.
24 
25 ## Main definitions and results
26 
27 * `gen π x`, `wordProd π w`: the image of `x ∈ L` and the ordered product of a word.
28 * `pbwPiece π n`: the span of products of at most `n` Lie elements, that is `Fₙ`.
29 * `pbwFiltration π hπ`: the PBW filtration of `R` as a `RingFiltration k R`.
30 * `wordProd_sub_mem_pbwPiece`: permuting a word changes its product only in lower
31   filtration.
32 
33 The symbol presentation by the polynomial ring and the Krull dimension
34 consequences are in `AlgebraicAnalysis.Enveloping.Presentation`.
35 -/

48 /-- The image in `R` of a Lie element `x`, namely `π (ι x)`. -/
49 def gen (π : UniversalEnvelopingAlgebra k L →ₐ[k] R) (x : L) : R :=
50   π (ι k x)
51 
52 /-- The product `w₁ w₂ ⋯ wₙ` in `R` of the images of a word `w` of Lie elements, in the
53 written order. -/
54 def wordProd (π : UniversalEnvelopingAlgebra k L →ₐ[k] R) (w : List L) : R :=
55   (w.map (gen π)).prod
56 
57 /-- The `n`-th PBW piece `Fₙ`: the `k`-span of products of at most `n` Lie elements. -/
58 def pbwPiece (π : UniversalEnvelopingAlgebra k L →ₐ[k] R) (n : ℕ) : Submodule k R :=
59   Submodule.span k {r | ∃ w : List L, w.length ≤ n ∧ wordProd π w = r}
60 
61 variable (π : UniversalEnvelopingAlgebra k L →ₐ[k] R)
62 
63 theorem wordProd_nil : wordProd π ([] : List L) = 1 := by
64   simp [wordProd]
65 
66 theorem wordProd_cons (x : L) (w : List L) :
67     wordProd π (x :: w) = gen π x * wordProd π w := by
68   simp [wordProd]
69 
70 theorem wordProd_append (u v : List L) :
71     wordProd π (u ++ v) = wordProd π u * wordProd π v := by
72   simp [wordProd]
73 
74 theorem wordProd_singleton (x : L) : wordProd π [x] = gen π x := by
75   simp [wordProd]
76 
77 theorem wordProd_mem_pbwPiece {w : List L} {n : ℕ} (hw : w.length ≤ n) :
78     wordProd π w ∈ pbwPiece π n :=
79   Submodule.subset_span ⟨w, hw, rfl⟩
80 
81 theorem pbwPiece_mono {a b : ℕ} (h : a ≤ b) : pbwPiece π a ≤ pbwPiece π b :=
82   Submodule.span_mono fun _ ⟨w, hw, hr⟩ => ⟨w, hw.trans h, hr⟩
83 
84 theorem one_mem_pbwPiece (n : ℕ) : (1 : R) ∈ pbwPiece π n := by
85   simpa [wordProd_nil] using wordProd_mem_pbwPiece π (w := []) (n := n) (Nat.zero_le _)
86 
87 theorem gen_mem_pbwPiece_one (x : L) : gen π x ∈ pbwPiece π 1 := by
88   simpa [wordProd_singleton] using wordProd_mem_pbwPiece π (w := [x]) (n := 1) le_rfl
89 
90 theorem mul_mem_pbwPiece {i j : ℕ} {a b : R} (ha : a ∈ pbwPiece π i)
91     (hb : b ∈ pbwPiece π j) : a * b ∈ pbwPiece π (i + j) := by
92   have hle : pbwPiece π i * pbwPiece π j ≤ pbwPiece π (i + j) := by
93     rw [pbwPiece, pbwPiece, Submodule.span_mul_span]
94     refine Submodule.span_le.mpr ?_
95     rintro _ ⟨a, ⟨u, hu, rfl⟩, b, ⟨v, hv, rfl⟩, rfl⟩
96     exact Submodule.subset_span ⟨u ++ v, by simp only [List.length_append]; omega,
97       wordProd_append π u v⟩
98   exact hle (Submodule.mul_mem_mul ha hb)
99 
100 /-- The PBW filtration of `R` is submultiplicative. -/
101 theorem pbwPiece_mul_le (i j : ℕ) : pbwPiece π i * pbwPiece π j ≤ pbwPiece π (i + j) :=
102   Submodule.mul_le.mpr fun _ ha _ hb => mul_mem_pbwPiece π ha hb
103 
104 /-- The PBW pieces exhaust `R` when `π` is surjective: every element of `U(L)` is a sum of
105 products of Lie generators, by induction on the tensor algebra. -/
106 theorem iSup_pbwPiece_eq_top (hπ : Function.Surjective π) : ⨆ n, pbwPiece π n = ⊤ := by
107   rw [eq_top_iff]
108   intro r _
109   obtain ⟨u, rfl⟩ := hπ r
110   obtain ⟨t, rfl⟩ := Quotient.exists_rep u
111   have key : ∀ t : TensorAlgebra k L, ∃ n, π (mkAlgHom k L t) ∈ pbwPiece π n := by
112     intro t
113     induction t using TensorAlgebra.induction with
114     | algebraMap c =>
115       refine ⟨0, ?_⟩
116       have : π (mkAlgHom k L (algebraMap k (TensorAlgebra k L) c)) = algebraMap k R c := by
117         simp
118       rw [this, Algebra.algebraMap_eq_smul_one]
119       exact Submodule.smul_mem _ c (one_mem_pbwPiece π 0)
120     | ι x =>
121       refine ⟨1, ?_⟩
122       exact gen_mem_pbwPiece_one π x
123     | mul a b ha hb =>
124       obtain ⟨n, hn⟩ := ha
125       obtain ⟨m, hm⟩ := hb
126       refine ⟨n + m, ?_⟩
127       simp only [map_mul]
128       exact mul_mem_pbwPiece π hn hm
129     | add a b ha hb =>
130       obtain ⟨n, hn⟩ := ha
131       obtain ⟨m, hm⟩ := hb
132       refine ⟨max n m, ?_⟩
133       simp only [map_add]
134       exact Submodule.add_mem _ (pbwPiece_mono π (le_max_left n m) hn)
135         (pbwPiece_mono π (le_max_right n m) hm)
136   obtain ⟨n, hn⟩ := key t
137   exact Submodule.mem_iSup_of_mem n hn
138 
139 end Definitions

146 /-- **The PBW filtration.** For a surjective algebra map `π : U(L) → R`, the pieces
147 `Fₙ = span {products of at most n Lie elements}` form a `RingFiltration k R`. -/
148 noncomputable def pbwFiltration (π : UniversalEnvelopingAlgebra k L →ₐ[k] R)
149     (hπ : Function.Surjective π) : RingFiltration k R where
150   F := pbwPiece π
151   mono := fun _ _ h => pbwPiece_mono π h
152   one_mem := one_mem_pbwPiece π 0
153   mul_le := pbwPiece_mul_le π
154   exhaustive := iSup_pbwPiece_eq_top π hπ
155 
156 theorem pbwFiltration_F (π : UniversalEnvelopingAlgebra k L →ₐ[k] R) (hπ : Function.Surjective π)
157     (n : ℕ) : (pbwFiltration π hπ).F n = pbwPiece π n :=
158   rfl
159 
160 /-- **Bracket identity.** The commutator of two generators is the generator of their Lie
161 bracket, `[gen x, gen y] = gen [x, y]`; in particular it lies in `F₁`. -/
162 theorem gen_bracket (π : UniversalEnvelopingAlgebra k L →ₐ[k] R) (x y : L) :
163     gen π x * gen π y - gen π y * gen π x = gen π ⁅x, y⁆ := by
164   have h : ι k ⁅x, y⁆ = ι k x * ι k y - ι k y * ι k x := by
165     rw [LieHom.map_lie, Ring.lie_def]
166   simp only [gen, h, map_sub, map_mul]
167 
168 /-- Reordering lemma. Permuting the factors of a word changes its product only by an element of
169 the filtration piece `Fₙ` as soon as the word has length at most `n + 1`. The proof moves one
170 adjacent transposition at a time; a swap changes the product by `-[gen x, gen y] · (rest)`,
171 which lies in `F₁ · F_{len rest} ⊆ F_{len rest + 1}`. -/
172 theorem wordProd_sub_mem_pbwPiece (π : UniversalEnvelopingAlgebra k L →ₐ[k] R)
173     {w w' : List L} (h : w.Perm w') :
174     ∀ n : ℕ, w.length ≤ n + 1 → wordProd π w - wordProd π w' ∈ pbwPiece π n := by
175   induction h with
176   | nil =>
177     intro n _
178     simp [wordProd_nil]
179   | @cons x l₁ l₂ hl ih =>
180     intro n hn
181     cases n with
182     | zero =>
183       have h1 : l₁ = [] := List.length_eq_zero_iff.mp (by simp only [List.length_cons] at hn; omega)
184       have h2 : l₂ = [] := List.length_eq_zero_iff.mp (by rw [← hl.length_eq, h1]; rfl)
185       subst h1 h2
186       simp
187     | succ m =>
188       simp only [wordProd_cons, ← mul_sub]
189       have hm := ih m (by simp only [List.length_cons] at hn; omega)
190       exact pbwPiece_mono π (by omega) (mul_mem_pbwPiece π (gen_mem_pbwPiece_one π x) hm)
191   | @swap x y l =>
192     intro n hn
193     simp only [List.length_cons] at hn
194     simp only [wordProd_cons]
195     have e : gen π y * (gen π x * wordProd π l) - gen π x * (gen π y * wordProd π l) =
196         -(gen π ⁅x, y⁆ * wordProd π l) := by
197       rw [← gen_bracket π x y]
198       noncomm_ring
199     rw [e]
200     exact Submodule.neg_mem _ (pbwPiece_mono π (by omega)
201       (mul_mem_pbwPiece π (gen_mem_pbwPiece_one π ⁅x, y⁆) (wordProd_mem_pbwPiece π le_rfl)))
202   | @trans w₁ w₂ w₃ h₁ h₂ ih₁ ih₂ =>
203     intro n hn
204     have hn' : w₂.length ≤ n + 1 := by rw [← h₁.length_eq]; exact hn
205     have := Submodule.add_mem _ (ih₁ n hn) (ih₂ n hn')
206     simpa using this
207 
208 end Filtration
