12 # Surjective symbol presentation of the PBW filtration
13 
14 Let `b : Basis (Fin m) k L` be a basis of the Lie algebra `L` and let
15 `G = MvPolynomial (Fin m) k`, graded by total degree. A monomial `X^α` is sent to the
16 ordered product `b₀^{α₀} ⋯ b_{m-1}^{α_{m-1}}` in `R`, extended linearly. This gives
17 `ψ : G → R`, and its graded pieces `σₙ : Gₙ → Fₙ/Fₙ₋₁` form a surjective symbol
18 presentation of the PBW filtration `pbwFiltration π hπ` (`pbwSymbolPresentation`).
19 
20 The argument uses only the reordering lemma of `PBWFiltration`:
21 
22 * Multiplicativity. For homogeneous `g` of degree `i` and `h` of degree `j`,
23   `ψ(g h) - ψ(g) ψ(h)` lies in `F_{i+j-1}`. Expanding over monomials reduces this to
24   the rearrangement of the letters of `X^α X^β`.
25 * Surjectivity. Let `Eₙ = ψ(𝒜ₙ) + Fₙ₋₁`. Left multiplication by a basis generator sends
26   `Eₙ` into `Eₙ₊₁`, by the multiplicativity above with `X_j`. Induction on words then gives
27   `Fₙ ⊆ Eₙ`, that is, every class of `Fₙ/Fₙ₋₁` is the image of a degree-`n` polynomial.
28 
29 No injectivity of `σₙ` is claimed. The presentation gives right and left Noetherianity of
30 `R` and deviation at most `dim L` on both sides, by `RingFiltration.SymbolPresentation`
31 and `ringKrullDim k[X₀, …, X_{m-1}] = m`.
32 
33 ## Main results
34 
35 * `pbwSymbolPresentation`: the surjective symbol presentation of `pbwFiltration π hπ`.
36 * `isNoetherian_right_of_surjective`, `isNoetherianRing_of_surjective`,
37   `devLE_right_of_surjective`, `devLE_left_of_surjective`: for `L` finite-dimensional and a
38   surjective algebra map `π : U(L) → R`, `R` is Noetherian on both sides, and
39   `Module.DevLE` holds with bound `dim L` on both sides.
40 -/

157 /-- Permuting a word changes its product by an element of the filtration piece of its length. -/
158 theorem wordProd_sub_mem_below (hπ : Function.Surjective π) {w w' : List L} (h : w.Perm w') :
159     wordProd π w - wordProd π w' ∈ (pbwFiltration π hπ).below w.length := by
160   rcases Nat.eq_zero_or_pos w.length with h0 | hpos
161   · have h1 : w = [] := List.length_eq_zero_iff.mp h0
162     have h2 : w' = [] := List.length_eq_zero_iff.mp (by rw [← h.length_eq, h1]; rfl)
163     subst h1 h2
164     simp
165   · obtain ⟨n, hn⟩ : ∃ n, w.length = n + 1 := ⟨w.length - 1, by omega⟩
166     rw [hn, RingFiltration.below_succ, pbwFiltration_F]
167     exact wordProd_sub_mem_pbwPiece π h n hn.le

169 /-- Monomial multiplicativity: `ψ(X^{α+β}) ≡ ψ(X^α) ψ(X^β)` modulo lower filtration. -/
170 theorem wordProd_mul_sub_mem_below (hπ : Function.Surjective π) (α β : Fin m →₀ ℕ) :
171     wordProd π (letters b (α + β)) - wordProd π (letters b α) * wordProd π (letters b β) ∈
172       (pbwFiltration π hπ).below (letters b (α + β)).length := by
173   rw [← wordProd_append]
174   exact wordProd_sub_mem_below π hπ (letters_add_perm b α β)

176 /-- **Multiplicativity of the symbol.** For homogeneous `g` of degree `i` and `h` of degree `j`,
177 `ψ(g h) ≡ ψ(g) ψ(h)` modulo `F_{i+j-1}`. Expanding both sides over the monomials of `g` and `h`
178 reduces this to the monomial case. -/
179 theorem psi_mul_sub_mem_below (hπ : Function.Surjective π) {g h : MvPolynomial (Fin m) k}
180     {i j : ℕ} (hg : g ∈ homogeneousSubmodule (Fin m) k i)
181     (hh : h ∈ homogeneousSubmodule (Fin m) k j) :
182     psi b π (g * h) - psi b π g * psi b π h ∈ (pbwFiltration π hπ).below (i + j) := by
183   have hmul : psi b π (g * h) = ∑ α ∈ g.support, ∑ β ∈ h.support,
184       (g.coeff α * h.coeff β) • wordProd π (letters b (α + β)) := by
185     rw [MvPolynomial.mul_def]
186     simp only [Finsupp.sum, map_sum, psi_monomial]
187     rfl
188   have key : psi b π (g * h) - psi b π g * psi b π h =
189       ∑ α ∈ g.support, ∑ β ∈ h.support,
190         (g.coeff α * h.coeff β) •
191           (wordProd π (letters b (α + β)) - wordProd π (letters b α) * wordProd π (letters b β)) := by
192     rw [hmul, psi_eq_sum b π g, psi_eq_sum b π h, Finset.sum_mul_sum, ← Finset.sum_sub_distrib]
193     refine Finset.sum_congr rfl fun α _ => ?_
194     rw [← Finset.sum_sub_distrib]
195     refine Finset.sum_congr rfl fun β _ => ?_
196     rw [smul_mul_smul_comm, smul_sub]
197   rw [key]
198   refine Submodule.sum_mem _ fun α hα => Submodule.sum_mem _ fun β hβ => Submodule.smul_mem _ _ ?_
199   have hα' : (letters b α).length = i := by
200     rw [length_letters_eq_weight]
201     exact (mem_homogeneousSubmodule i g).mp hg (mem_support_iff.mp hα)
202   have hβ' : (letters b β).length = j := by
203     rw [length_letters_eq_weight]
204     exact (mem_homogeneousSubmodule j h).mp hh (mem_support_iff.mp hβ)
205   have hdeg : (letters b (α + β)).length = i + j := by
206     rw [(letters_add_perm b α β).length_eq, List.length_append, hα', hβ']
207   have := wordProd_mul_sub_mem_below b π hπ α β
208   rwa [hdeg] at this

210 /-- The subspace `Eₙ = ψ(𝒜ₙ) + Fₙ₋₁` of `R`. Its image in `Fₙ/Fₙ₋₁` is all of it. -/
211 def E (hπ : Function.Surjective π) (n : ℕ) : Submodule k R :=
212   Submodule.map (psi b π) (homogeneousSubmodule (Fin m) k n) ⊔ (pbwFiltration π hπ).below n

216 theorem E_le_pbwPiece (n : ℕ) : E b π hπ n ≤ pbwPiece π n := by
217   refine sup_le ?_ ?_

224 theorem pbwPiece_le_E_succ (n : ℕ) : pbwPiece π n ≤ E b π hπ (n + 1) := by

243 /-- Left multiplication by a basis generator sends `E_n` into `E_{n+1}`. Here `ψ(X_j g)` is
244 the degree-`n+1` term and `ψ(X_j) ψ(g) ≡ gen(b_j) ψ(g)`, both modulo lower filtration. -/
245 theorem gen_basis_mul_E (j : Fin m) {n : ℕ} {e : R} (he : e ∈ E b π hπ n) :
246     gen π (b j) * e ∈ E b π hπ (n + 1) := by
247   obtain ⟨y, hy, z, hz, rfl⟩ := Submodule.mem_sup.mp he
248   rw [mul_add]
249   refine Submodule.add_mem _ ?_ ?_
250   · obtain ⟨g, hg, rfl⟩ := Submodule.mem_map.mp hy
251     have hX : (X j : MvPolynomial (Fin m) k) ∈ homogeneousSubmodule (Fin m) k 1 :=
252       (mem_homogeneousSubmodule 1 _).mpr (isHomogeneous_X k j)
253     have hXg : X j * g ∈ homogeneousSubmodule (Fin m) k (n + 1) := by
254       rw [mem_homogeneousSubmodule, add_comm]
255       exact (isHomogeneous_X k j).mul ((mem_homogeneousSubmodule n g).mp hg)
256     have hA := psi_mul_sub_mem_below b π hπ hX hg
257     rw [add_comm 1 n] at hA
258     have hB : psi b π (X j) - gen π (b j) ∈ (pbwFiltration π hπ).below 1 := by
259       have h1 := wordProd_sub_mem_below π hπ (letters_single_perm b j)
260       rw [(letters_single_perm b j).length_eq, List.length_singleton] at h1
261       rw [psi_X, ← wordProd_singleton π (b j)]
262       exact h1
263     have hBg : (psi b π (X j) - gen π (b j)) * psi b π g ∈ pbwPiece π n := by
264       have h := mul_mem_pbwPiece π hB (psi_mem_pbwPiece b π hg)
265       simpa only [Nat.zero_add] using h
266     have key : gen π (b j) * psi b π g =
267         psi b π (X j * g) - (psi b π (X j * g) - psi b π (X j) * psi b π g) -
268           (psi b π (X j) - gen π (b j)) * psi b π g := by
269       noncomm_ring
270     rw [key]
271     refine Submodule.sub_mem _ (Submodule.sub_mem _ ?_ ?_) ?_
272     · exact Submodule.mem_sup_left (Submodule.mem_map_of_mem hXg)
273     · exact Submodule.mem_sup_right hA
274     · exact Submodule.mem_sup_right hBg
275   · have h := (pbwFiltration π hπ).mul_below_le 1 n
276       (Submodule.mul_mem_mul (gen_mem_pbwPiece_one π (b j)) hz)
277     rw [add_comm 1 n] at h
278     exact Submodule.mem_sup_right h

302 /-- The graded symbol `σₙ : 𝒜ₙ → Fₙ/Fₙ₋₁` induced by `ψ`. -/
303 noncomputable def sym (n : ℕ) :
304     homogeneousSubmodule (Fin m) k n →ₗ[k] (pbwFiltration π hπ).GradedQuotient n :=
305   ((pbwFiltration π hπ).belowInF n).mkQ ∘ₗ
306     LinearMap.codRestrict ((pbwFiltration π hπ).F n)
307       ((psi b π).comp (homogeneousSubmodule (Fin m) k n).subtype)
308       (fun g => psi_mem_pbwPiece b π g.2)

310 theorem sym_apply (n : ℕ) (g : homogeneousSubmodule (Fin m) k n) :
311     sym b hπ n g = Submodule.Quotient.mk ⟨psi b π g, psi_mem_pbwPiece b π g.2⟩ :=
312   rfl

314 theorem sym_surjective (n : ℕ) : Function.Surjective (sym b hπ n) := by
315   intro q
316   obtain ⟨⟨u, hu⟩, rfl⟩ := Submodule.Quotient.mk_surjective _ q
317   obtain ⟨y, hy, z, hz, hyz⟩ := Submodule.mem_sup.mp (pbwPiece_le_E b hπ n hu)
318   obtain ⟨g, hg, rfl⟩ := Submodule.mem_map.mp hy
319   refine ⟨⟨g, hg⟩, ?_⟩
320   rw [sym_apply]
321   refine (Submodule.Quotient.eq _).mpr ?_
322   rw [(pbwFiltration π hπ).mem_belowInF]
323   show psi b π g - u ∈ _
324   have hd : psi b π g - u = -z := by rw [← hyz]; abel
325   rw [hd]
326   exact Submodule.neg_mem _ hz
