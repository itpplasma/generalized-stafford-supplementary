24lemma derivation_one (D : OreDivisionDerivation B) : D 1 = 0 := by
25 have h := D.leibniz 1 1
26 simp only [one_mul, mul_one] at h
27 symm
28 calc
29 0 = D 1 - D 1 := (sub_self _).symm
30 _ = (D 1 + D 1) - D 1 := by rw [← h]
31 _ = D 1 := by abel

Apply the coefficient derivation to every coefficient.

33/-- Apply the coefficient derivation to every coefficient. -/
34def coefficientDerivation (D : OreDivisionDerivation B) :
35 Polynomial B →+ Polynomial B where
36 toFun p := p.sum fun i b => monomial i (D b)
37 map_zero' := by simp [Polynomial.sum_def]
38 map_add' p q := by
39 change (p + q).sum (fun i b => monomial i (D b)) =
40 p.sum (fun i b => monomial i (D b)) +
41 q.sum (fun i b => monomial i (D b))
42 apply Polynomial.sum_add_index
43 · intro i
44 simp [D.map_zero]
45 · intro i a b
46 simp [D.map_add]
47
48@[simp] lemma coefficientDerivation_monomial
49 (D : OreDivisionDerivation B) (i : ℕ) (b : B) :
50 coefficientDerivation D (monomial i b) = monomial i (D b) := by
51 by_cases hb : b = 0
52 · subst b
53 simp [D.map_zero]
54 · simp [coefficientDerivation, hb, D.map_zero]
55
56@[simp] lemma coefficientDerivation_C_mul
57 (D : OreDivisionDerivation B) (b : B) (p : Polynomial B) :
58 coefficientDerivation D (C b * p) =
59 C b * coefficientDerivation D p + C (D b) * p := by
60 induction p using Polynomial.induction_on' with
61 | add p q hp hq =>
62 simp [map_add, hp, hq, mul_add, add_mul, add_assoc, add_left_comm,
63 add_comm]
64 | monomial n a =>
65 simp [D.leibniz, mul_add, add_mul, mul_assoc]

Coefficients act by ordinary left multiplication.

67/-- Coefficients act by ordinary left multiplication. -/
68def coefficientLeft : B →+* AddMonoid.End (Polynomial B) where
69 toFun b :=
70 { toFun := fun p => C b * p
71 map_zero' := by simp
72 map_add' := by intro p q; simp [mul_add] }
73 map_one' := by
74 apply AddMonoidHom.ext
75 intro p
76 change C (1 : B) * p = p
77 simp
78 map_mul' b c := by
79 apply AddMonoidHom.ext
80 intro p
81 change C (b * c) * p = C b * (C c * p)
82 simp [mul_assoc]
83 map_zero' := by
84 apply AddMonoidHom.ext
85 intro p
86 change C (0 : B) * p = 0
87 simp
88 map_add' b c := by
89 apply AddMonoidHom.ext
90 intro p
91 change C (b + c) * p = C b * p + C c * p
92 simp [add_mul]

The faithful left-regular representation of the one-variable Ore model.

103/-- The faithful left-regular representation of the one-variable Ore model. -/
104def faithfulAmbient (D : OreDivisionDerivation B) :
105 OreAmbient B (AddMonoid.End (Polynomial B)) D where
106 embed := coefficientLeft
107 x := leftOreShift D
108 relation := by
109 intro b
110 apply AddMonoidHom.ext
111 intro p
112 change (C b * p) * X + coefficientDerivation D (C b * p) =
113 C b * (p * X + coefficientDerivation D p) + C (D b) * p
114 rw [coefficientDerivation_C_mul]
115 noncomm_ring

117@[simp] lemma coefficientDerivation_X_pow
118 (D : OreDivisionDerivation B) (n : ℕ) :
119 coefficientDerivation D (X ^ n) = 0 := by
120 rw [Polynomial.X_pow_eq_monomial, coefficientDerivation_monomial,
121 derivation_one, monomial_zero_right]

123@[simp] lemma leftOreShift_X_pow (D : OreDivisionDerivation B) (n : ℕ) :
124 leftOreShift D (X ^ n) = X ^ (n + 1) := by
125 change X ^ n * X + coefficientDerivation D (X ^ n) = X ^ (n + 1)
126 rw [coefficientDerivation_X_pow, add_zero, pow_succ]

128lemma leftOreShift_pow_apply_one (D : OreDivisionDerivation B) (n : ℕ) :
129 ((leftOreShift D) ^ n) (1 : Polynomial B) = X ^ n := by
130 induction n with
131 | zero => simp
132 | succ n ih =>
133 rw [pow_succ']
134 change leftOreShift D (((leftOreShift D) ^ n) 1) = X ^ (n + 1)
135 rw [ih]
136 exact leftOreShift_X_pow D n

138lemma faithful_eval_apply_one (D : OreDivisionDerivation B)
139 (p : Polynomial B) :
140 OreAmbient.eval D (faithfulAmbient D) p (1 : Polynomial B) = p := by
141 rw [OreAmbient.eval]
142 change (p.sum fun i b =>
143 coefficientLeft b * (leftOreShift D) ^ i) 1 = p
144 rw [Polynomial.sum_def]
145 calc
146 (∑ n ∈ p.support,
147 coefficientLeft (p.coeff n) * (leftOreShift D) ^ n) 1 =
148 (∑ n ∈ p.support,
149 ⇑(coefficientLeft (p.coeff n) * (leftOreShift D) ^ n)) 1 := by
150 exact congrFun
151 (AddMonoidHom.coe_finsetSum
152 (fun n => coefficientLeft (p.coeff n) * (leftOreShift D) ^ n)
153 p.support) (1 : Polynomial B)
154 _ = ∑ n ∈ p.support,
155 (coefficientLeft (p.coeff n) * (leftOreShift D) ^ n) 1 := by
156 exact Finset.sum_apply (1 : Polynomial B) p.support
157 (fun n => ⇑(coefficientLeft (p.coeff n) * (leftOreShift D) ^ n))
158 _ = ∑ n ∈ p.support,
159 C (p.coeff n) * (((leftOreShift D) ^ n) 1) := by rfl
160 _ = p := by
161 simp_rw [leftOreShift_pow_apply_one]
162 simpa only [Polynomial.sum_def] using Polynomial.sum_C_mul_X_pow_eq p

164theorem faithful_eval_injective (D : OreDivisionDerivation B) :
165 Function.Injective (OreAmbient.eval D (faithfulAmbient D)) := by

The image of the faithful normal-form representation.

201/-- The image of the faithful normal-form representation. -/
202def faithfulRange (D : OreDivisionDerivation B) :
203 Subring (AddMonoid.End (Polynomial B)) where
204 carrier := Set.range (OreAmbient.eval D (faithfulAmbient D))
205 zero_mem' := ⟨0, OreAmbient.eval_zero D (faithfulAmbient D)⟩
206 one_mem' := ⟨1, faithful_eval_one D⟩
207 add_mem' := by
208 rintro _ _ ⟨p, rfl⟩ ⟨q, rfl⟩
209 exact ⟨p + q, OreAmbient.eval_add D (faithfulAmbient D) p q⟩
210 neg_mem' := by
211 rintro _ ⟨p, rfl⟩
212 exact ⟨-p, map_neg (OreAmbient.evalAddHom D (faithfulAmbient D)) p⟩
213 mul_mem' := by
214 rintro _ _ ⟨p, rfl⟩ ⟨q, rfl⟩
215 exact ⟨rightMul D p q, OreAmbient.eval_rightMul D (faithfulAmbient D) p q⟩

221/-- A coefficient-left polynomial regarded as an element of the Ore ring. -/
222def normalForm (D : OreDivisionDerivation B) (p : Polynomial B) : NormalOre D :=

The universal map out of the concrete Ore ring.

334/-- The universal map out of the concrete Ore ring. -/
335def oreLift (D : OreDivisionDerivation B) (O : OreAmbient B A D) :
336 NormalOre D →+* A where
337 toFun z := OreAmbient.eval D O ((normalFormAddEquiv D).symm z)
338 map_zero' := by
339 change OreAmbient.eval D O ((normalFormAddEquiv D).symm 0) = 0
340 rw [(normalFormAddEquiv D).symm.map_zero]
341 exact OreAmbient.eval_zero D O
342 map_one' := by
343 have h : (normalFormAddEquiv D).symm (1 : NormalOre D) = 1 := by
344 apply (normalFormAddEquiv D).injective
345 simp [normalFormAddEquiv, normalFormAddHom]
346 change OreAmbient.eval D O ((normalFormAddEquiv D).symm 1) = 1
347 rw [h]
348 exact ambient_eval_one D O
349 map_add' z w := by
350 change OreAmbient.eval D O ((normalFormAddEquiv D).symm (z + w)) =
351 OreAmbient.eval D O ((normalFormAddEquiv D).symm z) +
352 OreAmbient.eval D O ((normalFormAddEquiv D).symm w)
353 have h := (normalFormAddEquiv D).symm.toAddHom.map_add z w
354 change (normalFormAddEquiv D).symm (z + w) =
355 (normalFormAddEquiv D).symm z + (normalFormAddEquiv D).symm w at h
356 rw [h]
357 exact OreAmbient.eval_add D O _ _
358 map_mul' z w := by
359 let p := (normalFormAddEquiv D).symm z
360 let q := (normalFormAddEquiv D).symm w
361 have hz : normalForm D p = z := by
362 change (normalFormAddEquiv D) p = z
363 exact (normalFormAddEquiv D).apply_symm_apply z
364 have hw : normalForm D q = w := by
365 change (normalFormAddEquiv D) q = w
366 exact (normalFormAddEquiv D).apply_symm_apply w
367 have hpq : (normalFormAddEquiv D).symm (z * w) = rightMul D p q := by
368 apply (normalFormAddEquiv D).injective
369 rw [(normalFormAddEquiv D).apply_symm_apply]
370 symm
371 change normalForm D (rightMul D p q) = z * w
372 rw [normalForm_mul, hz, hw]
373 change OreAmbient.eval D O ((normalFormAddEquiv D).symm (z * w)) =
374 OreAmbient.eval D O p * OreAmbient.eval D O q
375 rw [hpq]
376 exact OreAmbient.eval_rightMul D O p q

A ring map out of NormalOre D is uniquely determined by the coefficient map and the image of the Ore variable.

394/-- A ring map out of `NormalOre D` is uniquely determined by the coefficient
395map and the image of the Ore variable. -/
396theorem oreLift_unique (D : OreDivisionDerivation B) (O : OreAmbient B A D)
397 (g : NormalOre D →+* A)
398 (hCoefficient : ∀ b, g (normalCoefficient D b) = O.embed b)
399 (hVariable : g (normalVariable D) = O.x) :
400 g = oreLift D O := by
401 ext z
402 rcases normalForm_surjective D z with ⟨p, rfl⟩
403 rw [oreLift_normalForm]
404 induction p using Polynomial.induction_on' with
405 | add p q hp hq =>
406 rw [normalForm_add, g.map_add, OreAmbient.eval_add, hp, hq]
407 | monomial n b =>
408 rw [normalForm_monomial, g.map_mul, g.map_pow, hCoefficient, hVariable,
409 OreAmbient.eval_monomial]
