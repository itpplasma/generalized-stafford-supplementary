7 /-!
8 # Right Ore conditions and right Noetherian domains
9 
10 Right ideals and right modules use scalars in the opposite ring. The ascending
11 chain argument uses the finite sums of `b^j aR`: if `aR` and `bR` have zero
12 intersection, cancellation makes every new summand contribute a new element.
13 -/

22 /-- The common-right-multiple condition is Mathlib's left Ore condition on
23 all nonzero elements of the opposite ring. `Nonempty` forgets the chosen
24 numerator and denominator functions in `OreSet`. -/
25 theorem rightOreCondition_iff_nonempty_oreSet :
26     RightOreCondition R ↔ Nonempty (OreLocalization.OreSet (Rᵐᵒᵖ)⁰) := by
27   rw [OreLocalization.nonempty_oreSet_iff_of_noZeroDivisors]
28   constructor
29   · intro h r s
30     by_cases hr : r = 0
31     · exact ⟨0, 1, by simp [hr]⟩
32     obtain ⟨x, y, hxy, hne⟩ := h (a := unop r) (b := unop s.val)
33       (by simpa using hr) (by simpa using nonZeroDivisors.coe_ne_zero s)
34     have hx : x ≠ 0 := by intro hx; simp [hx] at hne
35     exact ⟨op y, ⟨op x, mem_nonZeroDivisors_iff_ne_zero.mpr (by simpa using hx)⟩,
36       unop_injective hxy⟩
37   · intro h a b ha hb
38     obtain ⟨y, x, hxy⟩ := h (op a)
39       ⟨op b, mem_nonZeroDivisors_iff_ne_zero.mpr (by simpa using hb)⟩
40     refine ⟨unop x.val, unop y, congrArg unop hxy, mul_ne_zero ha ?_⟩
41     simpa using nonZeroDivisors.coe_ne_zero x

43 /-- Finite partial sums of `aR + baR + ⋯`, written recursively to expose
44 cancellation of the lowest power of `b`. -/
45 private def partialSum (a b : R) : ℕ → Submodule Rᵐᵒᵖ R
46   | 0 => ⊥
47   | n + 1 => principalRightIdeal a ⊔ (partialSum a b n).map (leftMulLinear b)
48 
49 omit [IsDomain R] in
50 private theorem partialSum_mono (a b : R) : Monotone (partialSum a b) := by
51   apply monotone_nat_of_le_succ
52   intro n
53   induction n with
54   | zero => exact bot_le
55   | succ n ih => exact sup_le_sup_left (Submodule.map_mono ih) _
56 
57 omit [IsDomain R] in
58 private theorem pow_mul_mem_partialSum (a b : R) (n : ℕ) :
59     b ^ n * a ∈ partialSum a b (n + 1) := by
60   induction n with
61   | zero => exact Submodule.mem_sup_left (by exact ⟨1, by simp [leftMulLinear]⟩)
62   | succ n ih =>
63     apply Submodule.mem_sup_right
64     exact ⟨b ^ n * a, ih, by simp [leftMulLinear, pow_succ', mul_assoc]⟩
65 
66 private theorem pow_mul_not_mem_partialSum (a b : R) (ha : a ≠ 0) (hb : b ≠ 0)
67     (hdis : ∀ x y : R, a * x = b * y → a * x = 0) (n : ℕ) :
68     b ^ n * a ∉ partialSum a b n := by
69   induction n with
70   | zero => simpa [partialSum] using ha
71   | succ n ih =>
72     intro hm
73     obtain ⟨u, ⟨x, rfl⟩, v, ⟨y, hy, rfl⟩, huv⟩ := Submodule.mem_sup.mp hm
74     change a * x + b * y = b ^ (n + 1) * a at huv
75     have heq : a * x = b * (b ^ n * a - y) := by
76       rw [mul_sub, ← mul_assoc, ← pow_succ']
77       exact eq_sub_of_add_eq huv
78     have hax := hdis x _ heq
79     rw [hax, zero_add, pow_succ', mul_assoc] at huv
80     exact ih ((mul_left_cancel₀ hb huv) ▸ hy)

82 /-- A right Noetherian domain has nonzero common right multiples.
83 The Noetherian hypothesis is on the right regular module, not the left one. -/
84 theorem rightOreCondition_of_isNoetherian [IsNoetherian Rᵐᵒᵖ R] :
85     RightOreCondition R := by
86   intro a b ha hb
87   by_contra h
88   have hdis : ∀ x y : R, a * x = b * y → a * x = 0 := by
89     intro x y hxy
90     by_contra hne
91     exact h ⟨x, y, hxy, hne⟩
92   obtain ⟨n, hn⟩ := monotone_stabilizes_iff_noetherian.mpr
93     (inferInstance : IsNoetherian Rᵐᵒᵖ R)
94     ⟨partialSum a b, partialSum_mono a b⟩
95   have heq : partialSum a b n = partialSum a b (n + 1) := hn (n + 1) (Nat.le_succ n)
96   apply pow_mul_not_mem_partialSum a b ha hb hdis n
97   rw [heq]
98   exact pow_mul_mem_partialSum a b n

100 /-- The ring-class version of the right Noetherian Ore theorem. -/
101 theorem rightOreCondition_of_isNoetherianRing [IsNoetherianRing Rᵐᵒᵖ] :
102     RightOreCondition R := by
103   let : IsNoetherian Rᵐᵒᵖ R := isNoetherian_of_injective
104     (opLinearEquiv Rᵐᵒᵖ : R ≃ₗ[Rᵐᵒᵖ] Rᵐᵒᵖ).toLinearMap
105     (opLinearEquiv Rᵐᵒᵖ).injective
106   exact rightOreCondition_of_isNoetherian

108 /-- A choice of Mathlib Ore data for a right Noetherian domain. This is not
109 a global instance, so it does not override other choices of Ore data. -/
110 @[instance_reducible]
111 noncomputable def oreSetOppositeOfIsNoetherian [IsNoetherian Rᵐᵒᵖ R] :
112     OreLocalization.OreSet (Rᵐᵒᵖ)⁰ :=

125 /-- Torsion passes to a submodule, or any module embedded in a torsion module. -/
126 theorem isTorsionRight_of_injective (f : M →ₗ[Rᵐᵒᵖ] N) (hf : Function.Injective f)
127     (hN : IsTorsionRight (R := R) (M := N)) : IsTorsionRight (R := R) (M := M) := by
128   intro m
129   obtain ⟨s, hs, hsm⟩ := hN (f m)
130   exact ⟨s, hs, hf (by simpa using hsm)⟩
131 
132 /-- Torsion passes to any quotient presented by a surjective linear map. -/
133 theorem isTorsionRight_of_surjective (f : M →ₗ[Rᵐᵒᵖ] N) (hf : Function.Surjective f)
134     (hM : IsTorsionRight (R := R) (M := M)) : IsTorsionRight (R := R) (M := N) := by
135   intro n
136   obtain ⟨m, rfl⟩ := hf n
137   obtain ⟨s, hs, hsm⟩ := hM m
138   exact ⟨s, hs, by simpa using congrArg f hsm⟩

140 /-- Torsion of a principal quotient implies torsion of every larger ideal's quotient. -/
141 theorem quotient_isTorsion_of_principal_le [IsDomain R]
142     (hOre : OreRightIntersection.RightOreCondition R) (q : R) (hq : q ≠ 0)
143     (I : Submodule Rᵐᵒᵖ R) (hI : principalRightIdeal q ≤ I) :
144     IsTorsionRight (R := R) (M := R ⧸ I) := by
145   let := (OreRightIntersection.rightOreCondition_iff_nonempty_oreSet.mp hOre).some
146   exact isTorsionRight_of_surjective (Submodule.factor hI)
147     (Submodule.factor_surjective hI) (principal_quotient_isTorsion q hq)

149 /-- Over a right Ore domain, torsion elements form a right submodule. -/
150 def torsionSubmodule [IsDomain R] (hOre : OreRightIntersection.RightOreCondition R) :
151     Submodule Rᵐᵒᵖ M where
152   carrier := {m | ∃ s : R, s ≠ 0 ∧ op s • m = 0}
153   zero_mem' := ⟨1, one_ne_zero, smul_zero _⟩
154   add_mem' := by
155     rintro m n ⟨a, ha, hma⟩ ⟨b, hb, hnb⟩
156     obtain ⟨x, y, hxy, hne⟩ := hOre ha hb
157     refine ⟨a * x, hne, ?_⟩
158     rw [smul_add]
159     have hm : op (a * x) • m = 0 := by
160       rw [op_mul, mul_smul, hma, smul_zero]
161     have hn : op (a * x) • n = 0 := by
162       rw [hxy, op_mul, mul_smul, hnb, smul_zero]
163     rw [hm, hn, add_zero]
164   smul_mem' := by
165     rintro r m ⟨a, ha, hma⟩
166     by_cases hr : unop r = 0
167     · refine ⟨1, one_ne_zero, ?_⟩
168       have : r = 0 := unop_injective hr
169       simp [this]
170     obtain ⟨x, y, hxy, hne⟩ := hOre hr ha
171     have hx : x ≠ 0 := by intro hx; simp [hx] at hne
172     refine ⟨x, hx, ?_⟩
173     rw [← mul_smul]
174     change op (unop r * x) • m = 0
175     rw [hxy, op_mul, mul_smul, hma, smul_zero]
176 
177 end AlgebraicAnalysis.DenominatorTorsion
