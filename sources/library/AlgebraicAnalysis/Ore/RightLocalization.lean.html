53/-- Every element of a right Ore localization has a right denominator in `S`
54which is a unit and clears the fraction. -/
55theorem rightOre_clear
56 {R : Type u} [Ring R] {S : Submonoid R}
57 [OreLocalization.OreSet (oppositeSubmonoid S)]
58 (q : RightOreLocalization R S) :
59 ∃ a s : R, s ∈ S ∧ IsUnit (rightNumeratorRingHom (S := S) s) ∧
60 q * rightNumeratorRingHom (S := S) s = rightNumeratorRingHom (S := S) a := by
61 generalize hx : q.unop = x
62 induction x using OreLocalization.ind with
63 | _ a s =>
64 refine ⟨a.unop, s.val.unop, s.property, ?_, ?_⟩
65 · exact isUnit_op.mpr (OreLocalization.numerator_isUnit
66 (R := Rᵐᵒᵖ) (S := oppositeSubmonoid S) s)
67 · apply MulOpposite.unop_injective
68 rw [show q = MulOpposite.op (a /ₒ s) by
69 apply MulOpposite.unop_injective
