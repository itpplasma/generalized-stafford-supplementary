Finite right quotients over noncommutative Ore coefficients

Monic right division leaves coefficient-left remainders. A signed reverse normal-ordering identity rewrites them in a finite right-coefficient window. The natural scalar ring is therefore Bop, not B; this removes the commutativity restriction from the older finite-quotient consumer.

The final section instantiates the construction on the outer momentum layer of the iterated Weyl tower and uses the literal generators d and xNd.

100 normalForm D (X ^ ij.2))))

107 normalForm D (X ^ ij.2)))) =

The finite right-coefficient window of order less than n.

137/-- The finite right-coefficient window of order less than `n`. -/
138def rightCoefficientWindow (D : OreDivisionDerivation B) (n : ℕ) :
139 Submodule Bᵐᵒᵖ (NormalOre D) :=
140 Submodule.span Bᵐᵒᵖ
141 (Set.range fun j : Fin n => normalForm D (X ^ (j : ℕ)))
142
143lemma normalForm_monomial_mem_rightCoefficientWindow
144 (D : OreDivisionDerivation B) (b : B) {j n : ℕ} (hj : j < n) :
145 normalForm D (monomial j b) ∈ rightCoefficientWindow D n := by
146 rw [normalForm_monomial_reverse]
147 apply Submodule.sum_mem
148 intro ij hij
149 rw [← Nat.cast_smul_eq_nsmul Bᵐᵒᵖ]
150 apply Submodule.smul_mem
151 have hijSum : ij.1 + ij.2 = j :=
152 Finset.HasAntidiagonal.mem_antidiagonal.mp hij
153 have hijLt : ij.2 < n := by omega
154 have hpow : normalForm D (X ^ ij.2) ∈ rightCoefficientWindow D n := by
155 apply Submodule.subset_span
156 exact ⟨⟨ij.2, hijLt⟩, rfl⟩
157 have hscalar :
158 (MulOpposite.op ((D^[ij.1]) b)) • normalForm D (X ^ ij.2) ∈
159 rightCoefficientWindow D n :=
160 Submodule.smul_mem _ _ hpow
161 by_cases heven : Even ij.1
162 · rw [heven.neg_one_pow]
163 simpa only [one_mul] using hscalar
164 · have hodd : Odd ij.1 := Nat.not_even_iff_odd.mp heven
165 rw [hodd.neg_one_pow]
166 simpa only [neg_one_mul] using
167 (rightCoefficientWindow D n).neg_mem hscalar

169theorem normalForm_mem_rightCoefficientWindow_of_degree_lt
170 (D : OreDivisionDerivation B) (p : Polynomial B) (n : ℕ)
171 (hp : p = 0 ∨ p.natDegree < n) :
172 normalForm D p ∈ rightCoefficientWindow D n := by
173 rcases hp with rfl | hp
174 · simp
175 · change normalFormAddHom D p ∈ rightCoefficientWindow D n
176 rw [← Polynomial.sum_monomial_eq p, Polynomial.sum_def,
177 map_sum (normalFormAddHom D)]
178 apply Submodule.sum_mem
179 intro j hj
180 apply normalForm_monomial_mem_rightCoefficientWindow D
181 exact lt_of_le_of_lt (Polynomial.le_natDegree_of_mem_supp j hj) hp

A right ideal viewed as a coefficient-opposite submodule.

183/-- A right ideal viewed as a coefficient-opposite submodule. -/
184def rightIdealAsCoeffSubmodule (D : OreDivisionDerivation B)
185 (I : Submodule (NormalOre D)ᵐᵒᵖ (NormalOre D)) :
186 Submodule Bᵐᵒᵖ (NormalOre D) where
187 carrier := I
188 zero_mem' := I.zero_mem
189 add_mem' := I.add_mem
190 smul_mem' := by
191 intro b a ha
192 change a * normalCoefficient D b.unop ∈ I
193 rw [← op_smul_eq_mul]
194 exact I.smul_mem (MulOpposite.op (normalCoefficient D b.unop)) ha

209abbrev TwoGeneratorQuotient (D : OreDivisionDerivation B)

219lemma normalForm_rightMul_H_mem_twoGeneratorCoeffSubmodule

229theorem rightCoefficientWindow_quotient_surjective
230 [Nontrivial B] (D : OreDivisionDerivation B)
231 (H J : Polynomial B) (hH : H.Monic) :
232 Function.Surjective
233 ((twoGeneratorCoeffSubmodule D H J).mkQ.comp
234 (rightCoefficientWindow D H.natDegree).subtype) := by
235 intro y
236 obtain ⟨a, rfl⟩ :=
237 (twoGeneratorCoeffSubmodule D H J).mkQ_surjective y
238 obtain ⟨p, rfl⟩ := normalForm_surjective D a
239 obtain ⟨q, r, hdecomp, hr⟩ := right_division_exists D H p hH
240 have hrWindow : normalForm D r ∈
241 rightCoefficientWindow D H.natDegree :=
242 normalForm_mem_rightCoefficientWindow_of_degree_lt D r H.natDegree hr
243 refine ⟨⟨normalForm D r, hrWindow⟩, ?_⟩
244 change (twoGeneratorCoeffSubmodule D H J).mkQ (normalForm D r) =
245 (twoGeneratorCoeffSubmodule D H J).mkQ (normalForm D p)
246 apply (Submodule.Quotient.eq (twoGeneratorCoeffSubmodule D H J)).2
247 have hmultiple : normalForm D (rightMul D H q) ∈
248 twoGeneratorCoeffSubmodule D H J :=
249 normalForm_rightMul_H_mem_twoGeneratorCoeffSubmodule D H J q
250 rw [hdecomp, normalForm_add]
251 simpa using (twoGeneratorCoeffSubmodule D H J).neg_mem hmultiple

253theorem twoGeneratorQuotient_finite
254 [Nontrivial B] (D : OreDivisionDerivation B)
255 (H J : Polynomial B) (hH : H.Monic) :
256 Module.Finite Bᵐᵒᵖ (TwoGeneratorQuotient D H J) := by
257 letI : Module.Finite Bᵐᵒᵖ (rightCoefficientWindow D H.natDegree) :=
258 Module.Finite.span_of_finite Bᵐᵒᵖ (Set.finite_range
259 (fun j : Fin H.natDegree => normalForm D (X ^ (j : ℕ))))
260 exact Module.Finite.of_surjective
261 ((twoGeneratorCoeffSubmodule D H J).mkQ.comp
262 (rightCoefficientWindow D H.natDegree).subtype)
263 (rightCoefficientWindow_quotient_surjective D H J hH)
