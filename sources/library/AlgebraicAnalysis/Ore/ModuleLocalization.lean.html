9 # Ore localization of modules: exactness, torsion kernel and uniform dimension
10 
11 Let `S` be a left Ore set in a ring `A` and `X` a left `A`-module. Mathlib's
12 `OreLocalization S X`, written `X[S⁻¹]`, consists of fractions `s⁻¹ x` (written
13 `x /ₒ s`) and is a module over the Ore localization `A[S⁻¹]`. This file adds
14 the functorial and dimension-theoretic facts used for Goldie rank.
15 
16 * `OreModule.toLocalization S X : X →ₗ[A] X[S⁻¹]`, `x ↦ x /ₒ 1`, has kernel the
17   `S`-torsion elements (`toLocalization_eq_zero_iff`).
18 * `OreModule.localizationMap S f : X[S⁻¹] →ₗ[A[S⁻¹]] Y[S⁻¹]` is the localization of an
19   `A`-linear map `f`. Localization is exact: it preserves injective maps,
20   surjective maps and exactness in the middle (`localizationMap_injective`,
21   `localizationMap_surjective`, `localizationMap_exact`); in other words `A[S⁻¹]` is flat.
22 * `OreModule.finite`: the localization of a finitely generated module is
23   finitely generated.
24 * `OreModule.uniformDimGE_iff_of_injective`: if `N` embeds `A`-linearly in an
25   `A[S⁻¹]`-module `P` such that every element of `P` is multiplied into `N` by
26   an element of `S`, then `N` (over `A`) and `P` (over `A[S⁻¹]`) contain direct
27   sums of the same numbers of nonzero submodules. The submodules are compared
28   through the `S`-saturation `OreModule.saturation`.
29 
30 For right modules over a ring `R` and a right Ore set of `R`, apply these facts
31 with `A = Rᵐᵒᵖ` (see `AlgebraicAnalysis.Goldie.FractionModule`).
32 
33 ## References
34 
35 J. C. McConnell and J. C. Robson, *Noncommutative Noetherian Rings*, 2.1.16–2.1.17
36 (flatness and the torsion kernel) and 2.2.13 (uniform dimension is preserved);
37 K. R. Goodearl and R. B. Warfield, *An Introduction to Noncommutative Noetherian
38 Rings*, 2nd ed., Chapter 10. Nothing here is new.

55 /-- The canonical `A`-linear map `X → X[S⁻¹]`, `x ↦ x /ₒ 1`. -/
56 def toLocalization : X →ₗ[A] OreLocalization S X where
57   toFun x := x /ₒ (1 : S)
58   map_add' _ _ := add_oreDiv.symm
59   map_smul' a x := (smul_oreDiv_one a x).symm

72 /-- **The torsion kernel.** An element `x` maps to zero in `X[S⁻¹]` if and only
73 if `s • x = 0` for some `s ∈ S`. -/
74 theorem toLocalization_eq_zero_iff {x : X} :
75     toLocalization S X x = 0 ↔ ∃ s ∈ S, s • x = 0 := by
76   rw [toLocalization_apply]
77   constructor
78   · intro h
79     rw [OreLocalization.zero_def, oreDiv_eq_iff] at h
80     obtain ⟨u, v, h1, h2⟩ := h
81     have huv : (u : A) = v := by simpa using h2
82     refine ⟨u, u.2, ?_⟩
83     rw [huv, ← h1, smul_zero]
84   · rintro ⟨s, hs, hsx⟩
85     rw [OreLocalization.expand' x 1 ⟨s, hs⟩, Submonoid.smul_def, hsx, zero_oreDiv]
86 
87 /-- A fraction `s⁻¹ x` vanishes if and only if `x` maps to zero. -/
88 theorem oreDiv_eq_zero_iff {x : X} {s : S} :
89     x /ₒ s = 0 ↔ toLocalization S X x = 0 := by
90   constructor
91   · intro h
92     rw [← smul_oreDiv_self, h, smul_zero]
93   · intro h
94     have : x /ₒ s = ((1 : A) /ₒ s) • toLocalization S X x := by
95       rw [toLocalization_apply, OreLocalization.one_div_smul, one_mul]
96     rw [this, h, smul_zero]

99 /-- The localization `X[S⁻¹] → Y[S⁻¹]`, `s⁻¹ x ↦ s⁻¹ f(x)`, of an `A`-linear map
100 `f : X → Y`. -/
101 def localizationMap (f : X →ₗ[A] Y) : OreLocalization S X →ₗ[OreLocalization S A] OreLocalization S Y where
102   toFun := liftExpand (fun x s => f x /ₒ s) fun x t s ht => by
103     rw [map_smul]
104     exact OreLocalization.expand (f x) s t ht
105   map_add' p q := by
106     induction p using OreLocalization.ind with | _ x s =>
107     induction q using OreLocalization.ind with | _ y t =>
108     rw [oreDiv_add_oreDiv, liftExpand_of, liftExpand_of, liftExpand_of, map_add,
109       Submonoid.smul_def, map_smul, map_smul, oreDiv_add_oreDiv, Submonoid.smul_def]
110   map_smul' q p := by
111     induction q using OreLocalization.ind with | _ r t =>
112     induction p using OreLocalization.ind with | _ x s =>
113     rw [oreDiv_smul_oreDiv, RingHom.id_apply, liftExpand_of, liftExpand_of, map_smul,
114       oreDiv_smul_oreDiv]

126 /-- Localization preserves injective maps. -/
127 theorem localizationMap_injective {f : X →ₗ[A] Y} (hf : Function.Injective f) :
128     Function.Injective (localizationMap S f) := by
129   rw [injective_iff_map_eq_zero]
130   intro p hp
131   induction p using OreLocalization.ind with | _ x s =>
132   rw [localizationMap_oreDiv, oreDiv_eq_zero_iff, toLocalization_eq_zero_iff] at hp
133   obtain ⟨t, ht, htx⟩ := hp
134   rw [oreDiv_eq_zero_iff, toLocalization_eq_zero_iff]
135   exact ⟨t, ht, hf (by rw [map_smul, htx, map_zero])⟩
136 
137 /-- Localization preserves surjective maps. -/
138 theorem localizationMap_surjective {f : X →ₗ[A] Y} (hf : Function.Surjective f) :
139     Function.Surjective (localizationMap S f) := by
140   intro q
141   induction q using OreLocalization.ind with | _ y s =>
142   obtain ⟨x, rfl⟩ := hf y
143   exact ⟨x /ₒ s, rfl⟩

145 /-- **Ore localization is exact** (`A[S⁻¹]` is a flat right `A`-module): the
146 localization of an exact sequence `X → Y → Z` is exact. -/
147 theorem localizationMap_exact {f : X →ₗ[A] Y} {g : Y →ₗ[A] Z} (hfg : Function.Exact f g) :
148     Function.Exact (localizationMap S f) (localizationMap S g) := by
149   intro q
150   induction q using OreLocalization.ind with | _ y s =>
151   rw [localizationMap_oreDiv, oreDiv_eq_zero_iff, toLocalization_eq_zero_iff]
152   constructor
153   · rintro ⟨t, ht, hty⟩
154     obtain ⟨x, hx⟩ := (hfg (t • y)).mp (by rw [map_smul, hty])
155     refine ⟨x /ₒ (⟨t, ht⟩ * s), ?_⟩
156     rw [localizationMap_oreDiv, hx, OreLocalization.expand' y s ⟨t, ht⟩, Submonoid.smul_def]
157   · rintro ⟨p, hp⟩
158     induction p using OreLocalization.ind with | _ x u =>
159     rw [localizationMap_oreDiv, oreDiv_eq_iff] at hp
160     obtain ⟨a, b, h1, h2⟩ := hp
161     refine ⟨a, a.2, ?_⟩
162     rw [← map_smul, ← Submonoid.smul_def, h1, map_smul, hfg.apply_apply_eq_zero, smul_zero]

165 /-- The localization of a finitely generated module is finitely generated over
166 the localized ring. -/
167 theorem finite [Module.Finite A X] : Module.Finite (OreLocalization S A) (OreLocalization S X) := by
168   classical
169   obtain ⟨g, hg⟩ := Module.Finite.fg_top (R := A) (M := X)
170   refine ⟨⟨g.image (toLocalization S X), ?_⟩⟩
171   rw [eq_top_iff]
172   rintro q -
173   induction q using OreLocalization.ind with | _ x s =>
174   have hx : toLocalization S X x ∈ (Submodule.span (OreLocalization S A)
175       ((g.image (toLocalization S X) : Finset _) : Set (OreLocalization S X))).restrictScalars A := by
176     refine Submodule.span_le_restrictScalars A (OreLocalization S A) _ ?_
177     rw [Finset.coe_image, ← Submodule.map_span]
178     exact Submodule.mem_map_of_mem (hg ▸ Submodule.mem_top)
179   have : x /ₒ s = ((1 : A) /ₒ s) • toLocalization S X x := by
180     rw [toLocalization_apply, OreLocalization.one_div_smul, one_mul]
181   rw [this]
182   exact Submodule.smul_mem _ _ hx

191 /-- In an `A[S⁻¹]`-module, `a ∈ A` acts as the fraction `a /ₒ 1`. -/
192 theorem oreDiv_one_smul_eq (a : A) (x : P) : (a /ₒ (1 : S) : OreLocalization S A) • x = a • x := by
193   calc (a /ₒ (1 : S) : OreLocalization S A) • x = (a • (1 : OreLocalization S A)) • x := by
194         rw [OreLocalization.one_def, smul_oreDiv_one, smul_eq_mul, mul_one]
195     _ = a • x := by rw [smul_assoc, one_smul]
196 
197 /-- Elements of `S` act injectively on an `A[S⁻¹]`-module. -/
198 theorem eq_zero_of_smul_eq_zero {s : A} (hs : s ∈ S) {x : P} (h : s • x = 0) : x = 0 := by
199   have hu : IsUnit (s /ₒ (1 : S) : OreLocalization S A) := numerator_isUnit ⟨s, hs⟩
200   rw [← oreDiv_one_smul_eq S] at h
201   rw [← one_smul (OreLocalization S A) x, ← hu.unit.inv_mul, mul_smul]
202   change hu.unit⁻¹ • ((s /ₒ (1 : S) : OreLocalization S A) • x) = 0
203   rw [h, smul_zero]

206 /-- The `S`-saturation of an `A`-submodule `K` of an `A[S⁻¹]`-module `P`: the
207 elements `x` with `s • x ∈ K` for some `s ∈ S`. It is the smallest
208 `A[S⁻¹]`-submodule containing `K`. -/
209 def saturation (K : Submodule A P) : Submodule (OreLocalization S A) P where
210   carrier := {x | ∃ s ∈ S, s • x ∈ K}
211   zero_mem' := ⟨1, S.one_mem, by rw [smul_zero]; exact K.zero_mem⟩
212   add_mem' := by
213     rintro x y ⟨s, hs, hx⟩ ⟨s', hs', hy⟩
214     have h := ore_eq s ⟨s', hs'⟩
215     refine ⟨oreDenom s ⟨s', hs'⟩ * s, S.mul_mem (oreDenom s ⟨s', hs'⟩).2 hs, ?_⟩
216     rw [smul_add]
217     refine K.add_mem ?_ ?_
218     · rw [mul_smul]; exact K.smul_mem _ hx
219     · rw [h, mul_smul]; exact K.smul_mem _ hy
220   smul_mem' := by
221     rintro q x ⟨s, hs, hx⟩
222     induction q using OreLocalization.ind with | _ r t =>
223     have hrt : (t : A) • ((r /ₒ t) • x) = r • x := by
224       rw [← oreDiv_one_smul_eq S, ← mul_smul, OreLocalization.mul_cancel, oreDiv_one_smul_eq]
225     have h := ore_eq r ⟨s, hs⟩
226     refine ⟨oreDenom r ⟨s, hs⟩ * t, S.mul_mem (oreDenom r ⟨s, hs⟩).2 t.2, ?_⟩
227     change (oreDenom r ⟨s, hs⟩ * t : A) • ((r /ₒ t) • x) ∈ K
228     rw [mul_smul, hrt, ← mul_smul, h, mul_smul]
229     exact K.smul_mem _ hx

233 theorem mem_saturation {K : Submodule A P} {x : P} :
234     x ∈ saturation (S := S) K ↔ ∃ s ∈ S, s • x ∈ K := Iff.rfl
235 
236 theorem le_saturation (K : Submodule A P) {x : P} (hx : x ∈ K) : x ∈ saturation (S := S) K :=
237   ⟨1, S.one_mem, by rwa [one_smul]⟩
238 
239 theorem saturation_mono {K K' : Submodule A P} (h : K ≤ K') :
240     saturation (S := S) K ≤ saturation K' := fun _ ⟨s, hs, hx⟩ => ⟨s, hs, h hx⟩
241 
242 theorem saturation_inf (K K' : Submodule A P) :
243     saturation (S := S) K ⊓ saturation K' ≤ saturation (K ⊓ K') := by
244   rintro x ⟨⟨s, hs, hx⟩, ⟨s', hs', hx'⟩⟩
245   have h := ore_eq s ⟨s', hs'⟩
246   refine ⟨oreDenom s ⟨s', hs'⟩ * s, S.mul_mem (oreDenom s ⟨s', hs'⟩).2 hs, ?_, ?_⟩
247   · change (oreDenom s ⟨s', hs'⟩ * s : A) • x ∈ K
248     rw [mul_smul]; exact K.smul_mem _ hx
249   · change (oreDenom s ⟨s', hs'⟩ * s : A) • x ∈ K'
250     rw [h, mul_smul]; exact K'.smul_mem _ hx'
251 
252 theorem saturation_bot : saturation (S := S) (⊥ : Submodule A P) = ⊥ := by
253   rw [eq_bot_iff]
254   rintro x ⟨s, hs, hx⟩
255   exact eq_zero_of_smul_eq_zero S hs ((Submodule.mem_bot A).mp hx)
256 
257 end Saturation

268 /-- A direct sum of `n` nonzero `A`-submodules of `N` gives, through an
269 injective `A`-linear map `φ : N → P` and saturation, a direct sum of `n`
270 nonzero `A[S⁻¹]`-submodules of `P`. -/
271 theorem uniformDimGE_of_injective {φ : N →ₗ[A] P} (hφ : Function.Injective φ) {n : ℕ}
272     (h : UniformDimGE A N n) : UniformDimGE (OreLocalization S A) P n := by
273   obtain ⟨K, hK, hne⟩ := h
274   refine ⟨fun i => saturation (S := S) ((K i).map φ), ?_, fun i h0 => ?_⟩
275   · rw [iSupIndep_def] at hK ⊢
276     intro i
277     rw [disjoint_iff, eq_bot_iff]
278     have hsup : (⨆ (j) (_ : j ≠ i), saturation (S := S) ((K j).map φ)) ≤
279         saturation ((⨆ (j) (_ : j ≠ i), K j).map φ) :=
280       iSup₂_le fun j hj => saturation_mono (Submodule.map_mono (le_iSup₂ (f := fun j _ => K j) j hj))
281     refine (inf_le_inf_left _ hsup).trans ((saturation_inf _ _).trans ?_)
282     rw [← Submodule.map_inf φ hφ, disjoint_iff.mp (hK i), Submodule.map_bot, saturation_bot]
283   · obtain ⟨x, hx, hx0⟩ := Submodule.exists_mem_ne_zero_of_ne_bot (hne i)
284     have : φ x ∈ saturation (S := S) ((K i).map φ) := le_saturation _ (Submodule.mem_map_of_mem hx)
285     simp only at h0
286     rw [h0, Submodule.mem_bot, ← map_zero φ] at this
287     exact hx0 (hφ this)

289 /-- A direct sum of `n` nonzero `A[S⁻¹]`-submodules of `P` pulls back along
290 `φ : N → P` to a direct sum of `n` nonzero `A`-submodules of `N`, provided `φ` is
291 injective and every element of `P` is multiplied into the image of `φ` by an
292 element of `S`. -/
293 theorem uniformDimGE_of_essential {φ : N →ₗ[A] P} (hφ : Function.Injective φ)
294     (hess : ∀ x : P, ∃ s ∈ S, s • x ∈ LinearMap.range φ) {n : ℕ}
295     (h : UniformDimGE (OreLocalization S A) P n) : UniformDimGE A N n := by
296   obtain ⟨K, hK, hne⟩ := h
297   refine ⟨fun i => ((K i).restrictScalars A).comap φ, ?_, fun i h0 => ?_⟩
298   · rw [iSupIndep_def] at hK ⊢
299     intro i
300     rw [Submodule.disjoint_def]
301     intro y hy hy'
302     have hsup : (⨆ (j) (_ : j ≠ i), ((K j).restrictScalars A).comap φ) ≤
303         ((⨆ (j) (_ : j ≠ i), K j).restrictScalars A).comap φ :=
304       iSup₂_le fun j hj => Submodule.comap_mono
305         ((Submodule.restrictScalars_le A).mpr (le_iSup₂ (f := fun j _ => K j) j hj))
306     have := (Submodule.disjoint_def.mp (hK i)) (φ y) hy (hsup hy')
307     exact hφ (this.trans (map_zero φ).symm)
308   · obtain ⟨x, hx, hx0⟩ := Submodule.exists_mem_ne_zero_of_ne_bot (hne i)
309     obtain ⟨s, hs, y, hy⟩ := hess x
310     have hyK : y ∈ ((K i).restrictScalars A).comap φ := by
311       change φ y ∈ (K i).restrictScalars A
312       rw [hy]
313       exact Submodule.smul_mem _ s ((Submodule.restrictScalars_mem A (K i) x).mpr hx)
314     simp only at h0
315     rw [h0, Submodule.mem_bot] at hyK
316     rw [hyK, map_zero] at hy
317     exact hx0 (eq_zero_of_smul_eq_zero S hs hy.symm)

319 /-- **Uniform dimension is preserved by localization** (McConnell–Robson 2.2.13).
320 Let `φ : N → P` be an injective `A`-linear map into an `A[S⁻¹]`-module such that
321 every `x ∈ P` has some `s ∈ S` with `s • x` in the image of `φ`. Then `N`
322 contains a direct sum of `n` nonzero `A`-submodules if and only if `P` contains a
323 direct sum of `n` nonzero `A[S⁻¹]`-submodules. -/
324 theorem uniformDimGE_iff_of_injective {φ : N →ₗ[A] P} (hφ : Function.Injective φ)
325     (hess : ∀ x : P, ∃ s ∈ S, s • x ∈ LinearMap.range φ) {n : ℕ} :
326     UniformDimGE A N n ↔ UniformDimGE (OreLocalization S A) P n :=
327   ⟨uniformDimGE_of_injective hφ, uniformDimGE_of_essential hφ hess⟩
328 
329 end UniformDimension
