8 # Locally small modules and dense submodules
9 
10 All modules are left modules over an arbitrary ring; use the opposite ring for
11 right modules. Local smallness tests finitely generated submodules, rather than
12 asserting a deviation bound on an infinite direct sum. Extension closure uses
13 a Noetherian scalar ring. The small radical is small for Noetherian modules.
14 
15 These are new proofs for the interfaces in generalized-stafford's research
16 notes `docs/intrinsic-finite-krull.md`, §2, and
17 `.lake/research/active/formal-scope-BC.md`, steps 2 and 4; no source-code
18 extraction or downstream declaration replacement is involved.
19 -/

29 /-- Every finitely generated submodule belongs to the strict deviation stratum. -/
30 def LocSmall : Prop := ∀ U : Submodule R M, U.FG → Small R e U
31 
32 /-- A submodule is dense when its quotient belongs to the strict stratum. -/
33 def Dense {M : Type v} [AddCommGroup M] [Module R M]
34     (U : Submodule R M) : Prop := Small R e (M ⧸ U)
35 
36 /-- The sum of all small submodules. It is small when the ambient module is Noetherian. -/
37 def smallRadical : Submodule R M := ⨆ U : {U : Submodule R M // Small R e U}, U.val

41 theorem Small.map {U : Submodule R M} (h : Small R e U) (f : M →ₗ[R] N) :
42     Small R e (U.map f) :=
43   h.of_surjective (f.submoduleMap U) (by
44     rintro ⟨y, x, hx, rfl⟩
45     exact ⟨⟨x, hx⟩, rfl⟩)

47 theorem Small.of_le {U V : Submodule R M} (h : Small R e V) (hUV : U ≤ V) :
48     Small R e U :=
49   h.of_injective (Submodule.inclusion hUV) (Submodule.inclusion_injective hUV)

51 theorem Small.sup {U V : Submodule R M} (hU : Small R e U) (hV : Small R e V) :
52     Small R e (U ⊔ V : Submodule R M) := by
53   rw [Submodule.sup_eq_range]
54   exact (hU.prod hV).of_surjective (U.subtype.coprod V.subtype).rangeRestrict (by rintro ⟨y, x, rfl⟩; exact ⟨x, rfl⟩)

56 theorem Small.finset_sup {ι : Type*} (s : Finset ι) (U : ι → Submodule R M)

58   Finset.sup_induction (p := fun U : Submodule R M => Small R e U)

61 /-- Pulling back a small submodule of a quotient gives a small extension. -/
62 theorem Small.comap_mkQ {p : Submodule R M} (hp : Small R e p)
63     {q : Submodule R (M ⧸ p)} (hq : Small R e q) :
64     Small R e (q.comap p.mkQ) := by
65   let f : q.comap p.mkQ →ₗ[R] q :=
66     (p.mkQ.comp (q.comap p.mkQ).subtype).codRestrict q (fun x => x.property)
67   have hf : Function.Surjective f := by
68     rintro ⟨x, hx⟩
69     obtain ⟨y, rfl⟩ := p.mkQ_surjective x
70     exact ⟨⟨y, hx⟩, rfl⟩
71   have hk : Small R e (LinearMap.ker f) := by
72     apply hp.of_injective
73       (((q.comap p.mkQ).subtype.comp (LinearMap.ker f).subtype).codRestrict p ?_) ?_
74     · intro x
75       exact (Submodule.Quotient.mk_eq_zero p).mp
76         (congrArg Subtype.val x.property)
77     · intro x y h
78       apply Subtype.ext
79       apply Subtype.ext
80       exact congrArg (fun z : p => (z : M)) h
81   exact (small_iff_submodule_quotient e (LinearMap.ker f)).mpr
82     ⟨hk, (small_linearEquiv_iff e (f.quotKerEquivOfSurjective hf)).mpr hq⟩

84 theorem le_smallRadical {U : Submodule R M} (hU : Small R e U) :
85     U ≤ smallRadical R e M := le_iSup (fun U : {U : Submodule R M // Small R e U} => U.val) ⟨U, hU⟩

87 /-- A finitely generated submodule of a sum of small submodules is small. -/
88 theorem small_of_fg_le_iSup {ι : Type*} (U : ι → Submodule R M)
89     (hU : ∀ i, Small R e (U i)) {V : Submodule R M} (hV : V.FG)
90     (hVU : V ≤ ⨆ i, U i) : Small R e V := by
91   classical
92   obtain ⟨s, hs⟩ := (Submodule.fg_iff_compact V).mp hV |>.exists_finset_of_le_iSup U hVU
93   rw [← Finset.sup_eq_iSup] at hs
94   exact (Small.finset_sup s U (fun i _ => hU i)).of_le hs

96 theorem locSmall_smallRadical : LocSmall R e (smallRadical R e M) := by
97   intro U hU
98   have h := small_of_fg_le_iSup
99     (fun V : {V : Submodule R M // Small R e V} => V.val) (fun V => V.property)
100     (hU.map (smallRadical R e M).subtype) (Submodule.map_subtype_le _ _)
101   exact h.of_injective ((smallRadical R e M).subtype.submoduleMap U)
102     (by intro x y h; apply Subtype.ext; apply Subtype.ext; have hh := congrArg Subtype.val h; exact hh)

104 theorem Small.locSmall (h : Small R e M) : LocSmall R e M := fun U _ => h.submodule U

106 /-- Local smallness agrees with smallness on finitely generated modules. -/
107 theorem locSmall_iff_small [Module.Finite R M] : LocSmall R e M ↔ Small R e M := by
108   refine ⟨fun h => ?_, Small.locSmall⟩
109   exact (h ⊤ Module.Finite.fg_top).of_surjective (⊤ : Submodule R M).subtype
110     (by intro x; exact ⟨⟨x, Submodule.mem_top⟩, rfl⟩)

112 /-- In particular, no additional Noetherian modules are killed by local extension. -/
113 theorem locSmall_iff_small_of_isNoetherian [IsNoetherian R M] :
114     LocSmall R e M ↔ Small R e M := locSmall_iff_small

116 theorem small_smallRadical [IsNoetherian R M] : Small R e (smallRadical R e M) :=
117   locSmall_iff_small.mp locSmall_smallRadical

119 theorem LocSmall.of_injective (h : LocSmall R e N) (f : M →ₗ[R] N)

125 theorem LocSmall.submodule (h : LocSmall R e M) (U : Submodule R M) : LocSmall R e U :=

128 /-- Pointwise small envelopes are enough: finite generators use a finite sum of them. -/
129 theorem locSmall_of_small_envelopes
130     (h : ∀ x : M, ∃ U : Submodule R M, Small R e U ∧ x ∈ U) : LocSmall R e M := by
131   have ht : smallRadical R e M = ⊤ := by
132     apply top_unique
133     intro x _
134     obtain ⟨U, hU, hx⟩ := h x
135     exact le_smallRadical hU hx
136   intro U hU
137   exact small_of_fg_le_iSup
138     (fun V : {V : Submodule R M // Small R e V} => V.val) (fun V => V.property)
139     hU (by change U ≤ smallRadical R e M; rw [ht]; exact le_top)
140 
141 theorem LocSmall.small_envelope (h : LocSmall R e M) (x : M) :
142     ∃ U : Submodule R M, Small R e U ∧ x ∈ U :=
143   ⟨R ∙ x, h _ (Submodule.fg_span_singleton x), Submodule.mem_span_singleton_self x⟩

145 theorem LocSmall.of_surjective (h : LocSmall R e M) (f : M →ₗ[R] N)
146     (hf : Function.Surjective f) : LocSmall R e N := by
147   apply locSmall_of_small_envelopes
148   intro y
149   obtain ⟨x, rfl⟩ := hf y
150   obtain ⟨U, hU, hx⟩ := h.small_envelope x
151   exact ⟨U.map f, hU.map f, Submodule.mem_map_of_mem hx⟩

162 /-- The Noetherian ring hypothesis controls the kernel inside each finite submodule. -/
163 theorem LocSmall.of_submodule_quotient [IsNoetherianRing R] (U : Submodule R M)
164     (hU : LocSmall R e U) (hQ : LocSmall R e (M ⧸ U)) : LocSmall R e M := by
165   intro V hV
166   have : Module.Finite R V := Module.Finite.of_fg hV
167   let f : V →ₗ[R] M ⧸ U := U.mkQ.comp V.subtype
168   have hk : Small R e (LinearMap.ker f) := by
169     have hh := hU.of_injective
170       (show LinearMap.ker f →ₗ[R] U from
171         { toFun := fun x => ⟨x.val.val, (Submodule.Quotient.mk_eq_zero U).mp x.property⟩
172           map_add' := fun _ _ => rfl
173           map_smul' := fun _ _ => rfl })
174       (by intro x y h; apply Subtype.ext; apply Subtype.ext; have hh := congrArg Subtype.val h; exact hh)
175     exact locSmall_iff_small.mp hh
176   have hr : Small R e (LinearMap.range f) := hQ _ (Submodule.fg_range f)
177   exact Small.of_submodule_quotient (LinearMap.ker f) hk
178     ((small_linearEquiv_iff e f.quotKerEquivRange).mpr hr)

180 /-- Arbitrary sums of locally small submodules are locally small. -/
181 theorem LocSmall.iSup {ι : Type*} (U : ι → Submodule R M)
182     (hU : ∀ i, LocSmall R e (U i)) : LocSmall R e (⨆ i, U i : Submodule R M) := by
183   have hle : (⨆ i, U i) ≤ smallRadical R e M := by
184     apply iSup_le
185     intro i x hx
186     obtain ⟨V, hV, hmem⟩ := (hU i).small_envelope ⟨x, hx⟩
187     exact le_smallRadical (hV.map (U i).subtype) (Submodule.mem_map_of_mem hmem)
188   exact locSmall_smallRadical.of_injective (Submodule.inclusion hle)
189     (Submodule.inclusion_injective hle)

191 /-- Arbitrary direct sums, without a finite-index hypothesis. -/
192 theorem LocSmall.directSum {ι : Type*} {P : ι → Type v}
193     [∀ i, AddCommGroup (P i)] [∀ i, Module R (P i)]
194     (hP : ∀ i, LocSmall R e (P i)) : LocSmall R e (⨁ i, P i) := by
195   classical
196   let U (i : ι) : Submodule R (⨁ i, P i) := LinearMap.range (DirectSum.lof R ι P i)
197   have hU : ∀ i, LocSmall R e (U i) := fun i =>
198     (hP i).of_surjective (DirectSum.lof R ι P i).rangeRestrict (by rintro ⟨y, x, rfl⟩; exact ⟨x, rfl⟩)
199   have ht : (⨆ i, U i) = ⊤ := by
200     apply top_unique
201     suffices ∀ x : (⨁ i, P i), x ∈ ⨆ i, U i from fun x _ => this x
202     intro x
203     induction x using DirectSum.induction_on with
204     | zero => exact Submodule.zero_mem _
205     | add x y hx hy => exact Submodule.add_mem _ hx hy
206     | of i x => exact le_iSup U i ⟨x, rfl⟩
207   have h := LocSmall.iSup U hU
208   rw [ht] at h
209   exact h.of_surjective (⊤ : Submodule R (⨁ i, P i)).subtype
210     (by intro x; exact ⟨⟨x, Submodule.mem_top⟩, rfl⟩)

219 /-- A Noetherian module modulo its radical has no nonzero small submodule. -/
220 theorem eq_bot_of_small_quotient_radical [IsNoetherian R M]
221     (U : Submodule R (M ⧸ smallRadical R e M)) (hU : Small R e U) : U = ⊥ := by
222   have hle := le_smallRadical
223     ((small_smallRadical (R := R) (e := e) (M := M)).comap_mkQ hU)
224   have he : U.comap (smallRadical R e M).mkQ = smallRadical R e M :=
225     le_antisymm hle (Submodule.le_comap_mkQ _ _)
226   have := congrArg (Submodule.map (smallRadical R e M).mkQ) he
227   simpa only [Submodule.map_comap_eq_of_surjective (Submodule.mkQ_surjective _),
228     Submodule.mkQ_map_self] using this

238 /-- Inverse images of dense submodules are dense, with no surjectivity assumption. -/
239 theorem Dense.comap {V : Submodule R N} (hV : Dense R e V) (f : M →ₗ[R] N) :
240     Dense R e (V.comap f) := by
241   let g := V.mkQ.comp f
242   have hk : g.ker = V.comap f := by ext x; simp [g]
243   rw [Dense, ← hk]
244   exact (small_linearEquiv_iff e g.quotKerEquivRange).mpr (hV.submodule _)

246 /-- Surjective images of dense submodules are dense. -/
247 theorem Dense.map {U : Submodule R M} (hU : Dense R e U) (f : M →ₗ[R] N)
248     (hf : Function.Surjective f) : Dense R e (U.map f) := by

255 /-- Removing a small summand from a dense sum preserves density. -/
256 theorem Dense.of_small_sup {U V : Submodule R M} (h : Dense R e (U ⊔ V))
257     (hU : Small R e U) : Dense R e V := by
258   apply Small.of_submodule_quotient (U.map V.mkQ) (hU.map V.mkQ)
259   apply (small_linearEquiv_iff e (Submodule.quotientQuotientEquivQuotientSup V U)).mpr
260   rw [sup_comm V U]
261   exact h

263 /-- Finite intersections preserve density. -/
264 theorem Dense.inf {U V : Submodule R M} (hU : Dense R e U) (hV : Dense R e V) :
265     Dense R e (U ⊓ V) := by

267   have hk : LinearMap.ker f = U ⊓ V := by ext x; simp [f]

271 /-- Density composes for nested submodules. -/
272 theorem Dense.map_subtype {U : Submodule R M} (hU : Dense R e U)
273     {V : Submodule R U} (hV : Dense R e V) : Dense R e (V.map U.subtype) := by
274   let W := V.map U.subtype
275   let f : M ⧸ W →ₗ[R] M ⧸ U := Submodule.mapQ W U LinearMap.id
276     (Submodule.map_subtype_le U V)
277   have hf : Function.Surjective f := by
278     intro y
279     obtain ⟨x, rfl⟩ := Submodule.mkQ_surjective U y
280     exact ⟨W.mkQ x, rfl⟩
281   have hr : Small R e (LinearMap.ker f) := by
282     let g : U →ₗ[R] LinearMap.ker f :=
283       { toFun := fun x => ⟨W.mkQ x, by simp [f]⟩
284         map_add' := fun _ _ => rfl
285         map_smul' := fun _ _ => rfl }
286     have hg : V ≤ LinearMap.ker g := by
287       intro x hx
288       apply Subtype.ext
289       exact (Submodule.Quotient.mk_eq_zero W).mpr ⟨x, hx, rfl⟩
290     apply hV.of_surjective (V.liftQ g hg)
291     intro y
292     obtain ⟨x, hx⟩ := Submodule.mkQ_surjective W y.val
293     have hxu : x ∈ U := (Submodule.Quotient.mk_eq_zero U).mp (by
294       have : f y.val = 0 := y.property
295       rw [← hx] at this
296       exact this)
297     exact ⟨V.mkQ ⟨x, hxu⟩, Subtype.ext hx⟩
298   exact Small.of_submodule_quotient (LinearMap.ker f) hr
299     ((small_linearEquiv_iff e (f.quotKerEquivOfSurjective hf)).mpr hU)
