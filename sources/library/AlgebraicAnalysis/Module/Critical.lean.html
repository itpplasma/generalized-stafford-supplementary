9 # Critical modules and the small radical
10 
11 All scalars act on the left; use the opposite ring for right modules.
12 The strict dimension stratum is `Module.Small`, including its zero convention.
13 The small radical API comes from `LocSmall`; its smallness uses Noetherianity,
14 but criticality itself does not.
15 -/

24 /-- A nonzero module of exact finite dimension `e` all of whose proper
25 nonzero-submodule quotients lie in the strict stratum below `e`. -/
26 structure Critical (R : Type u) (e : ℕ) (M : Type v) [Ring R]
27     [AddCommGroup M] [Module R M] : Prop where
28   nontrivial : Nontrivial M
29   devLE : DevLE R e M
30   not_small : ¬ Small R e M
31   small_quotient : ∀ p : Submodule R M, p ≠ ⊥ → Small R e (M ⧸ p)
32 
33 /-- Criticality transports along a linear equivalence. -/
34 theorem Critical.of_linearEquiv {e : ℕ} (h : Critical R e M) (f : M ≃ₗ[R] N) :
35     Critical R e N := by
36   have := h.nontrivial
37   refine ⟨f.symm.toEquiv.nontrivial, (devLE_linearEquiv_iff e f).mp h.devLE,
38     fun hs => h.not_small ((small_linearEquiv_iff e f).mpr hs), ?_⟩
39   intro q hq
40   have hp : q.comap (f : M →ₗ[R] N) ≠ ⊥ := by
41     intro he
42     have := congrArg (Submodule.map (f : M →ₗ[R] N)) he
43     simp only [Submodule.map_comap_eq_of_surjective f.surjective, Submodule.map_bot] at this
44     exact hq this
45   exact (small_linearEquiv_iff e (Submodule.Quotient.equiv (q.comap (f : M →ₗ[R] N)) q f
46     (Submodule.map_comap_eq_of_surjective f.surjective q))).mp (h.small_quotient _ hp)
47 
48 /-- Criticality is invariant under linear equivalence. -/
49 theorem critical_linearEquiv_iff (e : ℕ) (f : M ≃ₗ[R] N) :
50     Critical R e M ↔ Critical R e N :=
51   ⟨fun h => h.of_linearEquiv f, fun h => h.of_linearEquiv f.symm⟩

53 /-- A nonzero submodule of a critical module cannot be small. -/
54 theorem Critical.not_small_submodule {e : ℕ} (h : Critical R e M)
55     {p : Submodule R M} (hp : p ≠ ⊥) : ¬Small R e p := by
56   intro hs
57   exact h.not_small (Small.of_submodule_quotient p hs (h.small_quotient p hp))

59 /-- A Noetherian critical module has zero small radical at its own dimension. -/
60 theorem Critical.smallRadical_eq_bot {e : ℕ} (h : Critical R e M) [IsNoetherian R M] :
61     smallRadical R e M = ⊥ := by
62   by_contra he
63   exact h.not_small_submodule he (small_smallRadical (e := e))

65 /-- Intersecting any two nonzero submodules of a critical module stays nonzero
66 (the usual uniformity condition). -/
67 theorem Critical.inf_ne_bot {e : ℕ} (h : Critical R e M)
68     {p q : Submodule R M} (hp : p ≠ ⊥) (hq : q ≠ ⊥) : p ⊓ q ≠ ⊥ := by
69   intro he
70   have hd : Disjoint q p := disjoint_iff.mpr (by simpa [inf_comm] using he.le)
71   have hi : Function.Injective (p.mkQ.comp q.subtype) := by
72     rw [← LinearMap.ker_eq_bot, LinearMap.ker_comp, Submodule.ker_mkQ]
73     exact Submodule.disjoint_iff_comap_eq_bot.mp hd
74   exact h.not_small_submodule hq ((h.small_quotient p hp).of_injective _ hi)
75 
76 /-- Every nonzero submodule of a critical module is critical of the same dimension. -/
77 theorem Critical.submodule {e : ℕ} (h : Critical R e M)
78     (p : Submodule R M) (hp : p ≠ ⊥) : Critical R e p := by
79   refine ⟨p.nontrivial_iff_ne_bot.mpr hp, h.devLE.submodule p,
80     h.not_small_submodule hp, ?_⟩
81   intro q hq
82   have hmap : q.map p.subtype ≠ ⊥ := by
83     intro he
84     have := congrArg (Submodule.comap p.subtype) he
85     simp only [Submodule.comap_map_eq_of_injective (f := p.subtype) Subtype.val_injective,
86       Submodule.comap_bot, Submodule.ker_subtype] at this
87     exact hq this
88   let f := q.mapQ (q.map p.subtype) p.subtype (Submodule.le_comap_map _ _)
89   have hi : Function.Injective f := by
90     rw [← LinearMap.ker_eq_bot, Submodule.ker_mapQ,
91       Submodule.comap_map_eq_of_injective Subtype.val_injective]
92     exact Submodule.mkQ_map_self q
93   exact (h.small_quotient _ hmap).of_injective f hi

95 /-- Dimension-zero criticality is ordinary simplicity. -/
96 theorem critical_zero_of_isSimpleModule [IsSimpleModule R M] : Critical R 0 M := by
97   have := IsSimpleModule.nontrivial R M
98   refine ⟨inferInstance, (by
99       intro c hc
100       by_cases hz : ∃ i, c i = ⊥
101       · obtain ⟨i, hi⟩ := hz
102         exact ⟨i, fun j hj => (le_bot_iff.mp (hi ▸ hc hj)).trans hi.symm⟩
103       · push Not at hz
104         exact ⟨0, fun i _ => ((eq_bot_or_eq_top (c i)).resolve_left (hz i)).trans
105           ((eq_bot_or_eq_top (c 0)).resolve_left (hz 0)).symm⟩),
106     fun h => not_subsingleton M h, ?_⟩
107   intro p hp
108   have : p = ⊤ := (eq_bot_or_eq_top p).resolve_left hp
109   subst p
110   exact small_of_subsingleton 0

112 /-- A finite-deviation bound makes the relation of having a non-small
113 successive quotient well-founded. This is the termination argument for
114 critical-submodule descent. -/
115 theorem DevLE.wellFounded_not_small_quotient {e : ℕ} (h : DevLE R e M) :
116     WellFounded (fun q p : Submodule R M =>
117       q ≤ p ∧ ¬Small R e (p ⧸ q.comap p.subtype)) := by
118   apply wellFounded_iff_isEmpty_descending_chain.mpr
119   refine ⟨fun ⟨c, hc⟩ => ?_⟩
120   have hanti : Antitone c := antitone_nat_of_succ_le (fun i => (hc i).1)
121   cases e with
122   | zero =>
123     obtain ⟨i, hi⟩ := h c hanti
124     have he : c (i + 1) = c i := hi _ (Nat.le_succ i)
125     apply (hc i).2
126     rw [he]
127     have : (c i).comap (c i).subtype = ⊤ := Submodule.comap_subtype_self _
128     rw [this]
129     exact small_of_subsingleton 0
130   | succ e =>
131     obtain ⟨i, hi⟩ := (devLE_succ_iff e).mp h c hanti
132     exact (hc i).2 (hi i le_rfl)

134 /-- In a module with no nonzero small submodule, finite deviation produces
135 an `e`-critical submodule. No Noetherian assumption is needed for descent. -/
136 theorem DevLE.exists_critical_submodule_of_pure {e : ℕ} (h : DevLE R e M)
137     [Nontrivial M] (hpure : ∀ p : Submodule R M, p ≠ ⊥ → ¬Small R e p) :
138     ∃ p : Submodule R M, Critical R e p := by
139   obtain ⟨p, hp, hmin⟩ := h.wellFounded_not_small_quotient.has_min
140     {p : Submodule R M | p ≠ ⊥} ⟨⊤, top_ne_bot⟩
141   refine ⟨p, p.nontrivial_iff_ne_bot.mpr hp, h.submodule p, hpure p hp, ?_⟩
142   intro q hq
143   by_contra hs
144   have hmap : q.map p.subtype ≠ ⊥ := by
145     intro he
146     apply hq
147     have := congrArg (Submodule.comap p.subtype) he
148     simpa only [Submodule.comap_map_eq_of_injective (f := p.subtype)
149       Subtype.val_injective, Submodule.comap_bot, Submodule.ker_subtype] using this
150   apply hmin (q.map p.subtype) hmap
151   refine ⟨Submodule.map_subtype_le p q, ?_⟩
152   rw [Submodule.comap_map_eq_of_injective (f := p.subtype) Subtype.val_injective]
153   exact hs

155 /-- There is a critical submodule of the least dimension occurring among
156 nonzero submodules. In particular its dimension is at most the given bound.
157 The result is stronger than the Noetherian version: descent alone suffices. -/
158 theorem DevLE.exists_critical_submodule_minimal {n : ℕ} (h : DevLE R n M)
159     [Nontrivial M] : ∃ e ≤ n, ∃ p : Submodule R M, Critical R e p ∧
160       ∀ (q : Submodule R M), q ≠ ⊥ → ∀ d, DevLE R d q → e ≤ d := by
161   classical
162   have hex : ∃ e, ∃ p : Submodule R M, p ≠ ⊥ ∧ DevLE R e p :=
163     ⟨n, ⊤, top_ne_bot, h.submodule ⊤⟩
164   let e := Nat.find hex
165   obtain ⟨a, ha, hda⟩ := Nat.find_spec hex
166   have hleast : ∀ (q : Submodule R M), q ≠ ⊥ → ∀ d, DevLE R d q → e ≤ d := by
167     intro q hq d hd
168     exact Nat.find_min' hex ⟨q, hq, hd⟩
169   have hpure : ∀ q : Submodule R a, q ≠ ⊥ → ¬Small R e q := by
170     intro q hq hs
171     have hmap : q.map a.subtype ≠ ⊥ := by
172       intro he
173       apply hq
174       have := congrArg (Submodule.comap a.subtype) he
175       simpa only [Submodule.comap_map_eq_of_injective (f := a.subtype)
176         Subtype.val_injective, Submodule.comap_bot, Submodule.ker_subtype] using this
177     cases he : e with
178     | zero => exact hq (Submodule.subsingleton_iff_eq_bot.mp (by simpa only [he, small_zero_iff] using hs))
179     | succ d =>
180       have hd : DevLE R d (q.map a.subtype) :=
181         (devLE_linearEquiv_iff d (Submodule.equivMapOfInjective a.subtype
182           Subtype.val_injective q)).mp (by simpa only [he, small_succ_iff] using hs)
183       have := hleast _ hmap d hd
184       omega
185   have := a.nontrivial_iff_ne_bot.mpr ha
186   obtain ⟨p, hp⟩ := hda.exists_critical_submodule_of_pure hpure
187   exact ⟨e, Nat.find_min' hex ⟨⊤, top_ne_bot, h.submodule ⊤⟩,
188     p.map a.subtype, hp.of_linearEquiv
189       (Submodule.equivMapOfInjective a.subtype Subtype.val_injective p), hleast⟩
190 
191 /-- Every nonzero finite-deviation module has a critical submodule. -/
192 theorem DevLE.exists_critical_submodule {n : ℕ} (h : DevLE R n M) [Nontrivial M] :
193     ∃ e ≤ n, ∃ p : Submodule R M, Critical R e p := by
194   obtain ⟨e, he, p, hp, _⟩ := h.exists_critical_submodule_minimal
195   exact ⟨e, he, p, hp⟩
