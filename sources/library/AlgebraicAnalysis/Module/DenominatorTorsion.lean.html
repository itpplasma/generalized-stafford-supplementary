This is the unconditional part of packet 9. It separates the algebraic quotient argument from the still-unformalized triangular PBW reduction. An explicit clearing witness for each vector implies torsion of the quotient; the principal right-ideal case is proved directly from the opposite Ore condition. No stage freeness or noncommutative flatness is postulated.

Right-module torsion, with the right scalar displayed as ops.

29/-- Right-module torsion, with the right scalar displayed as `op s`. -/
30def IsTorsionRight : Prop :=
31 ∀ m : M, ∃ s : R, s ≠ 0 ∧ (op s) • m = 0

A denominator-clearing witness for a right submodule quotient.

33/-- A denominator-clearing witness for a right submodule quotient. -/
34def HasDenominatorClearance (N : Submodule Rᵐᵒᵖ M) : Prop :=
35 ∀ m : M, ∃ s : R, s ≠ 0 ∧ (op s) • m ∈ N
36
37theorem quotient_isTorsion_of_clearance
38 (N : Submodule Rᵐᵒᵖ M)
39 (hclear : HasDenominatorClearance (R := R) N) :
40 IsTorsionRight (R := R) (M := M ⧸ N) := by
41 intro z
42 refine Submodule.Quotient.induction_on N z ?_
43 intro m
44 rcases hclear m with ⟨s, hs, hsm⟩
45 refine ⟨s, hs, ?_⟩
46 change N.mkQ ((op s) • m) = 0
47 rw [Submodule.mkQ_apply, Submodule.Quotient.mk_eq_zero]
48 exact hsm
49
50end AbstractClearance

Left multiplication by q is a right-R-linear map.

57/-- Left multiplication by `q` is a right-`R`-linear map. -/
58def leftMulLinear (q : R) : R →ₗ[Rᵐᵒᵖ] R where
59 toFun x := q * x
60 map_add' x y := by
61 change q * (x + y) = q * x + q * y
62 rw [mul_add]
63 map_smul' a x := by
64 change q * (x * (unop a)) = (q * x) * (unop a)
65 rw [mul_assoc]

The right ideal qR, represented as the range of left multiplication.

67/-- The right ideal `qR`, represented as the range of left multiplication. -/
68def principalRightIdeal (q : R) : Submodule Rᵐᵒᵖ R :=
69 LinearMap.range (leftMulLinear q)
70
71@[nolint unusedArguments]
72theorem principalRightIdeal_mem (q x : R) :
73 q * x ∈ principalRightIdeal q := by
74 exact ⟨x, rfl⟩
75
76theorem principal_quotient_isTorsion (q : R) (hq : q ≠ 0) :
77 IsTorsionRight (R := R)
78 (M := R ⧸ principalRightIdeal q) := by
79 intro z
80 refine Submodule.Quotient.induction_on (principalRightIdeal q) z ?_
81 intro x
82 let qop : (Rᵐᵒᵖ)⁰ :=
83 ⟨op q, mem_nonZeroDivisors_iff_ne_zero.mpr (by simpa using hq)⟩
84 rcases OreLocalization.oreCondition (op x) qop with ⟨num, den, hOre⟩
85 let denR : R := unop (den : Rᵐᵒᵖ)
86 have hOre' : x * denR = q * unop num := by
87 have h := congrArg unop hOre
88 simpa only [unop_mul, unop_op, denR, qop] using h
89 refine ⟨denR, ?_, ?_⟩
90 · intro hden
91 have hden' : (den : Rᵐᵒᵖ) = 0 := by
92 apply unop_injective
93 simpa [denR] using hden
94 exact (nonZeroDivisors.coe_ne_zero den) hden'
95 change (principalRightIdeal q).mkQ ((op denR) • x) = 0
96 rw [Submodule.mkQ_apply, Submodule.Quotient.mk_eq_zero]
97 refine ⟨unop num, ?_⟩
98 change q * unop num = x * denR
99 exact hOre'.symm
100
101end PrincipalRightIdeal
