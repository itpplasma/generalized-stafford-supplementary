A critical step with an explicit common finite dimension ceiling.

25/-- A critical step with an explicit common finite dimension ceiling. -/
26def CriticalStep (R : Type u) (n : ℕ) (M : Type v) [Ring R]
27 [AddCommGroup M] [Module R M] (p q : Submodule R M) : Prop :=
28 p ≤ q ∧ ∃ d ≤ n, Critical R d (q ⧸ p.comap q.subtype)

A finite critical filtration, including its zero and whole-module endpoints.

30/-- A finite critical filtration, including its zero and whole-module endpoints. -/
31structure CriticalFiltration (R : Type u) (n : ℕ) (M : Type v) [Ring R]
32 [AddCommGroup M] [Module R M] extends RelSeries {(p, q) | CriticalStep R n M p q} where
33 head_eq_bot : toRelSeries.head = ⊥
34 last_eq_top : toRelSeries.last = ⊤

A lifted submodule of a quotient has precisely the expected factor.

36/-- A lifted submodule of a quotient has precisely the expected factor. -/
37noncomputable def comapMkQFactorEquiv (p : Submodule R M)
38 (a : Submodule R (M ⧸ p)) :
39 (a.comap p.mkQ ⧸ p.comap (a.comap p.mkQ).subtype) ≃ₗ[R] a := by
40 let f : a.comap p.mkQ →ₗ[R] a :=
41 (p.mkQ.comp (a.comap p.mkQ).subtype).codRestrict a (fun x => x.property)
42 have hf : Function.Surjective f := by
43 rintro ⟨x, hx⟩
44 obtain ⟨y, rfl⟩ := p.mkQ_surjective x
45 exact ⟨⟨y, hx⟩, rfl⟩
46 have hk : f.ker = p.comap (a.comap p.mkQ).subtype := by
47 ext x
48 change (⟨p.mkQ x.val, x.property⟩ : a) = 0 ↔ x.val ∈ p
49 rw [Subtype.ext_iff]
50 exact Submodule.Quotient.mk_eq_zero p
51 exact (Submodule.quotEquivOfEq _ _ hk.symm).trans (f.quotKerEquivOfSurjective hf)

Noetherian induction on submodules terminates the construction of critical factors in successive quotients.

53/-- Noetherian induction on submodules terminates the construction of critical
54factors in successive quotients. -/
55theorem DevLE.nonempty_criticalFiltration {n : ℕ} (h : DevLE R n M)
56 [IsNoetherian R M] : Nonempty (CriticalFiltration R n M) := by
57 classical
58 have aux : ∀ p : Submodule R M, ∃ s : RelSeries {(p, q) | CriticalStep R n M p q},
59 s.head = p ∧ s.last = ⊤ := by
60 intro p
61 induction p using (wellFounded_gt : WellFounded ((· > ·) :
62 Submodule R M → Submodule R M → Prop)).induction with
63 | h p ih =>
64 by_cases hp : p = ⊤
65 · exact ⟨RelSeries.singleton _ p, rfl, hp⟩
66 · have : Nontrivial (M ⧸ p) := Submodule.Quotient.nontrivial_iff.mpr hp
67 obtain ⟨d, hd, a, ha⟩ := (h.quotient p).exists_critical_submodule
68 let q := a.comap p.mkQ
69 have hpq : p < q := by
70 refine lt_of_le_of_ne (Submodule.le_comap_mkQ _ _) ?_
71 intro he
72 have hbot : a = ⊥ := by
73 have := congrArg (Submodule.map p.mkQ) he
74 simpa only [Submodule.map_comap_eq_of_surjective (Submodule.mkQ_surjective _),
75 Submodule.mkQ_map_self, q] using this.symm
76 exact (a.nontrivial_iff_ne_bot.mp ha.nontrivial) hbot
77 obtain ⟨s, hs, ht⟩ := ih q hpq
78 have hstep : CriticalStep R n M p s.head := by
79 rw [hs]
80 exact ⟨hpq.le, d, hd, ha.of_linearEquiv (comapMkQFactorEquiv p a).symm⟩
81 exact ⟨s.cons p hstep, by simp, by simpa using ht⟩

83 exact ⟨⟨s, hs, ht⟩⟩

87private def OrderSmall (e : ℕ) (α : Type*) [PartialOrder α] : Prop :=

136private theorem critical_upper_interval_small {e : ℕ} {a b x : Submodule R M}

A bounded monotone integer rank which detects every non-small interval.

165private def IntervalRank (e : ℕ) (a b : Submodule R M) (l : ℕ) : Prop :=
166 ∃ r : Set.Icc a b → ℕ, Monotone r ∧ (∀ x, r x ≤ l) ∧
167 ∀ x y, x ≤ y → r x = r y → OrderSmall e (Set.Icc x.val y.val)

169private theorem intervalRank_small {e : ℕ} {a b : Submodule R M}
170 (h : OrderSmall e (Set.Icc a b)) : IntervalRank e a b 0 := by
171 exact ⟨fun _ => 0, fun _ _ _ => le_rfl, fun _ => le_rfl,
172 fun x y _ _ => h.subinterval x.property.1 y.property.2⟩

174private theorem intervalRank_critical {e : ℕ} {a b : Submodule R M}
175 (hab : a ≤ b) (h : Critical R e (b ⧸ a.comap b.subtype)) :
176 IntervalRank e a b 1 := by
177 classical
178 let r : Set.Icc a b → ℕ := fun x => if x.val = a then 0 else 1
179 have hr : Monotone r := by
180 intro x y hxy
181 change x.val ≤ y.val at hxy
182 dsimp [r]
183 split_ifs with hx hy hy
184 · omega
185 · omega
186 · exact False.elim (hx (le_antisymm (hxy.trans_eq hy) x.property.1))
187 · omega
188 refine ⟨r, hr, fun x => by dsimp [r]; split_ifs <;> omega, ?_⟩
189 intro x y hxy he
190 by_cases hx : x.val = a
191 · have hy : y.val = a := by
192 by_contra hy
193 simp [r, hx, hy] at he
194 have hsmall : OrderSmall e (Set.Icc a a) := by
195 cases e with
196 | zero => exact (inferInstance : Subsingleton (Set.Icc a a))
197 | succ d => exact Order.devLE_of_subsingleton d
198 change OrderSmall e (Set.Icc x.val y.val)
199 rw [hx, hy]
200 exact hsmall
201 · exact (critical_upper_interval_small hab h
202 (lt_of_le_of_ne x.property.1 (Ne.symm hx)) x.property.2).subinterval

205private theorem intervalRank_add {e l k : ℕ} {a c b : Submodule R M}
206 (hac : a ≤ c) (hcb : c ≤ b) (h : IntervalRank e a c l)
207 (k' : IntervalRank e c b k) : IntervalRank e a b (l + k) := by
208 obtain ⟨r, hr, hrl, hre⟩ := h
209 obtain ⟨s, hs, hsk, hse⟩ := k'
210 let lo : Set.Icc a b → Set.Icc a c := fun x =>
211 ⟨x.val ⊓ c, le_inf x.property.1 hac, inf_le_right⟩
212 let hi : Set.Icc a b → Set.Icc c b := fun x =>
213 ⟨x.val ⊔ c, le_sup_right, sup_le x.property.2 hcb⟩
214 have hlo : Monotone lo := fun _ _ h => inf_le_inf_right c h
215 have hhi : Monotone hi := fun x y h =>
216 show x.val ⊔ c ≤ y.val ⊔ c from sup_le_sup_right (show x.val ≤ y.val from h) c
217 refine ⟨fun x => r (lo x) + s (hi x),
218 fun x y hxy => Nat.add_le_add (hr (hlo hxy)) (hs (hhi hxy)),
219 fun x => Nat.add_le_add (hrl _) (hsk _), ?_⟩
220 intro x y hxy he
221 change r (lo x) + s (hi x) = r (lo y) + s (hi y) at he
222 have helo : r (lo x) = r (lo y) := by
223 have := hr (hlo hxy)
224 have := hs (hhi hxy)
225 omega
226 have hehi : s (hi x) = s (hi y) := by
227 have := hr (hlo hxy)
228 have := hs (hhi hxy)
229 omega
230 have hprod := (hre _ _ (hlo hxy) helo).prod (hse _ _ (hhi hxy) hehi)
231 let f : Set.Icc x.val y.val →
232 Set.Icc (x.val ⊓ c) (y.val ⊓ c) × Set.Icc (x.val ⊔ c) (y.val ⊔ c) := fun z =>
233 (⟨z.val ⊓ c, inf_le_inf_right c z.property.1, inf_le_inf_right c z.property.2⟩,
234 ⟨z.val ⊔ c, sup_le_sup_right z.property.1 c, sup_le_sup_right z.property.2 c⟩)
235 have hf : StrictMono f := fun z w h =>
236 show f z < f w from (strictMono_inf_prod_sup (z := c) (show z.val < w.val from h))
237 cases e with
238 | zero =>
239 have hxy' : x.val = y.val := by
240 by_contra hne
241 have hlt := hf (show (⟨x.val, le_rfl, hxy⟩ : Set.Icc x.val y.val) <
242 ⟨y.val, hxy, le_rfl⟩ from lt_of_le_of_ne hxy (fun h => hne (congrArg (fun z : Set.Icc x.val y.val => z.val) h)))
243 exact (ne_of_lt hlt) (hprod.elim _ _)
