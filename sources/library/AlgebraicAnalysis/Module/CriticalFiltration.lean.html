9 /-!
10 # Finite critical filtrations
11 
12 A filtration is an actual finite ascending series of submodules from zero to
13  the whole module, with critical successive quotient modules. Dimensions need
14 not be ordered. Scalars may be noncommutative.
15 -/

25 /-- A critical step with an explicit common finite dimension ceiling. -/
26 def CriticalStep (R : Type u) (n : ℕ) (M : Type v) [Ring R]
27     [AddCommGroup M] [Module R M] (p q : Submodule R M) : Prop :=
28   p ≤ q ∧ ∃ d ≤ n, Critical R d (q ⧸ p.comap q.subtype)
29 
30 /-- A finite critical filtration, including its zero and whole-module endpoints. -/
31 structure CriticalFiltration (R : Type u) (n : ℕ) (M : Type v) [Ring R]
32     [AddCommGroup M] [Module R M] extends RelSeries {(p, q) | CriticalStep R n M p q} where
33   head_eq_bot : toRelSeries.head = ⊥
34   last_eq_top : toRelSeries.last = ⊤

36 /-- A lifted submodule of a quotient has precisely the expected factor. -/
37 noncomputable def comapMkQFactorEquiv (p : Submodule R M)
38     (a : Submodule R (M ⧸ p)) :
39     (a.comap p.mkQ ⧸ p.comap (a.comap p.mkQ).subtype) ≃ₗ[R] a := by
40   let f : a.comap p.mkQ →ₗ[R] a :=
41     (p.mkQ.comp (a.comap p.mkQ).subtype).codRestrict a (fun x => x.property)
42   have hf : Function.Surjective f := by
43     rintro ⟨x, hx⟩
44     obtain ⟨y, rfl⟩ := p.mkQ_surjective x
45     exact ⟨⟨y, hx⟩, rfl⟩
46   have hk : f.ker = p.comap (a.comap p.mkQ).subtype := by
47     ext x
48     change (⟨p.mkQ x.val, x.property⟩ : a) = 0 ↔ x.val ∈ p
49     rw [Subtype.ext_iff]
50     exact Submodule.Quotient.mk_eq_zero p
51   exact (Submodule.quotEquivOfEq _ _ hk.symm).trans (f.quotKerEquivOfSurjective hf)

53 /-- Noetherian induction on submodules terminates the construction of critical
54 factors in successive quotients. -/
55 theorem DevLE.nonempty_criticalFiltration {n : ℕ} (h : DevLE R n M)
56     [IsNoetherian R M] : Nonempty (CriticalFiltration R n M) := by
57   classical
58   have aux : ∀ p : Submodule R M, ∃ s : RelSeries {(p, q) | CriticalStep R n M p q},
59       s.head = p ∧ s.last = ⊤ := by
60     intro p
61     induction p using (wellFounded_gt : WellFounded ((· > ·) :
62         Submodule R M → Submodule R M → Prop)).induction with
63     | h p ih =>
64       by_cases hp : p = ⊤
65       · exact ⟨RelSeries.singleton _ p, rfl, hp⟩
66       · have : Nontrivial (M ⧸ p) := Submodule.Quotient.nontrivial_iff.mpr hp
67         obtain ⟨d, hd, a, ha⟩ := (h.quotient p).exists_critical_submodule
68         let q := a.comap p.mkQ
69         have hpq : p < q := by
70           refine lt_of_le_of_ne (Submodule.le_comap_mkQ _ _) ?_
71           intro he
72           have hbot : a = ⊥ := by
73             have := congrArg (Submodule.map p.mkQ) he
74             simpa only [Submodule.map_comap_eq_of_surjective (Submodule.mkQ_surjective _),
75               Submodule.mkQ_map_self, q] using this.symm
76           exact (a.nontrivial_iff_ne_bot.mp ha.nontrivial) hbot
77         obtain ⟨s, hs, ht⟩ := ih q hpq
78         have hstep : CriticalStep R n M p s.head := by
79           rw [hs]
80           exact ⟨hpq.le, d, hd, ha.of_linearEquiv (comapMkQFactorEquiv p a).symm⟩
81         exact ⟨s.cons p hstep, by simp, by simpa using ht⟩
82   obtain ⟨s, hs, ht⟩ := aux ⊥
83   exact ⟨⟨s, hs, ht⟩⟩

85 /- The following order helpers keep the chain-length argument independent
86 of choices of representatives in quotient modules. -/
87 private def OrderSmall (e : ℕ) (α : Type*) [PartialOrder α] : Prop :=
88   match e with
89   | 0 => Subsingleton α
90   | d + 1 => Order.DevLE d α

92 private theorem OrderSmall.of_strictMono {α β : Type*} [PartialOrder α]
93     [PartialOrder β] {e : ℕ} (h : OrderSmall e β) {f : α → β} (hf : StrictMono f)
94     (hi : Function.Injective f) : OrderSmall e α := by
95   cases e with
96   | zero => exact ⟨fun x y => hi (h.elim _ _)⟩
97   | succ d => exact Order.DevLE.of_strictMono h hf

99 private theorem OrderSmall.prod {α β : Type*} [PartialOrder α] [PartialOrder β]
100     {e : ℕ} (h : OrderSmall e α) (k : OrderSmall e β) : OrderSmall e (α × β) := by
101   cases e with
102   | zero => exact ⟨fun x y => Prod.ext (h.elim _ _) (k.elim _ _)⟩
103   | succ d => exact Order.DevLE.prod h k

105 private theorem orderSmall_submodule_iff (e : ℕ) :
106     OrderSmall e (Submodule R M) ↔ Small R e M := by
107   cases e with
108   | zero =>
109     constructor
110     · intro h
111       have he : (⊤ : Submodule R M) = ⊥ := h.elim _ _
112       have hz : ∀ x : M, x = 0 := fun x => by
113         have hx : x ∈ (⊥ : Submodule R M) := he ▸ Submodule.mem_top
114         exact hx
115       exact ⟨fun x y => (hz x).trans (hz y).symm⟩
116     · intro h
117       have : Subsingleton M := h
118       exact (inferInstance : Subsingleton (Submodule R M))
119   | succ d => rfl

121 private theorem orderSmall_interval_iff (e : ℕ) {a b : Submodule R M} (hab : a ≤ b) :
122     OrderSmall e (Set.Icc a b) ↔ Small R e (b ⧸ a.comap b.subtype) := by
123   let f := Submodule.quotientIntervalOrderIso a b hab
124   rw [← orderSmall_submodule_iff]
125   exact ⟨fun h => h.of_strictMono f.strictMono f.injective,
126     fun h => h.of_strictMono f.symm.strictMono f.symm.injective⟩

128 private theorem OrderSmall.subinterval {e : ℕ} {a b x y : Submodule R M}
129     (h : OrderSmall e (Set.Icc a b)) (hax : a ≤ x) (hyb : y ≤ b) :
130     OrderSmall e (Set.Icc x y) := by
131   let f : Set.Icc x y → Set.Icc a b :=
132     fun z => ⟨z.val, hax.trans z.property.1, z.property.2.trans hyb⟩
133   exact h.of_strictMono (f := f) (fun _ _ h => h)
134     (fun _ _ h => Subtype.ext (congrArg (fun t : Set.Icc a b => t.val) h))

136 private theorem critical_upper_interval_small {e : ℕ} {a b x : Submodule R M}
137     (hab : a ≤ b) (h : Critical R e (b ⧸ a.comap b.subtype))
138     (hax : a < x) (hxb : x ≤ b) : OrderSmall e (Set.Icc x b) := by
139   let f := Submodule.quotientIntervalOrderIso a b hab
140   let q := f.symm ⟨x, hax.le, hxb⟩
141   have hq : q ≠ ⊥ := by
142     intro he
143     have ha : f.symm ⟨a, le_rfl, hab⟩ = ⊥ := by
144       apply le_bot_iff.mp
145       have hh : (⟨a, le_rfl, hab⟩ : Set.Icc a b) ≤ f ⊥ := (f ⊥).property.1
146       simpa only [f.symm_apply_apply] using f.symm.monotone hh
147     have hh : f.symm (⟨a, le_rfl, hab⟩ : Set.Icc a b) < q := f.symm.strictMono hax
148     rw [ha, he] at hh
149     exact (lt_irrefl _ hh)
150   have hs := (orderSmall_submodule_iff e).mpr (h.small_quotient q hq)
151   let g : Set.Icc x b → Submodule R ((b ⧸ a.comap b.subtype) ⧸ q) := fun z =>
152     (Submodule.comapMkQRelIso q).symm
153       ⟨f.symm ⟨z.val, hax.le.trans z.property.1, z.property.2⟩,
154         f.symm.monotone z.property.1⟩
155   have hg : StrictMono g := fun z w hzw =>
156     (Submodule.comapMkQRelIso q).symm.strictMono (f.symm.strictMono hzw)
157   have hi : Function.Injective g := by
158     intro z w hzw
159     have hh := (Submodule.comapMkQRelIso q).symm.injective hzw
160     have hh' := congrArg Subtype.val hh
161     exact Subtype.ext (congrArg (fun t : Set.Icc a b => t.val) (f.symm.injective hh'))
162   exact hs.of_strictMono hg hi

164 /-- A bounded monotone integer rank which detects every non-small interval. -/
165 private def IntervalRank (e : ℕ) (a b : Submodule R M) (l : ℕ) : Prop :=
166   ∃ r : Set.Icc a b → ℕ, Monotone r ∧ (∀ x, r x ≤ l) ∧
167     ∀ x y, x ≤ y → r x = r y → OrderSmall e (Set.Icc x.val y.val)
168 
169 private theorem intervalRank_small {e : ℕ} {a b : Submodule R M}
170     (h : OrderSmall e (Set.Icc a b)) : IntervalRank e a b 0 := by
171   exact ⟨fun _ => 0, fun _ _ _ => le_rfl, fun _ => le_rfl,
172     fun x y _ _ => h.subinterval x.property.1 y.property.2⟩

174 private theorem intervalRank_critical {e : ℕ} {a b : Submodule R M}
175     (hab : a ≤ b) (h : Critical R e (b ⧸ a.comap b.subtype)) :
176     IntervalRank e a b 1 := by
177   classical
178   let r : Set.Icc a b → ℕ := fun x => if x.val = a then 0 else 1
179   have hr : Monotone r := by
180     intro x y hxy
181     change x.val ≤ y.val at hxy
182     dsimp [r]
183     split_ifs with hx hy hy
184     · omega
185     · omega
186     · exact False.elim (hx (le_antisymm (hxy.trans_eq hy) x.property.1))
187     · omega
188   refine ⟨r, hr, fun x => by dsimp [r]; split_ifs <;> omega, ?_⟩
189   intro x y hxy he
190   by_cases hx : x.val = a
191   · have hy : y.val = a := by
192       by_contra hy
193       simp [r, hx, hy] at he
194     have hsmall : OrderSmall e (Set.Icc a a) := by
195       cases e with
196       | zero => exact (inferInstance : Subsingleton (Set.Icc a a))
197       | succ d => exact Order.devLE_of_subsingleton d
198     change OrderSmall e (Set.Icc x.val y.val)
199     rw [hx, hy]
200     exact hsmall
201   · exact (critical_upper_interval_small hab h
202       (lt_of_le_of_ne x.property.1 (Ne.symm hx)) x.property.2).subinterval
203         le_rfl y.property.2

205 private theorem intervalRank_add {e l k : ℕ} {a c b : Submodule R M}
206     (hac : a ≤ c) (hcb : c ≤ b) (h : IntervalRank e a c l)
207     (k' : IntervalRank e c b k) : IntervalRank e a b (l + k) := by
208   obtain ⟨r, hr, hrl, hre⟩ := h
209   obtain ⟨s, hs, hsk, hse⟩ := k'
210   let lo : Set.Icc a b → Set.Icc a c := fun x =>
211     ⟨x.val ⊓ c, le_inf x.property.1 hac, inf_le_right⟩
212   let hi : Set.Icc a b → Set.Icc c b := fun x =>
213     ⟨x.val ⊔ c, le_sup_right, sup_le x.property.2 hcb⟩
214   have hlo : Monotone lo := fun _ _ h => inf_le_inf_right c h
215   have hhi : Monotone hi := fun x y h =>
216     show x.val ⊔ c ≤ y.val ⊔ c from sup_le_sup_right (show x.val ≤ y.val from h) c
217   refine ⟨fun x => r (lo x) + s (hi x),
218     fun x y hxy => Nat.add_le_add (hr (hlo hxy)) (hs (hhi hxy)),
219     fun x => Nat.add_le_add (hrl _) (hsk _), ?_⟩
220   intro x y hxy he
221   change r (lo x) + s (hi x) = r (lo y) + s (hi y) at he
222   have helo : r (lo x) = r (lo y) := by
223     have := hr (hlo hxy)
224     have := hs (hhi hxy)
225     omega
226   have hehi : s (hi x) = s (hi y) := by
227     have := hr (hlo hxy)
228     have := hs (hhi hxy)
229     omega
230   have hprod := (hre _ _ (hlo hxy) helo).prod (hse _ _ (hhi hxy) hehi)
231   let f : Set.Icc x.val y.val →
232       Set.Icc (x.val ⊓ c) (y.val ⊓ c) × Set.Icc (x.val ⊔ c) (y.val ⊔ c) := fun z =>
233     (⟨z.val ⊓ c, inf_le_inf_right c z.property.1, inf_le_inf_right c z.property.2⟩,
234       ⟨z.val ⊔ c, sup_le_sup_right z.property.1 c, sup_le_sup_right z.property.2 c⟩)
235   have hf : StrictMono f := fun z w h =>
236     show f z < f w from (strictMono_inf_prod_sup (z := c) (show z.val < w.val from h))
237   cases e with
238   | zero =>
239     have hxy' : x.val = y.val := by
240       by_contra hne
241       have hlt := hf (show (⟨x.val, le_rfl, hxy⟩ : Set.Icc x.val y.val) <
242           ⟨y.val, hxy, le_rfl⟩ from lt_of_le_of_ne hxy (fun h => hne (congrArg (fun z : Set.Icc x.val y.val => z.val) h)))
243       exact (ne_of_lt hlt) (hprod.elim _ _)
244     rw [hxy']
245     exact (inferInstance : Subsingleton (Set.Icc y.val y.val))
246   | succ d => exact Order.DevLE.of_strictMono hprod hf

249 private noncomputable def stepWeight (e : ℕ) (a b : Submodule R M) : ℕ := by
250   classical
251   exact if Small R e (b ⧸ a.comap b.subtype) then 0 else 1

253 private noncomputable def seriesWeight (e : ℕ)
254     (s : RelSeries {(a, b) | CriticalStep R e M a b}) : ℕ :=
255   ∑ i : Fin s.length, stepWeight e (s i.castSucc) (s i.succ)

257 private theorem intervalRank_step {e : ℕ} {a b : Submodule R M}
258     (h : CriticalStep R e M a b) : IntervalRank e a b (stepWeight e a b) := by
259   classical
260   by_cases hs : Small R e (b ⧸ a.comap b.subtype)
261   · simpa only [stepWeight, ite_eq_left hs] using
262       intervalRank_small ((orderSmall_interval_iff e h.1).mpr hs)
263   · obtain ⟨d, hd, hc⟩ := h.2
264     have he : e ≤ d := by
265       by_contra he
266       have hsmall : Small R (d + 1) (b ⧸ a.comap b.subtype) := hc.devLE
267       exact hs (hsmall.mono (Nat.succ_le_of_lt (Nat.lt_of_not_ge he)))
268     have hde : d = e := le_antisymm hd he
269     subst d
270     simpa only [stepWeight, ite_eq_right hs] using intervalRank_critical h.1 hc

291 private theorem intervalRank_series {e : ℕ}
292     (s : RelSeries {(a, b) | CriticalStep R e M a b}) :
293     IntervalRank e s.head s.last (seriesWeight e s) := by
294   induction s using RelSeries.inductionOn' with
295   | singleton a =>
296     have hs : OrderSmall e (Set.Icc a a) := by
297       cases e with

311 /-- The quotient module at one step. -/
312 abbrev Factor (i : Fin F.length) :=
313   F.toRelSeries i.succ ⧸
314     (F.toRelSeries i.castSucc).comap (F.toRelSeries i.succ).subtype
315 
316 /-- Every factor has a critical dimension bounded by the ceiling. -/
317 theorem factor_critical (i : Fin F.length) : ∃ d ≤ n, Critical R d (F.Factor i) :=
318   (F.step i).2
319 
320 /-- The series is ascending. -/
321 theorem monotone : Monotone F.toRelSeries :=
322   (Fin.monotone_iff_le_succ).mpr (fun i => (F.step i).1)
323 
324 /-- Critical factors are nonzero, hence the series is strictly ascending. -/
325 theorem strictMono : StrictMono F.toRelSeries := by
326   apply Fin.strictMono_iff_lt_succ.mpr
327   intro i
328   refine lt_of_le_of_ne (F.step i).1 ?_
329   intro he
330   obtain ⟨d, _, hd⟩ := F.factor_critical i
331   have hz : (F.toRelSeries i.castSucc).comap (F.toRelSeries i.succ).subtype = ⊤ := by
332     rw [he]
333     exact Submodule.comap_subtype_self _
334   exact hd.not_small (by
335     change Small R d (F.toRelSeries i.succ ⧸ _)
336     rw [hz]
337     exact small_of_subsingleton d)

339 /-- The number of factors surviving the strict stratum `e`. -/
340 noncomputable def topLength (e : ℕ) : ℕ := by
341   classical
342   exact ∑ i : Fin F.length, if Small R e (F.Factor i) then 0 else 1

344 /-- Lower-dimensional factors vanish; critical factors of dimension at least
345 `e` survive. This also fixes the stratum indexing at zero. -/
346 theorem factor_not_small_iff (i : Fin F.length) {d e : ℕ}
347     (hd : Critical R d (F.Factor i)) : ¬Small R e (F.Factor i) ↔ e ≤ d := by
348   constructor
349   · intro hs
350     by_contra he
351     have hsmall : Small R (d + 1) (F.Factor i) := hd.devLE
352     exact hs (hsmall.mono (Nat.succ_le_of_lt (Nat.lt_of_not_ge he)))
353   · intro he hs
354     exact hd.not_small (hs.mono he)

356 /-- There are at most as many surviving factors as total factors. -/
357 theorem topLength_le_length (e : ℕ) : F.topLength e ≤ F.length := by
358   classical
359   unfold topLength
360   calc
361     _ ≤ ∑ _ : Fin F.length, 1 := Finset.sum_le_sum (fun i _ => by split_ifs <;> omega)
362     _ = F.length := by simp

364 /-- An integer rank on the submodule lattice detects every non-small interval.
365 The bound counts precisely the critical factors at the common top dimension. -/
366 theorem exists_bounded_rank {e : ℕ} (F : CriticalFiltration R e M) :
367     ∃ r : Submodule R M → ℕ, Monotone r ∧ (∀ p, r p ≤ F.topLength e) ∧
368       ∀ p q, p ≤ q → r p = r q → Small R e (q ⧸ p.comap q.subtype) := by
369   have hr := intervalRank_series F.toRelSeries
370   have hw : seriesWeight e F.toRelSeries = F.topLength e := rfl
371   rw [hw, F.head_eq_bot, F.last_eq_top] at hr
372   obtain ⟨r, hm, hb, hs⟩ := hr
373   let j : Submodule R M → Set.Icc (⊥ : Submodule R M) ⊤ := fun p => ⟨p, bot_le, le_top⟩
374   refine ⟨r ∘ j, fun p q h => hm h, fun p => hb _, ?_⟩
375   intro p q hpq he
376   exact (orderSmall_interval_iff e hpq).mp (hs (j p) (j q) hpq he)

378 /-- Every finite portion of any ascending chain has at most `topLength`
379 non-small successive factors. Thus the bound is independent of the chain. -/
380 theorem non_small_chain_count_le {e : ℕ} (F : CriticalFiltration R e M)
381     (c : ℕ → Submodule R M) (hc : Monotone c) (k : ℕ) :
382     ((Finset.range k).filter (fun i =>
383       ¬Small R e (c (i + 1) ⧸ (c i).comap (c (i + 1)).subtype))).card ≤ F.topLength e := by
384   classical
385   obtain ⟨r, hm, hb, hs⟩ := F.exists_bounded_rank
386   have aux : ∀ k, ((Finset.range k).filter (fun i =>
387       ¬Small R e (c (i + 1) ⧸ (c i).comap (c (i + 1)).subtype))).card ≤ r (c k) := by
388     intro k
389     induction k with
390     | zero => simp
391     | succ k ih =>
392       have hle := hm (hc (Nat.le_succ k))
393       by_cases hsmall : Small R e (c (k + 1) ⧸ (c k).comap (c (k + 1)).subtype)
394       · simpa [Finset.range_add_one, Finset.filter_insert, hsmall, Finset.mem_filter] using ih.trans hle
395       · have hlt : r (c k) < r (c (k + 1)) :=
396           lt_of_le_of_ne hle (fun he => hsmall (hs _ _ (hc (Nat.le_succ k)) he))
397         have hbound : ((Finset.range k).filter (fun i =>
398             ¬Small R e (c (i + 1) ⧸ (c i).comap (c (i + 1)).subtype))).card + 1 ≤
399             r (c (k + 1)) := by omega
400         simpa [Finset.range_add_one, Finset.filter_insert, hsmall, Finset.mem_filter] using hbound
401   exact (aux k).trans (hb _)

403 /-- In particular, a chain with every factor non-small has bounded length. -/
404 theorem non_small_chain_length_le {e : ℕ} (F : CriticalFiltration R e M)
405     (c : ℕ → Submodule R M) (hc : Monotone c) (k : ℕ)
406     (h : ∀ i < k, ¬Small R e (c (i + 1) ⧸ (c i).comap (c (i + 1)).subtype)) :
407     k ≤ F.topLength e := by
408   classical
409   have he : (Finset.range k).filter (fun i =>
410       ¬Small R e (c (i + 1) ⧸ (c i).comap (c (i + 1)).subtype)) = Finset.range k :=
411     Finset.filter_eq_self.mpr (fun i hi => h i (Finset.mem_range.mp hi))
412   simpa only [he, Finset.card_range] using F.non_small_chain_count_le c hc k
