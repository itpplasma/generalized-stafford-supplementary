13 # Finite deviation of modules
14 
15 The scalar ring is arbitrary; right modules are obtained by using `Rᵐᵒᵖ`.
16 `Module.Small R 0 M` means that `M` is zero, whereas `Small R (e+1) M`
17 means deviation at most `e`. No dimension is assigned to modules without a
18 finite bound. The exact-sequence maximum is expressed as an equivalence of
19 bounds, so no existence of a dimension is implicit.
20 -/

28 /-- The submodule lattice has finite deviation at most `n`. -/
29 def DevLE (R : Type u) (n : ℕ) (M : Type v) [Ring R] [AddCommGroup M]
30     [Module R M] : Prop := Order.DevLE n (Submodule R M)
31 
32 /-- The finite strict stratum: zero is the zero module; stratum `e+1`
33 consists of modules with deviation at most `e`. -/
34 def Small (R : Type u) (e : ℕ) (M : Type v) [Ring R] [AddCommGroup M]
35     [Module R M] : Prop :=
36   match e with
37   | 0 => Subsingleton M
38   | e + 1 => DevLE R e M

47 /-- Deviation zero agrees with Mathlib's Artinian condition. -/
48 theorem devLE_zero_iff_isArtinian : DevLE R 0 M ↔ IsArtinian R M := by
49   rw [← monotone_stabilizes_iff_artinian]
50   constructor
51   · intro h c
52     obtain ⟨n, hn⟩ := h c c.monotone
53     exact ⟨n, fun m hm => (hn m hm).symm⟩
54   · intro h c hc
55     obtain ⟨n, hn⟩ := h ⟨c, hc⟩
56     exact ⟨n, fun m hm => (hn m hm).symm⟩

131 /-- The finite-deviation maximum formula for a submodule and its quotient. -/
132 theorem devLE_iff_submodule_quotient (n : ℕ) (p : Submodule R M) :
133     DevLE R n M ↔ DevLE R n p ∧ DevLE R n (M ⧸ p) := by
134   refine ⟨fun h => ⟨h.submodule p, h.quotient p⟩, fun h => ?_⟩
135   exact (Order.DevLE.prod h.1 h.2).of_strictMono (Submodule.strictMono_comap_prod_map p)

137 /-- Zero is kept distinct from Artinian modules in the exact maximum formula. -/
138 theorem small_iff_submodule_quotient (e : ℕ) (p : Submodule R M) :
139     Small R e M ↔ Small R e p ∧ Small R e (M ⧸ p) := by
140   cases e with
141   | zero =>
142     refine ⟨fun h => ⟨h.submodule p, h.quotient p⟩, fun h => ?_⟩
143     have hp : Subsingleton p := h.1
144     have hq : Subsingleton (M ⧸ p) := h.2
145     apply subsingleton_of_forall_eq 0
146     intro x
147     have hx : x ∈ p := (Submodule.Quotient.mk_eq_zero p).mp (Subsingleton.elim _ _)
148     exact congrArg Subtype.val (show (⟨x, hx⟩ : p) = 0 from Subsingleton.elim _ _)
149   | succ n => exact devLE_iff_submodule_quotient n p

160 /-- The maximum formula for an arbitrary short exact sequence of modules. -/
161 theorem devLE_iff_of_exact {P : Type*} [AddCommGroup P] [Module R P]
162     (n : ℕ) (f : N →ₗ[R] M) (g : M →ₗ[R] P) (hf : Function.Injective f)
163     (hg : Function.Surjective g) (hfg : Function.Exact f g) :
164     DevLE R n M ↔ DevLE R n N ∧ DevLE R n P := by
165   have hek : LinearMap.range f = LinearMap.ker g := (LinearMap.exact_iff.mp hfg).symm
166   let e : N ≃ₗ[R] LinearMap.ker g :=
167     (LinearEquiv.ofInjective f hf).trans (LinearEquiv.ofEq _ _ hek)
168   rw [devLE_iff_submodule_quotient n (LinearMap.ker g),
169     ← devLE_linearEquiv_iff n e,
170     devLE_linearEquiv_iff n (g.quotKerEquivOfSurjective hg)]
171 
172 theorem small_iff_of_exact {P : Type*} [AddCommGroup P] [Module R P]
173     (e : ℕ) (f : N →ₗ[R] M) (g : M →ₗ[R] P) (hf : Function.Injective f)
174     (hg : Function.Surjective g) (hfg : Function.Exact f g) :
175     Small R e M ↔ Small R e N ∧ Small R e P := by
176   have hek : LinearMap.range f = LinearMap.ker g := (LinearMap.exact_iff.mp hfg).symm
177   let k : N ≃ₗ[R] LinearMap.ker g :=
178     (LinearEquiv.ofInjective f hf).trans (LinearEquiv.ofEq _ _ hek)
179   rw [small_iff_submodule_quotient e (LinearMap.ker g),
180     ← small_linearEquiv_iff e k,
181     small_linearEquiv_iff e (g.quotKerEquivOfSurjective hg)]

183 /-- The intersection/projection map for the split exact sequence of a product. -/
184 private theorem strictMono_prod_submodules : StrictMono
185     (fun p : Submodule R (M × N) =>
186       (p.comap (LinearMap.inl R M N), p.map (LinearMap.snd R M N))) := by
187   intro p q hpq
188   refine lt_of_le_of_ne ⟨Submodule.comap_mono hpq.le, Submodule.map_mono hpq.le⟩ ?_
189   intro he
190   have he₁ : p.comap (LinearMap.inl R M N) = q.comap (LinearMap.inl R M N) :=
191     congrArg Prod.fst he
192   have he₂ : p.map (LinearMap.snd R M N) = q.map (LinearMap.snd R M N) :=
193     congrArg Prod.snd he
194   apply hpq.ne
195   apply le_antisymm hpq.le
196   intro x hx
197   obtain ⟨y, hy, hyx⟩ := (show x.2 ∈ p.map (LinearMap.snd R M N) from
198     he₂.symm ▸ Submodule.mem_map_of_mem hx)
199   change y.2 = x.2 at hyx
200   have hd : (x.1 - y.1, (0 : N)) ∈ q := by
201     have := q.sub_mem hx (hpq.le hy)
202     simpa [Prod.sub_def, hyx] using this
203   have hd' : (x.1 - y.1, (0 : N)) ∈ p := by
204     have : x.1 - y.1 ∈ p.comap (LinearMap.inl R M N) :=
205       he₁.symm ▸ hd
206     exact this
207   have := p.add_mem hd' hy
208   simpa [Prod.add_def, hyx] using this

220 /-- Finite products (equivalently finite direct sums) preserve a bound. -/
221 theorem DevLE.pi {ι : Type w} [Finite ι] {P : ι → Type v}
222     [∀ i, AddCommGroup (P i)] [∀ i, Module R (P i)] {n : ℕ}
223     (h : ∀ i, DevLE R n (P i)) : DevLE R n (∀ i, P i) := by
224   apply Module.pi_induction' R
225     (fun X _ _ => DevLE R n X) (fun X _ _ => DevLE R n X)
226     (fun f h => (devLE_linearEquiv_iff n f).mp h)
227     (fun f h => (devLE_linearEquiv_iff n f).mp h)
228     (devLE_of_subsingleton n) (fun h h' => h.prod h') P h
229 
230 theorem Small.pi {ι : Type w} [Finite ι] {P : ι → Type v}
231     [∀ i, AddCommGroup (P i)] [∀ i, Module R (P i)] {e : ℕ}

233   cases e with
234   | zero => exact ⟨fun f g => funext fun i => (h i).elim _ _⟩

237 /-- Finite direct sums preserve a deviation bound. No infinite-sum closure is asserted. -/
238 theorem DevLE.directSum {ι : Type w} [Finite ι] {P : ι → Type v}
239     [∀ i, AddCommGroup (P i)] [∀ i, Module R (P i)] {n : ℕ}
240     (h : ∀ i, DevLE R n (P i)) : DevLE R n (⨁ i, P i) := by
241   classical
242   let := Fintype.ofFinite ι
243   exact (devLE_linearEquiv_iff n (DirectSum.linearEquivFunOnFintype R ι P)).mpr
244     (DevLE.pi h)

265 /-- The interval `[B,A]` is the lattice of submodules of `A/B`, where the
266 denominator is pulled back along the inclusion of `A` in the ambient module. -/
267 def quotientIntervalOrderIso (B A : Submodule R M) (hBA : B ≤ A) :
268     Submodule R (A ⧸ B.comap A.subtype) ≃o Set.Icc B A where
269   toFun C := ⟨(C.comap (B.comap A.subtype).mkQ).map A.subtype,
270     by
271       have h := Submodule.map_mono (le_comap_mkQ (B.comap A.subtype) C)
272         (f := A.subtype)
273       simpa only [map_comap_subtype, inf_eq_right.mpr hBA] using h,
274     map_subtype_le A _⟩
275   invFun U := (U.val.comap A.subtype).map (B.comap A.subtype).mkQ
276   left_inv C := by
277     change (((C.comap (B.comap A.subtype).mkQ).map A.subtype).comap A.subtype).map
278       (B.comap A.subtype).mkQ = C
279     rw [comap_map_eq_of_injective (f := A.subtype) Subtype.val_injective]
280     exact map_comap_eq_of_surjective (Submodule.mkQ_surjective _) C
281   right_inv U := by
282     apply Subtype.ext
283     dsimp
284     rw [comap_map_mkQ, sup_eq_right.mpr (comap_mono U.property.1),
285       map_comap_subtype, inf_eq_right.mpr U.property.2]
286   map_rel_iff' := by
287     intro C D
288     change (C.comap _).map A.subtype ≤ (D.comap _).map A.subtype ↔ C ≤ D
289     rw [map_le_map_iff_of_injective Subtype.val_injective,
290       comap_le_comap_iff_of_surjective (Submodule.mkQ_surjective _)]

303 /-- The successor bound in terms of successive quotient modules. -/
304 theorem devLE_succ_iff (n : ℕ) : DevLE R (n + 1) M ↔
305     ∀ c : ℕ → Submodule R M, Antitone c → ∃ N, ∀ i, N ≤ i →
306       DevLE R n (c i ⧸ (c (i + 1)).comap (c i).subtype) := by
307   unfold DevLE
308   rw [Order.devLE_succ_iff]
309   constructor <;> intro h c hc <;> obtain ⟨N, hN⟩ := h c hc
310   · exact ⟨N, fun i hi => (Submodule.devLE_interval_iff n _ _ (hc (Nat.le_succ i))).mp
311       (hN i hi)⟩
312   · exact ⟨N, fun i hi => (Submodule.devLE_interval_iff n _ _ (hc (Nat.le_succ i))).mpr
313       (hN i hi)⟩
