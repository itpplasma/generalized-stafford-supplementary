23 /-- The right annihilator of a vector, as a right ideal. -/
24 def rightAnnihilator (w : W) : Submodule Rᵐᵒᵖ R where
25   carrier := {r | op r • w = 0}
26   zero_mem' := zero_smul _ _
27   add_mem' := by
28     intro r s hr hs
29     change op r • w = 0 at hr
30     change op s • w = 0 at hs
31     change op (r + s) • w = 0
32     rw [op_add, add_smul, hr, hs, add_zero]
33   smul_mem' := by
34     intro r s hs
35     change op (s * unop r) • w = 0
36     rw [op_mul, op_unop, mul_smul, hs, smul_zero]

38 /-- A finite-dimensional subspace of an ordinary torsion bimodule has a
39 common nonzero right annihilator. The scalar actions need only commute. -/
40 theorem exists_common_annihilator (hOre : RightOreCondition R)
41     (hW : IsTorsionRight (R := R) (M := W))
42     (Y : Submodule E W) [Module.Finite E Y] :
43     ∃ c : R, c ≠ 0 ∧ ∀ y ∈ Y, op c • y = 0 := by
44   classical
45   let B := Module.finBasis E Y
46   obtain ⟨c, hc, hcB⟩ := exists_mem_finset_rightIdeals Finset.univ
47     (fun i => rightAnnihilator (R := R) (B i : W))
48     (fun i _ => by
49       obtain ⟨r, hr, hri⟩ := hW (B i : W)
50       exact ⟨r, hri, hr⟩) hOre
51   refine ⟨c, hc, ?_⟩
52   intro y hy
53   have heq := congrArg (fun z : Y => (z : W)) (B.sum_repr ⟨y, hy⟩)
54   simp only [Submodule.coe_sum, Submodule.coe_smul] at heq
55   rw [← heq, Finset.smul_sum]
56   apply Finset.sum_eq_zero
57   intro i _
58   rw [smul_comm, show op c • (B i : W) = 0 from hcB i (Finset.mem_univ i), smul_zero]

60 /-- The two-sided annihilator of the cyclic right module generated by `g`.
61 Its definition quantifies over `gR`, avoiding a choice of generators. -/
62 def cyclicAnnihilator (g : W) : TwoSidedIdeal R :=
63   TwoSidedIdeal.mk' {c | ∀ r : R, op (r * c) • g = 0}
64     (by simp)
65     (by
66       intro x y hx hy r
67       rw [mul_add, op_add, add_smul, hx r, hy r, add_zero])
68     (by intro x hx r; rw [mul_neg, op_neg, neg_smul, hx r, neg_zero])
69     (by intro x y hy r; simpa only [← mul_assoc] using hy (r * x))
70     (by
71       intro x y hx r
72       rw [← mul_assoc, op_mul, mul_smul, hx r, smul_zero])

74 /-- In a simple right Ore domain, a nonzero vector in a torsion bimodule
75 escapes every finite-dimensional left subspace even with a prescribed nonzero
76 right tail: `g R b` cannot be contained in that subspace. -/
77 theorem exists_smul_mul_not_mem [IsSimpleRing R]
78     (hOre : RightOreCondition R) (hW : IsTorsionRight (R := R) (M := W))
79     {g : W} (hg : g ≠ 0) {b : R} (hb : b ≠ 0)
80     (Y : Submodule E W) [Module.Finite E Y] :
81     ∃ r : R, op (r * b) • g ∉ Y := by
82   classical
83   by_contra! h
84   obtain ⟨c, hc, hcY⟩ := exists_common_annihilator hOre hW Y
85   have hbc : b * c ∈ cyclicAnnihilator g := by
86     simp only [cyclicAnnihilator, TwoSidedIdeal.mem_mk', Set.mem_ofPred_eq]
87     intro r
88     rw [← mul_assoc, op_mul, mul_smul]
89     exact hcY _ (h r)
90   have h1 := IsSimpleRing.one_mem_of_ne_zero_mem (cyclicAnnihilator g)
91     (mul_ne_zero hb hc) hbc
92   have hzero := h1 1
93   exact hg (by simpa using hzero)

105 /-- A pencil with central coefficients, as an `E`-linear map. -/
106 def centralPencil (κ : k →+* E) (hκ : ∀ c a, κ c * a = a * κ c)
107     (P Q : D →ₗ[E] W) (c : k) : D →ₗ[E] W where
108   toFun d := P d + κ c • Q d
109   map_add' x y := by simp only [map_add, smul_add]; abel
110   map_smul' a x := by
111     simp only [map_smul, RingHom.id_apply, smul_add, smul_smul, hκ]

113 /-- The injectivity formulation of the central pencil theorem. -/
114 theorem centralPencil_injective_finite [Module.Finite E D]
115     (κ : k →+* E) (hκ : ∀ c a, κ c * a = a * κ c)
116     (P Q : D →ₗ[E] W) (hP : Function.Injective P) :
117     {c | ¬ Function.Injective (centralPencil κ hκ P Q c)}.Finite := by
118   refine (AlgebraicAnalysis.pencil_ker_finite κ hκ P Q hP).subset ?_
119   intro c hc
120   by_contra hn
121   apply hc
122   apply LinearMap.ker_eq_bot.mp
123   apply LinearMap.ker_eq_bot'.mpr
124   intro d hd
125   by_contra hd0
126   exact hn ⟨d, hd0, hd⟩

128 /-- A pencil is also generically injective when its second coefficient is
129 injective. Zero may now be exceptional. -/
130 theorem reversed_centralPencil_injective_finite [Module.Finite E D]
131     (κ : k →+* E) (hκ : ∀ c a, κ c * a = a * κ c)
132     (P Q : D →ₗ[E] W) (hQ : Function.Injective Q) :
133     {c | ¬ Function.Injective (centralPencil κ hκ P Q c)}.Finite := by
134   have hf := Set.Finite.preimage (fun _ _ _ _ h => inv_injective h)
135     (centralPencil_injective_finite κ hκ Q P hQ)
136   refine (hf.union (Set.finite_singleton 0)).subset ?_
137   intro c hc
138   by_cases hc0 : c = 0
139   · exact Or.inr hc0
140   left
141   intro hreverse
142   apply hc
143   intro x y hxy
144   apply hreverse
145   change P x + κ c • Q x = P y + κ c • Q y at hxy
146   change Q x + κ (c⁻¹) • P x = Q y + κ (c⁻¹) • P y
147   have hcE : κ c ≠ 0 := by simpa using κ.injective.ne hc0
148   have h := congrArg (fun w : W => (κ c)⁻¹ • w) hxy
149   simpa only [smul_add, inv_smul_smul₀ hcE, map_inv₀, add_comm] using h

159 /-- The maximal-rank argument for a linear family. Its escape hypothesis
160 requires a value outside every finite-dimensional subspace for each nonzero
161 domain vector. This separates the linear argument from torsion and simplicity. -/
162 theorem exists_injective_of_escape
163     (σ : k →+* R) (κ : k →+* E) (hκ : ∀ c a, κ c * a = a * κ c)
164     (F : R → D →ₗ[E] W)
165     (hF : ∀ r s c d, F (r + σ c * s) d = F r d + κ c • F s d)
166     (hescape : ∀ d : D, d ≠ 0 → ∀ Y : Submodule E W,
167       Module.Finite E Y → ∃ r : R, F r d ∉ Y) :
168     ∃ r : R, Function.Injective (F r) := by
169   classical
170   have hbounded : BddAbove (Set.range fun r => Module.finrank E (LinearMap.range (F r))) :=
171     ⟨Module.finrank E D, by rintro _ ⟨r, rfl⟩; exact (F r).finrank_range_le⟩
172   obtain ⟨n, ⟨r₀, rfl⟩, hmax⟩ :=
173     hbounded.exists_isGreatest_of_nonempty ⟨_, 0, rfl⟩
174   by_contra! hnone
175   have hker : LinearMap.ker (F r₀) ≠ ⊥ := by
176     intro h
177     exact hnone r₀ (LinearMap.ker_eq_bot.mp h)
178   obtain ⟨d, hdker, hd⟩ := Submodule.exists_mem_ne_zero_of_ne_bot hker
179   have hd0 : F r₀ d = 0 := hdker
180   let Y := LinearMap.range (F r₀)
181   obtain ⟨r, hr⟩ := hescape d hd Y inferInstance
182   let w := F r d
183   obtain ⟨j, hj⟩ := (F r₀).rangeRestrict.exists_rightInverse_of_surjective
184     (F r₀).range_rangeRestrict
185   have hj' (y : Y) : F r₀ (j y) = (y : W) :=
186     congrArg Subtype.val (LinearMap.congr_fun hj y)
187   let A : Y × E →ₗ[E] W := Y.subtype.comp (LinearMap.fst E Y E) +
188     (LinearMap.toSpanSingleton E W w).comp (LinearMap.snd E Y E)
189   let Q : Y × E →ₗ[E] W := (F r).comp (j.comp (LinearMap.fst E Y E))
190   have hA : Function.Injective A := by
191     apply LinearMap.ker_eq_bot.mp
192     apply LinearMap.ker_eq_bot'.mpr
193     rintro ⟨y, a⟩ ha
194     change (y : W) + a • w = 0 at ha
195     have ha0 : a = 0 := by
196       by_contra ha0
197       apply hr
198       have hm : a • w ∈ Y := (eq_neg_of_add_eq_zero_right ha) ▸ Y.neg_mem y.property
199       exact (Y.smul_mem_iff ha0).mp hm
200     subst a
201     simp only [zero_smul, add_zero] at ha
202     exact Prod.ext (Subtype.ext ha) rfl
203   obtain ⟨c, hc⟩ := ((centralPencil_injective_finite κ hκ A Q hA).union
204     (Set.finite_singleton 0)).exists_notMem
205   have hc0 : c ≠ 0 := fun h => hc (Or.inr h)
206   have hcE : κ c ≠ 0 := by simpa using κ.injective.ne hc0
207   have hAc : Function.Injective (centralPencil κ hκ A Q c) := by
208     by_contra hn
209     exact hc (Or.inl hn)
210   let J : Y × E →ₗ[E] D := j.comp (LinearMap.fst E Y E) +
211     (LinearMap.toSpanSingleton E D ((κ c)⁻¹ • d)).comp (LinearMap.snd E Y E)
212   have hcomp : (F (r₀ + σ c * r)).comp J = centralPencil κ hκ A Q c := by
213     apply LinearMap.ext
214     rintro ⟨y, a⟩
215     change F (r₀ + σ c * r) (j y + a • ((κ c)⁻¹ • d)) =
216       (y : W) + a • w + κ c • F r (j y)
217     rw [map_add, map_smul, map_smul, hF, hj', hF, hd0, zero_add]
218     rw [smul_smul, smul_smul, mul_assoc, inv_mul_cancel₀ hcE, mul_one]
219     abel
220   have hJ : Function.Injective ((F (r₀ + σ c * r)).comp J) := hcomp ▸ hAc
221   have hinj : Function.Injective ((F (r₀ + σ c * r)).rangeRestrict.comp J) := by
222     intro x y hxy
223     exact hJ (congrArg Subtype.val hxy)
224   have hdim := LinearMap.finrank_le_finrank_of_injective hinj
225   have hbound := hmax (Set.mem_range_self (r₀ + σ c * r))
226   change Module.finrank E (LinearMap.range (F (r₀ + σ c * r))) ≤ Module.finrank E Y at hbound
227   rw [Module.finrank_prod, Module.finrank_self] at hdim
228   omega

230 /-- A finite-dimensional injective map into a torsion bimodule can be
231 followed by right multiplication with a prescribed nonzero tail and remain
232 injective. Scalars in `k` act compatibly on the two sides. -/
233 theorem exists_injective_right_mul [IsSimpleRing R]
234     (hOre : RightOreCondition R) (hW : IsTorsionRight (R := R) (M := W))
235     (σ : k →+* R) (κ : k →+* E) (hκ : ∀ c a, κ c * a = a * κ c)
236     (hscalar : ∀ c (w : W), op (σ c) • w = κ c • w)
237     (P : D →ₗ[E] W) (hP : Function.Injective P) {b : R} (hb : b ≠ 0) :
238     ∃ r : R, Function.Injective ((rightMulLinear (E := E) (r * b)).comp P) := by
239   apply exists_injective_of_escape σ κ hκ
240     (fun r => (rightMulLinear (E := E) (r * b)).comp P)
241   · intro r s c d
242     change op ((r + σ c * s) * b) • P d =
243       op (r * b) • P d + κ c • (op (s * b) • P d)
244     rw [add_mul, op_add, add_smul, mul_assoc]
245     congr 1
246     rw [op_mul, mul_smul, hscalar, smul_comm]
247   · intro d hd Y hY
248     let := hY
249     exact exists_smul_mul_not_mem hOre hW
250       (fun h => hd (hP (h.trans (map_zero P).symm))) hb Y
251 
252 end Rank

264 /-- Finitely many linear families, possibly over different division rings,
265 have a common injective value if each has an injective value and their
266 restrictions to scalar lines are central pencils. -/
267 theorem exists_simultaneously_injective
268     (σ : k →+* R) (κ : ∀ i, k →+* Eᵢ i)
269     (hκ : ∀ i c a, κ i c * a = a * κ i c)
270     (F : ∀ i, R → Dᵢ i →ₗ[Eᵢ i] Wᵢ i)
271     (hF : ∀ i r t c d, F i (r + σ c * t) d = F i r d + κ i c • F i t d)
272     (s : Finset ι) (hgood : ∀ i ∈ s, ∃ r, Function.Injective (F i r)) :
273     ∃ r, ∀ i ∈ s, Function.Injective (F i r) := by
274   classical
275   induction s using Finset.induction_on with
276   | empty => exact ⟨0, by simp⟩
277   | @insert i s hi ih =>
278     obtain ⟨r₀, hr₀⟩ := ih (fun j hj => hgood j (Finset.mem_insert_of_mem hj))
279     obtain ⟨r₁, hr₁⟩ := hgood i (Finset.mem_insert_self i s)
280     have hold : (⋃ j ∈ s,
281         {c | ¬ Function.Injective (centralPencil (κ j) (hκ j) (F j r₀) (F j r₁) c)}).Finite :=
282       s.finite_toSet.biUnion fun j hj =>
283         centralPencil_injective_finite (κ j) (hκ j) _ _ (hr₀ j hj)
284     have hnew := reversed_centralPencil_injective_finite (κ i) (hκ i)
285       (F i r₀) (F i r₁) hr₁
286     obtain ⟨c, hc⟩ := (hold.union hnew).exists_notMem
287     refine ⟨r₀ + σ c * r₁, ?_⟩
288     intro j hj
289     have hp : Function.Injective
290         (centralPencil (κ j) (hκ j) (F j r₀) (F j r₁) c) := by
291       by_contra hn
292       by_cases hji : j = i
293       · subst j
294         exact hc (Or.inr hn)
295       · have hjs : j ∈ s := (Finset.mem_insert.mp hj).resolve_left hji
296         exact hc (Or.inl (Set.mem_iUnion.mpr ⟨j, Set.mem_iUnion.mpr ⟨hjs, hn⟩⟩))
297     intro x y hxy
298     apply hp
299     change F j r₀ x + κ j c • F j r₁ x = F j r₀ y + κ j c • F j r₁ y
300     simpa only [hF] using hxy

304 /-- Finite affine source selection in torsion bimodules. There is one
305 direction `H` such that every map with source `B + σ(c) H a` is injective
306 outside a finite set of central scalars. Offsets and division rings may vary
307 with the test. The prescribed right tail is `a bᵢ`. -/
308 theorem exists_affine_direction [IsSimpleRing R]
309     (hOre : RightOreCondition R)
310     (hW : ∀ i, IsTorsionRight (R := R) (M := Wᵢ i))
311     (σ : k →+* R) (κ : ∀ i, k →+* Eᵢ i)
312     (hκ : ∀ i c a, κ i c * a = a * κ i c)
313     (hscalar : ∀ i c (w : Wᵢ i), op (σ c) • w = κ i c • w)
314     (P Z : ∀ i, Dᵢ i →ₗ[Eᵢ i] Wᵢ i) (hP : ∀ i, Function.Injective (P i))
315     (s : Finset ι) (b : ι → R) (hb : ∀ i ∈ s, b i ≠ 0)
316     {a : R} (ha : a ≠ 0) (B : R) :
317     ∃ H : R, {c : k | ∃ i ∈ s, ¬ Function.Injective
318       (fun d => Z i d + op ((B + σ c * H * a) * b i) • P i d)}.Finite := by
319   classical
320   let F (i : ι) (r : R) : Dᵢ i →ₗ[Eᵢ i] Wᵢ i :=
321     (rightMulLinear (E := Eᵢ i) (r * (a * b i))).comp (P i)
322   have hF (i : ι) (r t : R) (c : k) (d : Dᵢ i) :
323       F i (r + σ c * t) d = F i r d + κ i c • F i t d := by
324     change op ((r + σ c * t) * (a * b i)) • P i d =
325       op (r * (a * b i)) • P i d + κ i c • (op (t * (a * b i)) • P i d)
326     rw [add_mul, op_add, add_smul, mul_assoc]
327     congr 1
328     rw [op_mul, mul_smul, hscalar, smul_comm]
329   obtain ⟨H, hH⟩ := exists_simultaneously_injective σ κ hκ F hF s (fun i hi =>
330     exists_injective_right_mul hOre (hW i) σ (κ i) (hκ i) (hscalar i)
331       (P i) (hP i) (mul_ne_zero ha (hb i hi)))
332   let A (i : ι) : Dᵢ i →ₗ[Eᵢ i] Wᵢ i :=
333     Z i + (rightMulLinear (E := Eᵢ i) (B * b i)).comp (P i)
334   have hfinite : (⋃ i ∈ s,
335       {c | ¬ Function.Injective (centralPencil (κ i) (hκ i) (A i) (F i H) c)}).Finite :=
336     s.finite_toSet.biUnion fun i hi =>
337       reversed_centralPencil_injective_finite (κ i) (hκ i) _ _ (hH i hi)
338   refine ⟨H, hfinite.subset ?_⟩
339   rintro c ⟨i, hi, hbad⟩
340   refine Set.mem_iUnion.mpr ⟨i, Set.mem_iUnion.mpr ⟨hi, ?_⟩⟩
341   have heq : (fun d => Z i d + op ((B + σ c * H * a) * b i) • P i d) =
342       centralPencil (κ i) (hκ i) (A i) (F i H) c := by
343     funext d
344     change Z i d + op ((B + σ c * H * a) * b i) • P i d =
345       (Z i d + op (B * b i) • P i d) + κ i c • (op (H * (a * b i)) • P i d)
346     have hscaled : op (σ c * (H * (a * b i))) • P i d =
347         κ i c • (op (H * (a * b i)) • P i d) := by
348       rw [op_mul, mul_smul, hscalar, smul_comm]
349     rw [add_mul, op_add, add_smul, mul_assoc, mul_assoc, hscaled]
350     abel
351   change ¬ Function.Injective (centralPencil (κ i) (hκ i) (A i) (F i H) c)
352   rwa [← heq]

354 /-- The actual source `B + H a` supplied by finite affine selection. -/
355 theorem exists_affine_source [IsSimpleRing R]
356     (hOre : RightOreCondition R)

358     (σ : k →+* R) (κ : ∀ i, k →+* Eᵢ i)

361     (P Z : ∀ i, Dᵢ i →ₗ[Eᵢ i] Wᵢ i) (hP : ∀ i, Function.Injective (P i))
