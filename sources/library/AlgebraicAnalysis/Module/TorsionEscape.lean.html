The right annihilator of a vector, as a right ideal.

23/-- The right annihilator of a vector, as a right ideal. -/
24def rightAnnihilator (w : W) : Submodule Rᵐᵒᵖ R where
25 carrier := {r | op r • w = 0}
26 zero_mem' := zero_smul _ _
27 add_mem' := by
28 intro r s hr hs
29 change op r • w = 0 at hr
30 change op s • w = 0 at hs
31 change op (r + s) • w = 0
32 rw [op_add, add_smul, hr, hs, add_zero]
33 smul_mem' := by
34 intro r s hs
35 change op (s * unop r) • w = 0
36 rw [op_mul, op_unop, mul_smul, hs, smul_zero]

A finite-dimensional subspace of an ordinary torsion bimodule has a common nonzero right annihilator. The scalar actions need only commute.

38/-- A finite-dimensional subspace of an ordinary torsion bimodule has a
39common nonzero right annihilator. The scalar actions need only commute. -/
40theorem exists_common_annihilator (hOre : RightOreCondition R)
41 (hW : IsTorsionRight (R := R) (M := W))
42 (Y : Submodule E W) [Module.Finite E Y] :
43 ∃ c : R, c ≠ 0 ∧ ∀ y ∈ Y, op c • y = 0 := by
44 classical
45 let B := Module.finBasis E Y
46 obtain ⟨c, hc, hcB⟩ := exists_mem_finset_rightIdeals Finset.univ
47 (fun i => rightAnnihilator (R := R) (B i : W))
48 (fun i _ => by
49 obtain ⟨r, hr, hri⟩ := hW (B i : W)
50 exact ⟨r, hri, hr⟩) hOre
51 refine ⟨c, hc, ?_⟩

In a simple right Ore domain, a nonzero vector in a torsion bimodule escapes every finite-dimensional left subspace even with a prescribed nonzero right tail: gRb cannot be contained in that subspace.

74/-- In a simple right Ore domain, a nonzero vector in a torsion bimodule
75escapes every finite-dimensional left subspace even with a prescribed nonzero
76right tail: `g R b` cannot be contained in that subspace. -/
77theorem exists_smul_mul_not_mem [IsSimpleRing R]
78 (hOre : RightOreCondition R) (hW : IsTorsionRight (R := R) (M := W))
79 {g : W} (hg : g ≠ 0) {b : R} (hb : b ≠ 0)
80 (Y : Submodule E W) [Module.Finite E Y] :
81 ∃ r : R, op (r * b) • g ∉ Y := by
82 classical
83 by_contra! h
84 obtain ⟨c, hc, hcY⟩ := exists_common_annihilator hOre hW Y
85 have hbc : b * c ∈ cyclicAnnihilator g := by
86 simp only [cyclicAnnihilator, TwoSidedIdeal.mem_mk', Set.mem_ofPred_eq]

88 rw [← mul_assoc, op_mul, mul_smul]

The maximal-rank argument for a linear family. Its escape hypothesis requires a value outside every finite-dimensional subspace for each nonzero domain vector. This separates the linear argument from torsion and simplicity.

159/-- The maximal-rank argument for a linear family. Its escape hypothesis
160requires a value outside every finite-dimensional subspace for each nonzero
161domain vector. This separates the linear argument from torsion and simplicity. -/
162theorem exists_injective_of_escape
163 (σ : k →+* R) (κ : k →+* E) (hκ : ∀ c a, κ c * a = a * κ c)
164 (F : R → D →ₗ[E] W)
165 (hF : ∀ r s c d, F (r + σ c * s) d = F r d + κ c • F s d)
166 (hescape : ∀ d : D, d ≠ 0 → ∀ Y : Submodule E W,
167 Module.Finite E Y → ∃ r : R, F r d ∉ Y) :
168 ∃ r : R, Function.Injective (F r) := by
169 classical
170 have hbounded : BddAbove (Set.range fun r => Module.finrank E (LinearMap.range (F r))) :=
171 ⟨Module.finrank E D, by rintro _ ⟨r, rfl⟩; exact (F r).finrank_range_le⟩
172 obtain ⟨n, ⟨r₀, rfl⟩, hmax⟩ :=
173 hbounded.exists_isGreatest_of_nonempty ⟨_, 0, rfl⟩
174 by_contra! hnone
175 have hker : LinearMap.ker (F r₀) ≠ ⊥ := by
176 intro h
177 exact hnone r₀ (LinearMap.ker_eq_bot.mp h)
178 obtain ⟨d, hdker, hd⟩ := Submodule.exists_mem_ne_zero_of_ne_bot hker
179 have hd0 : F r₀ d = 0 := hdker
180 let Y := LinearMap.range (F r₀)
181 obtain ⟨r, hr⟩ := hescape d hd Y inferInstance
182 let w := F r d
183 obtain ⟨j, hj⟩ := (F r₀).rangeRestrict.exists_rightInverse_of_surjective
184 (F r₀).range_rangeRestrict
185 have hj' (y : Y) : F r₀ (j y) = (y : W) :=
186 congrArg Subtype.val (LinearMap.congr_fun hj y)
187 let A : Y × E →ₗ[E] W := Y.subtype.comp (LinearMap.fst E Y E) +
188 (LinearMap.toSpanSingleton E W w).comp (LinearMap.snd E Y E)
189 let Q : Y × E →ₗ[E] W := (F r).comp (j.comp (LinearMap.fst E Y E))
190 have hA : Function.Injective A := by
191 apply LinearMap.ker_eq_bot.mp
192 apply LinearMap.ker_eq_bot'.mpr
193 rintro ⟨y, a⟩ ha
194 change (y : W) + a • w = 0 at ha
195 have ha0 : a = 0 := by
196 by_contra ha0
197 apply hr
198 have hm : a • w ∈ Y := (eq_neg_of_add_eq_zero_right ha) ▸ Y.neg_mem y.property
199 exact (Y.smul_mem_iff ha0).mp hm
200 subst a
201 simp only [zero_smul, add_zero] at ha
202 exact Prod.ext (Subtype.ext ha) rfl
203 obtain ⟨c, hc⟩ := ((centralPencil_injective_finite κ hκ A Q hA).union
204 (Set.finite_singleton 0)).exists_notMem
205 have hc0 : c ≠ 0 := fun h => hc (Or.inr h)
206 have hcE : κ c ≠ 0 := by simpa using κ.injective.ne hc0
207 have hAc : Function.Injective (centralPencil κ hκ A Q c) := by
208 by_contra hn
209 exact hc (Or.inl hn)
210 let J : Y × E →ₗ[E] D := j.comp (LinearMap.fst E Y E) +
211 (LinearMap.toSpanSingleton E D ((κ c)⁻¹ • d)).comp (LinearMap.snd E Y E)
212 have hcomp : (F (r₀ + σ c * r)).comp J = centralPencil κ hκ A Q c := by
213 apply LinearMap.ext
214 rintro ⟨y, a⟩
215 change F (r₀ + σ c * r) (j y + a • ((κ c)⁻¹ • d)) =
216 (y : W) + a • w + κ c • F r (j y)
217 rw [map_add, map_smul, map_smul, hF, hj', hF, hd0, zero_add]
218 rw [smul_smul, smul_smul, mul_assoc, inv_mul_cancel₀ hcE, mul_one]
219 abel
220 have hJ : Function.Injective ((F (r₀ + σ c * r)).comp J) := hcomp ▸ hAc
221 have hinj : Function.Injective ((F (r₀ + σ c * r)).rangeRestrict.comp J) := by
222 intro x y hxy
223 exact hJ (congrArg Subtype.val hxy)
224 have hdim := LinearMap.finrank_le_finrank_of_injective hinj
225 have hbound := hmax (Set.mem_range_self (r₀ + σ c * r))
226 change Module.finrank E (LinearMap.range (F (r₀ + σ c * r))) ≤ Module.finrank E Y at hbound
227 rw [Module.finrank_prod, Module.finrank_self] at hdim
228 omega
