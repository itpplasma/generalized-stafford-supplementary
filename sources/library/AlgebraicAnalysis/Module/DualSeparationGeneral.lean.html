29 /-- A partial map to a critical module descends through a quotient when its
30 restriction to the kernel vanishes. -/
31 theorem exists_factor_mkQ_of_critical (hC : Module.Critical R e C)
32     (U : Submodule R M) (f : PartialHom R e M C)
33     (hf : f.comp (ofLinearMap U.subtype) = 0) :
34     ∃ g : PartialHom R e (M ⧸ U) C, f = g.comp (ofLinearMap U.mkQ) := by
35   induction f using Quotient.inductionOn with | _ f =>
36   have hz := (mk_eq_zero_iff_of_critical hC
37     (f.comp (.ofLinearMap U.subtype))).mp hf
38   let q := U.mkQ.submoduleMap f.val.domain
39   have hq : Function.Surjective q := by
40     rintro ⟨_, x, hx, rfl⟩
41     exact ⟨⟨x, hx⟩, rfl⟩
42   have hk : LinearMap.ker q ≤ LinearMap.ker f.val.toFun := by
43     intro x hx
44     have hxu : x.val ∈ U := (Submodule.Quotient.mk_eq_zero U).mp
45       (congrArg Subtype.val hx)
46     have hd : (⟨x.val, hxu⟩ : U) ∈
47         (f.comp (.ofLinearMap U.subtype)).val.domain := by
48       apply (Representative.mem_pullback _ _ Submodule.mem_top).mpr
49       exact x.property
50     exact LinearMap.congr_fun hz ⟨⟨x.val, hxu⟩, hd⟩
51   let b : f.val.domain.map U.mkQ →ₗ[R] C :=
52     ((LinearMap.ker q).liftQ f.val.toFun hk).comp
53       (q.quotKerEquivOfSurjective hq).symm.toLinearMap
54   have hb (x : f.val.domain) : b (q x) = f.val x := by
55     dsimp [b]
56     rw [LinearMap.quotKerEquivOfSurjective_symm_apply]
57     rfl
58   let g : Representative R e (M ⧸ U) C :=
59     ⟨⟨f.val.domain.map U.mkQ, b⟩, f.property.map _ U.mkQ_surjective⟩
60   refine ⟨mk g, Quotient.sound ?_⟩
61   refine ⟨f.val.domain, f.property, le_rfl, ?_, ?_⟩
62   · intro x hx
63     exact (Representative.mem_pullback _ _ Submodule.mem_top).mpr ⟨x, hx, rfl⟩
64   · intro x
65     exact (hb x).symm

70 /-- Pull a normalized germ back to the original source, retaining the reduced
71 target. Surjectivity of the radical quotient makes this operation injective. -/
72 def pullSource (f : Hom R e M C) :
73     PartialHom R e M (C ⧸ smallRadical R e C) :=
74   PartialHom.comp f (PartialHom.ofLinearMap (smallRadical R e M).mkQ)

76 theorem pullSource_injective : Function.Injective
77     (pullSource (R := R) (e := e) (M := M) (C := C)) := by
78   intro f g h

82   rw [LinearMap.range_eq_top.mpr (Submodule.mkQ_surjective _)]
83   exact small_of_subsingleton e

96 /-- Every partial map into the reduced target defines a normalized

98 theorem exists_pullSource [IsNoetherian R M] [IsNoetherian R C]
99     (f : PartialHom R e M (C ⧸ smallRadical R e C)) :
100     ∃ g : Hom R e M C, pullSource g = f := by
101   induction f using Quotient.inductionOn with | _ f =>
102   let p : Pair R e M C :=
103     ⟨f.val.domain, f.property, smallRadical R e C, small_smallRadical, f.val.toFun⟩
104   refine ⟨p.toHom, Quotient.sound ?_⟩
105   refine ⟨f.val.domain, f.property, ?_, le_rfl, ?_⟩
106   · intro x hx
107     exact (Representative.mem_pullback _ _ Submodule.mem_top).mpr ⟨x, hx, rfl⟩
108   · intro x
109     change p.normalize.val (p.domainMap x) = f.val x
110     rw [Pair.normalize_apply]
111     change Submodule.mapQ (smallRadical R e C) (smallRadical R e C)
112       LinearMap.id le_rfl (f.val x) = f.val x
113     rw [Submodule.mapQ_id]

116 /-- Left exactness of germ Hom along an ordinary submodule, for a critical
117 target. The factorization is unique, since the quotient germ is epic. -/
118 theorem comp_subtype_eq_zero_iff [IsNoetherian R M] [IsNoetherian R C]
119     (hC : Module.Critical R e C) (U : Submodule R M) (f : Hom R e M C) :
120     comp f (ofLinearMap U.subtype) = 0 ↔
121       ∃ g : Hom R e (M ⧸ U) C, f = comp g (ofLinearMap U.mkQ) := by
122   constructor
123   · intro hf
124     have hz : (pullSource f).comp (PartialHom.ofLinearMap U.subtype) = 0 := by
125       rw [← pullSource_comp_ofLinearMap, hf, pullSource_zero]
126     obtain ⟨g, hg⟩ := PartialHom.exists_factor_mkQ_of_critical
127       (critical_quotient_smallRadical hC) U (pullSource f) hz
128     obtain ⟨g, rfl⟩ := exists_pullSource g
129     refine ⟨g, pullSource_injective ?_⟩
130     rw [pullSource_comp_ofLinearMap]
131     exact hg
132   · rintro ⟨g, rfl⟩
133     rw [comp_assoc, ← ofLinearMap_comp]
134     have hz : U.mkQ.comp U.subtype = 0 := by ext x; simp
135     rw [hz, ofLinearMap_zero]
136     exact PartialHom.comp_zero g

138 /-- Precomposition is linear over the target's germ endomorphism ring. -/
139 def precompLinear (x : Hom R e N M) :
140     Hom R e M C →ₗ[End R e C] Hom R e N C where
141   toFun f := comp f x
142   map_add' f g := add_comp f g x
143   map_smul' a f := comp_assoc a f x

155 /-- The left-exact sequence `0 → Hom(M/U,C) → Hom(M,C) → Hom(U,C)`. -/
156 theorem range_quotientLinear_eq_ker_restrictLinear
157     [IsNoetherian R M] [IsNoetherian R C]
158     (hC : Module.Critical R e C) (U : Submodule R M) :
159     LinearMap.range (quotientLinear (C := C) (e := e) U) =
160       LinearMap.ker (restrictLinear (C := C) (e := e) U) := by
161   ext f
162   change (∃ g, comp g (ofLinearMap U.mkQ) = f) ↔
163     comp f (ofLinearMap U.subtype) = 0
164   rw [comp_subtype_eq_zero_iff hC]
165   constructor <;> rintro ⟨g, hg⟩ <;> exact ⟨g, hg.symm⟩

167 /-- Surjective ordinary maps remain epic after normalization. -/
168 theorem epic_ofLinearMap_of_surjective (a : N →ₗ[R] M)
169     (ha : Function.Surjective a) : Epic (ofLinearMap (e := e) a) := by
170   apply (PartialHom.epic_ofLinearMap_iff _).mpr
171   have hr : Function.Surjective (reducedMap (e := e) a) := by
172     intro y
173     obtain ⟨y, rfl⟩ := Submodule.mkQ_surjective (smallRadical R e M) y
174     obtain ⟨x, rfl⟩ := ha y
175     exact ⟨(smallRadical R e N).mkQ x, rfl⟩
176   rw [LinearMap.range_eq_top.mpr hr]
177   exact small_of_subsingleton e

184 /-- A linear equivalence of sources induces a Schur-linear equivalence. -/
185 def precompEquiv (a : M ≃ₗ[R] N) :
186     Hom R e N C ≃ₗ[End R e C] Hom R e M C where
187   __ := precompLinear (ofLinearMap a.toLinearMap)
188   invFun f := comp f (ofLinearMap a.symm.toLinearMap)
189   left_inv f := by
190     change comp (comp f _) _ = f
191     rw [comp_assoc, ← ofLinearMap_comp, a.comp_symm,
192       ofLinearMap_id, comp_id]
193   right_inv f := by
194     change comp (comp f _) _ = f
195     rw [comp_assoc, ← ofLinearMap_comp, a.symm_comp,
196       ofLinearMap_id, comp_id]

198 /-- Germ Hom out of a small source vanishes. -/
199 theorem subsingleton_of_small (hM : Small R e M) :
200     Subsingleton (Hom R e M C) := by
201   have ht : smallRadical R e M = ⊤ := by
202     apply top_unique
203     apply le_smallRadical
204     exact (small_linearEquiv_iff e (Submodule.topEquiv : (⊤ : Submodule R M) ≃ₗ[R] M)).mpr hM
205   have : Subsingleton (M ⧸ smallRadical R e M) := by rw [ht]; infer_instance
206   constructor
207   intro f g
208   induction f using Quotient.inductionOn with | _ f =>
209   induction g using Quotient.inductionOn with | _ g =>
210   apply Quotient.sound
211   apply Representative.rel_of_eq
212   intro x hx hy
213   have hxf : (⟨x, hx⟩ : f.val.domain) = 0 := Subsingleton.elim _ _
214   have hxg : (⟨x, hy⟩ : g.val.domain) = 0 := Subsingleton.elim _ _
215   rw [hxf, hxg]
216   exact f.val.toFun.map_zero.trans g.val.toFun.map_zero.symm

218 /-- Finiteness and the dimension inequality for one left-exact Schur sequence. -/
219 theorem finite_finrank_le_of_submodule [IsNoetherian R M] [IsNoetherian R C]
220     (hC : Module.Critical R e C) (U : Submodule R M)
221     [Module.Finite (End R e C) (Hom R e U C)]
222     [Module.Finite (End R e C) (Hom R e (M ⧸ U) C)] :
223     letI := End.divisionRing hC
224     Module.Finite (End R e C) (Hom R e M C) ∧
225       Module.finrank (End R e C) (Hom R e M C) ≤
226         Module.finrank (End R e C) (Hom R e U C) +
227           Module.finrank (End R e C) (Hom R e (M ⧸ U) C) := by
228   letI := End.divisionRing hC
229   let a := quotientLinear (C := C) (e := e) U
230   let b := restrictLinear (C := C) (e := e) U
231   have hr : LinearMap.range a = LinearMap.ker b :=
232     range_quotientLinear_eq_ker_restrictLinear hC U
233   haveI : Module.Finite (End R e C) (LinearMap.ker b) := by
234     rw [← hr]
235     infer_instance
236   haveI : Module.Finite (End R e C) (Hom R e M C) :=
237     Module.Finite.of_exact (f := (LinearMap.ker b).subtype) (g := b.rangeRestrict)
238       (LinearMap.exact_iff.mpr (by ext x; simp))
239       (by rintro ⟨y, x, rfl⟩; exact ⟨x, rfl⟩)
240   refine ⟨inferInstance, ?_⟩
241   have hk : Module.finrank (End R e C) (LinearMap.ker b) ≤
242       Module.finrank (End R e C) (Hom R e (M ⧸ U) C) := by
243     rw [← hr]
244     exact a.finrank_range_le
245   have hb := (LinearMap.range b).finrank_le
246   have he := b.finrank_range_add_finrank_ker
247   omega

279 /-- Every surviving factor at the ceiling is critical at that ceiling. -/
280 theorem critical_of_criticalStep_not_small {p q : Submodule R M}
281     (h : Module.CriticalStep R e M p q)
282     (hs : ¬Small R e (q ⧸ p.comap q.subtype)) :
283     Module.Critical R e (q ⧸ p.comap q.subtype) := by
284   obtain ⟨d, hd, hc⟩ := h.2
285   have he : e ≤ d := by
286     by_contra he
287     exact hs ((show Small R (d + 1) (q ⧸ p.comap q.subtype) from hc.devLE).mono

289   have hde : d = e := le_antisymm hd he

295 private theorem finite_finrank_le_series [IsNoetherian R M] [IsNoetherian R C]
296     (hC : Module.Critical R e C)
297     (s : RelSeries {(p, q) | Module.CriticalStep R e M p q})
298     (hs : Small R e s.head) :
299     letI := End.divisionRing hC
300     Module.Finite (End R e C) (Hom R e s.last C) ∧
301       Module.finrank (End R e C) (Hom R e s.last C) ≤ schurWeight s := by
302   classical
303   letI := End.divisionRing hC
304   induction s using RelSeries.inductionOn' with
305   | singleton p =>
306     rw [RelSeries.last_singleton]
307     change Small R e p at hs
308     haveI := subsingleton_of_small (C := C) hs
309     exact ⟨inferInstance, by simp [schurWeight, Module.finrank_zero_of_subsingleton]⟩
310   | snoc s b h ih =>
311     rw [RelSeries.head_snoc] at hs
312     rw [RelSeries.last_snoc]
313     obtain ⟨hfin, hdim⟩ := ih hs
314     letI := hfin
315     let U := s.last.comap b.subtype
316     let a : U ≃ₗ[R] s.last := Submodule.comapSubtypeEquivOfLe h.1
317     let t := precompEquiv (C := C) (e := e) a
318     haveI : Module.Finite (End R e C) (Hom R e U C) :=
319       Module.Finite.of_surjective t.toLinearMap t.surjective
320     have hu : Module.finrank (End R e C) (Hom R e U C) ≤ schurWeight s := by
321       rw [← t.finrank_eq]
322       exact hdim
323     have hfactor : Module.Finite (End R e C) (Hom R e (b ⧸ U) C) ∧
324         Module.finrank (End R e C) (Hom R e (b ⧸ U) C) ≤
325           if Small R e (b ⧸ U) then 0 else 1 := by
326       by_cases hf : Small R e (b ⧸ U)
327       · haveI := subsingleton_of_small (C := C) hf
328         exact ⟨inferInstance, by simp [hf, Module.finrank_zero_of_subsingleton]⟩
329       · have hc := critical_of_criticalStep_not_small h hf
330         exact ⟨finite_of_critical hc hC, by simpa [hf] using finrank_le_one_of_critical hc hC⟩
331     letI := hfactor.1
332     obtain ⟨hf, hd⟩ := finite_finrank_le_of_submodule hC U
333     refine ⟨hf, ?_⟩
334     rw [schurWeight_snoc]
335     exact hd.trans (Nat.add_le_add hu hfactor.2)

433   let t := precompEquiv (C := C) (e := e)
434     (Submodule.topEquiv : (⊤ : Submodule R M) ≃ₗ[R] M)
435   exact h (t f) (fun he => hf (t.injective (he.trans (map_zero _).symm)))
436 
437 end Hom

444 /-- The finite index family of critical factors surviving at the ceiling.
445 Repeated isomorphic factors may occur. -/
446 abbrev TopIndex (F : CriticalFiltration R e M) :=
447   {i : Fin F.length // ¬Small R e (F.Factor i)}
448 
449 /-- The critical module representing a retained type. -/
450 abbrev TopFactor (F : CriticalFiltration R e M) (i : F.TopIndex) := F.Factor i.val
451 
452 /-- Each retained type is critical at the common ceiling. -/
453 theorem topFactor_critical (F : CriticalFiltration R e M) (i : F.TopIndex) :
454     Critical R e (F.TopFactor i) :=
455   Germ.critical_of_criticalStep_not_small (F.step i.val) i.property
456 
457 /-- The number of retained indices is exactly the top length. -/
458 theorem card_topIndex (F : CriticalFiltration R e M) :
459     Fintype.card F.TopIndex = F.topLength e := by
460   classical
461   simp only [TopIndex, Fintype.card_subtype, topLength,
462     Finset.card_eq_sum_ones, Finset.sum_filter]
463   apply Finset.sum_congr rfl
464   intro i _
465   by_cases hs : Small R e (F.Factor i) <;> simp [hs]

467 end Module.CriticalFiltration

475 /-- Every critical target with nonzero dual Hom is germ-isomorphic to a
476 retained factor. This identifies the finite list of critical types. -/
477 theorem exists_topFactor_iso_of_ne_zero (hC : Module.Critical R e C)
478     (F : Module.CriticalFiltration R e M) (f : Hom R e M C) (hf : f ≠ 0) :
479     ∃ i : F.TopIndex, ∃ g : Hom R e (F.TopFactor i) C,
480       ∃ a : Hom R e C (F.TopFactor i), comp a g = id ∧ comp g a = id := by
481   obtain ⟨i, hi, g, hg⟩ := exists_factor_hom_of_ne_zero hC F f hf
482   let j : F.TopIndex := ⟨i, hi⟩
483   obtain ⟨a, hag, hga⟩ := exists_inverse_of_critical (F.topFactor_critical j) hC g hg
484   exact ⟨j, g, a, hag, hga⟩

496 /-- Finite critical dual separation. A germ with arbitrary source is epic
497 modulo small modules exactly when precomposition is injective on the finitely
498 many critical Schur spaces supplied by a critical filtration. -/
499 theorem epic_iff_injective_comp_topFactors (hM : Module.DevLE R e M)
500     (F : Module.CriticalFiltration R e M) (x : Hom R e N M) :
501     Epic x ↔ ∀ i : F.TopIndex, Function.Injective
502       (fun h : Hom R e M (F.TopFactor i) => comp h x) := by
503   classical
504   refine ⟨fun hx _ _ _ he => hx.right_cancel _ _ he, ?_⟩
505   intro hinj
506   by_contra hx
507   induction x using Quotient.inductionOn with | _ x =>
508   have hns : ¬Small R e ((M ⧸ smallRadical R e M) ⧸ x.val.toFun.range) := hx
509   have hd := ((hM.quotient (smallRadical R e M)).quotient x.val.toFun.range)
510   obtain ⟨p, hp⟩ := hd.exists_critical_quotient hns
511   let Q := ((M ⧸ smallRadical R e M) ⧸ x.val.toFun.range) ⧸ p
512   let b : (M ⧸ smallRadical R e M) →ₗ[R] Q := p.mkQ.comp x.val.toFun.range.mkQ
513   let k : M →ₗ[R] Q := b.comp (smallRadical R e M).mkQ
514   have hks : Function.Surjective k :=
515     p.mkQ_surjective.comp (x.val.toFun.range.mkQ_surjective.comp
516       (Submodule.mkQ_surjective _))
517   have hkn : ofLinearMap (e := e) k ≠ 0 := by
518     intro hz
519     have hs := (ofLinearMap_eq_zero_iff k).mp hz
520     rw [LinearMap.range_eq_top.mpr hks] at hs
521     exact hp.not_small (hs.of_surjective (⊤ : Submodule R Q).subtype
522       (fun y => ⟨⟨y, Submodule.mem_top⟩, rfl⟩))
523   have hkred : reducedMap (e := e) k = (smallRadical R e Q).mkQ.comp b := by
524     apply LinearMap.ext
525     intro y
526     obtain ⟨y, rfl⟩ := Submodule.mkQ_surjective (smallRadical R e M) y
527     rfl
528   have hkill : comp (ofLinearMap k) (PartialHom.mk x) = 0 := by
529     apply Quotient.sound
530     apply Representative.rel_of_eq
531     intro y hy _
532     change reducedMap k (x.val ⟨y, x.pullback_le _ hy⟩) = 0
533     rw [hkred]
534     have hz : x.val.toFun.range.mkQ (x.val ⟨y, x.pullback_le _ hy⟩) = 0 :=
535       (Submodule.Quotient.mk_eq_zero _).mpr ⟨_, rfl⟩
536     change (smallRadical R e Q).mkQ (p.mkQ (x.val.toFun.range.mkQ _)) = 0
537     rw [hz, map_zero, map_zero]
538   obtain ⟨i, g, a, hag, hga⟩ := exists_topFactor_iso_of_ne_zero hp F (ofLinearMap k) hkn
539   let h : Hom R e M (F.TopFactor i) := comp a (ofLinearMap k)
540   have hne : h ≠ 0 := by
541     intro hz
542     have he := congrArg (fun t => comp g t) hz
543     change comp g (comp a (ofLinearMap k)) = comp g 0 at he
544     rw [← comp_assoc, hga, id_comp] at he
545     exact hkn (he.trans (PartialHom.comp_zero _))
546   apply hne
547   apply hinj i
548   change comp (comp a (ofLinearMap k)) (PartialHom.mk x) = comp 0 (PartialHom.mk x)
549   rw [comp_assoc, hkill]
550   exact (PartialHom.comp_zero _).trans (PartialHom.zero_comp _).symm
551
