8 # Deviation at omega and quotients by injective endomorphisms
9 
10 `Module.DevLEOmega` asks for eventually finite-deviation successive intervals,
11 without a uniform finite bound on these intervals. The descending ranges of
12 an injective endomorphism have isomorphic successive intervals. Consequently
13 its cokernel has finite deviation under this omega bound, and loses one
14 stratum under a finite bound. Left multiplication on a right regular module
15 specializes this to `R/dR` in a domain. No Noetherian or Ore hypothesis is
16 needed for these conclusions.
17 -/

23 /-- Deviation at most omega, expressed without assigning an ordinal dimension. -/
24 def DevLEOmega (R : Type u) (M : Type v) [Ring R] [AddCommGroup M] [Module R M] : Prop :=
25   ∀ c : ℕ → Submodule R M, Antitone c → ∃ N, ∀ i, N ≤ i →
26     ∃ n, Order.DevLE n (Set.Icc (c (i + 1)) (c i))

30 theorem DevLE.omega {n : ℕ} (h : DevLE R n M) : DevLEOmega R M :=
31   fun _c _ => ⟨0, fun _i _ => ⟨n, Order.DevLE.interval h _ _⟩⟩

33 /-- Each successive range interval of an injective endomorphism is the
34 submodule lattice of its cokernel. -/
35 noncomputable def quotientRangePowOrderIso (f : M →ₗ[R] M) (hf : Function.Injective f)
36     (i : ℕ) : Submodule R (M ⧸ LinearMap.range f) ≃o
37       Set.Icc (LinearMap.range (f ^ (i + 1))) (LinearMap.range (f ^ i)) := by
38   have hfi : Function.Injective (f ^ i) := Module.End.iterate_injective hf i
39   have he : LinearMap.range (f ^ (i + 1)) = (LinearMap.range f).map (f ^ i) := by
40     rw [Module.End.iterate_succ, LinearMap.range_comp]
41   refine
42     { toFun := fun U => ⟨(U.comap (LinearMap.range f).mkQ).map (f ^ i),
43         ?_, ?_⟩
44       invFun := fun U => (U.val.comap (f ^ i)).map (LinearMap.range f).mkQ
45       left_inv := ?_
46       right_inv := ?_
47       map_rel_iff' := ?_ }
48   · rw [he]
49     exact Submodule.map_mono (Submodule.le_comap_mkQ _ _)
50   · rintro x ⟨y, _, rfl⟩
51     exact ⟨y, rfl⟩
52   · intro U
53     dsimp
54     rw [Submodule.comap_map_eq_of_injective hfi]
55     exact Submodule.map_comap_eq_of_surjective (Submodule.mkQ_surjective _) U
56   · intro U
57     apply Subtype.ext
58     dsimp
59     have hl : LinearMap.range f ≤ U.val.comap (f ^ i) := by
60       rw [← Submodule.map_le_iff_le_comap, ← he]
61       exact U.property.1
62     rw [Submodule.comap_map_mkQ, sup_eq_right.mpr hl,
63       Submodule.map_comap_eq, inf_eq_right.mpr U.property.2]
64   · intro U V
65     change (U.comap _).map (f ^ i) ≤ (V.comap _).map (f ^ i) ↔ U ≤ V
66     rw [Submodule.map_le_map_iff_of_injective hfi,
67       Submodule.comap_le_comap_iff_of_surjective (Submodule.mkQ_surjective _)]

69 /-- An injective endomorphism lowers a positive finite deviation bound on its cokernel. -/
70 theorem DevLE.quotient_range_of_injective {n : ℕ} (h : DevLE R (n + 1) M)
71     (f : M →ₗ[R] M) (hf : Function.Injective f) : DevLE R n (M ⧸ LinearMap.range f) := by
72   obtain ⟨N, hN⟩ := h f.iterateRange f.iterateRange.monotone
73   exact (Order.devLE_orderIso_iff n (quotientRangePowOrderIso f hf N)).mpr
74     (hN N le_rfl)
75 
76 /-- The omega bound gives a finite bound on the cokernel of an injective endomorphism. -/
77 theorem DevLEOmega.quotient_range_of_injective (h : DevLEOmega R M)
78     (f : M →ₗ[R] M) (hf : Function.Injective f) :
79     ∃ n, DevLE R n (M ⧸ LinearMap.range f) := by
80   obtain ⟨N, hN⟩ := h f.iterateRange f.iterateRange.monotone
81   obtain ⟨n, hn⟩ := hN N le_rfl
82   exact ⟨n, (Order.devLE_orderIso_iff n (quotientRangePowOrderIso f hf N)).mpr hn⟩
83 
84 /-- The strict-stratum version includes the Artinian-to-zero endpoint. -/
85 theorem Small.quotient_range_of_injective {e : ℕ} (h : Small R (e + 1) M)
86     (f : M →ₗ[R] M) (hf : Function.Injective f) : Small R e (M ⧸ LinearMap.range f) := by
87   cases e with
88   | zero =>
89     have : IsArtinian R M := devLE_zero_iff_isArtinian.mp h
90     have hs : Function.Surjective f := IsArtinian.surjective_of_injective_endomorphism f hf
91     exact Submodule.Quotient.subsingleton_iff.mpr (LinearMap.range_eq_top.mpr hs)
92   | succ e => exact DevLE.quotient_range_of_injective h f hf

98 /-- The successive intervals of `R ⊇ dR ⊇ d²R ⊇ …`, with the right-sided
99 convention explicit, are all the submodule lattice of `R/dR`. -/
100 noncomputable def principalRightPowOrderIso {d : D} (hd : d ≠ 0) (i : ℕ) :
101     Submodule Dᵐᵒᵖ (D ⧸ LinearMap.range (LinearMap.mulLeft Dᵐᵒᵖ d)) ≃o
102       Set.Icc (LinearMap.range (LinearMap.mulLeft Dᵐᵒᵖ (d ^ (i + 1))))
103         (LinearMap.range (LinearMap.mulLeft Dᵐᵒᵖ (d ^ i))) := by
104   rw [← LinearMap.pow_mulLeft Dᵐᵒᵖ D d (i + 1),
105     ← LinearMap.pow_mulLeft Dᵐᵒᵖ D d i]
106   exact quotientRangePowOrderIso (LinearMap.mulLeft Dᵐᵒᵖ d)
107     (fun _ _ hxy => mul_left_cancel₀ hd hxy) i
108 
109 /-- In a domain, the right principal quotient drops a finite deviation stratum. -/
110 theorem Small.principal_right_quotient {e : ℕ} (h : Small Dᵐᵒᵖ (e + 1) D)
111     {d : D} (hd : d ≠ 0) :
112     Small Dᵐᵒᵖ e (D ⧸ LinearMap.range (LinearMap.mulLeft Dᵐᵒᵖ d)) :=
113   h.quotient_range_of_injective (LinearMap.mulLeft Dᵐᵒᵖ d)
114     (fun _ _ hxy => mul_left_cancel₀ hd hxy)
115 
116 /-- Under the omega deviation bound, every nonzero principal right quotient
117 has finite deviation. Units are allowed and yield the zero quotient. -/
118 theorem DevLEOmega.principal_right_quotient (h : DevLEOmega Dᵐᵒᵖ D)
119     {d : D} (hd : d ≠ 0) :
120     ∃ n, DevLE Dᵐᵒᵖ n (D ⧸ LinearMap.range (LinearMap.mulLeft Dᵐᵒᵖ d)) :=
121   h.quotient_range_of_injective (LinearMap.mulLeft Dᵐᵒᵖ d)
122     (fun _ _ hxy => mul_left_cancel₀ hd hxy)
