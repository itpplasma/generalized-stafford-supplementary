9 # Deviation of a commutative Noetherian ring is bounded by its Krull dimension
10 
11 For a commutative Noetherian ring `A` of finite Krull dimension `m`, the
12 lattice of ideals of `A` has Gabriel–Rentschler deviation at most `m`:
13 `Module.devLE_of_ringKrullDim_le`. This is the inequality
14 `Kdim A ≤ cl.Kdim A` of [McConnell–Robson, *Noncommutative Noetherian Rings*,
15 6.4.8] (equality holds there; only the upper bound is formalized here).
16 
17 ## Proof
18 
19 * `Module.devLE_iff_small_factors` restates the deviation bound uniformly in
20   terms of the strict strata `Module.Small`: `DevLE n M` iff every descending
21   chain eventually has successive factors in `Small n`.
22 * `Module.small_of_le_annihilator`: if `Small e (A ⧸ q)` for every prime
23   `q ⊇ J`, then every finitely generated module killed by `J` is in `Small e`.
24   This uses Mathlib's prime filtration
25   `IsNoetherianRing.induction_on_isQuotientEquivQuotientPrime` and the
26   extension rule `Module.small_iff_of_exact`.
27 * `Module.small_quotient_prime_of_coheight_lt`: by induction on `n`, if the
28   prime `p` has coheight `< n` in `PrimeSpectrum A` then `Small n (A ⧸ p)`.
29   For a descending chain of ideals of `A ⧸ p` that never reaches zero, choose
30   `a ∉ p` representing a nonzero element of `c (i+1)`; then `a` and `p` kill
31   `c i / c (i+1)`, so every prime in its prime filtration strictly contains
32   `p` and has smaller coheight.
33 
34 No dimension is assigned; `Order.coheight` and `ringKrullDim` are Mathlib's.
35 -/

45 /-- The successive factors of a descending chain lie in the strict stratum
46 `Small n` from some point on exactly when the deviation is at most `n`. At
47 `n = 0` the factors are zero, i.e. the chain stabilizes. -/
48 theorem devLE_iff_small_factors (n : ℕ) : DevLE R n M ↔
49     ∀ c : ℕ → Submodule R M, Antitone c → ∃ N, ∀ i, N ≤ i →
50       Small R n (c i ⧸ (c (i + 1)).comap (c i).subtype) := by
51   cases n with
52   | succ n => exact devLE_succ_iff n
53   | zero =>
54     have key : ∀ (c : ℕ → Submodule R M), Antitone c → ∀ i,
55         Subsingleton (c i ⧸ (c (i + 1)).comap (c i).subtype) ↔ c (i + 1) = c i := by
56       intro c hc i
57       rw [Submodule.Quotient.subsingleton_iff, Submodule.comap_subtype_eq_top]
58       exact ⟨fun h => le_antisymm (hc (Nat.le_succ i)) h, fun h => h.ge⟩
59     constructor
60     · intro h c hc
61       obtain ⟨N, hN⟩ := h c hc
62       refine ⟨N, fun i hi => (key c hc i).mpr ?_⟩
63       exact (hN (i + 1) (Nat.le_succ_of_le hi)).trans (hN i hi).symm
64     · intro h c hc
65       obtain ⟨N, hN⟩ := h c hc
66       refine ⟨N, fun i hi => ?_⟩
67       induction i, hi using Nat.le_induction with
68       | base => rfl
69       | succ i hi ih => exact ((key c hc i).mp (hN i hi)).trans ih
70 
71 end Strata

75 /-- **Prime-filtration bound.** If every quotient `A ⧸ q` by a prime `q ⊇ J`
76 lies in the stratum `Small e`, then so does every finitely generated module
77 annihilated by `J`. -/
78 theorem small_of_le_annihilator (e : ℕ) (J : Ideal A)
79     (hJ : ∀ q : PrimeSpectrum A, J ≤ q.asIdeal → Small A e (A ⧸ q.asIdeal))
80     {M : Type v} [AddCommGroup M] [Module A M] [Module.Finite A M]
81     (hM : J ≤ Module.annihilator A M) : Small A e M := by
82   revert hM
83   induction ‹Module.Finite A M› using IsNoetherianRing.induction_on_isQuotientEquivQuotientPrime
84     with
85   | subsingleton N =>
86     intro _
87     cases e with
88     | zero => exact (small_zero_iff).mpr inferInstance
89     | succ e => exact devLE_of_subsingleton e
90   | quotient N q f =>
91     intro hN
92     refine (small_linearEquiv_iff e f).mpr (hJ q fun r hr => ?_)
93     have hr' : r ∈ Module.annihilator A (A ⧸ q.asIdeal) := f.annihilator_eq ▸ hN hr
94     have h1 := (Module.mem_annihilator.mp hr') 1
95     rwa [Algebra.smul_def, mul_one, Ideal.Quotient.algebraMap_eq,
96       Ideal.Quotient.eq_zero_iff_mem] at h1
97   | exact N₁ N₂ N₃ f g hf hg hfg h₁ h₃ =>
98     intro hN
99     exact (small_iff_of_exact e f g hf hg hfg).mpr
100       ⟨h₁ (hN.trans (f.annihilator_le_of_injective hf)),
101         h₃ (hN.trans (g.annihilator_le_of_surjective hg))⟩

104 /-- In `A ⧸ p`, multiplication by a representative `a` of an element of a
105 submodule `C` maps every element into `C`. -/
106 private theorem smul_mem_of_mk_mem {p : Ideal A} (C : Submodule A (A ⧸ p)) {a : A}
107     (ha : Ideal.Quotient.mk p a ∈ C) (m : A ⧸ p) : a • m ∈ C := by
108   obtain ⟨b, rfl⟩ := Ideal.Quotient.mk_surjective m
109   have : a • Ideal.Quotient.mk p b = b • Ideal.Quotient.mk p a := by
110     rw [Algebra.smul_def, Algebra.smul_def, Ideal.Quotient.algebraMap_eq, ← map_mul, ← map_mul,
111       mul_comm]
112   rw [this]
113   exact C.smul_mem b ha

115 /-- **Prime quotients.** If the prime `p` has coheight strictly below `n` in
116 `PrimeSpectrum A`, then the cyclic module `A ⧸ p` lies in the stratum
117 `Small n`, i.e. has deviation at most `n - 1`. -/
118 theorem small_quotient_prime_of_coheight_lt (n : ℕ) :
119     ∀ p : PrimeSpectrum A, Order.coheight p < n → Small A n (A ⧸ p.asIdeal) := by
120   induction n with
121   | zero => intro p hp; exact absurd hp (not_lt_bot)
122   | succ n ih =>
123     intro p hp
124     have hpn : Order.coheight p ≤ n :=
125       (ENat.lt_add_one_iff (ENat.natCast_ne_top n)).mp (by exact_mod_cast hp)
126     show DevLE A n (A ⧸ p.asIdeal)
127     rw [devLE_iff_small_factors]
128     intro c hc
129     by_cases hbot : ∃ N, c N = ⊥
130     · obtain ⟨N, hN⟩ := hbot
131       refine ⟨N, fun i hi => ?_⟩
132       have hci : c i = ⊥ := le_bot_iff.mp (hN ▸ hc hi)
133       have : Subsingleton (c i) := by rw [hci]; infer_instance
134       have : Subsingleton (c i ⧸ (c (i + 1)).comap (c i).subtype) :=
135         Submodule.Quotient.mk_surjective _ |>.subsingleton
136       cases n with
137       | zero => exact (small_zero_iff).mpr inferInstance
138       | succ e => exact devLE_of_subsingleton e
139     push Not at hbot
140     refine ⟨0, fun i _ => ?_⟩
141     obtain ⟨x, hx, hx0⟩ := Submodule.exists_mem_ne_zero_of_ne_bot (hbot (i + 1))
142     obtain ⟨a, rfl⟩ := Ideal.Quotient.mk_surjective x
143     have hap : a ∉ p.asIdeal := fun h => hx0 (Ideal.Quotient.eq_zero_iff_mem.mpr h)
144     set J : Ideal A := p.asIdeal ⊔ Ideal.span {a}
145     apply small_of_le_annihilator n J
146     · intro q hq
147       have hlt : p < q := by
148         refine lt_of_le_of_ne (fun r hr => hq (Ideal.mem_sup_left hr)) fun he => hap ?_
149         rw [he]
150         exact hq (Ideal.mem_sup_right (Ideal.mem_span_singleton_self a))
151       have hfin : Order.coheight q < ⊤ :=
152         ((Order.coheight_anti hlt.le).trans hpn).trans_lt (ENat.natCast_lt_top n)
153       exact ih q ((Order.coheight_strictAnti hlt hfin).trans_le hpn)
154     · rw [sup_le_iff]
155       constructor
156       · intro r hr
157         rw [Module.mem_annihilator]
158         intro m
159         obtain ⟨⟨y, hy⟩, rfl⟩ := Submodule.Quotient.mk_surjective _ m
160         rw [← Submodule.Quotient.mk_smul, Submodule.Quotient.mk_eq_zero]
161         obtain ⟨b, rfl⟩ := Ideal.Quotient.mk_surjective y
162         have : r • Ideal.Quotient.mk p.asIdeal b = 0 := by
163           rw [Algebra.smul_def, Ideal.Quotient.algebraMap_eq, ← map_mul,
164             Ideal.Quotient.eq_zero_iff_mem]
165           exact p.asIdeal.mul_mem_right b hr
166         simp only [Submodule.mem_comap, Submodule.subtype_apply, SetLike.val_smul, this]
167         exact (c (i + 1)).zero_mem
168       · rw [Ideal.span_le]
169         rintro r (rfl : r = a)
170         rw [SetLike.mem_coe, Module.mem_annihilator]
171         intro m
172         obtain ⟨⟨y, hy⟩, rfl⟩ := Submodule.Quotient.mk_surjective _ m
173         rw [← Submodule.Quotient.mk_smul, Submodule.Quotient.mk_eq_zero]
174         exact smul_mem_of_mk_mem (c (i + 1)) hx y

176 /-- **[MR 6.4.8], upper bound.** A commutative Noetherian ring of Krull
177 dimension at most `m` has ideal lattice of deviation at most `m`. -/
178 theorem devLE_of_ringKrullDim_le (m : ℕ) (h : ringKrullDim A ≤ m) : DevLE A m A := by
179   have hS : Small A (m + 1) A := by
180     refine small_of_le_annihilator (m + 1) ⊥ (fun q _ => ?_) bot_le
181     refine small_quotient_prime_of_coheight_lt (m + 1) q ?_
182     have hq : (Order.coheight q : WithBot ℕ∞) ≤ m :=
183       (Order.coheight_le_krullDim q).trans h
184     have hq' : Order.coheight q ≤ m := by exact_mod_cast hq
185     exact lt_of_le_of_lt hq' (by exact_mod_cast Nat.lt_succ_self m)
186   exact hS

188 /-- Finite Krull dimension gives deviation at most omega. -/
189 theorem devLE_exists_of_ringKrullDim_ne_top (h : ringKrullDim A ≠ ⊤) :
190     ∃ m : ℕ, DevLE A m A := by
191   rcases hA : ringKrullDim A with _ | _ | m
192   · exact ⟨0, devLE_of_ringKrullDim_le 0 (by rw [hA]; exact bot_le)⟩
193   · exact absurd hA h
194   · exact ⟨m, devLE_of_ringKrullDim_le m (by rw [hA]; rfl)⟩
195 
196 end Module
