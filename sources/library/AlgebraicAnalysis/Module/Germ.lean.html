7 /-!
8 # Dense partial linear maps and germs
9 
10 The partial-map calculus is category-free. Equality is witnessed on a common
11 dense submodule. Quotienting the target by the small radical gives normalized
12 germs for Noetherian modules; right modules use opposite scalars throughout.
13 The pair correspondence below requires Noetherian source and target; no
14 quotient-valued correspondence for arbitrary modules is asserted.
15 -/

34 /-- Agreement after a common dense restriction. -/
35 def Rel (f g : Representative R e M N) : Prop :=
36   ∃ U : Submodule R M, Dense R e U ∧ ∃ hf : U ≤ f.val.domain,
37     ∃ hg : U ≤ g.val.domain,
38       ∀ x : U, f.val ⟨x, hf x.property⟩ = g.val ⟨x, hg x.property⟩
39 
40 theorem Rel.refl (f : Representative R e M N) : Rel f f :=
41   ⟨f.val.domain, f.property, le_rfl, le_rfl, fun _ => rfl⟩
42 
43 theorem Rel.symm {f g : Representative R e M N} (h : Rel f g) : Rel g f := by
44   obtain ⟨U, hU, hf, hg, h⟩ := h
45   exact ⟨U, hU, hg, hf, fun x => (h x).symm⟩
46 
47 theorem Rel.trans {f g h : Representative R e M N} (hfg : Rel f g) (hgh : Rel g h) :
48     Rel f h := by
49   obtain ⟨U, hU, hf, hg, huv⟩ := hfg
50   obtain ⟨V, hV, hg', hh, hvw⟩ := hgh
51   exact ⟨U ⊓ V, hU.inf hV, inf_le_left.trans hf, inf_le_right.trans hh,
52     fun x => (huv ⟨x, x.property.1⟩).trans (hvw ⟨x, x.property.2⟩)⟩

88 theorem mem_pullback (f : Representative R e M N) (U : Submodule R N)

90     x ∈ f.pullback U ↔ f.val ⟨x, hx⟩ ∈ U := by

116 theorem Rel.comp {g g' : Representative R e N P} {f f' : Representative R e M N}
117     (hg : Rel g g') (hf : Rel f f') : Rel (g.comp f) (g'.comp f') := by
118   obtain ⟨U, hU, hUf, hUf', hff'⟩ := hf
119   obtain ⟨V, hV, hVg, hVg', hgg'⟩ := hg
120   let W := U ⊓ f.pullback V
121   have hwf (x : W) : f.val ⟨x, hUf x.property.1⟩ ∈ V :=
122     (f.mem_pullback V _).mp x.property.2
123   have hwf' (x : W) : f'.val ⟨x, hUf' x.property.1⟩ ∈ V := by
124     rw [← hff' ⟨x, x.property.1⟩]; exact hwf x
125   refine ⟨W, hU.inf (f.dense_pullback hV), ?_, ?_, ?_⟩
126   · intro x hx
127     exact (f.mem_pullback _ (hUf hx.1)).mpr (hVg (hwf ⟨x, hx⟩))
128   · intro x hx
129     exact (f'.mem_pullback _ (hUf' hx.1)).mpr (hVg' (hwf' ⟨x, hx⟩))
130   · intro x
131     change g.val _ = g'.val _
132     have h := hgg' ⟨f.val ⟨x, hUf x.property.1⟩, hwf x⟩
133     refine h.trans ?_
134     congr 1
135     exact Subtype.ext (hff' ⟨x, x.property.1⟩)

147 /-- Restricting a dense-range map to a dense submodule keeps its range dense. -/
148 theorem dense_range_restrict (f : Representative R e M N) (U : Submodule R M)
149     (hU : Dense R e U) (hUf : U ≤ f.val.domain)
150     (hf : Dense R e f.val.toFun.range) :
151     Dense R e (f.val.toFun.comp (Submodule.inclusion hUf)).range := by
152   have h := hf.map_subtype ((hU.comap f.val.domain.subtype).map
153     f.val.toFun.rangeRestrict (by rintro ⟨y, x, rfl⟩; exact ⟨x, rfl⟩))
154   have he : (((U.comap f.val.domain.subtype).map f.val.toFun.rangeRestrict).map
155       f.val.toFun.range.subtype) = (f.val.toFun.comp (Submodule.inclusion hUf)).range := by
156     ext y
157     constructor
158     · rintro ⟨z, ⟨x, hx, rfl⟩, rfl⟩
159       exact ⟨⟨x.val, hx⟩, rfl⟩
160     · rintro ⟨x, rfl⟩
161       exact ⟨f.val.toFun.rangeRestrict ⟨x, hUf x.property⟩,
162         ⟨⟨x, hUf x.property⟩, x.property, rfl⟩, rfl⟩
163   rwa [he] at h

165 /-- Dense range is independent of the representative. -/
166 theorem Rel.dense_range_iff {f g : Representative R e M N} (h : Rel f g) :
167     Dense R e f.val.toFun.range ↔ Dense R e g.val.toFun.range := by
168   suffices ∀ (f g : Representative R e M N), Rel f g →
169       Dense R e f.val.toFun.range → Dense R e g.val.toFun.range from
170     ⟨this f g h, this g f h.symm⟩
171   intro f g h hf
172   obtain ⟨U, hU, hUf, hUg, he⟩ := h
173   have hmap : f.val.toFun.comp (Submodule.inclusion hUf) =
174       g.val.toFun.comp (Submodule.inclusion hUg) := by ext x; exact he x
175   have hd := f.dense_range_restrict U hU hUf hf
176   rw [hmap] at hd
177   exact hd.mono (LinearMap.range_comp_le_range _ _)

379 /-- Dense range gives the categorical right-cancellation property. -/
380 theorem Epic.right_cancel {f : PartialHom R e M N} (hf : Epic f)
381     (g h : PartialHom R e N P) (he : g.comp f = h.comp f) : g = h := by
382   induction f using Quotient.inductionOn with | _ f =>
383   induction g using Quotient.inductionOn with | _ g =>
384   induction h using Quotient.inductionOn with | _ h =>
385   obtain ⟨U, hU, hUg, hUh, hgh⟩ := Quotient.exact he
386   have hUf : U ≤ f.val.domain := hUg.trans (f.pullback_le _)
387   let a := f.val.toFun.comp (Submodule.inclusion hUf)
388   have hag : a.range ≤ g.val.domain := by
389     rintro y ⟨x, rfl⟩
390     exact (f.mem_pullback _ _).mp (hUg x.property)
391   have hah : a.range ≤ h.val.domain := by
392     rintro y ⟨x, rfl⟩
393     exact (f.mem_pullback _ _).mp (hUh x.property)
394   apply Quotient.sound
395   refine ⟨a.range, f.dense_range_restrict U hU hUf hf, hag, hah, ?_⟩
396   rintro ⟨_, x, rfl⟩
397   exact hgh x

440 theorem ofLinearMap_eq_zero_iff [IsNoetherian R N] (f : M →ₗ[R] N) :

450     have hx := LinearMap.mem_range_self (reducedMap (e := e) f)

452     rw [hz] at hx
453     exact hx

459       exact (Submodule.Quotient.mk_eq_zero _).mpr

561 theorem ker_domainMap_le [IsNoetherian R M] [IsNoetherian R N] :
562     LinearMap.ker p.domainMap ≤ LinearMap.ker p.targetMap := by
563   have hk : Small R e (LinearMap.ker p.domainMap) :=
564     (small_smallRadical (R := R) (e := e) (M := M)).of_injective
565       { toFun := fun x => ⟨x.val.val, (Submodule.Quotient.mk_eq_zero _).mp
566           (show (smallRadical R e M).mkQ x.val.val = 0 from
567             congrArg Subtype.val (show p.domainMap x.val = 0 from x.property))⟩
568         map_add' := fun _ _ => rfl
569         map_smul' := fun _ _ => rfl }
570       (by intro x y h; apply Subtype.ext; apply Subtype.ext
571           have hh := congrArg Subtype.val h; exact hh)
572   have hz := eq_bot_of_small_quotient_radical
573     ((LinearMap.ker p.domainMap).map p.targetMap) (hk.map p.targetMap)
574   intro x hx
575   have hh : p.targetMap x ∈ (LinearMap.ker p.domainMap).map p.targetMap :=
576     Submodule.mem_map_of_mem hx
577   rw [hz] at hh
578   exact hh

616 /-- Original representatives agree after a dense restriction and a small
617 common enlargement of their target denominators. -/
618 def Rel (p q : Pair R e M N) : Prop :=
619   ∃ U : Submodule R M, Dense R e U ∧ ∃ hp : U ≤ p.domain, ∃ hq : U ≤ q.domain,
620     ∃ T : Submodule R N, Small R e T ∧ ∃ hpT : p.target ≤ T, ∃ hqT : q.target ≤ T,
621       ∀ x : U, Submodule.mapQ p.target T LinearMap.id hpT (p.map ⟨x, hp x.property⟩) =
622         Submodule.mapQ q.target T LinearMap.id hqT (q.map ⟨x, hq x.property⟩)

624 private theorem enlarge_trans (A B C : Submodule R N) (hAB : A ≤ B) (hBC : B ≤ C)
625     (x : N ⧸ A) : Submodule.mapQ B C LinearMap.id hBC
626       (Submodule.mapQ A B LinearMap.id hAB x) =
627       Submodule.mapQ A C LinearMap.id (hAB.trans hBC) x := by
628   obtain ⟨x, rfl⟩ := Submodule.mkQ_surjective A x
629   rfl

631 /-- Normalization has exactly the prescribed equivalence relation. -/
632 theorem toHom_eq_toHom_iff [IsNoetherian R M] [IsNoetherian R N] (q : Pair R e M N) :
633     p.toHom = q.toHom ↔ Rel p q := by
634   constructor
635   · intro h
636     obtain ⟨W, hW, hWp, hWq, he⟩ := Quotient.exact h
637     let U := (p.domain ⊓ q.domain) ⊓ W.comap (smallRadical R e M).mkQ
638     refine ⟨U, (p.dense_domain.inf q.dense_domain).inf (hW.comap _),
639       inf_le_left.trans inf_le_left, inf_le_left.trans inf_le_right,
640       smallRadical R e N, small_smallRadical, le_smallRadical p.small_target,
641       le_smallRadical q.small_target, ?_⟩
642     intro x
643     have h := he ⟨(smallRadical R e M).mkQ x, x.property.2⟩
644     change p.targetMap ⟨x, x.property.1.1⟩ = q.targetMap ⟨x, x.property.1.2⟩
645     rw [← p.normalize_apply ⟨x, x.property.1.1⟩,
646       ← q.normalize_apply ⟨x, x.property.1.2⟩]
647     exact h
648   · rintro ⟨U, hU, hUp, hUq, T, hT, hpT, hqT, he⟩
649     apply Quotient.sound
650     refine ⟨U.map (smallRadical R e M).mkQ,
651       hU.map _ (Submodule.mkQ_surjective _), Submodule.map_mono hUp,
652       Submodule.map_mono hUq, ?_⟩
653     rintro ⟨_, y, hy, rfl⟩
654     change p.normalize.val (p.domainMap ⟨y, hUp hy⟩) =
655       q.normalize.val (q.domainMap ⟨y, hUq hy⟩)
656     rw [p.normalize_apply, q.normalize_apply]
657     have h := congrArg (Submodule.mapQ T (smallRadical R e N) LinearMap.id
658       (le_smallRadical hT)) (he ⟨y, hy⟩)
659     simpa only [targetMap, LinearMap.comp_apply, enlarge_trans] using h

661 /-- Every normalized germ comes from a pair in the original modules. -/
662 theorem toHom_surjective [IsNoetherian R M] [IsNoetherian R N] :
663     Function.Surjective (fun p : Pair R e M N => p.toHom) := by
664   intro h
665   refine Quotient.inductionOn (motive := fun h => ∃ p : Pair R e M N, p.toHom = h) h ?_
666   intro f
667   let U := f.val.domain.comap (smallRadical R e M).mkQ
668   let a : U →ₗ[R] f.val.domain :=
669     { toFun := fun x => ⟨(smallRadical R e M).mkQ x, x.property⟩
670       map_add' := fun _ _ => rfl
671       map_smul' := fun _ _ => rfl }
672   let p : Pair R e M N :=
673     ⟨U, f.property.comap _, smallRadical R e N, small_smallRadical, f.val.toFun.comp a⟩
674   refine ⟨p, Quotient.sound (Representative.rel_of_eq _ _ ?_)⟩
675   intro x hp hf
676   obtain ⟨y, hy, he⟩ := hp
677   have hx : (⟨x, ⟨y, hy, he⟩⟩ : p.normalize.val.domain) = p.domainMap ⟨y, hy⟩ :=
678     Subtype.ext he.symm
679   rw [hx, p.normalize_apply]
680   change Submodule.mapQ (smallRadical R e N) (smallRadical R e N) LinearMap.id le_rfl
681     (f.val ⟨(smallRadical R e M).mkQ y, hy⟩) = f.val ⟨x, hf⟩
682   rw [Submodule.mapQ_id]
683   change f.val ⟨(smallRadical R e M).mkQ y, hy⟩ = f.val ⟨x, hf⟩
684   congr 1
685   exact Subtype.ext he
