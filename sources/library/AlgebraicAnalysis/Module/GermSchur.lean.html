8 /-!
9 # Schur's lemma for critical germs
10 
11 Multiplication of germ endomorphisms is composition, with the left factor
12 applied last. All module structures use left scalars; right modules are
13 represented using the opposite ring.

17 `.lake/research/active/formal-scope-BC.md`, steps 5–6; no Lean declarations
18 are extracted or duplicated from a downstream repository.

30 /-- A nonzero map between critical modules of the same dimension is injective. -/
31 theorem critical_injective_of_ne_zero (hC : Critical R e C) (hD : Critical R e D)
32     (f : C →ₗ[R] D) (hf : f ≠ 0) : Function.Injective f := by
33   rw [← LinearMap.ker_eq_bot]
34   by_contra hk
35   have hs : Small R e f.range :=
36     (small_linearEquiv_iff e f.quotKerEquivRange).mp (hC.small_quotient _ hk)
37   have hr : f.range = ⊥ := by
38     by_contra hr
39     exact hD.not_small_submodule hr hs
40   exact hf (LinearMap.range_eq_bot.mp hr)

44 /-- For a critical target, a representative is zero as a germ exactly when
45 its underlying map is zero. -/
46 theorem mk_eq_zero_iff_of_critical (hD : Critical R e D)
47     (f : Representative R e C D) : mk f = 0 ↔ f.val.toFun = 0 := by
48   constructor
49   · intro hf
50     obtain ⟨U, hU, hUf, _, he⟩ := Quotient.exact hf
51     have hk : U ≤ (LinearMap.ker f.val.toFun).map f.val.domain.subtype := by
52       intro x hx
53       exact ⟨⟨x, hUf hx⟩, he ⟨x, hx⟩, rfl⟩
54     have hd := (hU.mono hk).comap f.val.domain.subtype
55     rw [Submodule.comap_map_eq_of_injective Subtype.val_injective] at hd
56     have hs := (small_linearEquiv_iff e f.val.toFun.quotKerEquivRange).mp hd
57     have hr : f.val.toFun.range = ⊥ := by
58       by_contra hr
59       exact hD.not_small_submodule hr hs
60     exact LinearMap.range_eq_bot.mp hr
61   · intro hf
62     apply Quotient.sound
63     apply Representative.rel_of_eq
64     intro x hx hy
65     change f.val.toFun _ = 0
66     rw [hf]; rfl

76 /-- Nonzero germs between critical modules admit two-sided inverses. -/
77 theorem exists_inverse_of_critical (hC : Critical R e C) (hD : Critical R e D)
78     (f : PartialHom R e C D) (hf : f ≠ 0) :
79     ∃ g : PartialHom R e D C,
80       g.comp f = ofLinearMap LinearMap.id ∧ f.comp g = ofLinearMap LinearMap.id := by
81   induction f using Quotient.inductionOn with | _ f =>
82   have hd : f.val.domain ≠ ⊥ := by
83     intro hd
84     apply hf
85     apply (mk_eq_zero_iff_of_critical hD f).mpr
86     ext x
87     have hx : x = 0 := Subtype.ext (hd.le x.property)
88     rw [hx, map_zero, LinearMap.zero_apply]
89   have hi := critical_injective_of_ne_zero (hC.submodule _ hd) hD f.val.toFun
90     (fun hz => hf ((mk_eq_zero_iff_of_critical hD f).mpr hz))
91   let a := LinearEquiv.ofInjective f.val.toFun hi
92   let g : Representative R e D C :=
93     ⟨⟨f.val.toFun.range, f.val.domain.subtype.comp a.symm.toLinearMap⟩,
94       epic_of_ne_zero_of_critical hD (mk f) hf⟩
95   refine ⟨mk g, ?_, ?_⟩
96   · apply Quotient.sound
97     apply Representative.rel_of_eq
98     intro x hx hy
99     have hx' : x ∈ f.val.domain := f.pullback_le _ hx
100     exact congrArg Subtype.val (a.symm_apply_apply ⟨x, hx'⟩)
101   · apply Quotient.sound
102     apply Representative.rel_of_eq
103     intro x hx hy
104     have hx' : x ∈ f.val.toFun.range := g.pullback_le _ hx
105     exact congrArg Subtype.val (a.apply_symm_apply ⟨x, hx'⟩)
106 
107 end PartialHom

142 /-- The Schur division ring of a Noetherian critical module. This is an
143 explicit structure because criticality is a proposition, not a typeclass. -/
144 noncomputable abbrev divisionRing [IsNoetherian R C] (hC : Critical R e C) :
145     DivisionRing (End R e C) := by
146   letI : Nontrivial (End R e C) := by
147     apply nontrivial_of_ne (1 : End R e C) 0
148     intro h
149     have hz : Hom.ofLinearMap (e := e) (LinearMap.id : C →ₗ[R] C) = 0 := by
150       rw [Hom.ofLinearMap_id]
151       exact h
152     have hs := (Hom.ofLinearMap_eq_zero_iff _).mp hz
153     rw [LinearMap.range_id] at hs
154     exact hC.not_small (hs.of_surjective (⊤ : Submodule R C).subtype
155       (fun x => ⟨⟨x, Submodule.mem_top⟩, rfl⟩))
156   exact DivisionRing.ofIsUnitOrEqZero fun f => by
157     by_cases hf : f = 0
158     · exact Or.inr hf
159     · obtain ⟨g, hgf, hfg⟩ := Hom.exists_inverse_of_critical hC hC f hf
160       exact Or.inl ⟨⟨f, g, hfg, hgf⟩, rfl⟩
161 
162 end End

190 /-- Multiplication by a central scalar, as a linear map. -/
191 def centralScalar (a : Subring.center R) : C →ₗ[R] C where
192   toFun x := (a : R) • x
193   map_add' := smul_add (a : R)
194   map_smul' r x := by
195     change (a : R) • (r • x) = r • ((a : R) • x)
196     simp only [smul_smul, (Subring.mem_center_iff.mp a.property r)]
197 
198 /-- Central ring scalars act as germ endomorphisms. -/
199 def centralMap : Subring.center R →+* End R e C where
200   toFun a := Hom.ofLinearMap (centralScalar a)
201   map_one' := by
202     have he : centralScalar (C := C) (1 : Subring.center R) = LinearMap.id :=
203       LinearMap.ext fun x => one_smul R x
204     change Hom.ofLinearMap _ = Hom.id
205     rw [he, Hom.ofLinearMap_id]
206   map_mul' a b := by
207     change Hom.ofLinearMap _ = Hom.comp (Hom.ofLinearMap _) (Hom.ofLinearMap _)
208     rw [← Hom.ofLinearMap_comp]
209     apply congrArg (Hom.ofLinearMap (R := R) (e := e))
210     exact LinearMap.ext fun x => mul_smul (a : R) (b : R) x
211   map_zero' := by
212     have he : centralScalar (C := C) (0 : Subring.center R) = 0 :=
213       LinearMap.ext fun x => zero_smul R x
214     change Hom.ofLinearMap _ = 0
215     rw [he, Hom.ofLinearMap_zero]
216   map_add' a b := by
217     change Hom.ofLinearMap _ = Hom.ofLinearMap _ + Hom.ofLinearMap _
218     rw [← Hom.ofLinearMap_add]
219     apply congrArg (Hom.ofLinearMap (R := R) (e := e))
220     exact LinearMap.ext fun x => add_smul (a : R) (b : R) x
221 
222 /-- Central scalar maps commute with every germ, including partial maps. -/
223 theorem centralMap_commute (a : Subring.center R) (f : End R e C) :
224     Commute (centralMap (R := R) (e := e) (C := C) a) f := by
225   have he : Hom.reducedMap (e := e)
226       (centralScalar (C := C) a) =
227       centralScalar (C := C ⧸ smallRadical R e C) a := by
228     apply LinearMap.ext
229     intro x
230     obtain ⟨x, rfl⟩ := Submodule.mkQ_surjective (smallRadical R e C) x
231     rfl
232   change PartialHom.comp (PartialHom.ofLinearMap _) f =
233     PartialHom.comp f (PartialHom.ofLinearMap _)
234   rw [he]
235   induction f using Quotient.inductionOn with | _ f =>
236   apply Quotient.sound
237   apply Representative.rel_of_eq
238   intro x hx hy
239   have hx' := f.pullback_le _ hx
240   change (a : R) • f.val.toFun _ = f.val.toFun _
241   have hm := f.val.toFun.map_smul (a : R) (⟨x, hx'⟩ : f.val.domain)
242   exact hm.symm

273 def leftMul (r : R) : R →ₗ[Rᵐᵒᵖ] R where

445     (U : Submodule Rᵐᵒᵖ (Section R e C)) (hU : LocSmall Rᵐᵒᵖ e U) : U = ⊥ := by
446   by_contra hn
447   let V := U.comap ofElement
448   let f : V →ₗ[Rᵐᵒᵖ] U :=
449     (ofElement.comp V.subtype).codRestrict U (fun x => x.property)
450   have hi : Function.Injective f := by
451     intro x y he
452     exact Subtype.ext (ofElement_injective hC (congrArg Subtype.val he))
453   exact hC.not_small_submodule (comap_ofElement_ne_bot hC hn)
454     (locSmall_iff_small.mp (hU.of_injective f hi))

456 /-- Over a domain, ordinary elementwise torsion passes from a critical
457 module to its section. The scalar product used is literally `i * s`. -/
458 theorem torsion [NoZeroDivisors R] [IsNoetherian Rᵐᵒᵖ C]
459     (hC : Critical Rᵐᵒᵖ e C)
460     (ht : ∀ c : C, ∃ s : R, s ≠ 0 ∧ op s • c = 0) :
461     ∀ w : Section R e C, ∃ s : R, s ≠ 0 ∧ op s • w = 0 := by
462   intro w
463   by_cases hw : w = 0
464   · obtain ⟨s, hs, _⟩ := ht 0
465     exact ⟨s, hs, by rw [hw, smul_zero]⟩
466   obtain ⟨i, c, hc, he⟩ := exists_smul_eq_ofElement hC w hw
467   have hi : i ≠ 0 := by
468     intro hi
469     rw [hi, op_zero, zero_smul] at he
470     exact hc (ofElement_injective hC (he.symm.trans (map_zero ofElement).symm))
471   obtain ⟨s, hs, hsc⟩ := ht c
472   refine ⟨i * s, mul_ne_zero hi hs, ?_⟩
473   rw [← right_action_assoc, he, ← map_smul, hsc, map_zero]
