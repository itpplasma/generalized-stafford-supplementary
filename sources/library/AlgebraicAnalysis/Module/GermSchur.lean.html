Schur's lemma for critical germs

Multiplication of germ endomorphisms is composition, with the left factor applied last. All module structures use left scalars; right modules are represented using the opposite ring.

These are new proofs of the interfaces in generalized-stafford's docs/categorical-localization.md, §§1–2, and .lake/research/active/formal-scope-BC.md, steps 5–6; no Lean declarations are extracted or duplicated from a downstream repository.

A nonzero map between critical modules of the same dimension is injective.

30/-- A nonzero map between critical modules of the same dimension is injective. -/
31theorem critical_injective_of_ne_zero (hC : Critical R e C) (hD : Critical R e D)
32 (f : C →ₗ[R] D) (hf : f ≠ 0) : Function.Injective f := by
33 rw [← LinearMap.ker_eq_bot]
34 by_contra hk
35 have hs : Small R e f.range :=
36 (small_linearEquiv_iff e f.quotKerEquivRange).mp (hC.small_quotient _ hk)
37 have hr : f.range = ⊥ := by
38 by_contra hr
39 exact hD.not_small_submodule hr hs
40 exact hf (LinearMap.range_eq_bot.mp hr)

For a critical target, a representative is zero as a germ exactly when its underlying map is zero.

44/-- For a critical target, a representative is zero as a germ exactly when
45its underlying map is zero. -/
46theorem mk_eq_zero_iff_of_critical (hD : Critical R e D)
47 (f : Representative R e C D) : mk f = 0 ↔ f.val.toFun = 0 := by
48 constructor
49 · intro hf
50 obtain ⟨U, hU, hUf, _, he⟩ := Quotient.exact hf
51 have hk : U ≤ (LinearMap.ker f.val.toFun).map f.val.domain.subtype := by
52 intro x hx
53 exact ⟨⟨x, hUf hx⟩, he ⟨x, hx⟩, rfl⟩
54 have hd := (hU.mono hk).comap f.val.domain.subtype
55 rw [Submodule.comap_map_eq_of_injective Subtype.val_injective] at hd
56 have hs := (small_linearEquiv_iff e f.val.toFun.quotKerEquivRange).mp hd
57 have hr : f.val.toFun.range = ⊥ := by
58 by_contra hr
59 exact hD.not_small_submodule hr hs
60 exact LinearMap.range_eq_bot.mp hr
61 · intro hf
62 apply Quotient.sound
63 apply Representative.rel_of_eq
64 intro x hx hy
65 change f.val.toFun _ = 0
66 rw [hf]; rfl

Nonzero germs between critical modules admit two-sided inverses.

76/-- Nonzero germs between critical modules admit two-sided inverses. -/
77theorem exists_inverse_of_critical (hC : Critical R e C) (hD : Critical R e D)
78 (f : PartialHom R e C D) (hf : f ≠ 0) :
79 ∃ g : PartialHom R e D C,
80 g.comp f = ofLinearMap LinearMap.id ∧ f.comp g = ofLinearMap LinearMap.id := by
81 induction f using Quotient.inductionOn with | _ f =>
82 have hd : f.val.domain ≠ ⊥ := by
83 intro hd
84 apply hf
85 apply (mk_eq_zero_iff_of_critical hD f).mpr
86 ext x
87 have hx : x = 0 := Subtype.ext (hd.le x.property)
88 rw [hx, map_zero, LinearMap.zero_apply]
89 have hi := critical_injective_of_ne_zero (hC.submodule _ hd) hD f.val.toFun
90 (fun hz => hf ((mk_eq_zero_iff_of_critical hD f).mpr hz))
91 let a := LinearEquiv.ofInjective f.val.toFun hi
92 let g : Representative R e D C :=
93 ⟨⟨f.val.toFun.range, f.val.domain.subtype.comp a.symm.toLinearMap⟩,
94 epic_of_ne_zero_of_critical hD (mk f) hf⟩
95 refine ⟨mk g, ?_, ?_⟩
96 · apply Quotient.sound
97 apply Representative.rel_of_eq
98 intro x hx hy
99 have hx' : x ∈ f.val.domain := f.pullback_le _ hx
100 exact congrArg Subtype.val (a.symm_apply_apply ⟨x, hx'⟩)
101 · apply Quotient.sound
102 apply Representative.rel_of_eq
103 intro x hx hy
104 have hx' : x ∈ f.val.toFun.range := g.pullback_le _ hx
105 exact congrArg Subtype.val (a.apply_symm_apply ⟨x, hx'⟩)

The Schur division ring of a Noetherian critical module. This is an explicit structure because criticality is a proposition, not a typeclass.

142/-- The Schur division ring of a Noetherian critical module. This is an
143explicit structure because criticality is a proposition, not a typeclass. -/
144noncomputable abbrev divisionRing [IsNoetherian R C] (hC : Critical R e C) :
145 DivisionRing (End R e C) := by
146 letI : Nontrivial (End R e C) := by
147 apply nontrivial_of_ne (1 : End R e C) 0
148 intro h
149 have hz : Hom.ofLinearMap (e := e) (LinearMap.id : C →ₗ[R] C) = 0 := by
150 rw [Hom.ofLinearMap_id]
151 exact h
152 have hs := (Hom.ofLinearMap_eq_zero_iff _).mp hz
153 rw [LinearMap.range_id] at hs
154 exact hC.not_small (hs.of_surjective (⊤ : Submodule R C).subtype
155 (fun x => ⟨⟨x, Submodule.mem_top⟩, rfl⟩))
156 exact DivisionRing.ofIsUnitOrEqZero fun f => by
157 by_cases hf : f = 0
158 · exact Or.inr hf
159 · obtain ⟨g, hgf, hfg⟩ := Hom.exists_inverse_of_critical hC hC f hf

Central ring scalars act as germ endomorphisms.

198/-- Central ring scalars act as germ endomorphisms. -/
199def centralMap : Subring.center R →+* End R e C where
200 toFun a := Hom.ofLinearMap (centralScalar a)
201 map_one' := by
202 have he : centralScalar (C := C) (1 : Subring.center R) = LinearMap.id :=
203 LinearMap.ext fun x => one_smul R x
204 change Hom.ofLinearMap _ = Hom.id
205 rw [he, Hom.ofLinearMap_id]

213 LinearMap.ext fun x => zero_smul R x
214 change Hom.ofLinearMap _ = 0

223theorem centralMap_commute (a : Subring.center R) (f : End R e C) :
224 Commute (centralMap (R := R) (e := e) (C := C) a) f := by

Left multiplication on the right regular module.

272/-- Left multiplication on the right regular module. -/
273def leftMul (r : R) : R →ₗ[Rᵐᵒᵖ] R where
274 toFun s := r * s
275 map_add' := mul_add r
276 map_smul' a s := (mul_assoc r s (unop a)).symm
277
278@[simp] theorem leftMul_one : leftMul (1 : R) = LinearMap.id := by ext; simp [leftMul]
279@[simp] theorem leftMul_zero : leftMul (0 : R) = 0 := by ext; simp [leftMul]
280theorem leftMul_add (r s : R) : leftMul (r + s) = leftMul r + leftMul s := by
281 ext; exact add_mul _ _ _
282theorem leftMul_mul (r s : R) : leftMul (r * s) = (leftMul r).comp (leftMul s) := by
283 ext; exact mul_assoc _ _ _
284
285instance : SMul Rᵐᵒᵖ (Section R e C) :=
286 ⟨fun r w => Hom.comp w (Hom.ofLinearMap (leftMul (unop r)))⟩
287
288instance : Module Rᵐᵒᵖ (Section R e C) where
289 one_smul w := by
290 change Hom.comp w (Hom.ofLinearMap (leftMul 1)) = w
291 rw [leftMul_one, Hom.ofLinearMap_id, Hom.comp_id]
292 mul_smul r s w := by
293 change Hom.comp w (Hom.ofLinearMap (leftMul (unop s * unop r))) = _
294 rw [leftMul_mul, Hom.ofLinearMap_comp, ← Hom.comp_assoc]
295 rfl
296 smul_add r w v := Hom.add_comp _ _ _
297 smul_zero r := show Hom.comp 0 _ = 0 from PartialHom.zero_comp _
298 add_smul r s w := by
299 change Hom.comp w (Hom.ofLinearMap (leftMul (unop r + unop s))) = _
300 rw [leftMul_add, Hom.ofLinearMap_add, Hom.comp_add]
301 rfl
302 zero_smul w := by
303 change Hom.comp w (Hom.ofLinearMap (leftMul 0)) = 0
304 rw [leftMul_zero, Hom.ofLinearMap_zero]
305 exact PartialHom.comp_zero w

456/-- Over a domain, ordinary elementwise torsion passes from a critical
457module to its section. The scalar product used is literally `i * s`. -/
458theorem torsion [NoZeroDivisors R] [IsNoetherian Rᵐᵒᵖ C]
459 (hC : Critical Rᵐᵒᵖ e C)
460 (ht : ∀ c : C, ∃ s : R, s ≠ 0 ∧ op s • c = 0) :
461 ∀ w : Section R e C, ∃ s : R, s ≠ 0 ∧ op s • w = 0 := by
462 intro w
463 by_cases hw : w = 0
464 · obtain ⟨s, hs, _⟩ := ht 0
465 exact ⟨s, hs, by rw [hw, smul_zero]⟩
466 obtain ⟨i, c, hc, he⟩ := exists_smul_eq_ofElement hC w hw
467 have hi : i ≠ 0 := by
468 intro hi

470 exact hc (ofElement_injective hC (he.symm.trans (map_zero ofElement).symm))
471 obtain ⟨s, hs, hsc⟩ := ht c
