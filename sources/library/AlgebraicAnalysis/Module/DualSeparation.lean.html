9 # Critical quotients and dual separation

27 /-- A non-small Noetherian module of deviation at most `e` has an
28 `e`-critical quotient. The quotient is obtained by maximality, not by
29 critical-submodule descent. -/
30 theorem DevLE.exists_critical_quotient [IsNoetherian R M]
31     (hM : DevLE R e M) (hne : ¬Small R e M) :
32     ∃ p : Submodule R M, Critical R e (M ⧸ p) := by
33   classical
34   have hbot : ¬Small R e (M ⧸ (⊥ : Submodule R M)) := by
35     intro hs
36     exact hne ((small_linearEquiv_iff e
37       ((⊥ : Submodule R M).quotEquivOfEqBot rfl)).mp hs)
38   obtain ⟨p, hp, hmax⟩ := wellFounded_gt.has_min
39     {p : Submodule R M | ¬Small R e (M ⧸ p)} ⟨⊥, hbot⟩
40   have hn : Nontrivial (M ⧸ p) := not_subsingleton_iff_nontrivial.mp
41     (fun hs => hp (by let := hs; exact small_of_subsingleton e))
42   refine ⟨p, hn, hM.quotient p, hp, ?_⟩
43   intro q hq
44   let t := q.comap p.mkQ
45   have hpt : p ≤ t := Submodule.le_comap_mkQ _ _
46   have heq : t.map p.mkQ = q :=
47     Submodule.map_comap_eq_of_surjective (Submodule.mkQ_surjective p) q
48   have hlt : p < t := lt_of_le_of_ne hpt (by
49     intro he
50     have he' := congrArg (Submodule.map p.mkQ) he
51     rw [Submodule.mkQ_map_self, heq] at he'
52     exact hq he'.symm)
53   have ht : Small R e (M ⧸ t) := by
54     by_contra ht
55     exact hmax t ht hlt
56   have hs := (small_linearEquiv_iff e
57     (Submodule.quotientQuotientEquivQuotient p t hpt)).mpr ht
58   rwa [heq] at hs

64 /-- Evaluation at a germ is linear over the target's germ endomorphisms. -/
65 def compLinear (x : Hom R e M C) : End R e C →ₗ[End R e C] Hom R e M C where
66   toFun h := comp h x
67   map_add' h g := add_comp h g x
68   map_smul' a h := comp_assoc a h x

70 /-- A nonzero germ between critical modules identifies its dual Hom with
71 the target's Schur ring by composition. -/
72 theorem compLinear_bijective_of_critical [IsNoetherian R M]
73     (hM : Critical R e M) (hC : Critical R e C) (x : Hom R e M C) (hx : x ≠ 0) :
74     Function.Bijective (compLinear x) := by
75   obtain ⟨y, hyx, hxy⟩ := exists_inverse_of_critical hM hC x hx
76   constructor
77   · intro f g he
78     have h := congrArg (fun z : Hom R e M C => comp z y) he
79     change comp (comp f x) y = comp (comp g x) y at h
80     simpa only [comp_assoc, hxy, comp_id] using h
81   · intro f
82     refine ⟨comp f y, ?_⟩
83     change comp (comp f y) x = f
84     rw [comp_assoc, hyx, comp_id]

86 /-- For critical source and target, the Schur Hom space is finite. -/
87 theorem finite_of_critical [IsNoetherian R M]
88     (hM : Critical R e M) (hC : Critical R e C) :
89     Module.Finite (End R e C) (Hom R e M C) := by
90   classical
91   by_cases hs : Subsingleton (Hom R e M C)
92   · let := hs
93     infer_instance
94   · have := not_subsingleton_iff_nontrivial.mp hs
95     obtain ⟨x, hx⟩ := exists_ne (0 : Hom R e M C)
96     exact Module.Finite.of_surjective (compLinear x)
97       (compLinear_bijective_of_critical hM hC x hx).surjective

99 /-- A critical factor contributes at most one Schur dimension. -/
100 theorem finrank_le_one_of_critical [IsNoetherian R M]
101     (hM : Critical R e M) (hC : Critical R e C) :
102     letI := End.divisionRing hC
103     Module.finrank (End R e C) (Hom R e M C) ≤ 1 := by
104   classical
105   let := End.divisionRing hC
106   by_cases hs : Subsingleton (Hom R e M C)
107   · let := hs
108     rw [Module.finrank_zero_of_subsingleton]
109     exact Nat.zero_le _
110   · have := not_subsingleton_iff_nontrivial.mp hs
111     obtain ⟨x, hx⟩ := exists_ne (0 : Hom R e M C)
112     have h := LinearMap.finrank_le_finrank_of_surjective (f := compLinear x)
113       (compLinear_bijective_of_critical hM hC x hx).surjective
114     simpa only [Module.finrank_self] using h

116 /-- A germ into a critical module is epic exactly when it is nonzero. -/
117 theorem epic_iff_ne_zero_of_critical (hC : Critical R e C) (x : Hom R e M C) :
118     Epic x ↔ x ≠ 0 := by
119   refine ⟨?_, PartialHom.epic_of_ne_zero_of_critical (critical_quotient_smallRadical hC) x⟩
120   intro hx hz
121   rw [hz] at hx
122   have hs : Small R e (C ⧸ smallRadical R e C) := by
123     change Dense R e (0 : Representative R e
124       (M ⧸ smallRadical R e M) (C ⧸ smallRadical R e C)).val.toFun.range at hx
125     have hr : (0 : Representative R e
126       (M ⧸ smallRadical R e M) (C ⧸ smallRadical R e C)).val.toFun.range = ⊥ :=
127       LinearMap.range_zero
128     rw [hr] at hx
129     exact (small_linearEquiv_iff e
130       ((⊥ : Submodule R (C ⧸ smallRadical R e C)).quotEquivOfEqBot rfl)).mp hx
131   exact (critical_quotient_smallRadical hC).not_small hs

133 /-- Critical-target dual separation: testing against the target's own
134 Schur space suffices, for a germ with any source. -/
135 theorem epic_iff_injective_comp_of_critical (hC : Critical R e C) (x : Hom R e M C) :
136     Epic x ↔ Function.Injective (fun h : End R e C => comp h x) := by
137   refine ⟨fun hx g h he => hx.right_cancel g h he, ?_⟩
138   intro hi
139   apply (epic_iff_ne_zero_of_critical hC x).mpr
140   intro hz
141   have he : comp (id : End R e C) x = comp (0 : End R e C) x := by
142     rw [hz]
143     exact (PartialHom.comp_zero _).trans (PartialHom.zero_comp _).symm
144   have hid := hi he
145   let : DivisionRing (End R e C) := End.divisionRing hC
146   exact one_ne_zero hid
