31 # Central pencils over a division ring
32 
33 Let `E` be a division ring and `κ : k →+* E` a field embedding with central
34 image.  If `v₁, …, vₙ` are `E`-linearly independent in a left `E`-module and
35 `w₁, …, wₙ` are arbitrary, then `vᵢ + κ(c) wᵢ` stay independent for all but
36 finitely many `c ∈ k`.
37 
38 No determinants are used (they do not exist over a noncommutative `E`).  A
39 bad `c ≠ 0` produces an eigenvector of the `E`-linear map `π ∘ W` for the
40 central eigenvalue `-κ(c)⁻¹`, where `π` is a left inverse of the injective map
41 `P`.  Eigenvectors for distinct central eigenvalues are independent (this is
42 where centrality is used), so only finitely many bad values exist.
43 
44 We prove the statement in the form of a pair of linear maps `P, W : D → Z`
45 with `D` finite-dimensional and `P` injective (`pencil_ker_finite`), and in the
46 form of families (`division_pencil_finite`).
47 -/

57 /-- Eigenvectors of an `E`-linear endomorphism for pairwise distinct central
58 eigenvalues are `E`-linearly independent. -/
59 theorem central_eigenvectors_linearIndependent {ι : Type*} (A : D →ₗ[E] D) (μ : ι → E)
60     (hμ : Function.Injective μ) (hc : ∀ i (a : E), μ i * a = a * μ i)
61     (x : ι → D) (hx0 : ∀ i, x i ≠ 0) (hA : ∀ i, A (x i) = μ i • x i) :
62     LinearIndependent E x := by
63   classical
64   rw [linearIndependent_iff']
65   intro s
66   induction s using Finset.induction_on with
67   | empty => simp
68   | insert j s hj ih =>
69     intro g hg
70     have key : ∀ i, A (g i • x i) - μ j • (g i • x i) = (g i * (μ i - μ j)) • x i := by
71       intro i
72       rw [map_smul, hA, smul_smul, smul_smul, hc j, mul_sub, sub_smul]
73     have h1 : ∑ i ∈ s, (g i * (μ i - μ j)) • x i = 0 := by
74       have h2 : A (∑ i ∈ insert j s, g i • x i) - μ j • (∑ i ∈ insert j s, g i • x i) = 0 := by
75         rw [hg, map_zero, smul_zero, sub_zero]
76       rw [map_sum, Finset.smul_sum, ← Finset.sum_sub_distrib] at h2
77       simp only [key] at h2
78       rw [Finset.sum_insert hj, sub_self, mul_zero, zero_smul, zero_add] at h2
79       exact h2
80     have hs : ∀ i ∈ s, g i = 0 := by
81       intro i hi
82       have h3 := ih (fun i => g i * (μ i - μ j)) h1 i hi
83       have hne : μ i - μ j ≠ 0 := by
84         rw [sub_ne_zero]
85         intro h
86         exact hj (hμ h ▸ hi)
87       exact (mul_eq_zero.mp h3).resolve_right hne
88     have hgs : ∑ i ∈ s, g i • x i = 0 :=
89       Finset.sum_eq_zero fun i hi => by rw [hs i hi, zero_smul]
90     rw [Finset.sum_insert hj, hgs, add_zero] at hg
91     have hgj : g j = 0 := (smul_eq_zero.mp hg).resolve_right (hx0 j)
92     intro i hi
93     rcases Finset.mem_insert.mp hi with rfl | hi
94     · exact hgj
95     · exact hs i hi
96 
97 end Eigen

102 /-- If `P : D → Z` is injective and `D` is
103 finite-dimensional, then `P + κ(c) W` has a nonzero kernel for only finitely
104 many `c`. -/
105 theorem pencil_ker_finite {D Z : Type*} [AddCommGroup D] [Module E D] [Module.Finite E D]
106     [AddCommGroup Z] [Module E Z] (P W : D →ₗ[E] Z) (hP : Function.Injective P) :
107     {c : k | ∃ d : D, d ≠ 0 ∧ P d + κ c • W d = 0}.Finite := by
108   classical
109   obtain ⟨π, hπ⟩ := P.exists_leftInverse_of_injective (LinearMap.ker_eq_bot.mpr hP)
110   have hπP : ∀ d, π (P d) = d := fun d => LinearMap.congr_fun hπ d
111   set A : D →ₗ[E] D := π.comp W
112   set Bad : Set k := {c : k | c ≠ 0 ∧ ∃ d : D, d ≠ 0 ∧ P d + κ c • W d = 0}
113   have hsub : {c : k | ∃ d : D, d ≠ 0 ∧ P d + κ c • W d = 0} ⊆ Bad := by
114     rintro c ⟨d, hd, hPd⟩
115     refine ⟨?_, d, hd, hPd⟩
116     rintro rfl
117     apply hd
118     apply hP
119     rw [map_zero]
120     simpa using hPd
121   refine Set.Finite.subset ?_ hsub
122   let x : Bad → D := fun c => c.2.2.choose
123   have hx0 : ∀ c, x c ≠ 0 := fun c => c.2.2.choose_spec.1
124   have hxP : ∀ c : Bad, P (x c) + κ c • W (x c) = 0 := fun c => c.2.2.choose_spec.2
125   let μ : Bad → E := fun c => κ (-(c : k)⁻¹)
126   have hμ : Function.Injective μ := by
127     intro c c' h
128     apply Subtype.ext
129     have := κ.injective h
130     simpa using this
131   have hA : ∀ c : Bad, A (x c) = μ c • x c := by
132     intro c
133     have h1 : x c + κ c • A (x c) = 0 := by
134       have := congrArg π (hxP c)
135       rwa [map_add, map_smul, hπP, map_zero] at this
136     have hc0 : (c : k) ≠ 0 := c.2.1
137     have h2 : A (x c) = κ ((c : k)⁻¹) • (κ c • A (x c)) := by
138       rw [smul_smul, ← map_mul, inv_mul_cancel₀ hc0, map_one, one_smul]
139     rw [h2, eq_neg_of_add_eq_zero_right h1, smul_neg, ← neg_smul, ← map_neg]
140   have hli : LinearIndependent E x :=
141     central_eigenvectors_linearIndependent A μ hμ (fun c a => hκ _ a) x hx0 hA
142   have : Finite Bad := hli.finite_of_isNoetherian
143   exact Set.toFinite Bad

145 /-- If `v` is
146 `E`-linearly independent and `w` is arbitrary, then `v + κ(c) w` is
147 `E`-linearly independent for all but finitely many `c ∈ k`. -/
148 theorem division_pencil_finite {Z : Type*} [AddCommGroup Z] [Module E Z]
149     {ι : Type*} [Finite ι] (v w : ι → Z) (hv : LinearIndependent E v) :
150     {c : k | ¬ LinearIndependent E (fun i => v i + κ c • w i)}.Finite := by
151   refine (pencil_ker_finite κ hκ (Finsupp.linearCombination E v)
152     (Finsupp.linearCombination E w) hv).subset ?_
153   intro c hc
154   simp only [Set.mem_ofPred_eq, linearIndependent_iff, not_forall] at hc
155   obtain ⟨l, hl, hl0⟩ := hc
156   refine ⟨l, hl0, ?_⟩
157   rw [← hl, Finsupp.linearCombination_apply, Finsupp.linearCombination_apply,
158     Finsupp.linearCombination_apply, Finsupp.smul_sum, ← Finsupp.sum_add]
159   refine Finsupp.sum_congr fun i _ => ?_
160   rw [smul_add, smul_smul, smul_smul, hκ]
