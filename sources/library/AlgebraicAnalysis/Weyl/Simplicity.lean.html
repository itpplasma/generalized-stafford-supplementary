10 # Simplicity of the Weyl algebras in characteristic zero
11 
12 The pair stage `B[x][∂; d/dx]` of a simple `k`-algebra `B` is simple when
13 `k` has characteristic zero.  The argument is the classical one for the Weyl
14 algebra.  Let `I` be a nonzero two-sided ideal.
15 
16 * Commutation with the coordinate `x` lowers the `∂`-degree by one, with
17   leading coefficient a nonzero integer multiple, so `I` meets the coordinate
18   stage `B[x]`.
19 * Commutation with `∂` lowers the `x`-degree by one in the same way, so `I`
20   meets the base `B`.
21 * `I ∩ B` is a nonzero two-sided ideal of `B`, hence contains `1`.
22 
23 Induction over the tower `IteratedPairStage k n` and transport along the
24 presentation equivalence give `IsSimpleRing (WeylAlg k n)`.
25 
26 Provenance: this file is new in `AlgebraicAnalysis`.  The tower and
27 presentation it applies are the copied Weyl development of this package.
28 -/

40 /-- A nonzero natural number is not a left zero divisor in a `k`-algebra over a
41 field of characteristic zero. -/
42 theorem eq_zero_of_natCast_mul (k : Type*) [Field k] [CharZero k] {A : Type*} [Ring A]
43     [Algebra k A] {m : ℕ} (hm : m ≠ 0) {a : A}
44     (h : (m : A) * a = 0) : a = 0 := by
45   have hc : (m : k) ≠ 0 := Nat.cast_ne_zero.mpr hm
46   calc a = algebraMap k A ((m : k)⁻¹) * ((m : A) * a) := by
47         rw [← mul_assoc, ← map_natCast (algebraMap k A) m, ← map_mul,
48           inv_mul_cancel₀ hc, map_one, one_mul]
49     _ = 0 := by rw [h, mul_zero]
50 
51 /-- A nonzero natural number is not a right zero divisor in a `k`-algebra over a
52 field of characteristic zero. -/
53 theorem eq_zero_of_mul_natCast (k : Type*) [Field k] [CharZero k] {A : Type*} [Ring A]
54     [Algebra k A] {m : ℕ} (hm : m ≠ 0) {a : A}
55     (h : a * (m : A) = 0) : a = 0 := by
56   have hc : (m : k) ≠ 0 := Nat.cast_ne_zero.mpr hm
57   calc a = (a * (m : A)) * algebraMap k A ((m : k)⁻¹) := by
58         rw [← map_natCast (algebraMap k A) m, mul_assoc, ← map_mul,
59           mul_inv_cancel₀ hc, map_one, mul_one]
60     _ = 0 := by rw [h, zero_mul]
61 
62 end Torsion

77 /-- The central coordinate commutes with every element of the coordinate
78 stage.  This is the commutativity of `X` with `B[X]`, transported through the
79 normal form. -/
80 theorem variable_mul_coordinate_comm (s : CoordinateStage (B := B)) :
81     normalVariable (zeroDerivation : OreDivisionDerivation B) * s =
82       s * normalVariable (zeroDerivation : OreDivisionDerivation B) := by
83   obtain ⟨p, rfl⟩ := normalForm_surjective zeroDerivation s
84   rw [normalVariable, ← normalForm_mul_zeroDerivation, ← normalForm_mul_zeroDerivation,
85     (Polynomial.commute_X p).eq]

94 /-- The momentum and coordinate of a Weyl pair satisfy `∂ x = x ∂ + 1`. -/
95 theorem pairMomentum_mul_pairCoordinate :
96     pairMomentum (B := B) * pairCoordinate (B := B) =
97       pairCoordinate (B := B) * pairMomentum (B := B) + 1 := by
98   have h := normalVariable_mul_coefficient (outerD (B := B))
99     (normalVariable (zeroDerivation : OreDivisionDerivation B))
100   rw [coordinateDerivation_variable, map_one] at h
101   exact h

104 /-- Iterated commutator of the momentum with the coordinate:
105 `∂^(j+1) x = x ∂^(j+1) + (j+1) ∂^j`. -/
106 theorem pairMomentum_pow_mul_pairCoordinate (j : ℕ) :
107     pairMomentum (B := B) ^ (j + 1) * pairCoordinate (B := B) =
108       pairCoordinate (B := B) * pairMomentum (B := B) ^ (j + 1) +
109         ((j + 1 : ℕ) : PairStage (B := B)) * pairMomentum (B := B) ^ j := by
110   induction j with
111   | zero => simpa using pairMomentum_mul_pairCoordinate (B := B)
112   | succ j ih =>
113     have hc : pairMomentum (B := B) * ((j + 1 : ℕ) : PairStage (B := B)) =
114         ((j + 1 : ℕ) : PairStage (B := B)) * pairMomentum (B := B) :=
115       (Nat.cast_commute (j + 1) _).symm.eq
116     calc pairMomentum (B := B) ^ (j + 1 + 1) * pairCoordinate (B := B)
117         = pairMomentum (B := B) * (pairMomentum (B := B) ^ (j + 1) *
118             pairCoordinate (B := B)) := by rw [pow_succ', mul_assoc]
119       _ = pairMomentum (B := B) * (pairCoordinate (B := B) *
120             pairMomentum (B := B) ^ (j + 1) +
121             ((j + 1 : ℕ) : PairStage (B := B)) * pairMomentum (B := B) ^ j) := by
122           rw [ih]
123       _ = (pairMomentum (B := B) * pairCoordinate (B := B)) *
124             pairMomentum (B := B) ^ (j + 1) +
125             ((j + 1 : ℕ) : PairStage (B := B)) * (pairMomentum (B := B) *
126               pairMomentum (B := B) ^ j) := by
127           rw [mul_add, ← mul_assoc, ← mul_assoc, hc,
128             mul_assoc (((j + 1 : ℕ) : PairStage (B := B))) pairMomentum]
129       _ = (pairCoordinate (B := B) * pairMomentum (B := B) + 1) *
130             pairMomentum (B := B) ^ (j + 1) +
131             ((j + 1 : ℕ) : PairStage (B := B)) * pairMomentum (B := B) ^ (j + 1) := by
132           rw [pairMomentum_mul_pairCoordinate,
133             ← pow_succ' (pairMomentum (B := B)) j]
134       _ = pairCoordinate (B := B) * pairMomentum (B := B) ^ (j + 1 + 1) +
135             ((j + 1 + 1 : ℕ) : PairStage (B := B)) * pairMomentum (B := B) ^ (j + 1) := by
136           rw [add_mul, one_mul, mul_assoc,
137             ← pow_succ' (pairMomentum (B := B)) (j + 1), Nat.cast_succ (j + 1), add_mul,
138             one_mul, add_assoc, add_comm (pairMomentum (B := B) ^ (j + 1))]

141 /-- Commutation with the coordinate lowers the momentum degree of a normal
142 form: `x p(∂) - p(∂) x = -p'(∂)`. -/
143 theorem pairCoordinate_comm_normalForm (p : Polynomial (CoordinateStage (B := B))) :
144     pairCoordinate (B := B) * normalForm (outerD (B := B)) p -
145         normalForm (outerD (B := B)) p * pairCoordinate (B := B) =
146       normalForm (outerD (B := B)) (-derivative p) := by
147   induction p using Polynomial.induction_on' with
148   | add p q hp hq =>
149     rw [normalForm_add, derivative_add, neg_add, normalForm_add, ← hp, ← hq]
150     noncomm_ring
151   | monomial j s =>
152     rcases j with _ | j
153     · simp only [normalForm_monomial, pow_zero, mul_one, Nat.cast_zero, mul_zero,
154         derivative_monomial, monomial_zero_right, normalForm_zero, neg_zero]
155       exact sub_eq_zero.mpr (pairCoordinate_comm_coefficient s)
156     · have hx : pairCoordinate (B := B) * normalCoefficient (outerD (B := B)) s =
157           normalCoefficient (outerD (B := B)) s * pairCoordinate (B := B) :=
158         pairCoordinate_comm_coefficient s
159       rw [derivative_monomial, normalForm_neg, normalForm_monomial, normalForm_monomial,
160         Nat.add_sub_cancel, map_mul, map_natCast (normalCoefficient (outerD (B := B))),
161         show normalVariable (outerD (B := B)) = pairMomentum (B := B) from rfl]
162       have h1 : pairCoordinate (B := B) * (normalCoefficient (outerD (B := B)) s *
163           pairMomentum (B := B) ^ (j + 1)) =
164           normalCoefficient (outerD (B := B)) s *
165             (pairCoordinate (B := B) * pairMomentum (B := B) ^ (j + 1)) := by
166         rw [← mul_assoc, hx, mul_assoc]
167       have h2 : normalCoefficient (outerD (B := B)) s * pairMomentum (B := B) ^ (j + 1) *
168           pairCoordinate (B := B) =
169           normalCoefficient (outerD (B := B)) s * (pairMomentum (B := B) ^ (j + 1) *
170             pairCoordinate (B := B)) := by
171         rw [mul_assoc]
172       rw [h1, h2, pairMomentum_pow_mul_pairCoordinate j, mul_add, mul_assoc
173         (normalCoefficient (outerD (B := B)) s)]
174       abel
175 
176 end Stage

195 /-- **Coordinate descent.** A nonzero two-sided ideal of the pair stage meets the
196 coordinate stage `B[x]` in a nonzero element: commutation with the coordinate
197 `x` lowers the momentum degree of a normal form, with a nonzero integer
198 multiple of the leading coefficient. -/
199 theorem exists_coordinate_mem (k : Type*) [Field k] [CharZero k] [Algebra k B]
200     (I : Ideal (PairStage (B := B))) (hI : I.IsTwoSided) (hne : I ≠ ⊥) :
201     ∃ s : CoordinateStage (B := B), s ≠ 0 ∧
202       normalCoefficient (outerD (B := B)) s ∈ I := by
203   have key : ∀ d : ℕ, ∀ p : Polynomial (CoordinateStage (B := B)), p ≠ 0 →
204       p.natDegree ≤ d → normalForm (outerD (B := B)) p ∈ I →
205       ∃ s : CoordinateStage (B := B), s ≠ 0 ∧
206         normalCoefficient (outerD (B := B)) s ∈ I := by
207     intro d
208     induction d with
209     | zero =>
210       intro p hp hdeg hpI
211       refine ⟨p.coeff 0, ?_, ?_⟩
212       · intro h
213         apply hp
214         rw [eq_C_of_natDegree_le_zero hdeg, h, C_0]
215       · have hC : normalForm (outerD (B := B)) p =
216             normalCoefficient (outerD (B := B)) (p.coeff 0) := by
217           rw [← normalForm_C, ← eq_C_of_natDegree_le_zero hdeg]
218         rw [← hC]
219         exact hpI
220     | succ d ih =>
221       intro p hp hdeg hpI
222       by_cases hle : p.natDegree ≤ d
223       · exact ih p hp hle hpI
224       · have hnd : p.natDegree = d + 1 := by omega
225         have hlead : p.coeff (d + 1) ≠ 0 := by
226           rw [← hnd]
227           exact leadingCoeff_ne_zero.mpr hp
228         have hder : (derivative p).coeff d ≠ 0 := by
229           rw [coeff_derivative, ← Nat.cast_succ]
230           exact fun h => hlead
231             (coordinateStage_eq_zero_of_mul_natCast k (Nat.succ_ne_zero d) h)
232         have hne' : -(derivative p) ≠ 0 := by
233           refine neg_ne_zero.mpr ?_
234           intro h
235           apply hder
236           rw [h, coeff_zero]
237         have hmem : normalForm (outerD (B := B)) (-(derivative p)) ∈ I := by
238           rw [← pairCoordinate_comm_normalForm]
239           exact I.sub_mem (Ideal.mul_mem_left I _ hpI) (hI.mul_mem_of_left _ hpI)
240         have hdeg' : (-(derivative p)).natDegree ≤ d := by
241           rw [natDegree_neg]
242           have := natDegree_derivative_le p
243           omega
244         exact ih _ hne' hdeg' hmem
245   obtain ⟨r, hrI, hr0⟩ := Submodule.exists_mem_ne_zero_of_ne_bot hne
246   obtain ⟨p, rfl⟩ := normalForm_surjective (outerD (B := B)) r
247   have hp : p ≠ 0 := by
248     rintro rfl
249     exact hr0 (normalForm_zero _)
250   exact key p.natDegree p hp le_rfl hrI

253 /-- **Base descent.** If `B[x]` contains a nonzero element of a two-sided ideal of
254 the pair stage, then the base `B` contains a nonzero element of the same ideal
255 (after the scalar embedding): commutation with `∂` lowers the `x`-degree. -/
256 theorem exists_base_mem (k : Type*) [Field k] [CharZero k] [Algebra k B]
257     (I : Ideal (PairStage (B := B))) (hI : I.IsTwoSided) :
258     ∀ d : ℕ, ∀ q : Polynomial B, q ≠ 0 → q.natDegree ≤ d →
259       normalCoefficient (outerD (B := B))
260           (normalForm (zeroDerivation : OreDivisionDerivation B) q) ∈ I →
261       ∃ b : B, b ≠ 0 ∧
262         normalCoefficient (outerD (B := B))
263           (normalCoefficient (zeroDerivation : OreDivisionDerivation B) b) ∈ I := by
264   intro d
265   induction d with
266   | zero =>
267     intro q hq hdeg hqI
268     refine ⟨q.coeff 0, ?_, ?_⟩
269     · intro h
270       apply hq
271       rw [eq_C_of_natDegree_le_zero hdeg, h, C_0]
272     · have h0 : normalForm (zeroDerivation : OreDivisionDerivation B) q =
273           normalCoefficient (zeroDerivation : OreDivisionDerivation B) (q.coeff 0) := by
274         rw [← normalForm_C, ← eq_C_of_natDegree_le_zero hdeg]
275       rw [h0] at hqI
276       exact hqI
277   | succ d ih =>
278     intro q hq hdeg hqI
279     by_cases hle : q.natDegree ≤ d
280     · exact ih q hq hle hqI
281     · have hnd : q.natDegree = d + 1 := by omega
282       have hlead : q.coeff (d + 1) ≠ 0 := by
283         rw [← hnd]
284         exact leadingCoeff_ne_zero.mpr hq
285       have hder : (derivative q).coeff d ≠ 0 := by
286         rw [coeff_derivative, ← Nat.cast_succ]
287         exact fun h => hlead (eq_zero_of_mul_natCast k (Nat.succ_ne_zero d) h)
288       have hne' : derivative q ≠ 0 := fun h => hder (by rw [h, coeff_zero])
289       have h := normalVariable_mul_coefficient (outerD (B := B))
290         (normalForm (zeroDerivation : OreDivisionDerivation B) q)
291       have e := eq_sub_of_add_eq' h.symm
292       rw [coordinateDerivation_normalForm] at e
293       have hmem : normalCoefficient (outerD (B := B))
294           (normalForm (zeroDerivation : OreDivisionDerivation B) (derivative q)) ∈ I := by
295         rw [e]
296         exact I.sub_mem (Ideal.mul_mem_left I _ hqI) (hI.mul_mem_of_left _ hqI)
297       have hdeg' : (derivative q).natDegree ≤ d := by
298         have := natDegree_derivative_le q
299         omega
300       exact ih _ (by
301         intro h0
302         exact hne' h0) hdeg' hmem

305 /-- **The pair stage is simple.** If the coefficient ring `B` is simple and `k`
306 has characteristic zero, the pair stage `B[x][∂; d/dx]` is simple. -/
307 theorem isSimpleRing_pairStage (k : Type*) [Field k] [CharZero k] {B : Type*} [Ring B]
308     [Algebra k B] [IsSimpleRing B] : IsSimpleRing (PairStage (B := B)) := by
309   have hB : ∀ J : Ideal B, J.IsTwoSided → J = ⊥ ∨ J = ⊤ :=
310     (isSimpleRing_iff_isTwoSided_imp.mp inferInstance).2
311   haveI : Nontrivial (CoordinateStage (B := B)) := nontrivial_coordinateStage
312   haveI : Nontrivial (PairStage (B := B)) := nontrivial_pairStage
313   rw [isSimpleRing_iff_isTwoSided_imp]
314   refine ⟨inferInstance, fun I hI => ?_⟩
315   by_cases hbot : I = ⊥
316   · exact Or.inl hbot
317   right
318   obtain ⟨s, hs0, hsI⟩ := exists_coordinate_mem k I hI hbot
319   obtain ⟨q, rfl⟩ := normalForm_surjective zeroDerivation s
320   have hq0 : q ≠ 0 := by
321     rintro rfl
322     exact hs0 (normalForm_zero _)
323   obtain ⟨b, hb0, hbI⟩ := exists_base_mem k I hI q.natDegree q hq0 le_rfl hsI
324   have hJ : (Ideal.comap (pairCoefficient (B := B)) I).IsTwoSided := by
325     refine ⟨fun {a} c ha => ?_⟩
326     show pairCoefficient (B := B) (a * c) ∈ I
327     rw [map_mul]
328     exact hI.mul_mem_of_left _ ha
329   rcases hB _ hJ with h | h
330   · exfalso
331     apply hb0
332     have hbJ : b ∈ Ideal.comap (pairCoefficient (B := B)) I := hbI
333     rw [h] at hbJ
334     simpa using hbJ
335   · have h1 : (1 : B) ∈ Ideal.comap (pairCoefficient (B := B)) I := by
336       rw [h]
337       exact Submodule.mem_top
338     have h2 : pairCoefficient (B := B) 1 ∈ I := h1
339     rw [map_one] at h2
340     exact Ideal.eq_top_of_isUnit_mem I h2 isUnit_one

350 /-- The iterated pair stages over a field of characteristic zero are simple. -/
351 theorem isSimpleRing_iteratedPairStage (k : Type u) [Field k] [CharZero k] (n : ℕ) :
352     IsSimpleRing (IteratedPairStage k n) := by
353   induction n with
