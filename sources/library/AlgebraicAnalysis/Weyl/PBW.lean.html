11 /-!
12 # Ordered PBW basis from Ore normal forms
13 
14 The checked iterated Ore normal form gives a scalar-linear equivalence with
15 the commutative polynomial symbol space.  Transport across the independently
16 proved presented/iterated algebra equivalence yields a free-module basis
17 indexed by commutative monomials, without using monomial independence in the
18 quotient presentation.  Recursive all-degree formulas identify those basis
19 vectors with ordered products of the named Weyl generators.
20 -/

42 /-- Reindex the two newest symbol variables followed by the old variables to
43 the standard newest-first coordinate/momentum indexing. -/
44 def phaseSuccEquiv (n : Nat) :
45     ((PUnit.{1} ⊕ PUnit.{1}) ⊕ PhaseVar n) ≃ PhaseVar (n + 1) where
46   toFun
47     | .inl (.inl _) => .inr 0
48     | .inl (.inr _) => .inl 0
49     | .inr (.inl i) => .inl i.succ
50     | .inr (.inr i) => .inr i.succ
51   invFun
52     | .inl i => Fin.cases (.inl (.inr PUnit.unit))
53         (fun j => .inr (.inl j)) i
54     | .inr i => Fin.cases (.inl (.inl PUnit.unit))
55         (fun j => .inr (.inr j)) i
56   left_inv x := by
57     rcases x with (⟨⟨u⟩ | ⟨u⟩⟩ | ⟨i | i⟩) <;> rfl
58   right_inv x := by
59     rcases x with (i | i)
60     · exact Fin.cases rfl (fun _ => rfl) i
61     · exact Fin.cases rfl (fun _ => rfl) i

212 /-- Ordered monomials in the iterated tower: all older pairs occur first, and
213 within each pair the coordinate power precedes the momentum power. -/
214 def iteratedOrderedMonomial :
215     (n : ℕ) → (Fin n → ℕ) → (Fin n → ℕ) → IteratedPairStage k n
216   | 0, _, _ => 1
217   | n + 1, a, p =>
218       stageEmbedding k n
219           (iteratedOrderedMonomial n (fun i => a i.succ) (fun i => p i.succ)) *
220         stageCoordinate k n ^ a 0 * stageMomentum k n ^ p 0

483 def presentedCoefficientOrdered (n a p : ℕ) (z : PresentedWeyl k n) :
484     PresentedWeyl k (n + 1) :=
485   previousWeylEmbedding k n z * presentedCoordinate k n ^ a *
486     presentedMomentum k n ^ p
487 
488 /-- Normal-form coordinates of an arbitrary old presentation element followed
489 by powers of the newest named coordinate and momentum. -/
490 theorem presentedNormalFormLinearEquiv_previous_ordered
491     (n a p : ℕ) (z : PresentedWeyl k n) :
492     presentedNormalFormLinearEquiv k (n + 1)
493         (presentedCoefficientOrdered k n a p z) =
494       MvPolynomial.X (.inr (0 : Fin (n + 1))) ^ p *
495         MvPolynomial.X (.inl (0 : Fin (n + 1))) ^ a *
496         MvPolynomial.rename oldIndex
497           (presentedNormalFormLinearEquiv k n z) := by
498   letI : Algebra k (IteratedPairStage k n) := iteratedPairStageAlgebra k n
499   letI : Algebra k (CoordinateStage (B := IteratedPairStage k n)) :=
500     coordinateStageAlgebra
501   letI : Algebra k (PairStage (B := IteratedPairStage k n)) := pairStageAlgebra
502   rw [presentedCoefficientOrdered]
503   rw [presentedNormalFormLinearEquiv, LinearEquiv.trans_apply,
504     AlgEquiv.toLinearEquiv_apply]
505   have hmap : presentedIteratedEquiv k (n + 1)
506         (previousWeylEmbedding k n z *
507           presentedCoordinate k n ^ a * presentedMomentum k n ^ p) =
508       presentedIteratedEquiv k (n + 1) (previousWeylEmbedding k n z) *
509         presentedIteratedEquiv k (n + 1) (presentedCoordinate k n) ^ a *
510         presentedIteratedEquiv k (n + 1) (presentedMomentum k n) ^ p := by
511     simp only [map_mul, map_pow]
512   rw [hmap]
513   change iteratedNormalFormLinearEquiv k (n + 1)
514       (presentedToIterated k (n + 1) (previousWeylEmbedding k n z) *
515         presentedToIterated k (n + 1) (presentedCoordinate k n) ^ a *
516         presentedToIterated k (n + 1) (presentedMomentum k n) ^ p) = _
517   have hprevious := DFunLike.congr_fun (presentedToIterated_previous k n) z
518   rw [AlgHom.comp_apply, AlgHom.comp_apply] at hprevious
519   rw [hprevious, presentedToIterated_coordinate,
520     presentedToIterated_momentum]
521   change iteratedNormalFormLinearEquiv k (n + 1)
522     (iteratedCoefficientOrdered k n a p (presentedToIterated k n z)) = _
523   rw [iteratedNormalFormLinearEquiv_coefficient_ordered]
524   rfl

526 /-- Exact normal-ordering formula for two newest-pair blocks.  Every correction
527 term lowers both the coordinate and momentum exponents by the same amount. -/
528 theorem presentedCoefficientOrdered_mul
529     (n a b c d : ℕ) (z₁ z₂ : PresentedWeyl k n) :
530     presentedCoefficientOrdered k n a b z₁ *
531         presentedCoefficientOrdered k n c d z₂ =
532       ∑ i ∈ Finset.range (b + 1),
533         if i ≤ c then
534           (b.choose i * c.descFactorial i) •
535             presentedCoefficientOrdered k n (a + c - i) (b + d - i) (z₁ * z₂)
536         else 0 := by
537   let X := presentedCoordinate k n
538   let P := presentedMomentum k n
539   let Z₁ := previousWeylEmbedding k n z₁
540   let Z₂ := previousWeylEmbedding k n z₂
541   have hX : Commute X Z₂ := presentedCoordinate_commutes_previous k n z₂
542   have hP : Commute P Z₂ := presentedMomentum_commutes_previous k n z₂
543   have hblock : Commute (X ^ a * P ^ b) Z₂ :=
544     (hX.pow_left a).mul_left (hP.pow_left b)
545   have hPX : P * X - X * P = 1 := by
546     rw [show P * X = X * P + 1 from presentedMomentum_mul_coordinate k n]
547     noncomm_ring
548   rw [presentedCoefficientOrdered, presentedCoefficientOrdered]
549   change (Z₁ * X ^ a * P ^ b) * (Z₂ * X ^ c * P ^ d) = _
550   calc
551     (Z₁ * X ^ a * P ^ b) * (Z₂ * X ^ c * P ^ d) =
552         Z₁ * ((X ^ a * P ^ b) * Z₂) * (X ^ c * P ^ d) := by
553       noncomm_ring
554     _ = (Z₁ * Z₂ * X ^ a) * (P ^ b * X ^ c) * P ^ d := by
555       rw [hblock.eq]
556       noncomm_ring
557     _ = (Z₁ * Z₂ * X ^ a) *
558           (∑ i ∈ Finset.range (b + 1),
559             if i ≤ c then
560               (b.choose i * c.descFactorial i) •
561                 (X ^ (c - i) * P ^ (b - i))
562             else 0) * P ^ d := by
563       rw [OreAmbient.commutator_pow_mul_pow P X hPX b c]
564     _ = _ := by
565       rw [Finset.mul_sum, Finset.sum_mul]
566       apply Finset.sum_congr rfl
567       intro i hi
568       have hib : i ≤ b := by
569         have := Finset.mem_range.mp hi
570         omega
571       by_cases hic : i ≤ c
572       · simp only [if_pos hic, OreAmbient.mul_nsmul_left,
573           OreAmbient.nsmul_mul_right, mul_nsmul]
574         congr 1
575         congr 1
576         change (Z₁ * Z₂ * X ^ a) * (X ^ (c - i) * P ^ (b - i)) * P ^ d =
577           previousWeylEmbedding k n (z₁ * z₂) * X ^ (a + c - i) *
578             P ^ (b + d - i)
579         rw [map_mul]
580         rw [show a + c - i = a + (c - i) by omega,
581           show b + d - i = (b - i) + d by omega,
582           pow_add, pow_add]
583         noncomm_ring
584       · simp [hic]

586 -- Compile-time API contract: the exact contraction range, coefficient, and
587 -- exponent loss are part of the public normal-ordering statement.
588 example (n a b c d : ℕ) (z₁ z₂ : PresentedWeyl k n) :
589     presentedCoefficientOrdered k n a b z₁ *
590         presentedCoefficientOrdered k n c d z₂ =
591       ∑ i ∈ Finset.range (b + 1),
592         if i ≤ c then
593           (b.choose i * c.descFactorial i) •
594             presentedCoefficientOrdered k n (a + c - i) (b + d - i) (z₁ * z₂)
595         else 0 :=
596   presentedCoefficientOrdered_mul k n a b c d z₁ z₂

598 /-- Ordered monomials built recursively from the named generators in the
599 quotient presentation.  Older coordinate/momentum pairs occur first; within
600 each pair the coordinate power precedes the momentum power. -/
601 def presentedOrderedMonomial :
602     (n : ℕ) → (Fin n → ℕ) → (Fin n → ℕ) → PresentedWeyl k n
603   | 0, _, _ => 1
604   | n + 1, a, p =>
605       previousWeylEmbedding k n
606           (presentedOrderedMonomial n
607             (fun i => a i.succ) (fun i => p i.succ)) *
608         presentedCoordinate k n ^ a 0 * presentedMomentum k n ^ p 0
609 
610 /-- The presentation/Ore equivalence preserves the recursively ordered
611 monomials, not merely the degree-one generators. -/
612 theorem presentedIteratedEquiv_orderedMonomial :
613     ∀ (n : ℕ) (a p : Fin n → ℕ),
614       presentedIteratedEquiv k n (presentedOrderedMonomial k n a p) =
615         iteratedOrderedMonomial k n a p := by
616   intro n
617   induction n with
618   | zero =>
619       intro a p
620       simp [presentedOrderedMonomial, iteratedOrderedMonomial]
621   | succ n ih =>
622       intro a p
623       rw [presentedOrderedMonomial, iteratedOrderedMonomial, map_mul, map_mul,
624         map_pow, map_pow, presentedIteratedEquiv]
625       change
626         presentedToIterated k (n + 1)
627               (previousWeylEmbedding k n
628                 (presentedOrderedMonomial k n
629                   (fun i => a i.succ) (fun i => p i.succ))) *
630             presentedToIterated k (n + 1) (presentedCoordinate k n) ^ a 0 *
631           presentedToIterated k (n + 1) (presentedMomentum k n) ^ p 0 = _
632       have hprevious := DFunLike.congr_fun
633         (presentedToIterated_previous k n)
634         (presentedOrderedMonomial k n
635           (fun i => a i.succ) (fun i => p i.succ))
636       rw [AlgHom.comp_apply, AlgHom.comp_apply] at hprevious
637       have hih := ih (fun i => a i.succ) (fun i => p i.succ)
638       change presentedToIterated k n
639           (presentedOrderedMonomial k n
640             (fun i => a i.succ) (fun i => p i.succ)) = _ at hih
641       rw [hprevious, hih, presentedToIterated_coordinate,
642         presentedToIterated_momentum]
643       rfl

645 /-- Every ordered product of named presentation generators has the expected
646 commutative symbol monomial as its checked normal form. -/
647 theorem presentedOrderedMonomial_normal (n : ℕ) (a p : Fin n → ℕ) :
648     presentedNormalFormLinearEquiv k n
649         (presentedOrderedMonomial k n a p) =
650       MvPolynomial.monomial (phaseExponent a p) 1 := by
651   rw [presentedNormalFormLinearEquiv, LinearEquiv.trans_apply,
652     AlgEquiv.toLinearEquiv_apply, presentedIteratedEquiv_orderedMonomial,

655 /-- A free-module basis on the presented Weyl algebra, pulled back from the
656 standard monomial basis of the symbol polynomial space. -/
657 def presentedNormalFormBasis (n : Nat) :
658     Module.Basis (PhaseVar n →₀ ℕ) k (PresentedWeyl k n) :=
659   (MvPolynomial.basisMonomials (PhaseVar n) k).map
660     (presentedNormalFormLinearEquiv k n).symm

663     (m : PhaseVar n →₀ ℕ) :
664     presentedNormalFormBasis k n m =
665       (presentedNormalFormLinearEquiv k n).symm (MvPolynomial.monomial m 1) := by
666   rfl
