14 # Bernstein symbol of the Weyl algebra
15 
16 The Bernstein filtration of `A_n(k)` (all phase variables of weight one) has
17 associated graded ring the polynomial ring `k[x, ∂]` in `2n` variables, graded by
18 total degree.  This file packages that identification as a
19 `RingFiltration.GradedSymbol`, which yields:
20 
21 * right Noetherianity of `A_n(k)`, a second route to the tower argument; and
22 * `Module.DevLEOmega (A_n(k))ᵐᵒᵖ (A_n(k))`, the deviation bound `Kdim ≤ ω`
23   used by Corollary B of the paper.
24 
25 The degree-`N` symbol is the weighted-degree-`N` component of the normal-form
26 polynomial, `AlgebraicAnalysis.Weyl.principalComponentOnPiece`.
27 Its kernel is the strict lower Bernstein piece, which is exactly the
28 predecessor piece `below N` of the filtration.
29 -/

39 /-- The Bernstein filtration of `A_n(k)`: the piece of degree at most `N` is the
40 span of the normal-ordered words of total degree at most `N`. -/
41 @[reducible] noncomputable def filtration : RingFiltration k (WeylAlg k n) where
42   F N := bernsteinPiece k n N
43   mono := fun _ _ h => presentedWeightPiece_mono k (@bernsteinWeight n) h
44   one_mem := by
45     change (1 : WeylAlg k n) ∈ bernsteinPiece k n 0
46     rw [bernsteinPiece, mem_presentedWeightPiece]
47     intro m hm
48     rw [presentedNormalFormLinearEquiv_one] at hm
49     have hm0 : m = 0 := by
50       symm
51       simpa [MvPolynomial.coeff_one] using hm
52     subst m
53     simp [monomialWeight]
54   mul_le := fun i j => Submodule.mul_le.mpr fun _ hx _ hy =>
55     mul_mem_bernsteinPiece k hx hy
56   exhaustive := by
57     apply eq_top_iff.mpr
58     intro a _
59     obtain ⟨N, hN⟩ := exists_mem_bernsteinPiece k a
60     exact Submodule.mem_iSup_of_mem N hN

62 /-- The Bernstein grading of the polynomial symbol ring: homogeneous pieces of
63 total degree `N`. -/
64 abbrev grading : ℕ → Submodule k (SymbolRing k n) :=
65   fun N => MvPolynomial.weightedHomogeneousSubmodule k (@bernsteinWeight n) N
66 
67 /-- The degree-`N` symbol vanishes exactly on the strict lower Bernstein piece. -/
68 theorem principal_eq_zero_iff (N : ℕ)
69     (x : presentedWeightPiece k (@bernsteinWeight n) N) :
70     principalComponentOnPiece k (@bernsteinWeight n) N x = 0 ↔
71       (x : WeylAlg k n) ∈ presentedStrictLowerPiece k (@bernsteinWeight n) N := by
72   rw [← LinearMap.mem_ker, mem_ker_principalComponentOnPiece_iff]
73 
74 /-- The Bernstein principal symbol as a `RingFiltration.GradedSymbol`. -/
75 noncomputable def symbol : RingFiltration.GradedSymbol (filtration k n) (grading k n) where
76   gr N := principalComponentOnPiece k (@bernsteinWeight n) N
77   gr_surjective N := principalComponentOnPiece_surjective k (@bernsteinWeight n)
78   gr_eq_zero_iff N x :=
79     (principal_eq_zero_iff k n N x).trans (by cases N <;> exact Iff.rfl)
80   gr_mul i j x y := by
81     exact presentedPrincipalComponent_mul_bernstein k x.2 y.2
82 
83 /-- The Bernstein symbol algebra has finite Krull dimension. -/
84 theorem ringKrullDim_symbolRing_ne_top :
85     ringKrullDim (SymbolRing k n) ≠ ⊤ := by
86   rw [MvPolynomial.ringKrullDim_of_isNoetherianRing_of_finite,
87     ringKrullDim_eq_zero_of_field, zero_add, Nat.card_sum, Nat.card_fin]
88   intro h
89   exact ENat.natCast_ne_top (n + n) (Option.some_injective _ h)

91 /-- Right Noetherianity of `A_n(k)` from the Bernstein symbol. -/
92 theorem isNoetherian_right :
93     IsNoetherian (WeylAlg k n)ᵐᵒᵖ (WeylAlg k n) :=
94   letI := MvPolynomial.weightedGradedAlgebra k (@bernsteinWeight n)
95   (symbol k n).isNoetherian_right
96 
97 /-- Deviation at most `ω` on the right ideals of `A_n(k)`, from the Bernstein
98 symbol and the finite Krull dimension of the symbol ring. -/
99 theorem devLEOmega_right :
100     Module.DevLEOmega (WeylAlg k n)ᵐᵒᵖ (WeylAlg k n) :=
101   letI := MvPolynomial.weightedGradedAlgebra k (@bernsteinWeight n)
102   (symbol k n).devLEOmega_right (ringKrullDim_symbolRing_ne_top k n)
