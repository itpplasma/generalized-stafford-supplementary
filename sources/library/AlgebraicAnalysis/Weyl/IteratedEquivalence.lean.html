An element commuting with the image of every presented Weyl generator commutes with the image of the whole presented algebra.

106/-- An element commuting with the image of every presented Weyl generator
107commutes with the image of the whole presented algebra. -/
108theorem commutes_freeWeyl_image_of_generators
109 {ι A : Type*} [Ring A] [Algebra k A]
110 (omega : Matrix ι ι k)
111 (f : FreeWeyl k ι omega →ₐ[k] A) (a : A)
112 (h : ∀ i, a * f (freeWeylGenerator omega i) =
113 f (freeWeylGenerator omega i) * a) :
114 ∀ w, a * f w = f w * a := by
115 intro w
116 rcases RingQuot.mkAlgHom_surjective k (freeWeylRelation omega) w with
117 ⟨q, rfl⟩
118 let g : FreeAlgebra k ι →ₐ[k] A :=
119 f.comp (RingQuot.mkAlgHom k (freeWeylRelation omega))
120 change a * g q = g q * a
121 induction q using FreeAlgebra.induction k ι with
122 | grade0 c =>
123 rw [g.commutes]
124 exact (Algebra.commutes c a).symm
125 | grade1 i =>
126 change a * f (freeWeylGenerator omega i) =
127 f (freeWeylGenerator omega i) * a
128 exact h i
129 | mul u v hu hv =>
130 rw [map_mul]
131 calc
132 a * (g u * g v) = (a * g u) * g v := by rw [mul_assoc]
133 _ = (g u * a) * g v := by rw [hu]
134 _ = g u * (a * g v) := by rw [mul_assoc]
135 _ = g u * (g v * a) := by rw [hv]
136 _ = (g u * g v) * a := by rw [mul_assoc]
137 | add u v hu hv =>
138 rw [map_add, mul_add, add_mul, hu, hv]

141theorem iteratedCoordinate_commutator (n : Nat) :
142 ∀ i j : Fin n,
143 ringCommutator (iteratedCoordinate k n i) (iteratedCoordinate k n j) = 0 := by
144 induction n with

146 intro i

148 | succ n ih =>
149 intro i j

151 · refine Fin.cases ?_ (fun j => ?_) j
152 · simp [AlgebraicAnalysis.ringCommutator]
153 · simp only [iteratedCoordinate_zero, iteratedCoordinate_succ]
154 exact sub_eq_zero.mpr (stageCoordinate_mul_embedding k n _)
155 · refine Fin.cases ?_ (fun j => ?_) j
156 · simp only [iteratedCoordinate_zero, iteratedCoordinate_succ]
157 exact sub_eq_zero.mpr (stageCoordinate_mul_embedding k n _).symm
158 · simp only [iteratedCoordinate_succ]

Coordinate-momentum commutators are the negative Kronecker delta, in the convention used by Matrix.J.

198/-- Coordinate-momentum commutators are the negative Kronecker delta, in
199the convention used by `Matrix.J`. -/
200theorem iteratedCoordinate_momentum_commutator (n : Nat) :
201 ∀ i j : Fin n,
202 ringCommutator (iteratedCoordinate k n i) (iteratedMomentum k n j) =
203 if i = j then -1 else 0 := by
204 induction n with
205 | zero =>
206 intro i
207 exact Fin.elim0 i
208 | succ n ih =>
209 intro i j
210 refine Fin.cases ?_ (fun i => ?_) i
211 · refine Fin.cases ?_ (fun j => ?_) j
212 · change stageCoordinate k n * stageMomentum k n -
213 stageMomentum k n * stageCoordinate k n = -1
214 rw [stageMomentum_mul_coordinate]
215 noncomm_ring
216 · simp only [iteratedCoordinate_zero, iteratedMomentum_succ]
217 exact sub_eq_zero.mpr (stageCoordinate_mul_embedding k n _)
218 · refine Fin.cases ?_ (fun j => ?_) j
219 · simp only [iteratedCoordinate_succ, iteratedMomentum_zero]
220 simp only [Fin.succ_ne_zero, ↓reduceIte]
221 exact sub_eq_zero.mpr (stageMomentum_mul_embedding k n _).symm
222 · simp only [iteratedCoordinate_succ, iteratedMomentum_succ]
223 calc
224 ringCommutator
225 (stageAlgHom k n (iteratedCoordinate k n i))
226 (stageAlgHom k n (iteratedMomentum k n j)) =
227 stageAlgHom k n
228 (ringCommutator (iteratedCoordinate k n i)
229 (iteratedMomentum k n j)) :=
230 (map_commutator (stageAlgHom k n).toRingHom _ _).symm
231 _ = if i.succ = j.succ then -1 else 0 := by
232 rw [ih i j]
233 by_cases h : i = j <;> simp [h]

The combined generator family realizes the standard symplectic ringCommutator matrix.

235/-- The combined generator family realizes the standard symplectic
236ringCommutator matrix. -/
237theorem iteratedGenerator_commutator (n : Nat) :
238 ∀ a b : Fin n ⊕ Fin n,
239 ringCommutator (iteratedGenerator k n a) (iteratedGenerator k n b) =
240 algebraMap k (IteratedPairStage k n) (Matrix.J (Fin n) k a b) := by
241 intro a b
242 cases a with
243 | inl i =>
244 cases b with
245 | inl j =>
246 change ringCommutator (iteratedCoordinate k n i)
247 (iteratedCoordinate k n j) = _
248 rw [iteratedCoordinate_commutator]
249 simp [Matrix.J]
250 | inr j =>
251 change ringCommutator (iteratedCoordinate k n i)
252 (iteratedMomentum k n j) = _
253 rw [iteratedCoordinate_momentum_commutator]
254 by_cases h : i = j <;> simp [Matrix.J, h]
255 | inr i =>
256 cases b with
257 | inl j =>
258 change ringCommutator (iteratedMomentum k n i)
259 (iteratedCoordinate k n j) = _
260 rw [commutator_swap,
261 iteratedCoordinate_momentum_commutator]
262 by_cases h : j = i
263 · simp [Matrix.J, h]
264 · have hij : i ≠ j := fun e => h e.symm
265 simp [Matrix.J, h, hij]
266 | inr j =>
267 change ringCommutator (iteratedMomentum k n i)
268 (iteratedMomentum k n j) = _
269 rw [iteratedMomentum_commutator]
