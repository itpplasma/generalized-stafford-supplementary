7 /-!
8 # Scalar algebra structure on an Ore extension
9 
10 If the coefficient ring is a `k`-algebra and the derivation kills `k`, then
11 the canonical image of `k` is central in the normal-form Ore ring.
12 -/

25 /-- Scalars killed by the derivation remain central after adjoining the Ore
26 variable. -/
27 theorem normalScalar_commutes (D : OreDivisionDerivation B)
28     (hD : ∀ c : k, D (algebraMap k B c) = 0) (c : k) (z : NormalOre D) :
29     normalCoefficient D (algebraMap k B c) * z =
30       z * normalCoefficient D (algebraMap k B c) := by
31   rcases normalForm_surjective D z with ⟨p, rfl⟩
32   induction p using Polynomial.induction_on' with
33   | add p q hp hq =>
34       rw [normalForm_add, mul_add, add_mul, hp, hq]
35   | monomial n b =>
36       rw [normalForm_monomial]
37       let s : NormalOre D := normalCoefficient D (algebraMap k B c)
38       let a : NormalOre D := normalCoefficient D b
39       let x : NormalOre D := normalVariable D
40       have hsa : Commute s a := by
41         change normalCoefficient D (algebraMap k B c) * normalCoefficient D b =
42           normalCoefficient D b * normalCoefficient D (algebraMap k B c)
43         rw [← (normalCoefficient D).map_mul, ← (normalCoefficient D).map_mul,
44           Algebra.commutes c b]
45       have hsx : Commute s x := by
46         change normalCoefficient D (algebraMap k B c) * normalVariable D =
47           normalVariable D * normalCoefficient D (algebraMap k B c)
48         have h := normalVariable_mul_coefficient D (algebraMap k B c)
49         rw [hD c, map_zero, add_zero] at h
50         exact h.symm
51       exact (hsa.mul_right (hsx.pow_right n)).eq

53 /-- The scalar algebra structure induced by the canonical coefficient map. -/
54 def normalOreAlgebra (D : OreDivisionDerivation B)
55     (hD : ∀ c : k, D (algebraMap k B c) = 0) : Algebra k (NormalOre D) :=
56   RingHom.toAlgebra'
57     ((normalCoefficient D).comp (algebraMap k B))
58     (normalScalar_commutes D hD)

63       (normalCoefficient D).comp (algebraMap k B) := rfl
