9 /-!
10 # A coordinate-momentum pair over a coefficient ring
11 
12 Two successive Ore extensions adjoin a central coordinate and then its
13 derivation momentum.  This file packages the resulting three canonical
14 generator families and proves their exact relations.
15 -/

26 variable {B : Type*} [Ring B]
27 
28 /-- First adjoin a central coordinate, then a momentum differentiating it. -/
29 abbrev PairStage :=
30   NormalOre (coordinateDerivation :
31     OreDivisionDerivation (CoordinateStage (B := B)))
32 
33 /-- The zero derivation: the coordinate is central over the coefficient ring. -/
34 abbrev innerD : OreDivisionDerivation B := zeroDerivation
35 
36 /-- The derivation `d/dx` on the central-coordinate stage: the momentum acts on the coordinate. -/
37 abbrev outerD : OreDivisionDerivation (CoordinateStage (B := B)) :=
38   coordinateDerivation
39 
40 /-- Embed the old coefficient ring through both Ore stages. -/
41 def pairCoefficient : B →+* PairStage (B := B) :=
42   (normalCoefficient outerD).comp (normalCoefficient innerD)
43 
44 /-- The newly adjoined coordinate. -/
45 def pairCoordinate : PairStage (B := B) :=
46   normalCoefficient outerD (normalVariable innerD)
47 
48 /-- The newly adjoined momentum. -/
49 def pairMomentum : PairStage (B := B) :=
50   normalVariable outerD
51 
52 /-- The checked right PBW basis for the momentum stage over the coordinate
53 stage.  This is a consumer of the reusable one-stage right-PBW interface; the
54 pair-specific generators and Weyl relation are the presentation in
55 `AlgebraicAnalysis.Weyl.Presentation`. -/
56 def pairStageRightOrePBWBasis [Nontrivial B] :
57     Module.Basis ℕ (CoordinateStage (B := B))ᵐᵒᵖ (PairStage (B := B)) :=
58   rightOrePBWBasis coordinateDerivation
59 
60 /-- The new coordinate commutes with every old coefficient. -/
61 theorem pairCoordinate_mul_coefficient (b : B) :
62     pairCoordinate * pairCoefficient b = pairCoefficient b * pairCoordinate := by
63   change
64     normalCoefficient outerD (normalVariable innerD) *
65         normalCoefficient outerD (normalCoefficient innerD b) =
66       normalCoefficient outerD (normalCoefficient innerD b) *
67         normalCoefficient outerD (normalVariable innerD)
68   rw [← (normalCoefficient outerD).map_mul,
69     ← (normalCoefficient outerD).map_mul]
70   congr 1
71   have h := normalVariable_mul_coefficient innerD b
72   rw [zeroDerivation_apply, map_zero, add_zero] at h
73   exact h
74 
75 /-- The new momentum commutes with every old coefficient. -/
76 theorem pairMomentum_mul_coefficient (b : B) :
77     pairMomentum * pairCoefficient b = pairCoefficient b * pairMomentum := by
78   change
79     normalVariable outerD * normalCoefficient outerD (normalCoefficient innerD b) =
80       normalCoefficient outerD (normalCoefficient innerD b) * normalVariable outerD
81   have h := normalVariable_mul_coefficient outerD (normalCoefficient innerD b)
82   rw [coordinateDerivation_coefficient, map_zero, add_zero] at h
83   exact h
84 
85 /-- The new pair satisfies the Weyl relation in the manuscript convention. -/
86 theorem pairMomentum_mul_coordinate :
87     pairMomentum (B := B) * pairCoordinate =
88       pairCoordinate * pairMomentum + 1 := by
89   change
90     normalVariable outerD * normalCoefficient outerD (normalVariable innerD) =
91       normalCoefficient outerD (normalVariable innerD) * normalVariable outerD + 1
92   have h := normalVariable_mul_coefficient outerD
93     (normalVariable innerD : CoordinateStage (B := B))
94   rw [coordinateDerivation_variable, map_one] at h
95   exact h
96 
97 theorem pairCoordinate_commutator_momentum :
98     pairCoordinate (B := B) * pairMomentum - pairMomentum * pairCoordinate = -1 := by
99   rw [pairMomentum_mul_coordinate]
100   noncomm_ring

107 def coordinateStageAlgebra : Algebra k (CoordinateStage (B := B)) :=
108   AlgebraicAnalysis.Weyl.Ore.ScalarAlgebra.normalOreAlgebra innerD fun c => by
109     exact zeroDerivation_apply (algebraMap k B c)

111 theorem coordinateDerivation_algebraMap (c : k) :
112     @coordinateDerivation B _
113       (@algebraMap k (CoordinateStage (B := B)) _ _ coordinateStageAlgebra c) = 0 := by
114   change coordinateDerivation
115     (normalCoefficient innerD (algebraMap k B c)) = 0
116   exact coordinateDerivation_coefficient (algebraMap k B c)
