The additive normal-form equivalence is linear over every central scalar ring killed by the coefficient derivation. This is the scalar interface needed to turn the iterated Ore normal form into a PBW basis.

Polynomials are linearly equivalent to their coefficient finsupps.

30/-- Polynomials are linearly equivalent to their coefficient finsupps. -/
31def polynomialFinsuppLinearEquiv
32 {k R : Type*} [CommSemiring k] [Semiring R] [Algebra k R] :
33 Polynomial R ≃ₗ[k] (ℕ →₀ R) :=
34 { (Polynomial.toFinsuppIso R).toAddEquiv.trans
35 (AddMonoidAlgebra.coeffLinearEquiv k).toAddEquiv with
36 map_smul' := by
37 intro c p
38 rfl }

Apply a linear equivalence independently to every polynomial coefficient.

40/-- Apply a linear equivalence independently to every polynomial
41coefficient. -/
42def polynomialMapRangeLinearEquiv
43 {k R S : Type*} [CommSemiring k] [Semiring R] [Semiring S]
44 [Algebra k R] [Algebra k S] (e : R ≃ₗ[k] S) :
45 Polynomial R ≃ₗ[k] Polynomial S :=
46 (polynomialFinsuppLinearEquiv (k := k) (R := R)).trans
47 ((Finsupp.mapRange.linearEquiv e).trans
48 (polynomialFinsuppLinearEquiv (k := k) (R := S)).symm)
49
50@[simp] theorem polynomialMapRangeLinearEquiv_C
51 {k R S : Type*} [CommSemiring k] [Semiring R] [Semiring S]
52 [Algebra k R] [Algebra k S] (e : R ≃ₗ[k] S) (r : R) :
53 polynomialMapRangeLinearEquiv e (Polynomial.C r) =
54 Polynomial.C (e r) := by
55 apply (Polynomial.toFinsuppIso S).injective
56 simp [polynomialMapRangeLinearEquiv, polynomialFinsuppLinearEquiv]
57
58@[simp] theorem polynomialMapRangeLinearEquiv_monomial
59 {k R S : Type*} [CommSemiring k] [Semiring R] [Semiring S]
60 [Algebra k R] [Algebra k S] (e : R ≃ₗ[k] S) (n : ℕ) (r : R) :
61 polynomialMapRangeLinearEquiv e (Polynomial.monomial n r) =
62 Polynomial.monomial n (e r) := by
63 apply (Polynomial.toFinsuppIso S).injective
64 simp [polynomialMapRangeLinearEquiv, polynomialFinsuppLinearEquiv]
65
66@[simp] theorem polynomialMapRangeLinearEquiv_X
67 {k R S : Type*} [CommSemiring k] [Semiring R] [Semiring S]
68 [Algebra k R] [Algebra k S] (e : R ≃ₗ[k] S)
69 (h1 : e 1 = 1) :
70 polynomialMapRangeLinearEquiv e Polynomial.X = Polynomial.X := by

74@[simp] theorem polynomialMapRangeLinearEquiv_one
75 {k R S : Type*} [CommSemiring k] [Semiring R] [Semiring S]

77 (h1 : e 1 = 1) :

Flatten two nested univariate polynomial layers into two commuting symbol variables as an algebra equivalence over the coefficient ring.

101/-- Flatten two nested univariate polynomial layers into two commuting symbol
102variables as an algebra equivalence over the coefficient ring. -/
103def nestedPolynomialAlgEquiv (R : Type*) [CommRing R] :
104 Polynomial (Polynomial R) ≃ₐ[R]
105 MvPolynomial (PUnit.{1} ⊕ PUnit.{1}) R :=
106 (Polynomial.mapAlgEquiv ((MvPolynomial.pUnitAlgEquiv R).symm)).trans
107 ((((MvPolynomial.pUnitAlgEquiv (MvPolynomial PUnit.{1} R)).symm
108 ).restrictScalars R).trans
109 (MvPolynomial.sumAlgEquiv R PUnit.{1} PUnit.{1}).symm)

The scalar-linear form of nestedPolynomialAlgEquiv.

111/-- The scalar-linear form of `nestedPolynomialAlgEquiv`. -/
112def nestedPolynomialLinearEquiv
113 {k R : Type*} [CommRing k] [CommRing R] [Algebra k R] :
114 Polynomial (Polynomial R) ≃ₗ[k]

137private theorem nestedPolynomialLinearEquiv_mul
138 {k R : Type*} [CommRing k] [CommRing R] [Algebra k R]
139 (f g : Polynomial (Polynomial R)) :
140 nestedPolynomialLinearEquiv (k := k) (R := R) (f * g) =
141 nestedPolynomialLinearEquiv (k := k) (R := R) f *
142 nestedPolynomialLinearEquiv (k := k) (R := R) g := by
143 change nestedPolynomialAlgEquiv R (f * g) =
144 nestedPolynomialAlgEquiv R f * nestedPolynomialAlgEquiv R g
145 exact map_mul (nestedPolynomialAlgEquiv R) f g

147private theorem nestedPolynomialLinearEquiv_pow
148 {k R : Type*} [CommRing k] [CommRing R] [Algebra k R]
149 (f : Polynomial (Polynomial R)) (n : ℕ) :
150 nestedPolynomialLinearEquiv (k := k) (R := R) (f ^ n) =

The coefficient-left normal-form map respects central scalar multiplication.

190/-- The coefficient-left normal-form map respects central scalar
191multiplication. -/
192theorem normalForm_smul (c : k) (p : Polynomial B) :
193 normalForm D (c • p) = c • normalForm D p := by
194 induction p using Polynomial.induction_on' with
195 | add p q hp hq =>
196 rw [smul_add, normalForm_add, normalForm_add, hp, hq,
197 Algebra.smul_def, Algebra.smul_def, Algebra.smul_def, mul_add]
198 | monomial n b =>
199 rw [Polynomial.smul_monomial, normalForm_monomial,
200 normalForm_monomial]
201 rw [Algebra.smul_def c b]
202 have hs : c • (normalCoefficient D b * normalVariable D ^ n) =
203 algebraMap k (NormalOre D) c *
204 (normalCoefficient D b * normalVariable D ^ n) :=
205 Algebra.smul_def c _
206 rw [hs, hAlg]
207 change normalCoefficient D (algebraMap k B c * b) * normalVariable D ^ n =
208 normalCoefficient D (algebraMap k B c) *
209 (normalCoefficient D b * normalVariable D ^ n)
210 rw [map_mul, mul_assoc]

Ore normal forms are linearly equivalent to ordinary coefficient-left polynomials.

212/-- Ore normal forms are linearly equivalent to ordinary coefficient-left
213polynomials. -/
214def normalFormLinearEquiv : Polynomial B ≃ₗ[k] NormalOre D :=
215 { normalFormAddEquiv D with
216 map_smul' := normalForm_smul D hAlg }
217
218@[simp] theorem normalFormLinearEquiv_apply (p : Polynomial B) :
219 normalFormLinearEquiv D hAlg p = normalForm D p := rfl
220
221@[simp] theorem normalFormLinearEquiv_symm_normalForm (p : Polynomial B) :
222 (normalFormLinearEquiv D hAlg).symm (normalForm D p) = p :=
223 (normalFormLinearEquiv D hAlg).symm_apply_apply p
224
225@[simp] theorem normalFormLinearEquiv_symm_coefficient (b : B) :
226 (normalFormLinearEquiv D hAlg).symm (normalCoefficient D b) =
227 Polynomial.C b := by
228 rw [← normalForm_C, normalFormLinearEquiv_symm_normalForm]
229
230@[simp] theorem normalFormLinearEquiv_symm_variable :
231 (normalFormLinearEquiv D hAlg).symm (normalVariable D) =
232 Polynomial.X := by
233 rw [normalVariable, normalFormLinearEquiv_symm_normalForm]
234
235@[simp] theorem normalFormLinearEquiv_symm_one :
236 (normalFormLinearEquiv D hAlg).symm 1 = 1 := by

Expose the two coefficient-left polynomial layers of a pair stage and apply a chosen linear coordinate system to the old coefficients.

246/-- Expose the two coefficient-left polynomial layers of a pair stage and
247apply a chosen linear coordinate system to the old coefficients. -/
248def pairNormalFormLinearEquiv
249 {k B S : Type*} [CommRing k] [Ring B] [Algebra k B]
250 [CommRing S] [Algebra k S]
251 [Algebra k (CoordinateStage (B := B))]
252 [Algebra k (PairStage (B := B))]
253 (hInner : algebraMap k (CoordinateStage (B := B)) =
254 (normalCoefficient zeroDerivation).comp (algebraMap k B))
255 (hOuter : algebraMap k (PairStage (B := B)) =
256 (normalCoefficient coordinateDerivation).comp
257 (algebraMap k (CoordinateStage (B := B))))
258 (e : B ≃ₗ[k] S) :
259 PairStage (B := B) ≃ₗ[k] Polynomial (Polynomial S) :=
260 (normalFormLinearEquiv coordinateDerivation hOuter).symm.trans
261 (polynomialMapRangeLinearEquiv
262 ((normalFormLinearEquiv zeroDerivation hInner).symm.trans
263 (polynomialMapRangeLinearEquiv e)))

266 {k B S : Type*} [CommRing k] [Ring B] [Algebra k B]
267 [CommRing S] [Algebra k S]
268 [Algebra k (CoordinateStage (B := B))]
269 [Algebra k (PairStage (B := B))]
270 (hInner) (hOuter) (e : B ≃ₗ[k] S) (b : B) :
271 pairNormalFormLinearEquiv hInner hOuter e (pairCoefficient b) =
272 Polynomial.C (Polynomial.C (e b)) := by
273 simp [pairNormalFormLinearEquiv, pairCoefficient]

277 [CommRing S] [Algebra k S]

279 [Algebra k (PairStage (B := B))]

286@[simp] theorem pairNormalFormLinearEquiv_momentum
287 {k B S : Type*} [CommRing k] [Ring B] [Algebra k B]

The two-stage normal-form map sends a coefficient followed by powers of the new coordinate and momentum to the corresponding nested monomial.

301/-- The two-stage normal-form map sends a coefficient followed by powers of
302the new coordinate and momentum to the corresponding nested monomial. -/
303@[simp] theorem pairNormalFormLinearEquiv_orderedMonomial
304 {k B S : Type*} [CommRing k] [Ring B] [Algebra k B]
305 [CommRing S] [Algebra k S]
306 [Algebra k (CoordinateStage (B := B))]
307 [Algebra k (PairStage (B := B))]
308 (hInner : algebraMap k (CoordinateStage (B := B)) =
309 (normalCoefficient zeroDerivation).comp (algebraMap k B))
310 (hOuter : algebraMap k (PairStage (B := B)) =
311 (normalCoefficient coordinateDerivation).comp
312 (algebraMap k (CoordinateStage (B := B))))
313 (e : B ≃ₗ[k] S) (b : B) (a p : ℕ) :
314 pairNormalFormLinearEquiv hInner hOuter e
315 (pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ p) =
316 Polynomial.monomial p (Polynomial.monomial a (e b)) := by
317 have hnormal :
318 pairCoefficient b * pairCoordinate ^ a * pairMomentum ^ p =
319 normalForm coordinateDerivation
320 (Polynomial.monomial p

322 rw [normalForm_monomial, normalForm_monomial]

324 normalCoefficient coordinateDerivation (normalCoefficient zeroDerivation b) *
