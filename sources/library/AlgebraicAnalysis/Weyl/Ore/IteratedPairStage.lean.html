6 /-!
7 # Iterated coordinate-momentum pair stages
8 
9 This file recursively repeats the checked `PairStage` construction over an
10 arbitrary coefficient ring.  It records only the resulting tower and the
11 canonical data introduced at each successor; it does not identify the tower
12 with a presented Weyl algebra.
13 -/

23 /-- A type together with the ring structure used at the next Ore stage. -/
24 structure RingStage where
25   /-- The underlying type. -/
26   carrier : Type u
27   /-- The ring structure on the underlying type. -/
28   ring : Ring carrier
29 
30 variable (B : Type u) [Ring B]
31 
32 /-- The recursively constructed ring data after adjoining `n` Weyl pairs. -/
33 def iteratedPairData : Nat → RingStage
34   | 0 => ⟨B, inferInstance⟩
35   | n + 1 =>
36       let previous := iteratedPairData n
37       letI : Ring previous.carrier := previous.ring
38       ⟨PairStage (B := previous.carrier), inferInstance⟩
39 
40 /-- The ring obtained from `B` after recursively adjoining `n` checked
41 coordinate-momentum pairs. -/
42 def IteratedPairStage (n : Nat) : Type u :=
43   (iteratedPairData B n).carrier
44 
45 instance iteratedPairStageRing (n : Nat) : Ring (IteratedPairStage (B := B) n) :=
46   (iteratedPairData B n).ring
47 
48 /-- The zeroth stage is the original coefficient type. -/
49 theorem iteratedPairStage_zero : IteratedPairStage (B := B) 0 = B := rfl
50 
51 /-- Every successor is definitionally the checked `PairStage` construction
52 over its predecessor. -/
53 theorem iteratedPairStage_succ (n : Nat) :
54     IteratedPairStage (B := B) (n + 1) =
55       PairStage (B := IteratedPairStage (B := B) n) := rfl
56 
57 /-- The canonical embedding from stage `n` into stage `n + 1`. -/
58 def stageEmbedding (n : Nat) :
59     IteratedPairStage (B := B) n →+* IteratedPairStage (B := B) (n + 1) :=
60   pairCoefficient (B := IteratedPairStage (B := B) n)
61 
62 /-- The coordinate introduced at the successor of stage `n`. -/
63 def stageCoordinate (n : Nat) : IteratedPairStage (B := B) (n + 1) :=
64   pairCoordinate (B := IteratedPairStage (B := B) n)
65 
66 /-- The momentum introduced at the successor of stage `n`. -/
67 def stageMomentum (n : Nat) : IteratedPairStage (B := B) (n + 1) :=
68   pairMomentum (B := IteratedPairStage (B := B) n)
69 
70 /-- The coordinate introduced at stage `n + 1` commutes with the embedded
71 predecessor ring. -/
72 theorem stageCoordinate_mul_embedding (n : Nat)
73     (b : IteratedPairStage (B := B) n) :
74     stageCoordinate B n * stageEmbedding B n b =
75       stageEmbedding B n b * stageCoordinate B n :=
76   pairCoordinate_mul_coefficient b
