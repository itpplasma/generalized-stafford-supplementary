10 # Presentation of the Weyl algebra
11 
12 The Weyl algebra `A_n(k)` of `n` degrees of freedom is the free algebra on
13 `2n` generators `x_1, …, x_n, ∂_1, …, ∂_n` modulo the commutator relations
14 `[x_i, ∂_j] = -δ_ij` (the standard symplectic form `Matrix.J`), so that
15 `[∂_i, x_j] = δ_ij`.  This file records only the presentation and the ordered
16 PBW words used by the Weyl applications.
17 
18 Provenance: the definitions `freeWeylRelation`, `FreeWeyl`, `freeWeylGenerator`,
19 `PhaseVar`, `relation`, `WeylAlg`, `generator`, `oldIndex`, `freeOldMap`,
20 `freeOrderedMonomial`, `orderedMonomial` and `phaseDegree` are copied from
21 `Stafford38/ChallengeDefinitions.lean` of `itpplasma/stafford38-formal` at
22 commit `ac20b730fb368786e840320ebf6e62d19bbe29f3` (Apache-2.0).  The bodies are
23 unchanged; only the namespace moved from `Stafford` and `Stafford38Challenge`
24 to `AlgebraicAnalysis.Weyl`.  Consequently `AlgebraicAnalysis.Weyl.WeylAlg k n`
25 is, definitionally, the algebra `Stafford38Challenge.WeylAlg k n` of
26 stafford38-formal.  The challenge-level `bernsteinPiece`, `bernsteinDegree`
27 and fixed-source statements are not part of this extraction.
28 -/

32 variable {k ι : Type*} [Field k]
33 
34 /-- The commutator relations presenting a Weyl-type algebra: `[a, b] = ω i j`
35 for the generators `a = ι i`, `b = ι j` written in the free algebra. -/
36 def freeWeylRelation (omega : Matrix ι ι k)
37     (a b : FreeAlgebra k ι) : Prop :=
38   ∃ i j,
39     a = FreeAlgebra.ι k i * FreeAlgebra.ι k j -
40       FreeAlgebra.ι k j * FreeAlgebra.ι k i ∧
41     b = algebraMap k (FreeAlgebra k ι) (omega i j)
42 
43 /-- The free algebra modulo the commutator relations encoded by `omega`. -/
44 abbrev FreeWeyl (k : Type*) [Field k] (ι : Type*)
45     (omega : Matrix ι ι k) := RingQuot (freeWeylRelation omega)
46 
47 /-- The canonical generator of the presented Weyl-type algebra. -/
48 def freeWeylGenerator (omega : Matrix ι ι k) (i : ι) : FreeWeyl k ι omega :=
49   RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k i)
50 
51 /-- Index type of the coordinate `x_i` (left summand) and momentum `∂_i`
52 (right summand) generators. -/
53 abbrev PhaseVar (n : ℕ) := Fin n ⊕ Fin n
54 
55 /-- The Weyl commutator relation for the standard symplectic matrix. -/
56 abbrev relation {n : ℕ}
57     (omega : Matrix (PhaseVar n) (PhaseVar n) k)
58     (a b : FreeAlgebra k (PhaseVar n)) : Prop :=
59   freeWeylRelation omega a b
60 
61 /-- The standard presented Weyl algebra `A_n(k)`. -/
62 abbrev WeylAlg (k : Type*) [Field k] (n : ℕ) :=
63   FreeWeyl k (PhaseVar n) (Matrix.J (Fin n) k)
64 
65 /-- A named generator of the presented Weyl algebra. -/
66 abbrev generator (k : Type*) [Field k] (n : ℕ) (i : PhaseVar n) : WeylAlg k n :=
67   freeWeylGenerator (Matrix.J (Fin n) k) i
68 
69 /-- Insert an old phase variable after the newest coordinate or momentum. -/
70 def oldIndex {n : ℕ} : PhaseVar n → PhaseVar (n + 1)
71   | .inl i => .inl i.succ
72   | .inr i => .inr i.succ
73 
74 @[simp] theorem oldIndex_inl {n : ℕ} (i : Fin n) :
75     oldIndex (.inl i) = .inl i.succ := rfl
76 
77 @[simp] theorem oldIndex_inr {n : ℕ} (i : Fin n) :
78     oldIndex (.inr i) = .inr i.succ := rfl
79 
80 /-- Insert an old free-algebra word into the next-rank free algebra. -/
81 def freeOldMap (k : Type*) [Field k] (n : ℕ) :
82     FreeAlgebra k (PhaseVar n) →ₐ[k] FreeAlgebra k (PhaseVar (n + 1)) :=
83   FreeAlgebra.lift k (fun i => FreeAlgebra.ι k (oldIndex i))
84 
85 /-- Ordered PBW words, with old pairs followed by the newest pair. -/
86 def freeOrderedMonomial (k : Type*) [Field k] :
87     (n : ℕ) → (Fin n → ℕ) → (Fin n → ℕ) → FreeAlgebra k (PhaseVar n)
88   | 0, _, _ => 1
89   | n + 1, a, p =>
90       freeOldMap k n
91           (freeOrderedMonomial k n (fun i => a i.succ) (fun i => p i.succ)) *
92         FreeAlgebra.ι k (.inl (0 : Fin (n + 1))) ^ a 0 *
93         FreeAlgebra.ι k (.inr (0 : Fin (n + 1))) ^ p 0
94 
95 /-- The image of an ordered PBW word in the presented Weyl algebra. -/
96 def orderedMonomial (k : Type*) [Field k] (n : ℕ)
97     (a p : Fin n → ℕ) : WeylAlg k n :=
98   RingQuot.mkAlgHom k (relation (k := k) (n := n) (Matrix.J (Fin n) k))
99     (freeOrderedMonomial k n a p)
100 
101 /-- Total degree of a split coordinate/momentum exponent. -/
102 def phaseDegree {n : ℕ} (a p : Fin n → ℕ) : ℕ :=
103   (∑ i, a i) + ∑ i, p i
104 
105 /-- The canonical generators satisfy the defining relations: the commutator of
106 `freeWeylGenerator omega i` and `freeWeylGenerator omega j` is `omega i j`. -/
107 theorem freeWeylGenerator_commutator
108     (omega : Matrix ι ι k) (i j : ι) :
109     ringCommutator (freeWeylGenerator omega i) (freeWeylGenerator omega j) =
110       algebraMap k (FreeWeyl k ι omega) (omega i j) := by
111   have h := RingQuot.mkAlgHom_rel k
112     (show freeWeylRelation omega
113       (FreeAlgebra.ι k i * FreeAlgebra.ι k j -
114         FreeAlgebra.ι k j * FreeAlgebra.ι k i)
115       (algebraMap k (FreeAlgebra k ι) (omega i j)) from ⟨i, j, rfl, rfl⟩)
116   simp only [freeWeylGenerator, ringCommutator, RingQuot.mkAlgHom_coe]
117   calc
118     RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k i) *
119           RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k j) -
120         RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k j) *
121           RingQuot.mkAlgHom k (freeWeylRelation omega) (FreeAlgebra.ι k i) =
122       RingQuot.mkAlgHom k (freeWeylRelation omega)
123         (FreeAlgebra.ι k i * FreeAlgebra.ι k j -
124           FreeAlgebra.ι k j * FreeAlgebra.ι k i) := by
125             rw [← map_mul, ← map_mul, ← map_sub]
126     _ = RingQuot.mkAlgHom k (freeWeylRelation omega)
127         ((algebraMap k (FreeAlgebra k ι)) (omega i j)) := h
128     _ = algebraMap k (FreeWeyl k ι omega) (omega i j) := by
129       have hmap : algebraMap k (FreeWeyl k ι omega) =
130           (RingQuot.mkAlgHom k (freeWeylRelation omega)).toRingHom.comp
131             (algebraMap k (FreeAlgebra k ι)) := by
132         apply RingHom.ext
133         intro c
134         change algebraMap k (FreeWeyl k ι omega) c =
135           RingQuot.mkAlgHom k (freeWeylRelation omega)
136             (algebraMap k (FreeAlgebra k ι) c)
137         exact (RingQuot.mkAlgHom k (freeWeylRelation omega)).commutes c |>.symm
138       rw [hmap]
139       rfl
