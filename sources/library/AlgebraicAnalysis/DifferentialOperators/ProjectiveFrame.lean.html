12 # Differential operators when `Ω_{R/k}` is projective
13 
14 A **projective frame** of the commutative `k`-algebra `R`
15 (`DifferentialOperators.IsProjectiveFrame x E`) consists of algebra generators
16 `x₁, …, x_r` of `R` and derivations `E₁, …, E_r` with
17 
18   `d b = ∑ᵢ Eᵢ(b) d xᵢ` in `Ω_{R/k}` for every `b ∈ R`.
19 
20 Equivalently, `ω ↦ (Eᵢ ω)ᵢ` splits the surjection `Rʳ ↠ Ω_{R/k}`,
21 `eᵢ ↦ d xᵢ`. Unlike a coordinate frame, the `Eᵢ` need not be dual to the
22 `xᵢ`; the matrix `Eᵢ(xⱼ)` is only idempotent. A finitely generated algebra has
23 a projective frame exactly when `Ω_{R/k}` is projective
24 (`exists_isProjectiveFrame`), for instance when `R` is smooth over `k`.
25 
26 ## Main results
27 
28 * `IsProjectiveFrame.derivation_eq_sum`: every derivation `D : R → M` into an
29   `R`-module satisfies `D b = ∑ᵢ Eᵢ(b) D(xᵢ)`.
30 * `IsProjectiveFrame.nsmul_sub_sum_mem_orderBelow` (**Euler identity modulo
31   lower order**, any characteristic): for `P` of order at most `m`,
32   `m P - ∑ᵢ [P, xᵢ] Eᵢ` has order at most `m - 1`.
33 * Characteristic zero: the order filtration of `D_k(R)` is generated in degree
34   one by the `Eᵢ` over `R` (`IsProjectiveFrame.polynomialGenerators`), so
35   `R[ξ₁, …, ξ_r] ↠ gr D_k(R)`. Hence `D_k(R)` is generated by `R` and the
36   `Eᵢ`, and if `R` is Noetherian of Krull dimension at most `d`, `D_k(R)` is
37   right and left Noetherian with deviation at most `d + r` on both sides.
38 * `exists_isProjectiveFrame`, `ringKrullDim_le_of_adjoin_eq_top`: a finitely
39   generated algebra with projective `Ω` generated by `r` elements has a
40   projective frame of length `r` and Krull dimension at most `r`, so
41   (`devLE_right_of_adjoin_eq_top`, `devLE_left_of_adjoin_eq_top`) `D_k(R)` has
42   deviation at most `2r`.
43 
44 ## Proof of the Euler identity
45 
46 The map `b ↦ [P, b]` is a derivation modulo lower order: for `P` of order at
47 most `n + 1` the classes of `[P, b]` in `order n / orderBelow n` satisfy
48 Leibniz, since `[P, a] b - b [P, a] = [[P, a], b]` drops the order. Therefore
49 `[P, b] ≡ ∑ᵢ Eᵢ(b) [P, xᵢ]` (`commutator_sub_sum_mem_orderBelow`). The
50 induction then runs as for coordinate frames
51 (`IsCoordinateFrame.nsmul_sub_sum_mem_orderBelow`), using the commutator test
52 on algebra generators.
53

55

57 (differential operators of a differentially smooth algebra).
58 The argument via projective frames is AA-native.
59 -/

140 /-- A **projective frame** of the commutative `k`-algebra `R`: algebra
141 generators `x₁, …, x_r` and derivations `E₁, …, E_r` with
142 `d b = ∑ᵢ Eᵢ(b) d xᵢ` in `Ω_{R/k}`. The `Eᵢ` split `Rʳ ↠ Ω_{R/k}`, so
143 `Ω_{R/k}` is projective. -/
144 structure IsProjectiveFrame (x : σ → R) (E : σ → Derivation k R R) : Prop where
145   /-- The `xᵢ` generate `R` as a `k`-algebra. -/
146   adjoin_eq_top : Algebra.adjoin k (Set.range x) = ⊤
147   /-- `d b = ∑ᵢ Eᵢ(b) d xᵢ` in the Kähler differentials. -/
148   D_eq_sum : ∀ b, KaehlerDifferential.D k R b = ∑ i, E i b • KaehlerDifferential.D k R (x i)

162 /-- **Commutators modulo lower order.** For `P` of order at most `n + 1`,
163 `[P, b] - ∑ᵢ Eᵢ(b) [P, xᵢ]` has order at most `n - 1`. -/
164 theorem commutator_sub_sum_mem_orderBelow {n : ℕ} {P : End (k := k) (R := R)}
165     (hP : P ∈ order (n + 1)) (b : R) :
166     commutator P b - ∑ i, multiplication (E i b) * commutator P (x i) ∈ orderBelow n := by
167   have h := hF.derivation_eq_sum (commutatorDerivation hP) b
168   simp only [commutatorDerivation_apply, ← Submodule.Quotient.mk_smul,
169     smul_eq_multiplication_mul] at h
170   have hsum := map_sum (orderBelowSubmodule (k := k) (R := R) n).mkQ
171     (fun i => multiplication (E i b) * commutator P (x i)) Finset.univ
172   simp only [Submodule.mkQ_apply] at hsum
173   rw [← hsum, Submodule.Quotient.eq] at h
174   exact h
175 
176 /-- The same with the multiplications on the right: `[P, b] - ∑ᵢ [P, xᵢ] Eᵢ(b)`
177 has order at most `n - 1`. -/
178 theorem commutator_sub_sum_mul_mem_orderBelow {n : ℕ} {P : End (k := k) (R := R)}
179     (hP : P ∈ order (n + 1)) (b : R) :
180     commutator P b - ∑ i, commutator P (x i) * multiplication (E i b) ∈ orderBelow n := by
181   have hswap : ∀ i, multiplication (E i b) * commutator P (x i) -
182       commutator P (x i) * multiplication (E i b) ∈ orderBelow n := fun i => by
183     have h := mem_order_iff_commutator_mem_orderBelow.mp
184       ((mem_order_succ_iff P n).mp hP (x i)) (E i b)
185     rw [← neg_mem_iff, neg_sub]
186     exact h
187   have heq : commutator P b - ∑ i, commutator P (x i) * multiplication (E i b) =
188       (commutator P b - ∑ i, multiplication (E i b) * commutator P (x i)) +
189         ∑ i, (multiplication (E i b) * commutator P (x i) -
190           commutator P (x i) * multiplication (E i b)) := by
191     rw [Finset.sum_sub_distrib]
192     abel
193   rw [heq]
194   exact add_mem (hF.commutator_sub_sum_mem_orderBelow hP b)
195     (Submodule.sum_mem _ fun i _ => hswap i)
196 
197 /-- **Euler identity modulo lower order.** For an operator `P` of order at most
198 `m`, `m P - ∑ᵢ [P, xᵢ] Eᵢ` lies in `orderBelow m`. Holds in every
199 characteristic. -/
200 theorem nsmul_sub_sum_mem_orderBelow :
201     ∀ m, ∀ P ∈ order (k := k) (R := R) m,
202       m • P - ∑ i, commutator P (x i) * (E i).toLinearMap ∈ orderBelow m := by
203   intro m
204   induction m with
205   | zero =>
206       intro P hP
207       rw [mem_order_zero_iff] at hP
208       simp [hP]
209   | succ n ih =>
210       intro P hP
211       rw [orderBelow_succ, mem_order_iff_commutator_mem_orderBelow]
212       refine commutator_mem_of_adjoin hF.adjoin_eq_top _
213         (fun a _ hQ => multiplication_mul_mem_orderBelow a hQ)
214         (fun a _ hQ => mul_multiplication_mem_orderBelow a hQ) fun l => ?_
215       have key : commutator ((n + 1) • P - ∑ i, commutator P (x i) * (E i).toLinearMap) (x l) =
216           (n • commutator P (x l) -
217             ∑ i, commutator (commutator P (x l)) (x i) * (E i).toLinearMap) +
218           (commutator P (x l) -
219             ∑ i, commutator P (x i) * multiplication (E i (x l))) := by
220         rw [commutator_sub, commutator_nsmul, commutator_sum]
221         simp only [commutator_mul, commutator_derivation, Finset.sum_add_distrib,
222           commutator_commutator_comm P (x _) (x l)]
223         rw [succ_nsmul]
224         abel
225       rw [key]
226       exact add_mem (ih _ ((mem_order_succ_iff P n).mp hP (x l)))
227         (hF.commutator_sub_sum_mul_mem_orderBelow hP (x l))
228 
229 end IsProjectiveFrame

322     Submodule.span R (Set.range fun i => KaehlerDifferential.D k R (x i)) = ⊤ := by
323   set N := Submodule.span R (Set.range fun i => KaehlerDifferential.D k R (x i))
324   let T : Subalgebra k R :=
325     { carrier := {b | KaehlerDifferential.D k R b ∈ N}
326       mul_mem' := fun {a b} ha hb => by
327         rw [Set.mem_ofPred_eq, Derivation.leibniz]
328         exact add_mem (N.smul_mem a hb) (N.smul_mem b ha)
329       add_mem' := fun {a b} ha hb => by
330         rw [Set.mem_ofPred_eq, map_add]
331         exact add_mem ha hb

333         rw [Set.mem_ofPred_eq, Derivation.map_algebraMap]
334         exact zero_mem _ }

341 /-- **Projective frames exist** when `Ω_{R/k}` is projective: any finite family
342 of algebra generators extends to a projective frame, the derivations being
343 the coordinates of a splitting of `Rʳ ↠ Ω_{R/k}`. -/

345     {x : σ → R} (hx : Algebra.adjoin k (Set.range x) = ⊤) :
