12 # Differential operators on a polynomial ring
13 
14 Let `k` be a field and `R = k[x₁, …, xₙ]`. This file applies the coordinate
15 symbol presentation (`DifferentialOperators.CoordinateSymbol`) to the
16 coordinates `xᵢ` and the partial derivatives `∂ᵢ = ∂/∂xᵢ`, and identifies the
17 ring `D_k(R)` of all finite-order `k`-linear differential operators with the
18 Weyl algebra `A_n(k)` in characteristic zero.
19 
20 ## Main results
21 
22 For a field `k` of characteristic zero, `D = D_k(k[x₁, …, xₙ])`:
23 
24 * `PolynomialOperators.isNoetherian_right`, `isNoetherianRing`: `D` is right
25   and left Noetherian.
26 * `PolynomialOperators.devLE_right`, `devLE_left`: the right and left ideals
27   of `D` have deviation at most `2n`; `devLEOmega_right`, `devLEOmega_left`.
28 * `PolynomialOperators.adjoin_eq_top`: `D` is generated by the
29   multiplications by `xᵢ` and the `∂ᵢ`.
30 * `PolynomialOperators.weylEquiv : WeylAlg k n ≃ₐ[k] D`, `xᵢ ↦ xᵢ`,
31   `∂ᵢ ↦ ∂/∂xᵢ`. Surjectivity is the generation theorem; injectivity holds
32   because `A_n(k)` is simple (`Weyl.Simplicity.isSimpleRing_weylAlg`).
33 * `PolynomialOperators.isSimpleRing` (transported from `A_n(k)`), `isDomain`
34   (from the principal symbol, `IsCoordinateFrame.isDomain`), `infinite_center`.
35 
36 The Noetherian and deviation statements are proved from the order filtration
37 of `D` itself, not transported from the Weyl algebra.
38 -/

52 /-- The partial derivatives are dual to the coordinates. -/
53 theorem pderiv_X_eq_ite (i j : Fin n) :
54     pderiv i (X j : MvPolynomial (Fin n) k) = if i = j then 1 else 0 := by
55   rw [pderiv_X, Pi.single_apply]
56   simp only [eq_comm]
57 
58 /-- Partial derivatives commute. -/
59 theorem pderiv_mul_pderiv_comm (i j : Fin n) :
60     (pderiv (R := k) i).toLinearMap * (pderiv (R := k) j).toLinearMap =
61       (pderiv (R := k) j).toLinearMap * (pderiv (R := k) i).toLinearMap := by
62   have h : ⁅(pderiv i : Derivation k (MvPolynomial (Fin n) k) (MvPolynomial (Fin n) k)),
63       pderiv (R := k) j⁆ = 0 := by
64     apply MvPolynomial.derivation_ext
65     intro l
66     simp only [Derivation.commutator_apply, Derivation.zero_apply, pderiv_X_eq_ite]
67     split_ifs <;> simp
68   refine LinearMap.ext fun f => ?_
69   have hf := Derivation.congr_fun h f
70   simp only [Derivation.commutator_apply, Derivation.zero_apply] at hf
71   exact sub_eq_zero.mp hf
72 
73 theorem adjoin_range_X' :
74     Algebra.adjoin k (Set.range (X : Fin n → MvPolynomial (Fin n) k)) = ⊤ :=
75   MvPolynomial.adjoin_range_X
76 
77 /-- The coordinates and the partial derivatives form a coordinate frame of
78 `k[x₁, …, xₙ]`. -/
79 theorem isCoordinateFrame :
80     IsCoordinateFrame (X : Fin n → MvPolynomial (Fin n) k) (fun i => pderiv i) :=
81   .of_adjoin_eq_top adjoin_range_X' pderiv_X_eq_ite
82 
83 /-- The polynomial ring `k[x₁, …, xₙ]` has Krull dimension at most `n`. -/
84 theorem ringKrullDim_le : ringKrullDim (MvPolynomial (Fin n) k) ≤ n := by
85   rw [MvPolynomial.ringKrullDim_of_isNoetherianRing_of_finite, ringKrullDim_eq_zero_of_field,
86     zero_add, Nat.card_fin]

90 /-- `D_k(k[x])` is generated by the multiplications by the coordinates and the
91 partial derivatives (with `R = k[x]` itself among the generators). -/
92 theorem adjoin_eq_top :
93     Algebra.adjoin k (Set.range (multiplicationAlgHom k (MvPolynomial (Fin n) k)) ∪
94       Set.range fun i : Fin n => derivationElem (pderiv i)) = ⊤ :=
95   isCoordinateFrame.adjoin_eq_top

123 variable (k n) in
124 /-- The images of the Weyl generators: `inl i ↦ xᵢ`, `inr i ↦ ∂ᵢ`. -/
125 noncomputable def weylGenerator : PhaseVar n → DiffOp k n
126   | .inl i => multiplicationAlgHom k _ (X i)
127   | .inr i => derivationElem (pderiv i)
128 
129 omit [CharZero k] in
130 /-- The Weyl relations `[∂ᵢ, xⱼ] = δᵢⱼ`, `[xᵢ, xⱼ] = [∂ᵢ, ∂ⱼ] = 0`, in the sign
131 convention of `Matrix.J`. -/
132 theorem weylGenerator_commutator (i j : PhaseVar n) :
133     ringCommutator (weylGenerator k n i) (weylGenerator k n j) =
134       algebraMap k (DiffOp k n) (Matrix.J (Fin n) k i j) := by
135   apply Subtype.ext
136   rcases i with i | i <;> rcases j with j | j
137   · refine LinearMap.ext fun f => ?_
138     simp [weylGenerator, Matrix.J, mul_left_comm]
139   · refine LinearMap.ext fun f => ?_
140     simp [weylGenerator, Matrix.J, Matrix.one_apply, -pderiv_X, pderiv_X_eq_ite, eq_comm,
141       Derivation.leibniz]
142     split_ifs <;> simp
143   · refine LinearMap.ext fun f => ?_
144     simp [weylGenerator, Matrix.J, Matrix.one_apply, -pderiv_X, pderiv_X_eq_ite,
145       Derivation.leibniz]
146     split_ifs <;> simp
147   · simp only [weylGenerator, ringCommutator_apply, Subalgebra.coe_sub, Subalgebra.coe_mul,
148       coe_derivationElem, pderiv_mul_pderiv_comm i j, sub_self, Matrix.J]
149     simp

151 variable (k n) in
152 /-- The algebra map `A_n(k) → D_k(k[x₁, …, xₙ])`, `xᵢ ↦ xᵢ`, `∂ᵢ ↦ ∂/∂xᵢ`. -/
153 noncomputable def fromWeyl : WeylAlg k n →ₐ[k] DiffOp k n :=
154   Universal.freeWeylLift _ (weylGenerator k n) weylGenerator_commutator
155 
156 omit [CharZero k] in
157 @[simp] theorem fromWeyl_generator (i : PhaseVar n) :
158     fromWeyl k n (generator k n i) = weylGenerator k n i :=
159   Universal.freeWeylLift_generator _ _ _ i
160 
161 theorem fromWeyl_surjective : Function.Surjective (fromWeyl k n) := by
162   rw [← AlgHom.range_eq_top, eq_top_iff, ← adjoin_eq_top, Algebra.adjoin_le_iff]
163   rintro _ (⟨a, rfl⟩ | ⟨i, rfl⟩)
164   · have hX : Algebra.adjoin k (Set.range (X : Fin n → MvPolynomial (Fin n) k)) ≤
165         (fromWeyl k n).range.comap (multiplicationAlgHom k _) := by
166       rw [Algebra.adjoin_le_iff]
167       rintro _ ⟨i, rfl⟩
168       exact ⟨generator k n (.inl i), fromWeyl_generator _⟩
169     rw [adjoin_range_X'] at hX
170     exact hX (Algebra.mem_top (x := a))
171   · exact ⟨generator k n (.inr i), fromWeyl_generator _⟩
172 
173 theorem fromWeyl_injective : Function.Injective (fromWeyl k n) := by
174   have := Simplicity.isSimpleRing_weylAlg k n
175   exact (fromWeyl k n).toRingHom.injective

178 /-- **`D_k(k[x₁, …, xₙ]) ≅ A_n(k)`** in characteristic zero. -/
179 noncomputable def weylEquiv : WeylAlg k n ≃ₐ[k] DiffOp k n :=
180   AlgEquiv.ofBijective (fromWeyl k n) ⟨fromWeyl_injective, fromWeyl_surjective⟩

182 @[simp] theorem weylEquiv_apply (a : WeylAlg k n) : weylEquiv k n a = fromWeyl k n a := rfl

184 /-- **`D_k(k[x₁, …, xₙ])` is simple** in characteristic zero. -/
185 instance isSimpleRing : IsSimpleRing (DiffOp k n) :=
186   IsSimpleRing.of_ringEquiv (weylEquiv k n).toRingEquiv (Simplicity.isSimpleRing_weylAlg k n)

188 /-- **`D_k(k[x₁, …, xₙ])` is a domain** in characteristic zero: its principal
189 symbol identifies `gr D` with `k[x][ξ]` (`IsCoordinateFrame.isDomain`). -/
190 instance isDomain : IsDomain (DiffOp k n) :=
191   isCoordinateFrame.isDomain

193 /-- **`D_k(k[x₁, …, xₙ])` has an infinite centre** in characteristic zero: it
194 contains the scalars `k`. -/
195 theorem infinite_center : Infinite (Subring.center (DiffOp k n)) :=
