9 # `D_k(R)` is a domain when `R` is a domain
10 
11 Let `k` be a field of characteristic zero and `R` a commutative `k`-algebra
12 which is a domain. Then the ring `D_k(R)` of finite-order differential
13 operators is a domain (`DifferentialOperators.isDomain`). No finiteness,
14 smoothness or coordinate hypothesis is needed.
15 
16 ## Proof: polarization of the principal symbol
17 
18 Write `ad_b P = [P, b] = P b - b P`. The maps `ad_b` commute pairwise, are
19 additive in `b`, and satisfy the Leibniz rule
20 `ad_b (P Q) = P ad_b Q + (ad_b P) Q` (`commutatorMap_mul`), hence
21 `ad_b^N (P Q) = ∑ (N choose i) ad_b^i P · ad_b^{N-i} Q`
22 (`commutatorMap_pow_mul`). If `P` has order at most `m`, then `ad_b^m P` is
23 multiplication by an element `σ_P(b) ∈ R`, the principal symbol of `P`
24 evaluated on `db`.
25 
26 1. If `P` has exact order `m`, then `σ_P(b) ≠ 0` for some `b`
27    (`exists_symbol_ne_zero`). By induction: some `ad_c P` has exact order
28    `m - 1`, so `σ_{ad_c P}(b) ≠ 0` for some `b`; the value `σ_P(b + t c)` is a
29    polynomial in `t ∈ k` (`exists_polynomial_symbol`, binomial theorem for the
30    commuting `ad_b`, `ad_c`) whose `t`-coefficient is `m σ_{ad_c P}(b) ≠ 0`.
31    A nonzero polynomial over the domain `R` has finitely many roots and `k` is
32    infinite.
33 2. If `P`, `Q` have exact orders `m`, `n`, then
34    `ad_b^{m+n} (P Q) = (m+n choose m) ad_b^m P · ad_b^n Q`, which is
35    multiplication by `(m+n choose m) σ_P(b) σ_Q(b)`. Choosing `b` on a line
36    through points where `σ_P` and `σ_Q` do not vanish makes both nonzero at
37    once, again since a nonzero polynomial has finitely many roots. Hence
38    `P Q ≠ 0`.
39 
40 This is the classical fact that the principal symbol is multiplicative and
41 `gr D_k(R)` embeds into polynomial functions on `Ω_{R/k}`; compare
42 [McConnell–Robson, *Noncommutative Noetherian Rings*, 15.5.5]. The
43 polarization argument is AA-native.
44 -/

54 /-- `ad_b : P ↦ [P, b] = P b - b P`, as a `k`-linear endomorphism of `End_k(R)`. -/
55 def commutatorMap (b : R) : End (k := k) (R := R) →ₗ[k] End (k := k) (R := R) where
56   toFun P := commutator P b
57   map_add' P Q := commutator_add P Q b
58   map_smul' c P := commutator_smul c P b

80 /-- Leibniz rule: `ad_b (P Q) = P ad_b Q + (ad_b P) Q`. -/
81 theorem commutatorMap_mul (b : R) (P Q : End (k := k) (R := R)) :
82     commutatorMap b (P * Q) = P * commutatorMap b Q + commutatorMap b P * Q :=
83   commutator_mul P Q b

85 /-- General Leibniz rule:
86 `ad_b^N (P Q) = ∑_{i + j = N} (N choose i) ad_b^i P · ad_b^j Q`. -/
87 theorem commutatorMap_pow_mul (b : R) (P Q : End (k := k) (R := R)) (N : ℕ) :
88     (commutatorMap b ^ N) (P * Q) = ∑ ij ∈ Finset.antidiagonal N,
89       N.choose ij.1 • ((commutatorMap b ^ ij.1) P * (commutatorMap b ^ ij.2) Q) := by
90   induction N with
91   | zero => simp
92   | succ N ih =>
93       rw [Finset.sum_antidiagonal_choose_succ_nsmul
94         (fun i j => (commutatorMap b ^ i) P * (commutatorMap b ^ j) Q) N,
95         pow_succ', Module.End.mul_apply, ih, map_sum]
96       simp only [map_nsmul, commutatorMap_mul, smul_add, Finset.sum_add_distrib]
97       congr 1
98       · refine Finset.sum_congr rfl fun ij _ => ?_
99         rw [pow_succ', Module.End.mul_apply]
100       · refine Finset.sum_congr rfl fun ij hij => ?_
101         rw [Finset.mem_antidiagonal] at hij
102         rw [← Nat.choose_symm_of_eq_add hij.symm, pow_succ', Module.End.mul_apply]
103 
104 /-- `ad_b^j` lowers the order by `j`. -/
105 theorem commutatorMap_pow_mem_order (b : R) :
106     ∀ j m, ∀ P ∈ order (k := k) (R := R) (m + j), (commutatorMap b ^ j) P ∈ order m := by
107   intro j
108   induction j with
109   | zero => intro m P hP; simpa using hP
110   | succ j ih =>
111       intro m P hP
112       rw [pow_succ, Module.End.mul_apply]
113       exact ih m _ ((mem_order_succ_iff P (m + j)).mp (by rwa [← Nat.add_assoc] at hP) b)
114 
115 /-- `ad_b^{m+1}` kills operators of order at most `m`. -/
116 theorem commutatorMap_pow_eq_zero (b : R) {m : ℕ} {P : End (k := k) (R := R)}
117     (hP : P ∈ order m) : (commutatorMap b ^ (m + 1)) P = 0 := by
118   rw [pow_succ', Module.End.mul_apply]
119   exact (commutatorMap_pow_mem_order b m 0 P (by simpa using hP)) b

121 /-- `ad_b^j` kills operators of order less than `j`. -/
122 theorem commutatorMap_pow_eq_zero_of_lt (b : R) {m j : ℕ} (hmj : m < j)
123     {P : End (k := k) (R := R)} (hP : P ∈ order m) : (commutatorMap b ^ j) P = 0 := by
124   obtain ⟨l, rfl⟩ := Nat.exists_eq_add_of_lt hmj
125   rw [show m + l + 1 = l + (m + 1) by omega, pow_add, Module.End.mul_apply,
126     commutatorMap_pow_eq_zero b hP, map_zero]

128 /-- **The symbol is multiplicative.** If `P` has order at most `m` and `Q` order
129 at most `n`, then `ad_b^{m+n} (P Q) = (m+n choose m) ad_b^m P · ad_b^n Q`. -/
130 theorem commutatorMap_pow_mul_of_mem_order (b : R) {m n : ℕ} {P Q : End (k := k) (R := R)}
131     (hP : P ∈ order m) (hQ : Q ∈ order n) :
132     (commutatorMap b ^ (m + n)) (P * Q) =
133       (m + n).choose m • ((commutatorMap b ^ m) P * (commutatorMap b ^ n) Q) := by
134   rw [commutatorMap_pow_mul, Finset.sum_eq_single (m, n)]
135   · rintro ⟨i, j⟩ hij hne
136     rw [Finset.mem_antidiagonal] at hij
137     rcases lt_or_ge m i with hi | hi
138     · rw [commutatorMap_pow_eq_zero_of_lt b hi hP, zero_mul, smul_zero]
139     · have hj : n < j := by
140         rcases hi.lt_or_eq with hi | rfl
141         · omega
142         · exact absurd (by simp_all) hne
143       rw [commutatorMap_pow_eq_zero_of_lt b hj hQ, mul_zero, smul_zero]
144   · intro h
145     exact absurd (Finset.mem_antidiagonal.mpr (rfl : m + n = m + n)) h

153 /-- **Symbols along a line are polynomial.** If `P` has order at most `m`, there
154 is a polynomial `g ∈ R[T]` with `σ_P(b + t c) = g(t)` for all `t ∈ k`, whose
155 constant coefficient is `σ_P(b)` and whose linear coefficient is
156 `m σ_{ad_c P}(b)`, where `σ_P(b) = (ad_b^m P)(1)`. -/
157 theorem exists_polynomial_symbol {m : ℕ} {P : End (k := k) (R := R)} (hP : P ∈ order m)
158     (b c : R) : ∃ g : R[X], (∀ t : k,
159       (commutatorMap (b + t • c) ^ m) P 1 = g.eval (algebraMap k R t)) ∧
160       g.coeff 0 = (commutatorMap b ^ m) P 1 ∧
161       (∀ n, m = n + 1 → g.coeff 1 = (m : R) * (commutatorMap b ^ n) (commutatorMap c P) 1) := by
162   classical
163   let e : ℕ → R := fun i => m.choose i *
164     ((commutatorMap c ^ i * commutatorMap b ^ (m - i) :
165       Module.End k (End (k := k) (R := R))) P) 1
166   refine ⟨∑ i ∈ Finset.range (m + 1), C (e i) * X ^ i, fun t => ?_, ?_, fun n hn => ?_⟩
167   · have hc : Commute (t • commutatorMap (k := k) (R := R) c) (commutatorMap b) :=
168       (commutatorMap_commute c b).smul_left t
169     rw [commutatorMap_add, commutatorMap_smul, add_comm, hc.add_pow, LinearMap.sum_apply,
170       LinearMap.sum_apply, eval_finsetSum]
171     refine Finset.sum_congr rfl fun i _ => ?_
172     have hn : ((commutatorMap c ^ i * commutatorMap b ^ (m - i) *
173         ((m.choose i : ℕ) : Module.End k (End (k := k) (R := R))) :
174           Module.End k (End (k := k) (R := R))) P) =
175         m.choose i • ((commutatorMap c ^ i * commutatorMap b ^ (m - i) :
176           Module.End k (End (k := k) (R := R))) P) := by
177       rw [Module.End.mul_apply, Module.End.natCast_apply, map_nsmul]
178     rw [smul_pow, smul_mul_assoc, smul_mul_assoc, LinearMap.smul_apply, LinearMap.smul_apply, hn,
179       LinearMap.smul_apply]
180     simp only [eval_mul, eval_C, eval_pow, eval_X, e, nsmul_eq_mul]
181     rw [Algebra.smul_def, map_pow]
182     ring
183   · rw [finsetSum_coeff, Finset.sum_eq_single 0]
184     · simp [e]
185     · intro i _ hi
186       simp only [coeff_C_mul, coeff_X_pow]
187       rw [ite_eq_right_iff.mpr fun h => absurd h.symm hi, mul_zero]
188     · simp
189   · rw [finsetSum_coeff, Finset.sum_eq_single 1]
190     · subst hn
191       simp only [coeff_C_mul, e, Nat.choose_one_right, Nat.add_sub_cancel, pow_one, coeff_X_one,
192         mul_one]
193       rw [((commutatorMap_commute c b).pow_right n).eq, Module.End.mul_apply]
194     · intro i _ hi
195       simp [coeff_X_pow, Ne.symm hi]
196     · intro h
197       exact absurd (Finset.mem_range.mpr (by omega)) h

213 /-- A nonzero polynomial over `R` does not vanish at some point of `k`. -/
214 theorem exists_eval_algebraMap_ne_zero {g : R[X]} (hg : g ≠ 0) :
215     ∃ t : k, g.eval (algebraMap k R t) ≠ 0 := by

224 /-- **Nonvanishing of the symbol.** If `P` has order at most `m` but does not lie
225 in `orderBelow m` (exact order `m`), then `(ad_b^m P)(1) ≠ 0` for some `b`. -/
226 theorem exists_symbol_ne_zero :
227     ∀ m, ∀ P ∈ order (k := k) (R := R) m, P ∉ orderBelow m →
228       ∃ b : R, (commutatorMap b ^ m) P 1 ≠ 0 := by
229   intro m
230   induction m with
231   | zero =>
232       intro P hP hP0
233       refine ⟨0, fun h => hP0 ?_⟩
234       rw [orderBelow_zero, Submodule.mem_bot]
235       ext y
236       rw [apply_mul_of_mem_order_zero hP, show P 1 = 0 by simpa using h, zero_mul,
237         LinearMap.zero_apply]
238   | succ n ih =>
239       intro P hP hPn
240       obtain ⟨c, hc⟩ : ∃ c, commutator P c ∉ orderBelow n := by
241         by_contra h
242         push Not at h
243         exact hPn (mem_order_iff_commutator_mem_orderBelow.mpr h)
244       obtain ⟨b, hb⟩ := ih _ ((mem_order_succ_iff P n).mp hP c) hc
245       obtain ⟨g, hg, -, hg1⟩ := exists_polynomial_symbol (k := k) hP b c
246       have hg0 : g ≠ 0 := by
247         intro h0
248         have h1 := hg1 n rfl
249         rw [h0, coeff_zero] at h1
250         exact mul_ne_zero (natCast_ne_zero_of_charZero k _ (Nat.succ_ne_zero n)) hb h1.symm
251       obtain ⟨t, ht⟩ := exists_eval_algebraMap_ne_zero (k := k) hg0
252       exact ⟨b + t • c, by rwa [hg]⟩

264 /-- **A product of nonzero differential operators is nonzero** over a domain in
265 characteristic zero. -/
266 theorem mul_ne_zero_of_mem_algebra {P Q : End (k := k) (R := R)}
267     (hP : P ∈ algebra (k := k) (R := R)) (hQ : Q ∈ algebra (k := k) (R := R))
268     (hP0 : P ≠ 0) (hQ0 : Q ≠ 0) : P * Q ≠ 0 := by
269   obtain ⟨m, hPm, hPm'⟩ := exists_exact_order hP hP0
270   obtain ⟨n, hQn, hQn'⟩ := exists_exact_order hQ hQ0
271   obtain ⟨b, hb⟩ := exists_symbol_ne_zero m P hPm hPm'
272   obtain ⟨b', hb'⟩ := exists_symbol_ne_zero n Q hQn hQn'
273   obtain ⟨gP, hgP, hgP0, -⟩ := exists_polynomial_symbol (k := k) hPm b (b' - b)
274   obtain ⟨gQ, hgQ, -, -⟩ := exists_polynomial_symbol (k := k) hQn b (b' - b)
275   have hP' : gP ≠ 0 := fun h => hb (by rw [← hgP0, h, coeff_zero])
276   have hQ' : gQ ≠ 0 := fun h => hb' (by
277     have := hgQ 1
278     rw [one_smul, add_sub_cancel, h, eval_zero] at this
279     exact this)
280   obtain ⟨t, ht⟩ := exists_eval_algebraMap_ne_zero (k := k) (mul_ne_zero hP' hQ')
281   rw [eval_mul, ← hgP, ← hgQ] at ht
282   set b'' := b + t • (b' - b)
283   intro hPQ
284   have h := commutatorMap_pow_mul_of_mem_order (k := k) b'' hPm hQn
285   rw [hPQ, map_zero] at h
286   have h1 := LinearMap.congr_fun h 1
287   rw [LinearMap.zero_apply, LinearMap.smul_apply, Module.End.mul_apply,
288     apply_mul_of_mem_order_zero (commutatorMap_pow_mem_order b'' m 0 P (by simpa using hPm)),
289     nsmul_eq_mul] at h1
290   exact mul_ne_zero (natCast_ne_zero_of_charZero k _ (Nat.choose_pos (Nat.le_add_right m n)).ne') ht
291     h1.symm

293 /-- **`D_k(R)` is a domain** whenever `R` is a domain and `k` is a field of
294 characteristic zero. -/
295 theorem isDomain : IsDomain (algebra (k := k) (R := R)) := by
296   have : NoZeroDivisors (algebra (k := k) (R := R)) := ⟨fun {P Q} h => by
297     by_contra hne
298     push Not at hne
299     exact mul_ne_zero_of_mem_algebra P.2 Q.2 (fun h0 => hne.1 (Subtype.ext h0))
300       (fun h0 => hne.2 (Subtype.ext h0)) (congrArg Subtype.val h)⟩
301   exact NoZeroDivisors.to_isDomain (algebra (k := k) (R := R))
302 
303 end Domain
