11 # Differential operators on a localized polynomial ring
12 
13 Let `k` be a field, `S` a submonoid of `k[x₁, …, xₙ]` and `B = S⁻¹k[x₁, …, xₙ]`
14 (any `IsLocalization S B`). Examples: `k[x][f⁻¹]` (powers of `f`), the Laurent
15 polynomials, and the rational function field `k(x₁, …, xₙ)` (all nonzero
16 polynomials). The images of the coordinates `xᵢ` and the extended partial
17 derivatives `∂ᵢ` (`LocalizedPolynomialDerivations.localizedPderiv`) form a
18 coordinate frame of `B` (`isCoordinateFrame`), so the general coordinate
19 theory applies to `D = D_k(B)`.
20 
21 ## Main results
22 
23 In characteristic zero:
24 
25 * `isNoetherian_right`, `isNoetherianRing`: `D` is right and left Noetherian.
26 * `devLE_right`, `devLE_left`: right and left ideals of `D` have deviation at
27   most `2n` (since `dim B ≤ n`, `ringKrullDim_le`); `devLEOmega_right`,
28   `devLEOmega_left`.
29 * `adjoin_eq_top`: `D` is generated by `B` and the `∂ᵢ`.
30 
31 If moreover `0 ∉ S` (`S ≤ nonZeroDivisors`):
32 
33 * `eq_top_of_pderiv_mem`: the only nonzero ideal of `B` stable under all `∂ᵢ`
34   is `B` (lower the total degree of a numerator with the `∂ᵢ`).
35 * `isSimpleRing`: `D` is simple.
36 * `isDomain`: `D` is a domain (`gr D ≅ B[ξ]`).
37 * `infinite_center`: `D` has infinite centre.
38 
39 Also `AlgebraicAnalysis.ringKrullDim_le_of_isLocalization`: localization does
40 not increase Krull dimension.
41 
42 The arguments are AA-native; for the statements see [McConnell–Robson,
43 *Noncommutative Noetherian Rings*, 15.1.20, 15.3.8].
44 -/

48 /-- **Localization does not increase Krull dimension**: primes of `S⁻¹A`
49 embed order-preservingly into primes of `A` by contraction. -/
50 theorem ringKrullDim_le_of_isLocalization {A B : Type*} [CommRing A] [CommRing B] [Algebra A B]
51     (S : Submonoid A) [IsLocalization S B] : ringKrullDim B ≤ ringKrullDim A :=
52   Order.krullDim_le_of_strictMono
53     (fun p => ⟨p.asIdeal.under A, Ideal.IsPrime.under A p.asIdeal⟩)
54     fun _ _ h => (IsLocalization.orderEmbedding S B).strictMono h
55 
56 end AlgebraicAnalysis

68 /-- A partial derivative lowers the total degree by at least one. -/
69 theorem totalDegree_pderiv_le (p : MvPolynomial σ k) (i : σ) :
70     (pderiv i p).totalDegree ≤ p.totalDegree - 1 := by
71   refine Finset.sup_le fun m hm => ?_
72   rw [mem_support_iff, coeff_pderiv] at hm
73   have hc := left_ne_zero_of_mul hm
74   have hle := le_totalDegree (mem_support_iff.mpr hc)
75   rw [Finsupp.sum_add_index' (fun _ => rfl) (fun _ _ _ => rfl),
76     Finsupp.sum_single_index rfl] at hle
77   omega
78 
79 variable [CharZero k]
80 
81 /-- In characteristic zero, a polynomial with all partial derivatives zero is
82 constant. -/
83 theorem eq_C_of_pderiv_eq_zero {p : MvPolynomial σ k} (h : ∀ i, pderiv i p = 0) :
84     p = C (p.coeff 0) := by
85   classical
86   ext m
87   by_cases hm : m = 0
88   · subst hm
89     simp
90   obtain ⟨i, hi⟩ : ∃ i, m i ≠ 0 := by
91     by_contra hcon
92     push Not at hcon
93     exact hm (Finsupp.ext hcon)
94   rw [coeff_C]
95   simp only [Ne.symm hm, ↓reduceIte]
96   have hm' : m - Finsupp.single i 1 + Finsupp.single i 1 = m := by
97     ext l
98     simp only [Finsupp.coe_add, Finsupp.coe_tsub, Pi.add_apply, Pi.sub_apply,
99       Finsupp.single_apply]
100     split_ifs with hl
101     · subst hl
102       omega
103     · simp
104   have hcoeff := coeff_pderiv (i := i) p (m - Finsupp.single i 1)
105   rw [h i, hm'] at hcoeff
106   have hne : (((m - Finsupp.single i 1 : σ →₀ ℕ) i : ℕ) : k) + 1 ≠ 0 := Nat.cast_add_one_ne_zero _
107   rcases mul_eq_zero.mp (hcoeff.symm.trans (by simp)) with h0 | h0
108   · exact h0
109   · exact absurd h0 hne
110 
111 end Pderiv

117 variable {k : Type u} [Field k] {n : ℕ} (S : Submonoid (MvPolynomial (Fin n) k))
118   (B : Type u) [CommRing B] [Algebra (MvPolynomial (Fin n) k) B] [Algebra k B]
119   [IsScalarTower k (MvPolynomial (Fin n) k) B] [IsLocalization S B]

121 variable (k) in
122 /-- The ring `D_k(B)` of finite-order `k`-linear differential operators on
123 `B = S⁻¹k[x₁, …, xₙ]`. -/
124 abbrev DiffOp : Type u := algebra (k := k) (R := B)
125 
126 /-- The image of the coordinate `xᵢ` in `B`. -/
127 noncomputable abbrev coordinate (i : Fin n) : B :=
128   algebraMap (MvPolynomial (Fin n) k) B (X i)
129 
130 /-- Every polynomial lies in the subalgebra generated by the coordinates. -/
131 theorem algebraMap_mem_adjoin (p : MvPolynomial (Fin n) k) :
132     algebraMap (MvPolynomial (Fin n) k) B p ∈
133       Algebra.adjoin k (Set.range (coordinate (k := k) (n := n) B)) := by
134   have h := Algebra.adjoin_image k (IsScalarTower.toAlgHom k (MvPolynomial (Fin n) k) B)
135     (Set.range X)
136   rw [← Set.range_comp] at h
137   change _ ∈ Algebra.adjoin k (Set.range (IsScalarTower.toAlgHom k _ B ∘ X))
138   rw [h, MvPolynomial.adjoin_range_X]
139   exact ⟨p, Algebra.mem_top, rfl⟩
140 
141 include S
142 
143 /-- **The coordinates and the extended partial derivatives form a coordinate
144 frame of `B = S⁻¹k[x]`**: every element is `r/s` with `r, s` polynomials and
145 `s` a unit, and `∂ᵢ(xⱼ) = δᵢⱼ`. -/
146 theorem isCoordinateFrame :
147     IsCoordinateFrame (coordinate (k := k) (n := n) B) (localizedPderiv S B) where
148   exists_mul_eq b := by
149     obtain ⟨⟨r, s⟩, hrs⟩ := IsLocalization.surj S b
150     exact ⟨_, algebraMap_mem_adjoin B r, _, algebraMap_mem_adjoin B s,
151       IsLocalization.map_units B s, hrs⟩
152   dual := localizedPderiv_apply_algebraMap_X S B
153 
154 omit [Algebra k B] [IsScalarTower k (MvPolynomial (Fin n) k) B] in
155 /-- `B` is Noetherian. -/
156 theorem isNoetherianRing_base : IsNoetherianRing B :=
157   IsLocalization.isNoetherianRing S B inferInstance
158 
159 omit [Algebra k B] [IsScalarTower k (MvPolynomial (Fin n) k) B] in
160 /-- `dim S⁻¹k[x₁, …, xₙ] ≤ n`. -/
161 theorem ringKrullDim_le : ringKrullDim B ≤ n :=
162   (ringKrullDim_le_of_isLocalization S).trans PolynomialOperators.ringKrullDim_le

166 /-- `D_k(B)` is generated by `B` and the `∂ᵢ`. -/
167 theorem adjoin_eq_top :
168     Algebra.adjoin k (Set.range (multiplicationAlgHom k B) ∪
169       Set.range fun i => derivationElem (localizedPderiv S B i)) = ⊤ :=
170   (isCoordinateFrame S B).adjoin_eq_top

172 /-- `D_k(S⁻¹k[x₁, …, xₙ])` is right Noetherian. -/
173 theorem isNoetherian_right : IsNoetherian (DiffOp k B)ᵐᵒᵖ (DiffOp k B) :=
174   have := isNoetherianRing_base S B
175   (isCoordinateFrame S B).isNoetherian_right
176 
177 /-- `D_k(S⁻¹k[x₁, …, xₙ])` is left Noetherian. -/
178 theorem isNoetherianRing : IsNoetherianRing (DiffOp k B) :=
179   have := isNoetherianRing_base S B
180   (isCoordinateFrame S B).isNoetherianRing

182 /-- The right ideals of `D_k(S⁻¹k[x₁, …, xₙ])` have deviation at most `2n`. -/
183 theorem devLE_right : Module.DevLE (DiffOp k B)ᵐᵒᵖ (n + n) (DiffOp k B) := by
184   have := isNoetherianRing_base S B
185   simpa using (isCoordinateFrame S B).devLE_right (ringKrullDim_le S B)
186 
187 /-- The left ideals of `D_k(S⁻¹k[x₁, …, xₙ])` have deviation at most `2n`. -/
188 theorem devLE_left : Module.DevLE (DiffOp k B) (n + n) (DiffOp k B) := by
189   have := isNoetherianRing_base S B
190   simpa using (isCoordinateFrame S B).devLE_left (ringKrullDim_le S B)
191 
192 /-- The right ideals of `D_k(S⁻¹k[x₁, …, xₙ])` have deviation at most omega. -/
193 theorem devLEOmega_right : Module.DevLEOmega (DiffOp k B)ᵐᵒᵖ (DiffOp k B) :=

204 variable {S} (hS : S ≤ nonZeroDivisors (MvPolynomial (Fin n) k))
205 include hS

207 omit [CharZero k] [Algebra k B] [IsScalarTower k (MvPolynomial (Fin n) k) B] in
208 /-- `B` is a domain when `0 ∉ S`. -/
209 theorem isDomain_base : IsDomain B :=
210   IsLocalization.isDomain_of_le_nonZeroDivisors B hS
211 
212 /-- **The only nonzero ideal of `B` stable under all `∂ᵢ` is `B`.** A nonzero
213 `b ∈ J` has a nonzero polynomial numerator `r = b s ∈ J`; applying partial
214 derivatives lowers its total degree until a nonzero constant, a unit, lies
215 in `J`. -/
216 theorem eq_top_of_pderiv_mem (J : Ideal B) (hJ : J ≠ ⊥)
217     (hD : ∀ i, ∀ f ∈ J, localizedPderiv S B i f ∈ J) : J = ⊤ := by
218   have hinj := IsLocalization.injective B hS
219   have hpd : ∀ i (p : MvPolynomial (Fin n) k),
220       localizedPderiv S B i (algebraMap _ B p) = algebraMap _ B (pderiv i p) :=
221     fun i p => Derivation.congr_fun (localizedPderiv_compAlgebraMap S B i) p
222   have key : ∀ d, ∀ p : MvPolynomial (Fin n) k, p.totalDegree ≤ d → p ≠ 0 →
223       algebraMap _ B p ∈ J → J = ⊤ := by
224     intro d
225     induction d with
226     | zero =>
227         intro p hp hp0 hpJ
228         obtain ⟨c, rfl⟩ : ∃ c, p = C c := ⟨_, totalDegree_eq_zero_iff_eq_C.mp (by omega)⟩
229         have hc : c ≠ 0 := fun h => hp0 (by rw [h, C_0])
230         rw [← MvPolynomial.algebraMap_eq, ← IsScalarTower.algebraMap_apply] at hpJ
231         exact Ideal.eq_top_of_isUnit_mem J hpJ ((IsUnit.mk0 c hc).map (algebraMap k B))
232     | succ d ih =>
233         intro p hp hp0 hpJ
234         by_cases hall : ∀ i, pderiv i p = 0
235         · rw [eq_C_of_pderiv_eq_zero hall] at hpJ hp0
236           exact ih _ (by simp) hp0 hpJ
237         · push Not at hall
238           obtain ⟨i, hi⟩ := hall
239           refine ih _ ((totalDegree_pderiv_le p i).trans (by omega)) hi ?_
240           rw [← hpd]
241           exact hD i _ hpJ
242   obtain ⟨b, hbJ, hb0⟩ := Submodule.exists_mem_ne_zero_of_ne_bot hJ
243   obtain ⟨⟨r, s⟩, hrs⟩ := IsLocalization.surj S b
244   refine key _ r le_rfl (fun hr => hb0 ?_) (hrs ▸ J.mul_mem_right _ hbJ)
245   rw [hr, map_zero] at hrs
246   exact (IsLocalization.map_units B s).mul_left_eq_zero.mp hrs
247 
248 /-- **`D_k(S⁻¹k[x₁, …, xₙ])` is simple** in characteristic zero, for `0 ∉ S`. -/
249 theorem isSimpleRing : IsSimpleRing (DiffOp k B) :=
250   have := isDomain_base B hS
251   isSimpleRing_of_derivation fun J hJ hD => eq_top_of_pderiv_mem B hS J hJ fun _ => hD _
252 
253 /-- **`D_k(S⁻¹k[x₁, …, xₙ])` is a domain** in characteristic zero, for `0 ∉ S`. -/
254 theorem isDomain : IsDomain (DiffOp k B) :=
255   have := isDomain_base B hS
256   (isCoordinateFrame S B).isDomain
257 
258 omit [IsScalarTower k (MvPolynomial (Fin n) k) B] in
259 /-- **`D_k(S⁻¹k[x₁, …, xₙ])` has infinite centre** in characteristic zero, for
260 `0 ∉ S`. -/
261 theorem infinite_center : Infinite (Subring.center (DiffOp k B)) :=
262   have := isDomain_base B hS
263   have : Infinite k := CharZero.infinite k
264   DifferentialOperators.infinite_center k B
