9 # The order filtration of the ring of differential operators
10 
11 For a commutative `k`-algebra `R`, the ring `D = D_k(R)` of finite-order
12 `k`-linear differential operators (`DifferentialOperators.algebra`) is
13 filtered by order. This file packages the filtration and proves the two
14 basic structural facts of Grothendieck's theory, for every commutative `R`:
15 
16 * **Commutators drop the order.** If `P` has order at most `m` and `Q` order
17   at most `n`, then `PQ - QP` has order at most `m + n - 1`, and `PQ = QP`
18   when `m = n = 0` (`mul_sub_mul_mem_orderBelow`). Hence the associated graded
19   module of the order filtration is commutative
20   (`orderFiltration_grCommutative`).
21 * **First-order operators.** `D₁ = R ⊕ Der_k(R)` as `k`-modules:
22   `P ↦ (P 1, P - P(1))` (`orderOneEquiv`). In particular `D₁/D₀ ≅ Der_k(R)`.
23 
24 The statement `[D_m, D_n] ⊆ D_{m+n-1}` is written with `orderBelow`, the
25 order filtration shifted by one (`orderBelow 0 = 0`,
26 `orderBelow (n + 1) = order n`). An operator has order at most `n` exactly
27 when all its commutators with multiplications lie in `orderBelow n`
28 (`mem_order_iff_commutator_mem_orderBelow`).
29 
30 ## Main definitions
31 
32 * `DifferentialOperators.orderBelow`: the shifted order filtration.
33 * `DifferentialOperators.orderOneToDerivation`, `orderOneEquiv`: first-order
34   operators.
35 * `DifferentialOperators.orderFiltration`: the order filtration of
36   `D_k(R)` as a `RingFiltration k (algebra (k := k) (R := R))`.
37 
38 ## References
39 
40 [EGA IV] A. Grothendieck, *Éléments de géométrie algébrique IV*, 16.8.
41 [StacksD] The Stacks Project, Section 10.133 (tag 09CH).
42 The proofs are AA-native, by induction on the order using the Jacobi identity
43 `[[P, Q], a] = [P, [Q, a]] + [[P, a], Q]`.

118 /-- The order filtration shifted by one: `orderBelow 0 = 0` and
119 `orderBelow (n + 1) = order n`. -/
120 def orderBelow : ℕ → Submodule k (End (k := k) (R := R))
121   | 0 => ⊥
122   | n + 1 => order n
123 
124 @[simp] theorem orderBelow_zero : orderBelow (k := k) (R := R) 0 = ⊥ := rfl
125 
126 @[simp] theorem orderBelow_succ (n : ℕ) :
127     orderBelow (k := k) (R := R) (n + 1) = order n := rfl
128 
129 theorem orderBelow_le (n : ℕ) : orderBelow (k := k) (R := R) n ≤ order n := by
130   cases n with
131   | zero => exact bot_le
132   | succ n => exact order_mono_step n

134 /-- An operator has order at most `n` iff its commutators with all
135 multiplications lie in `orderBelow n`. -/
136 theorem mem_order_iff_commutator_mem_orderBelow {P : End (k := k) (R := R)} {n : ℕ} :
137     P ∈ order n ↔ ∀ a, commutator P a ∈ orderBelow n := by
138   cases n with
139   | zero => simp
140   | succ n => exact mem_order_succ_iff P n
141 
142 /-- `orderBelow n` is stable under composition with multiplications on the left. -/
143 theorem multiplication_mul_mem_orderBelow {Q : End (k := k) (R := R)} {n : ℕ} (a : R)
144     (hQ : Q ∈ orderBelow n) : multiplication a * Q ∈ orderBelow n := by
145   cases n with
146   | zero =>
147       rw [orderBelow_zero, Submodule.mem_bot] at hQ ⊢
148       rw [hQ, mul_zero]
149   | succ n => simpa using mul_mem_order (multiplication_mem_order_zero a) hQ
150 
151 /-- `orderBelow n` is stable under composition with multiplications on the right. -/
152 theorem mul_multiplication_mem_orderBelow {Q : End (k := k) (R := R)} {n : ℕ} (a : R)
153     (hQ : Q ∈ orderBelow n) : Q * multiplication a ∈ orderBelow n := by
154   cases n with
155   | zero =>
156       rw [orderBelow_zero, Submodule.mem_bot] at hQ ⊢
157       rw [hQ, zero_mul]
158   | succ n => simpa using mul_mem_order hQ (multiplication_mem_order_zero a)

160 /-! ## Commutators drop the order -/
161 
162 /-- **`[D_m, D_n] ⊆ D_{m+n-1}`.** If `P` has order at most `m` and `Q` order at
163 most `n`, then `PQ - QP` lies in `orderBelow (m + n)`: it has order at most
164 `m + n - 1`, and it vanishes when `m = n = 0`. -/
165 theorem mul_sub_mul_mem_orderBelow {P Q : End (k := k) (R := R)} {m n : ℕ}
166     (hP : P ∈ order m) (hQ : Q ∈ order n) : P * Q - Q * P ∈ orderBelow (m + n) := by
167   induction m generalizing n P Q with
168   | zero =>
169       rw [mem_order_zero_iff_eq_multiplication] at hP
170       have hcomm : P * Q - Q * P = -commutator Q (P 1) := by
171         rw [commutator, ← hP, neg_sub]
172       rw [hcomm, zero_add]
173       exact neg_mem (mem_order_iff_commutator_mem_orderBelow.mp hQ _)
174   | succ m ihm =>
175       induction n generalizing Q with
176       | zero =>
177           rw [mem_order_zero_iff_eq_multiplication] at hQ
178           have hcomm : P * Q - Q * P = commutator P (Q 1) := by
179             rw [commutator, ← hQ]
180           rw [hcomm, add_zero]
181           exact mem_order_iff_commutator_mem_orderBelow.mp hP _
182       | succ n ihn =>
183           rw [show m + 1 + (n + 1) = (m + n + 1) + 1 by omega, orderBelow_succ,
184             mem_order_succ_iff]
185           intro a
186           have hPa := (mem_order_succ_iff P m).mp hP a
187           have hQa := (mem_order_succ_iff Q n).mp hQ a
188           have key : commutator (P * Q - Q * P) a =
189               (P * commutator Q a - commutator Q a * P) +
190                 (commutator P a * Q - Q * commutator P a) := by
191             rw [commutator_sub, commutator_mul, commutator_mul]
192             abel
193           rw [key]
194           have h₁ := ihn hQa
195           have h₂ := ihm hPa hQ
196           rw [show m + 1 + n = (m + n) + 1 by omega, orderBelow_succ] at h₁
197           rw [show m + (n + 1) = (m + n) + 1 by omega, orderBelow_succ] at h₂
198           exact add_mem h₁ h₂

202 /-- A derivation is a differential operator of order at most one. -/
203 theorem derivation_mem_order_one (D : Derivation k R R) :
204     D.toLinearMap ∈ order (k := k) (R := R) 1 := by
205   rw [mem_order_succ_iff]
206   intro a
207   rw [commutator_derivation]
208   exact multiplication_mem_order_zero _
209 
210 /-- An operator of order at most one killing `1` satisfies the Leibniz rule. -/
211 theorem leibniz_of_mem_order_one {Q : End (k := k) (R := R)} (hQ : Q ∈ order 1)
212     (hQ1 : Q 1 = 0) (a b : R) : Q (a * b) = a * Q b + b * Q a := by
213   have h := (mem_order_zero_iff_eq_multiplication _).mp ((mem_order_succ_iff Q 0).mp hQ a)
214   have hb := LinearMap.congr_fun h b
215   simp only [commutator_apply, multiplication_apply, mul_one, hQ1, mul_zero, sub_zero] at hb
216   rw [mul_comm (Q a) b] at hb
217   exact sub_eq_iff_eq_add'.mp hb
218 
219 theorem sub_multiplication_mem_order_one {P : End (k := k) (R := R)} (hP : P ∈ order 1) :
220     P - multiplication (P 1) ∈ order 1 :=
221   sub_mem hP (order_mono_step 0 (multiplication_mem_order_zero _))

223 /-- The derivation part `P - P(1)` of a first-order operator `P`. -/
224 noncomputable def orderOneToDerivation :
225     order (k := k) (R := R) 1 →ₗ[k] Derivation k R R where
226   toFun P := Derivation.mk' (P.1 - multiplication (P.1 1)) fun a b => by
227     have h := leibniz_of_mem_order_one (Q := P.1 - multiplication (P.1 1))
228       (sub_multiplication_mem_order_one P.2) (by simp) a b
229     simpa only [smul_eq_mul] using h
230   map_add' P Q := by
231     ext a
232     simp only [Derivation.coe_mk', Submodule.coe_add, LinearMap.sub_apply, LinearMap.add_apply,
233       multiplication_apply, Derivation.add_apply]
234     ring
235   map_smul' c P := by
236     ext a
237     simp only [Derivation.coe_mk', Submodule.coe_smul, LinearMap.sub_apply, LinearMap.smul_apply,
238       multiplication_apply, RingHom.id_apply, Derivation.smul_apply]
239     rw [smul_sub, Algebra.smul_def, Algebra.smul_def, Algebra.smul_def]
240     ring

242 @[simp] theorem orderOneToDerivation_apply (P : order (k := k) (R := R) 1) (a : R) :
243     orderOneToDerivation P a = P.1 a - a * P.1 1 := by
244   simp [orderOneToDerivation, mul_comm]
245 
246 /-- Every first-order operator is a multiplication plus a derivation:
247 `P = P(1) + (P - P(1))`. -/
248 theorem eq_multiplication_add_orderOneToDerivation (P : order (k := k) (R := R) 1) :
249     P.1 = multiplication (P.1 1) + (orderOneToDerivation P).toLinearMap := by
250   ext a
251   simp [mul_comm]
252 
253 /-- **`D₁ = R ⊕ Der_k(R)`.** First-order operators are, as a `k`-module, the
254 direct sum of the multiplications and the derivations, by
255 `P ↦ (P 1, P - P(1))` with inverse `(a, D) ↦ a + D`. -/
256 noncomputable def orderOneEquiv :
257     order (k := k) (R := R) 1 ≃ₗ[k] R × Derivation k R R where
258   toFun P := (P.1 1, orderOneToDerivation P)
259   map_add' P Q := by simp
260   map_smul' c P := by simp
261   invFun p := ⟨multiplication p.1 + p.2.toLinearMap,
262     add_mem (order_mono_step 0 (multiplication_mem_order_zero _))
263       (derivation_mem_order_one p.2)⟩
264   left_inv P := by
265     apply Subtype.ext
266     exact (eq_multiplication_add_orderOneToDerivation P).symm
267   right_inv p := by
268     obtain ⟨a, D⟩ := p
269     refine Prod.ext (by simp) ?_
270     ext b
271     have h1 : D.toLinearMap 1 = 0 := D.map_one_eq_zero
272     simp only [orderOneToDerivation_apply, LinearMap.add_apply, multiplication_apply, mul_one,
273       h1, add_zero]
274     simp only [Derivation.coeFn_coe]
275     ring
276 
277 @[simp] theorem orderOneEquiv_apply (P : order (k := k) (R := R) 1) :
278     orderOneEquiv P = (P.1 1, orderOneToDerivation P) := rfl

282 /-- The order filtration of the ring `D_k(R)` of finite-order differential
283 operators: `F n` is the set of operators of order at most `n`. -/
284 def orderFiltration : RingFiltration k (algebra (k := k) (R := R)) where
285   F n := (order n).comap (algebra (k := k) (R := R)).val.toLinearMap
286   mono _ _ h _ hx := order_mono h hx
287   one_mem := (mem_order_zero_iff_eq_multiplication _).mpr (by ext; simp)
288   mul_le i j := Submodule.mul_le.mpr fun _ hx _ hy => mul_mem_order hx hy
289   exhaustive := by
290     refine eq_top_iff.mpr fun x _ => ?_
291     obtain ⟨n, hn⟩ := x.2
292     exact Submodule.mem_iSup_of_mem (ι := ℕ) n hn
293 
294 @[simp] theorem mem_orderFiltration_F {x : algebra (k := k) (R := R)} {n : ℕ} :
295     x ∈ (orderFiltration (k := k) (R := R)).F n ↔ (x : End (k := k) (R := R)) ∈ order n :=
296   Iff.rfl
297 
298 theorem mem_orderFiltration_below {x : algebra (k := k) (R := R)} {n : ℕ} :
299     x ∈ (orderFiltration (k := k) (R := R)).below n ↔
300       (x : End (k := k) (R := R)) ∈ orderBelow n := by
301   cases n with
302   | zero =>
303       simp only [RingFiltration.below_zero, Submodule.mem_bot, orderBelow_zero]
304       exact ⟨fun h => by rw [h]; rfl, fun h => Subtype.ext h⟩
305   | succ n => exact Iff.rfl
306 
307 /-- **The associated graded ring of `D_k(R)` is commutative**: for operators
308 of orders at most `i` and `j`, `PQ - QP` lies below order `i + j`. -/
309 theorem orderFiltration_grCommutative :
310     (orderFiltration (k := k) (R := R)).GrCommutative := by
311   intro i j x y hx hy
312   rw [mem_orderFiltration_below]
313   exact mul_sub_mul_mem_orderBelow hx hy
