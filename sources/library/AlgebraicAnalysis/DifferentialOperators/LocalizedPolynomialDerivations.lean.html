Extend a k-derivation through a localization A→B, by the formally-etale base-change equivalence for Kähler differentials.

29/-- Extend a `k`-derivation through a localization `A → B`, by the
30formally-etale base-change equivalence for Kähler differentials. -/
31noncomputable def extendDerivation
32 (S : Submonoid A) [IsLocalization S B]
33 (D : Derivation k A B) : Derivation k B B := by
34 letI : Algebra.FormallyEtale A B :=
35 Algebra.FormallyEtale.of_isLocalization (Rₘ := B) S
36 let base : B ⊗[A] KaehlerDifferential k A →ₗ[B] B :=
37 D.liftKaehlerDifferential.liftBaseChange B
38 let pull : KaehlerDifferential k B →ₗ[B] B :=
39 base.comp
40 (KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale
41 k A B).symm.toLinearMap
42 exact KaehlerDifferential.linearMapEquivDerivation k B pull

45theorem extendDerivation_compAlgebraMap

47 (D : Derivation k A B) :

A derivation of a localization is uniquely determined by its restriction to the original algebra.

57/-- A derivation of a localization is uniquely determined by its restriction
58to the original algebra. -/
59theorem derivation_ext_of_compAlgebraMap_eq
60 (S : Submonoid A) [IsLocalization S B]
61 {D₁ D₂ : Derivation k B B}
62 (h : D₁.compAlgebraMap A = D₂.compAlgebraMap A) :
63 D₁ = D₂ := by
64 letI : Algebra.FormallyEtale A B :=
65 Algebra.FormallyEtale.of_isLocalization (Rₘ := B) S
66 let e := KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale k A B
67 have hbase :
68 (D₁.liftKaehlerDifferential.restrictScalars A).comp
69 (KaehlerDifferential.map k k A B) =
70 (D₂.liftKaehlerDifferential.restrictScalars A).comp
71 (KaehlerDifferential.map k k A B) := by
72 apply Derivation.liftKaehlerDifferential_unique
73 apply Derivation.ext
74 intro x
75 simpa [KaehlerDifferential.map_D,
76 Derivation.liftKaehlerDifferential_comp_D] using
77 Derivation.congr_fun h x
78 have hpull :
79 D₁.liftKaehlerDifferential.comp e.toLinearMap =
80 D₂.liftKaehlerDifferential.comp e.toLinearMap := by
81 apply LinearMap.ext
82 intro z
83 induction z using TensorProduct.induction_on with
84 | zero => simp
85 | add x y hx hy => simp only [map_add, hx, hy]
86 | tmul a x =>
87 simp only [e,
88 KaehlerDifferential.tensorKaehlerEquivOfFormallyEtale_apply,
89 KaehlerDifferential.mapBaseChange_tmul,
90 LinearMap.comp_apply, LinearEquiv.coe_coe, LinearMap.map_smul]
91 exact congrArg (a • ·) (LinearMap.congr_fun hbase x)
92 apply Derivation.ext
93 intro x
94 have hmaps : D₁.liftKaehlerDifferential = D₂.liftKaehlerDifferential := by
95 apply LinearMap.ext
96 intro w
97 obtain ⟨z, rfl⟩ := e.surjective w
98 exact LinearMap.congr_fun hpull z
99 simpa [Derivation.liftKaehlerDifferential_comp_D] using
100 LinearMap.congr_fun hmaps (KaehlerDifferential.D k B x)

The polynomial ring in n coordinate variables over k.

106/-- The polynomial ring in `n` coordinate variables over `k`. -/
107abbrev PolynomialRing := MvPolynomial (Fin n) k

109private theorem pderiv_comm (i j : Fin n) (f : MvPolynomial (Fin n) k) :
110 MvPolynomial.pderiv i (MvPolynomial.pderiv j f) =
111 MvPolynomial.pderiv j (MvPolynomial.pderiv i f) := by
112 induction f using MvPolynomial.induction_on with
113 | C c => simp
114 | add p q hp hq => simp [hp, hq]
115 | mul_X p l hp =>
116 by_cases hij : i = j
117 · subst j
118 simp [hp, mul_comm]
119 · by_cases hil : i = l
120 · subst l
121 by_cases hjl : j = i
122 · exact False.elim (hij hjl.symm)
123 · simp [MvPolynomial.pderiv_mul, hp, hjl, hij, mul_comm] <;> ring
124 · by_cases hjl : j = l
125 · subst l
126 simp [MvPolynomial.pderiv_mul, hp, hil, hij, mul_comm] <;> ring
127 · simp [MvPolynomial.pderiv_mul, hp, hil, hjl, hij, mul_comm] <;> ring

129/-- The `i`th polynomial partial derivative, transported to a localization. -/
130noncomputable def localizedPderiv
131 (S : Submonoid (PolynomialRing (k := k) (n := n)))
132 (B : Type u) [CommRing B]
133 [Algebra (PolynomialRing (k := k) (n := n)) B]
134 [Algebra k B]
135 [IsScalarTower k (PolynomialRing (k := k) (n := n)) B]
136 [IsLocalization S B] (i : Fin n) : Derivation k B B :=
137 extendDerivation k _ B S
138 ((Algebra.linearMap _ B).compDer (MvPolynomial.pderiv i))

171theorem localizedPderiv_comm
172 (S : Submonoid (PolynomialRing (k := k) (n := n)))
173 (B : Type u) [CommRing B]
174 [Algebra (PolynomialRing (k := k) (n := n)) B]
175 [Algebra k B]
176 [IsScalarTower k (PolynomialRing (k := k) (n := n)) B]
177 [IsLocalization S B] (i j : Fin n) :
178 (localizedPderiv S B i).toLinearMap.comp
179 (localizedPderiv S B j).toLinearMap =
180 (localizedPderiv S B j).toLinearMap.comp
181 (localizedPderiv S B i).toLinearMap := by
182 have hcomm : ⁅localizedPderiv S B i, localizedPderiv S B j⁆ = 0 := by
183 apply derivation_ext_of_compAlgebraMap_eq k _ B S
184 apply Derivation.ext
185 intro f
186 change localizedPderiv S B i (localizedPderiv S B j
187 (algebraMap _ B f)) - localizedPderiv S B j (localizedPderiv S B i
188 (algebraMap _ B f)) = 0
189 have hi := Derivation.congr_fun
190 (localizedPderiv_compAlgebraMap (k := k) S B i) f
191 have hj := Derivation.congr_fun
192 (localizedPderiv_compAlgebraMap (k := k) S B j) f
193 change localizedPderiv S B i (algebraMap _ B f) = _ at hi
194 change localizedPderiv S B j (algebraMap _ B f) = _ at hj
195 change localizedPderiv S B i (algebraMap _ B f) =
196 algebraMap _ B (MvPolynomial.pderiv i f) at hi
197 change localizedPderiv S B j (algebraMap _ B f) =
198 algebraMap _ B (MvPolynomial.pderiv j f) at hj
199 rw [hj, hi]
200 have hij := Derivation.congr_fun
201 (localizedPderiv_compAlgebraMap (k := k) S B i)
202 (MvPolynomial.pderiv j f)
203 have hji := Derivation.congr_fun
204 (localizedPderiv_compAlgebraMap (k := k) S B j)
205 (MvPolynomial.pderiv i f)
206 change localizedPderiv S B i
207 (algebraMap _ B (MvPolynomial.pderiv j f)) = _ at hij
208 change localizedPderiv S B j
209 (algebraMap _ B (MvPolynomial.pderiv i f)) = _ at hji
210 change localizedPderiv S B i
211 (algebraMap _ B (MvPolynomial.pderiv j f)) =
212 algebraMap _ B (MvPolynomial.pderiv i (MvPolynomial.pderiv j f)) at hij
213 change localizedPderiv S B j
214 (algebraMap _ B (MvPolynomial.pderiv i f)) =
215 algebraMap _ B (MvPolynomial.pderiv j (MvPolynomial.pderiv i f)) at hji
216 rw [hij, hji]
217 rw [pderiv_comm i j f]
218 simp
219 apply LinearMap.ext
220 intro x
221 have hx := congrArg (fun d : Derivation k B B => d x) hcomm
222 exact sub_eq_zero.mp (by simpa [Derivation.commutator_apply,
223 LinearMap.comp_apply] using hx)
