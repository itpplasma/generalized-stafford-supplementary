Differential operators of order at most n.

44/-- Differential operators of order at most `n`. -/
45def order : ℕ → Submodule k (End (k := k) (R := R))
46 | 0 =>
47 { carrier := {P | ∀ a, commutator P a = 0}
48 zero_mem' := by simp [commutator]
49 add_mem' := by
50 intro P Q hP hQ a
51 have heq : commutator (P + Q) a = commutator P a + commutator Q a := by
52 ext x
53 simp [commutator_apply, mul_add, sub_eq_add_neg, add_assoc, add_comm,
54 add_left_comm]
55 calc
56 commutator (P + Q) a = commutator P a + commutator Q a := heq
57 _ = 0 := by rw [hP a, hQ a, add_zero]
58 smul_mem' := by
59 intro c P hP a
60 have heq : commutator (c • P) a = c • commutator P a := by
61 ext x
62 simp [commutator_apply, smul_sub]
63 rw [heq, hP a, smul_zero] }
64 | n + 1 =>
65 { carrier := {P | ∀ a, commutator P a ∈ order n}
66 zero_mem' := by simp [commutator]
67 add_mem' := by
68 intro P Q hP hQ a
69 have heq : commutator (P + Q) a = commutator P a + commutator Q a := by
70 ext x
71 simp [commutator_apply, mul_add, sub_eq_add_neg, add_assoc, add_comm,
72 add_left_comm]
73 rw [heq]
74 exact (order n).add_mem (hP a) (hQ a)
75 smul_mem' := by
76 intro c P hP a
77 have heq : commutator (c • P) a = c • commutator P a := by
78 ext x
79 simp [commutator_apply, smul_sub]
80 rw [heq]
81 exact (order n).smul_mem c (hP a) }

83@[simp] theorem mem_order_zero_iff (P : End (k := k) (R := R)) :
84 P ∈ order (k := k) (R := R) 0 ↔ ∀ a, commutator P a = 0 := Iff.rfl

87 P ∈ order (k := k) (R := R) (n + 1) ↔

92theorem mem_order_zero_iff_eq_multiplication (P : End (k := k) (R := R)) :
93 P ∈ order (k := k) (R := R) 0 ↔
94 P = multiplication (k := k) (P 1) := by
95 constructor
96 · intro h
97 ext a
98 have ha := LinearMap.congr_fun (h a) 1
99 simpa [mul_comm] using sub_eq_zero.mp (by simpa [commutator_apply] using ha)
100 · intro hP
101 rw [mem_order_zero_iff]
102 intro a
103 rw [hP]
104 ext x
105 simp [commutator_apply, mul_assoc, mul_comm]

107theorem order_mono_step (n : ℕ) :
108 order (k := k) (R := R) n ≤ order (n + 1) := by
109 induction n with
110 | zero =>
111 intro P hP
112 rw [mem_order_succ_iff]
113 intro a
114 rw [hP a]
115 exact (order 0).zero_mem
116 | succ n ih =>
117 intro P hP
118 rw [mem_order_succ_iff] at hP ⊢
119 intro a
120 exact ih (hP a)
121
122theorem order_mono {m n : ℕ} (h : m ≤ n) :
123 order (k := k) (R := R) m ≤ order n := by
124 induction n, h using Nat.le_induction with
125 | base => exact le_rfl
126 | succ n _ ih => exact ih.trans (order_mono_step n)

128theorem commutator_mul (P Q : End (k := k) (R := R)) (a : R) :
129 commutator (P * Q) a = P * commutator Q a + commutator P a * Q := by
130 ext x
131 simp [commutator_apply, Module.End.mul_apply]

133private theorem orderZero_mul {P Q : End (k := k) (R := R)} {n : ℕ}
134 (hP : P ∈ order (k := k) (R := R) 0) (hQ : Q ∈ order n) :
135 P * Q ∈ order n := by
136 induction n generalizing Q with
137 | zero =>
138 rw [mem_order_zero_iff_eq_multiplication] at hP hQ
139 rw [mem_order_zero_iff]
140 intro a
141 rw [hP, hQ]
142 ext x
143 simp [commutator_apply, mul_assoc, mul_comm, mul_left_comm]
144 | succ n ih =>
145 rw [mem_order_succ_iff] at hQ ⊢
146 intro a
147 rw [commutator_mul, hP a, zero_mul, add_zero]
148 exact ih (hQ a)

Orders add under composition.

150/-- Orders add under composition. -/
151theorem mul_mem_order {P Q : End (k := k) (R := R)} {m n : ℕ}
152 (hP : P ∈ order (k := k) (R := R) m) (hQ : Q ∈ order n) :
153 P * Q ∈ order (m + n) := by
154 induction m generalizing P n Q with
155 | zero => simpa using orderZero_mul hP hQ
156 | succ m ihm =>
157 induction n generalizing P Q with
158 | zero =>
159 rw [Nat.add_zero, mem_order_succ_iff] at hP ⊢
160 intro a
161 rw [commutator_mul, hQ a, mul_zero, zero_add]
162 simpa using ihm (hP a) hQ
163 | succ n ihn =>
164 have hQ' : Q ∈ order (n + 1) := hQ
165 rw [mem_order_succ_iff] at hP hQ
166 rw [Nat.add_succ, mem_order_succ_iff]
167 intro a
168 rw [commutator_mul]
169 exact (order ((m + 1) + n)).add_mem
170 (ihn hP (hQ a))
171 (by simpa only [Nat.succ_add, Nat.add_succ] using ihm (hP a) hQ')

The algebra of all finite-order k-linear differential operators on R.

173/-- The algebra of all finite-order `k`-linear differential operators on `R`. -/
174def algebra : Subalgebra k (End (k := k) (R := R)) where
175 carrier := {P | ∃ n, P ∈ order n}
176 zero_mem' := ⟨0, (order 0).zero_mem⟩
177 add_mem' := by
178 rintro P Q ⟨m, hP⟩ ⟨n, hQ⟩
179 refine ⟨max m n, (order (max m n)).add_mem ?_ ?_⟩
180 · exact order_mono (Nat.le_max_left _ _) hP
181 · exact order_mono (Nat.le_max_right _ _) hQ
182 mul_mem' := by
183 rintro P Q ⟨m, hP⟩ ⟨n, hQ⟩
184 exact ⟨m + n, mul_mem_order hP hQ⟩
185 one_mem' := by
186 refine ⟨0, (mem_order_zero_iff_eq_multiplication _).2 ?_⟩
187 ext x
188 simp [multiplication_apply]
189 algebraMap_mem' := by
190 intro c
191 refine ⟨0, (mem_order_zero_iff_eq_multiplication _).2 ?_⟩
192 ext x
193 simp [multiplication_apply]
