9 # The principal symbol of `D_k(R)` in coordinates
10 
11 A **coordinate frame** of a commutative `k`-algebra `R`
12 (`DifferentialOperators.IsCoordinateFrame x D`) consists of elements
13 `x₁, …, x_r` generating `R` up to units (every `b ∈ R` satisfies `b s = r`
14 with `r, s ∈ k[x]` and `s` a unit) and derivations `∂₁, …, ∂_r` dual to them,
15 `∂ᵢ(xⱼ) = δᵢⱼ`. No commutation of the `∂ᵢ` is assumed. Polynomial rings and
16 all their localizations carry such frames. This file proves:
17 
18 * **Commutator test** (`IsCoordinateFrame.mem_order_iff`): an operator has
19   order at most `n` once its commutators with the `xᵢ` lie in `orderBelow n`.
20   Since `[P, ab] = [P, a] b + a [P, b]`, the multiplications whose commutator
21   with `P` lies in `orderBelow n` form a subalgebra, and
22   `[P, b] s = [P, r] - b [P, s]` removes the unit denominators
23   (`commutator_mem_of_exists_mul_eq`).
24 * **Euler identity modulo lower order** (any characteristic): for `P` of order
25   at most `m`, `m P - ∑ᵢ [P, xᵢ] ∂ᵢ` has order at most `m - 1`
26   (`IsCoordinateFrame.nsmul_sub_sum_mem_orderBelow`).
27 * **Filtered generation** (characteristic zero): every operator of order at
28   most `n + 1` is `∑ᵢ Qᵢ ∂ᵢ` modulo order `n`, with `Qᵢ = [P, xᵢ]/(n+1)` of
29   order at most `n` (`sub_sum_mem_order`). Hence the order filtration is
30   generated in degree one by the `∂ᵢ` over `R` (`polynomialGenerators`),
31   `D_k(R)` is generated by `R` and the `∂ᵢ` (`adjoin_eq_top`), and the symbol
32   map `R[ξ₁, …, ξ_r] ↠ gr D_k(R)` is a surjective symbol presentation.
33 * Consequently, if `R` is Noetherian of Krull dimension at most `d`, then
34   `D_k(R)` is right and left Noetherian and its right and left ideals have
35   deviation at most `d + r` (`isNoetherian_right`, `isNoetherianRing`,
36   `devLE_right`, `devLE_left`, `devLEOmega_right`, `devLEOmega_left`).
37 
38 * **Principal symbol** (characteristic zero): the presentation is injective,
39   so `gr D_k(R) ≅ R[ξ]` (`gradedSymbol`), and `D_k(R)` is a domain when `R` is
40   (`isDomain`), by `RingFiltration.PolynomialGenerators.gradedSymbol` with the
41   dual relations `[∂ᵢ, xⱼ] = δᵢⱼ` (`derivationElem_commutator`).
42 
43 Simplicity is in `DifferentialOperators.CoordinateSimplicity`.
44 
45 ## References
46 
47 The Euler-identity argument is AA-native. For the statement see
48 [EGA IV, 16.11.2] and [McConnell–Robson, 15.5].
49 -/

70 /-- **Commutator test on generators up to units.** Let `S` be a submodule of
71 operators stable under composition with multiplications on both sides.
72 Suppose every `b ∈ R` satisfies `b s = r` with `r, s` in the subalgebra
73 generated by the `xᵢ` and `s` a unit (for instance, `R` is a localization of
74 `k[x]`). If `[P, xᵢ] ∈ S` for all `i`, then `[P, b] ∈ S` for every `b ∈ R`:
75 the multiplications with commutator in `S` form a subalgebra, and
76 `[P, b] s = [P, r] - b [P, s]`. -/
77 theorem commutator_mem_of_exists_mul_eq {σ : Type*} {x : σ → R}
78     (hx : ∀ b : R, ∃ r ∈ Algebra.adjoin k (Set.range x),
79       ∃ s ∈ Algebra.adjoin k (Set.range x), IsUnit s ∧ b * s = r)
80     (S : Submodule k (End (k := k) (R := R)))
81     (hl : ∀ a Q, Q ∈ S → multiplication a * Q ∈ S)
82     (hr : ∀ a Q, Q ∈ S → Q * multiplication a ∈ S)
83     {P : End (k := k) (R := R)} (hP : ∀ i, commutator P (x i) ∈ S) (b : R) :
84     commutator P b ∈ S := by
85   let T : Subalgebra k R :=
86     { carrier := {a | commutator P a ∈ S}
87       mul_mem' := fun {a b} ha hb => by
88         rw [Set.mem_ofPred_eq, commutator_mul_right]
89         exact add_mem (hr _ _ ha) (hl _ _ hb)
90       add_mem' := fun {a b} ha hb => by
91         rw [Set.mem_ofPred_eq, commutator_add_right]
92         exact add_mem ha hb
93       algebraMap_mem' := fun c => by
94         rw [Set.mem_ofPred_eq, commutator_algebraMap]
95         exact zero_mem _ }
96   have hT : Algebra.adjoin k (Set.range x) ≤ T :=
97     Algebra.adjoin_le (by rintro _ ⟨i, rfl⟩; exact hP i)
98   obtain ⟨r, hr', s, hs, ⟨u, rfl⟩, hbs⟩ := hx b
99   have hmul : commutator P b * multiplication (u : R) ∈ S := by
100     have h := commutator_mul_right P b u
101     rw [hbs] at h
102     rw [show commutator P b * multiplication (u : R) =
103       commutator P r - multiplication b * commutator P u by rw [h]; abel]
104     exact sub_mem (hT hr') (hl _ _ (hT hs))
105   have h1 := hr ((u⁻¹ : Rˣ) : R) _ hmul
106   rwa [mul_assoc, ← multiplication_mul, Units.mul_inv, show multiplication (k := k) (1 : R) = 1
107     from LinearMap.ext fun _ => one_mul _, mul_one] at h1
108 
109 /-- **Commutator test on generators.** Let `S` be a submodule of operators
110 stable under composition with multiplications on both sides. If `R` is
111 generated by the `xᵢ` and `[P, xᵢ] ∈ S` for all `i`, then `[P, a] ∈ S` for
112 every `a ∈ R`. -/
113 theorem commutator_mem_of_adjoin {σ : Type*} {x : σ → R}
114     (hx : Algebra.adjoin k (Set.range x) = ⊤) (S : Submodule k (End (k := k) (R := R)))
115     (hl : ∀ a Q, Q ∈ S → multiplication a * Q ∈ S)
116     (hr : ∀ a Q, Q ∈ S → Q * multiplication a ∈ S)
117     {P : End (k := k) (R := R)} (hP : ∀ i, commutator P (x i) ∈ S) (a : R) :
118     commutator P a ∈ S :=
119   commutator_mem_of_exists_mul_eq
120     (fun b => ⟨b, hx ▸ Algebra.mem_top, 1, Subalgebra.one_mem _, isUnit_one, mul_one b⟩)
121     S hl hr hP a
122 
123 end Commutant

129 /-- A **coordinate frame** of the commutative `k`-algebra `R`: elements
130 `x₁, …, x_r` which generate `R` up to units (every `b ∈ R` satisfies `b s = r`
131 with `r, s ∈ k[x₁, …, x_r] ⊆ R` and `s` a unit), together with derivations
132 `∂₁, …, ∂_r` dual to them, `∂ᵢ(xⱼ) = δᵢⱼ`. No commutation of the `∂ᵢ` is
133 assumed. Examples: a polynomial ring with its partial derivatives, and any
134 localization of it (`LocalizedPolynomialOperators`). -/
135 structure IsCoordinateFrame (x : σ → R) (D : σ → Derivation k R R) : Prop where
136   /-- Every element is a quotient of two elements of `k[x]`, by a unit. -/
137   exists_mul_eq : ∀ b : R, ∃ r ∈ Algebra.adjoin k (Set.range x),
138     ∃ s ∈ Algebra.adjoin k (Set.range x), IsUnit s ∧ b * s = r
139   /-- The derivations are dual to the coordinates. -/
140   dual : ∀ i j, D i (x j) = if i = j then 1 else 0

149   exists_mul_eq b := ⟨b, hx ▸ Algebra.mem_top, 1, Subalgebra.one_mem _, isUnit_one, mul_one b⟩

155 /-- **Commutator test.** An operator has order at most `n` iff its commutators
156 with the coordinates lie in `orderBelow n`. -/
157 theorem mem_order_iff {P : End (k := k) (R := R)} {n : ℕ} :
158     P ∈ order n ↔ ∀ i, commutator P (x i) ∈ orderBelow n := by
159   rw [mem_order_iff_commutator_mem_orderBelow]
160   exact ⟨fun h i => h (x i), fun h => commutator_mem_of_exists_mul_eq hF.exists_mul_eq _
161     (fun a _ hQ => multiplication_mul_mem_orderBelow a hQ)
162     (fun a _ hQ => mul_multiplication_mem_orderBelow a hQ) h⟩

164 /-- An operator commuting with all coordinates is a multiplication. -/
165 theorem mem_order_zero_of_commutator_eq_zero {P : End (k := k) (R := R)}
166     (hP : ∀ i, commutator P (x i) = 0) : P ∈ order 0 :=
167   hF.mem_order_iff.mpr fun i => by rw [hP i]; exact zero_mem _

171 /-- **Euler identity modulo lower order.** For an operator `P` of order at most
172 `m`, `m P - ∑ᵢ [P, xᵢ] ∂ᵢ` lies in `orderBelow m`. Holds in every
173 characteristic. -/
174 theorem nsmul_sub_sum_mem_orderBelow :
175     ∀ m, ∀ P ∈ order (k := k) (R := R) m,
176       m • P - ∑ i, commutator P (x i) * (D i).toLinearMap ∈ orderBelow m := by
177   intro m
178   induction m with
179   | zero =>
180       intro P hP
181       rw [mem_order_zero_iff] at hP
182       simp [hP]
183   | succ n ih =>
184       intro P hP
185       rw [orderBelow_succ, hF.mem_order_iff]
186       intro j
187       have hmul : ∀ i, commutator P (x i) * multiplication (D i (x j)) =
188           if i = j then commutator P (x j) else 0 := by
189         intro i
190         rw [hF.dual]
191         split_ifs with hij
192         · subst hij
193           ext y
194           simp
195         · ext y
196           simp
197       have key : commutator ((n + 1) • P - ∑ i, commutator P (x i) * (D i).toLinearMap) (x j) =
198           n • commutator P (x j) -
199             ∑ i, commutator (commutator P (x j)) (x i) * (D i).toLinearMap := by
200         rw [commutator_sub, commutator_nsmul, commutator_sum]
201         simp only [commutator_mul, commutator_derivation, hmul, Finset.sum_add_distrib,
202           Finset.sum_ite_eq', Finset.mem_univ, ite_true, commutator_commutator_comm P (x _) (x j)]
203         rw [succ_nsmul]
204         abel
205       rw [key]
206       exact ih _ ((mem_order_succ_iff P n).mp hP (x j))

251 /-- **Filtered generation by the coordinate derivations** (characteristic
252 zero). If `P` has order at most `n + 1`, then
253 `P - ∑ᵢ ((n+1)⁻¹ [P, xᵢ]) ∂ᵢ` has order at most `n`. -/
254 theorem sub_sum_mem_order {n : ℕ} {P : End (k := k) (R := R)} (hP : P ∈ order (n + 1)) :
255     P - ∑ i, (((n : k) + 1)⁻¹ • commutator P (x i)) * (D i).toLinearMap ∈ order n := by
256   have h := (order n).smul_mem ((n : k) + 1)⁻¹ (hF.nsmul_sub_sum_mem_orderBelow (n + 1) P hP)
257   have hne : ((n : k) + 1) ≠ 0 := by exact_mod_cast Nat.succ_ne_zero n
258   convert h using 1
259   rw [smul_sub, Finset.smul_sum, ← Nat.cast_smul_eq_nsmul k, smul_smul, Nat.cast_add,
260     Nat.cast_one, inv_mul_cancel₀ hne, one_smul]
261   simp only [smul_mul_assoc]
