12 # Smooth domains have no differential ideals; `D_k(A)` is simple
13 
14 Let `k` be a field of characteristic zero and `A` a smooth `k`-algebra which is
15 a domain. Then the only nonzero ideal of `A` stable under all `k`-derivations is
16 `A` itself (`eq_top_of_derivation_mem_of_smooth`), and therefore `D_k(A)` is a
17 simple ring (`isSimpleRing_of_smooth`, by `isSimpleRing_of_derivation`).
18 
19 ## Proof
20 
21 * **Polynomial rings** (`eq_top_of_forall_pderiv_mem`): partial derivatives
22   lower the total degree of a nonzero element of a differential ideal until a
23   nonzero constant is reached.
24 * **Quasi-finite algebras over a polynomial ring with dual derivations**
25   (`eq_top_of_derivation_mem_of_quasiFinite`). Let `R` be a domain and a
26   quasi-finite `k[X_σ]`-algebra with derivations `∂ᵢ`, `∂ᵢ(Xⱼ) = δᵢⱼ`. Then
27   `∂ᵢ(p) = ∂p/∂Xᵢ` on polynomials, so the kernel of `k[X] → R` is a
28   differential ideal, hence zero. If `J ⊆ R` is a nonzero `∂`-stable ideal
29   meeting no nonzero polynomial, a prime `Q ⊇ J` avoiding them lies over the
30   zero ideal, as does the prime `0 ⊊ Q`, contradicting incomparability of
31   primes in quasi-finite fibres. So `J` contains a nonzero polynomial `c`, and
32   `{p | p ∈ J}` is a nonzero differential ideal of `k[X]`, hence contains `1`.
33 * **Standard smooth algebras** (`eq_top_of_derivation_mem_of_isStandardSmooth`).
34   `Ω_{R/k}` has a basis `d s₁, …, d s_d` (Mathlib's
35   `IsStandardSmooth.iff_exists_basis_kaehlerDifferential`). Making `R` a
36   `k[X₁, …, X_d]`-algebra by `Xᵢ ↦ sᵢ`, the relative differentials vanish, so
37   `R` is unramified, hence quasi-finite, over `k[X]`; the dual basis gives the
38   derivations `∂ᵢ`.
39 * **Smooth algebras** (`eq_top_of_derivation_mem_of_smooth`). `Spec A` is
40   covered by standard opens `D(g)` with `A_g` standard smooth
41   (`Algebra.Smooth.exists_span_eq_top_isStandardSmooth`). A derivation of
42   `A_g` restricted to `A` becomes, after multiplication by a power of `g`, a
43   derivation of `A` (`exists_derivation_eq_pow_smul`), so `J A_g` is a
44   differential ideal of `A_g`, hence `A_g`, and some power of `g` lies in `J`.
45   If `J ⊆ 𝔪` maximal, choose `g ∉ 𝔪`.
46 
47 The route through quasi-finiteness and the local structure of smooth algebras
48 is AA-native. For the statement compare [McConnell–Robson, *Noncommutative
49 Noetherian Rings*, 15.3.8] and [Coutinho, *A Primer of Algebraic D-modules*,
50 Theorem 3.2.4 for polynomial rings].
51 -/

61 /-- **Polynomial rings have no differential ideals** in characteristic zero:
62 the only nonzero ideal of `k[X_σ]` stable under all `∂/∂Xᵢ` is `k[X_σ]`. -/
63 theorem eq_top_of_forall_pderiv_mem (J : Ideal (MvPolynomial σ k)) (hJ : J ≠ ⊥)
64     (hD : ∀ i, ∀ p ∈ J, pderiv i p ∈ J) : J = ⊤ := by
65   have key : ∀ d, ∀ p : MvPolynomial σ k, p.totalDegree ≤ d → p ≠ 0 → p ∈ J → J = ⊤ := by
66     intro d
67     induction d with
68     | zero =>
69         intro p hp hp0 hpJ
70         obtain ⟨c, rfl⟩ : ∃ c, p = C c := ⟨_, totalDegree_eq_zero_iff_eq_C.mp (by omega)⟩
71         have hc : c ≠ 0 := fun h => hp0 (by rw [h, C_0])
72         exact Ideal.eq_top_of_isUnit_mem J hpJ ((IsUnit.mk0 c hc).map C)
73     | succ d ih =>
74         intro p hp hp0 hpJ
75         by_cases hall : ∀ i, pderiv i p = 0
76         · rw [LocalizedPolynomialOperators.eq_C_of_pderiv_eq_zero hall] at hpJ hp0
77           exact ih _ (by simp) hp0 hpJ
78         · push Not at hall
79           obtain ⟨i, hi⟩ := hall
80           exact ih _ ((LocalizedPolynomialOperators.totalDegree_pderiv_le p i).trans (by omega)) hi
81             (hD i _ hpJ)
82   obtain ⟨p, hpJ, hp0⟩ := Submodule.exists_mem_ne_zero_of_ne_bot hJ
83   exact key _ p le_rfl hp0 hpJ
84 
85 end Polynomial

94 /-- Derivations dual to the coordinates act on polynomials as partial
95 derivatives: `∂ᵢ(p) = ∂p/∂Xᵢ`. -/
96 theorem derivation_algebraMap_eq_pderiv [DecidableEq σ]
97     (hdual : ∀ i j, D i (algebraMap (MvPolynomial σ k) R (X j)) = if i = j then 1 else 0)
98     (i : σ) (p : MvPolynomial σ k) :
99     D i (algebraMap _ R p) = algebraMap _ R (pderiv i p) := by
100   have h : (D i).compAlgebraMap (MvPolynomial σ k) =
101       (Algebra.linearMap (MvPolynomial σ k) R).compDer (pderiv i) := by
102     refine MvPolynomial.derivation_ext fun j => ?_
103     rw [Derivation.compAlgebraMap_apply, hdual]
104     change _ = algebraMap (MvPolynomial σ k) R (pderiv i (X j))
105     rw [pderiv_X]
106     by_cases hij : i = j
107     · subst hij; simp
108     · simp [hij, Ne.symm hij]
109   exact Derivation.congr_fun h p

113 /-- **Quasi-finite algebras over a polynomial ring have no differential
114 ideals.** Let `R` be a domain and a quasi-finite `k[X_σ]`-algebra with
115 derivations `∂ᵢ` dual to the `Xᵢ`. Then the only nonzero ideal of `R` stable
116 under all `∂ᵢ` is `R`. -/
117 theorem eq_top_of_derivation_mem_of_quasiFinite
118     (hdual : ∀ i j, D i (algebraMap (MvPolynomial σ k) R (X j)) = if i = j then 1 else 0)
119     (J : Ideal R) (hJ : J ≠ ⊥) (hD : ∀ i, ∀ f ∈ J, D i f ∈ J) : J = ⊤ := by
120   have hpd := derivation_algebraMap_eq_pderiv hdual
121   -- the pull-back of a differential ideal is differential
122   have hcomap : ∀ I : Ideal R, (∀ i, ∀ f ∈ I, D i f ∈ I) →
123       ∀ i, ∀ p ∈ I.comap (algebraMap (MvPolynomial σ k) R), pderiv i p ∈
124         I.comap (algebraMap (MvPolynomial σ k) R) := fun I hI i p hp => by
125     rw [Ideal.mem_comap, ← hpd]
126     exact hI i _ hp
127   -- the coordinates are algebraically independent
128   have hker : (⊥ : Ideal R).comap (algebraMap (MvPolynomial σ k) R) = ⊥ := by
129     by_contra h
130     have htop := eq_top_of_forall_pderiv_mem _ h (hcomap ⊥ fun i f hf => by
131       rw [Ideal.mem_bot] at hf ⊢
132       rw [hf, map_zero])
133     have h1 : (1 : MvPolynomial σ k) ∈ (⊥ : Ideal R).comap (algebraMap (MvPolynomial σ k) R) :=
134       htop ▸ Submodule.mem_top
135     rw [Ideal.mem_comap, map_one, Ideal.mem_bot] at h1
136     exact one_ne_zero h1
137   -- `J` contains the image of a nonzero polynomial
138   obtain ⟨c, hc0, hcJ⟩ : ∃ c : MvPolynomial σ k, c ≠ 0 ∧ algebraMap _ R c ∈ J := by
139     by_contra hcon
140     push Not at hcon
141     let M := Algebra.algebraMapSubmonoid R (nonZeroDivisors (MvPolynomial σ k))
142     obtain ⟨Q, hQ, hJQ, hQM⟩ := Ideal.exists_le_prime_disjoint J M (by
143       rw [Set.disjoint_left]
144       rintro _ hJ' ⟨c, hc, rfl⟩
145       exact hcon c (nonZeroDivisors.ne_zero hc) hJ')
146     have hunder : (⊥ : Ideal R).under (MvPolynomial σ k) = Q.under (MvPolynomial σ k) := by
147       rw [Ideal.under_def, Ideal.under_def, hker, eq_comm, eq_bot_iff]
148       intro p hp
149       rw [Ideal.mem_bot]
150       by_contra hp0
151       exact Set.disjoint_left.mp hQM hp ⟨p, mem_nonZeroDivisors_of_ne_zero hp0, rfl⟩
152     have hQ0 := Algebra.QuasiFinite.eq_of_le_of_under_eq (R := MvPolynomial σ k) ⊥ Q bot_le
153       hunder
154     exact hJ (eq_bot_iff.mpr (hQ0 ▸ hJQ))
155   -- the polynomials in `J` form a nonzero differential ideal
156   have htop := eq_top_of_forall_pderiv_mem _ (fun h => hc0 (by
157       have : c ∈ J.comap (algebraMap (MvPolynomial σ k) R) := hcJ
158       rwa [h, Ideal.mem_bot] at this)) (hcomap J hD)
159   have h1 : (1 : MvPolynomial σ k) ∈ J.comap (algebraMap (MvPolynomial σ k) R) :=
160     htop ▸ Submodule.mem_top
161   rw [Ideal.mem_comap, map_one] at h1
162   exact Ideal.eq_top_of_isUnit_mem J h1 isUnit_one
163 
164 end QuasiFinite

170 /-- **Standard smooth domains have no differential ideals** in characteristic
171 zero. -/
172 theorem eq_top_of_derivation_mem_of_isStandardSmooth [Algebra.IsStandardSmooth k R]
173     (J : Ideal R) (hJ : J ≠ ⊥) (hD : ∀ D : Derivation k R R, ∀ f ∈ J, D f ∈ J) : J = ⊤ := by
174   classical
175   obtain ⟨-, I, b, hb⟩ :=
176     (Algebra.IsStandardSmooth.iff_exists_basis_kaehlerDifferential (R := k) (S := R)).mp
177       inferInstance
178   have : Finite I := Module.Finite.finite_basis b
179   have : Fintype I := Fintype.ofFinite I
180   choose s hs using fun i => hb ⟨i, rfl⟩
181   let : Algebra (MvPolynomial I k) R := (MvPolynomial.aeval s).toAlgebra
182   have halg : ∀ p, algebraMap (MvPolynomial I k) R p = MvPolynomial.aeval s p := fun _ => rfl
183   have : IsScalarTower k (MvPolynomial I k) R := IsScalarTower.of_algebraMap_eq fun c => by
184     rw [halg, MvPolynomial.algebraMap_eq, MvPolynomial.aeval_C]
185   -- the relative differentials vanish
186   have : Subsingleton (Ω[R⁄MvPolynomial I k]) := by
187     have hzero : KaehlerDifferential.map k (MvPolynomial I k) R R = 0 := by
188       refine b.ext fun i => ?_
189       rw [← hs i, KaehlerDifferential.map_D, LinearMap.zero_apply, Algebra.algebraMap_self,
190         RingHom.id_apply, show s i = algebraMap (MvPolynomial I k) R (X i) by
191           rw [halg, MvPolynomial.aeval_X], Derivation.map_algebraMap]
192     refine ⟨fun ω ω' => ?_⟩
193     obtain ⟨η, rfl⟩ := KaehlerDifferential.map_surjective k (MvPolynomial I k) R ω
194     obtain ⟨η', rfl⟩ := KaehlerDifferential.map_surjective k (MvPolynomial I k) R ω'
195     rw [hzero, LinearMap.zero_apply, LinearMap.zero_apply]
196   have : Algebra.FormallyUnramified (MvPolynomial I k) R := ⟨inferInstance⟩
197   have : Algebra.FiniteType (MvPolynomial I k) R :=
198     Algebra.FiniteType.of_restrictScalars_finiteType k (MvPolynomial I k) R
199   have : Algebra.QuasiFinite (MvPolynomial I k) R := inferInstance
200   -- the dual derivations
201   let D : I → Derivation k R R := fun i => (b.coord i).compDer (KaehlerDifferential.D k R)
202   refine eq_top_of_derivation_mem_of_quasiFinite (D := D) (fun i j => ?_) J hJ
203     fun i => hD (D i)
204   rw [halg, MvPolynomial.aeval_X]
205   change b.coord i (KaehlerDifferential.D k R (s j)) = _
206   rw [hs, Module.Basis.coord_apply,
207     Module.Basis.repr_self, Finsupp.single_apply]
208   simp only [eq_comm]
209 
210 end StandardSmooth

217 /-- **Clearing denominators of a derivation.** Let `R = A[g⁻¹]` with `A → R`
218 injective and `Ω_{A/k}` finitely generated. For every `k`-derivation
219 `δ : A → R` there are `N` and a derivation `E` of `A` with `E = g^N δ`. -/
220 theorem exists_derivation_eq_pow_smul (g : A) [IsLocalization.Away g R]
221     (hinj : Function.Injective (algebraMap A R)) [Module.Finite A (Ω[A⁄k])]
222     (δ : Derivation k A R) :
223     ∃ N : ℕ, ∃ E : Derivation k A A, ∀ a, algebraMap A R (E a) = g ^ N • δ a := by
224   classical
225   set ψ := δ.liftKaehlerDifferential
226   obtain ⟨S, hS⟩ := Module.Finite.fg_top (R := A) (M := Ω[A⁄k])
227   obtain ⟨⟨_, N, rfl⟩, hN⟩ :=
228     IsLocalization.exist_integer_multiples_of_finset (Submonoid.powers g) (S.image ψ)
229   have hint : ∀ ω, IsLocalization.IsInteger A ((g ^ N : A) • ψ ω) := by
230     intro ω
231     have hω : ω ∈ Submodule.span A (S : Set (Ω[A⁄k])) := hS ▸ Submodule.mem_top
232     induction hω using Submodule.span_induction with
233     | mem ω hω => exact hN _ (Finset.mem_image_of_mem ψ hω)
234     | zero => rw [map_zero, smul_zero]; exact ⟨0, map_zero _⟩
235     | add ω ω' _ _ h h' => rw [map_add, smul_add]; exact IsLocalization.isInteger_add h h'
236     | smul a ω _ h => rw [map_smul, smul_comm]; exact IsLocalization.isInteger_smul h
237   let T : Submodule A R := LinearMap.range (Algebra.linearMap A R)
238   let ψ₁ : Ω[A⁄k] →ₗ[A] T := ((g ^ N : A) • ψ).codRestrict T fun ω => by
239     obtain ⟨a, ha⟩ := hint ω
240     exact ⟨a, ha⟩
241   let e : A ≃ₗ[A] T := LinearEquiv.ofInjective (Algebra.linearMap A R) hinj
242   refine ⟨N, (e.symm.toLinearMap ∘ₗ ψ₁).compDer (KaehlerDifferential.D k A), fun a => ?_⟩
243   have key : ∀ y : T, algebraMap A R (e.symm y) = y := fun y => by
244     conv_rhs => rw [← e.apply_symm_apply y]
245     rfl
246   change algebraMap A R (e.symm (ψ₁ (KaehlerDifferential.D k A a))) = _
247   rw [key]
248   simp [ψ₁, ψ]
249 
250 end Localization

257 /-- **Smooth domains have no differential ideals** in characteristic zero: the
258 only nonzero ideal of `A` stable under all `k`-derivations is `A`. -/
259 theorem eq_top_of_derivation_mem_of_smooth (J : Ideal A) (hJ : J ≠ ⊥)
260     (hD : ∀ D : Derivation k A A, ∀ f ∈ J, D f ∈ J) : J = ⊤ := by
261   obtain ⟨s, hs, hstd⟩ := Algebra.Smooth.exists_span_eq_top_isStandardSmooth k A
262   by_contra hJtop
263   obtain ⟨m, hm, hJm⟩ := Ideal.exists_le_maximal J hJtop
264   obtain ⟨g, hgs, hgm⟩ : ∃ g ∈ s, g ∉ m := by
265     by_contra h
266     push Not at h
267     exact hm.ne_top (eq_top_iff.mpr (hs ▸ Ideal.span_le.mpr h))
268   have hg0 : g ≠ 0 := fun h => hgm (h ▸ m.zero_mem)
269   let R := Localization.Away g
270   have hle : Submonoid.powers g ≤ nonZeroDivisors A :=
271     powers_le_nonZeroDivisors_of_noZeroDivisors hg0
272   have : IsDomain R := IsLocalization.isDomain_of_le_nonZeroDivisors R hle
273   have hinj : Function.Injective (algebraMap A R) := IsLocalization.injective R hle
274   have := hstd g hgs
275   -- the extension of `J` is a differential ideal of `A_g`
276   let J' : Ideal R := J.map (algebraMap A R)
277   have hJ' : J' ≠ ⊥ := by
278     obtain ⟨j, hjJ, hj0⟩ := Submodule.exists_mem_ne_zero_of_ne_bot hJ
279     intro h
280     have : algebraMap A R j ∈ J' := Ideal.mem_map_of_mem _ hjJ
281     rw [h, Ideal.mem_bot] at this
282     exact hj0 (hinj (this.trans (map_zero _).symm))
283   have hD' : ∀ D : Derivation k R R, ∀ f ∈ J', D f ∈ J' := by
284     intro D f hf
285     have hgen : ∀ j ∈ J, D (algebraMap A R j) ∈ J' := by
286       intro j hj
287       obtain ⟨N, E, hE⟩ := exists_derivation_eq_pow_smul g hinj (D.compAlgebraMap A)
288       have h1 : algebraMap A R (E j) ∈ J' := Ideal.mem_map_of_mem _ (hD E j hj)
289       rw [hE, Derivation.compAlgebraMap_apply, Algebra.smul_def] at h1
290       have hu : IsUnit (algebraMap A R (g ^ N)) :=
291         IsLocalization.map_units R (⟨g ^ N, N, rfl⟩ : Submonoid.powers g)
292       obtain ⟨u, hu⟩ := hu
293       have := J'.mul_mem_left (↑u⁻¹ : R) h1
294       rwa [← mul_assoc, ← hu, Units.inv_mul, one_mul] at this
295     induction hf using Submodule.span_induction with
296     | mem f hf =>
297         obtain ⟨j, hj, rfl⟩ := hf
298         exact hgen j hj
299     | zero => rw [map_zero]; exact J'.zero_mem
300     | add f f' _ _ h h' => rw [map_add]; exact J'.add_mem h h'
301     | smul r f hf h =>
302         rw [smul_eq_mul, Derivation.leibniz, smul_eq_mul, smul_eq_mul]
303         exact J'.add_mem (J'.mul_mem_left r h) (J'.mul_mem_right _ hf)
304   have htop := eq_top_of_derivation_mem_of_isStandardSmooth (k := k) J' hJ' hD'
305   -- a power of `g` lies in `J ⊆ 𝔪`
306   have h1 : (1 : R) ∈ J' := htop ▸ Submodule.mem_top
307   obtain ⟨⟨⟨j, hj⟩, ⟨_, n, rfl⟩⟩, hjn⟩ :=
308     (IsLocalization.mem_map_algebraMap_iff (Submonoid.powers g) R).mp h1
309   simp only [one_mul] at hjn
310   have hgn : g ^ n = j := hinj hjn
311   exact hgm (hm.isPrime.mem_of_pow_mem n (hgn ▸ hJm hj))
312 
313 /-- **`D_k(A)` is simple** for a smooth domain `A` over a field of
314 characteristic zero. -/
315 theorem isSimpleRing_of_smooth : IsSimpleRing (algebra (k := k) (R := A)) :=
316   isSimpleRing_of_derivation eq_top_of_derivation_mem_of_smooth
