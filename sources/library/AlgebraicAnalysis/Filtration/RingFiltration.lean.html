9 # Ascending ring filtrations
10 
11 This file defines the generic ascending filtration of a (possibly
12 noncommutative) ring `A` over a commutative base ring `R`, and develops the
13 foundational closure, associated-graded, order, and finiteness API that the
14 rest of a filtered-ring / D-module layer sits on.
15 
16 The pinned Mathlib has two filtration-shaped structures which do not cover
17 this notion: `Ideal.Filtration` (`Mathlib.RingTheory.Filtration`) is a
18 *decreasing* `I`-adic filtration of a module over a fixed ideal `I`, and
19 `IsRingFiltration`/`IsFiltration` (`Mathlib.RingTheory.FilteredAlgebra.Basic`)
20 are `SetLike`-generic classes that do not require a multiplicative unit in
21 degree `0` or exhaustiveness, and are stated for an arbitrary `Preorder`
22 index rather than for the specific `ℕ`-indexed, submodule-multiplication
23 shape used here. Neither is reused as the primary structure; both are noted
24 so a later contributor does not duplicate the search. What *is* reused is
25 Mathlib's submodule-multiplication semiring structure on `Submodule R A`
26 (`Mathlib.Algebra.Algebra.Operations`, available for `[CommSemiring R]
27 [Semiring A] [Algebra R A]`) and the quotient-bilinear-form lifting lemma
28 `LinearMap.liftQ₂` (`Mathlib.LinearAlgebra.Quotient.Bilinear`), which is
29 exactly the well-definedness tool the associated-graded multiplication needs.
30 
31 ## Main definitions
32 
33 * `RingFiltration R A`: an ascending, exhaustive, unital, submultiplicative
34   family `F : ℕ → Submodule R A`.
35 * `RingFiltration.GradedQuotient φ i`: the graded piece `F i / F (i - 1)`
36   (with the convention `F (-1) = ⊥`), together with `RingFiltration.gradedMul`,
37   the well-defined multiplication `Graded i → Graded j → Graded (i + j)`
38   induced on representatives.
39 * `RingFiltration.order`: for `a ≠ 0`, the least filtration degree containing
40   `a`.
41 * `RingFiltration.Separated`: no oscillation — once two consecutive filtration
42   pieces coincide, the filtration stays constant from there on.
43 * `RingFiltration.IsGoodFiltration`: each piece is a finite `R`-module; the
44   hypothesis a later Hilbert-function argument needs. Proved closed under sup
45   and product of pieces.
46 
47 No Weyl algebra, differential operator, or other application-specific
48 structure appears in this file.
49 -/

53 /-- An ascending filtration of a ring `A` by `R`-submodules: monotone,
54 containing `1` in degree `0`, submultiplicative (`F i * F j ≤ F (i + j)`,
55 using the submodule-multiplication semiring structure on `Submodule R A`),
56 and exhaustive (`⨆ i, F i = ⊤`). -/
57 structure RingFiltration (R A : Type*) [CommRing R] [Ring A] [Algebra R A] where
58   /-- The filtration pieces. -/
59   F : ℕ → Submodule R A
60   /-- The pieces are increasing. -/
61   mono : Monotone F
62   /-- The unit lies in degree `0`. -/
63   one_mem : (1 : A) ∈ F 0
64   /-- Submultiplicativity: `F i * F j ⊆ F (i + j)`. -/
65   mul_le : ∀ i j, F i * F j ≤ F (i + j)
66   /-- The filtration exhausts the ring. -/
67   exhaustive : ⨆ i, F i = ⊤
68

73 /-! ## Basic closure facts -/
74 
75 /-- A product of elements of filtration degree `i` and `j` has filtration
76 degree at most `i + j`. -/
77 theorem mul_mem_add {a b : A} {i j : ℕ} (ha : a ∈ φ.F i) (hb : b ∈ φ.F j) :
78     a * b ∈ φ.F (i + j) :=
79   φ.mul_le i j (Submodule.mul_mem_mul ha hb)
80 
81 /-- A power `a ^ n` of an element of filtration degree `d` has filtration
82 degree at most `d * n`. -/
83 theorem pow_mem {a : A} {d : ℕ} (ha : a ∈ φ.F d) (n : ℕ) : a ^ n ∈ φ.F (d * n) := by
84   induction n with
85   | zero => simpa using φ.one_mem
86   | succ n ih =>
87       have hstep : a ^ n * a ∈ φ.F (d * n + d) := φ.mul_mem_add ih ha
88       simpa [pow_succ, Nat.mul_succ] using hstep

90 /-! ## The associated graded pieces -/
91 
92 /-- The predecessor filtration piece, with the convention `F (-1) = ⊥` used
93 at degree `0`. -/
94 def below : ℕ → Submodule R A
95   | 0 => ⊥
96   | i + 1 => φ.F i
97 
98 @[simp] theorem below_zero : φ.below 0 = ⊥ := rfl
99 
100 theorem below_succ (i : ℕ) : φ.below (i + 1) = φ.F i := rfl
101 
102 /-- The predecessor piece, viewed as a submodule of `F i` itself. -/
103 def belowInF (i : ℕ) : Submodule R (φ.F i) :=
104   (φ.below i).comap (φ.F i).subtype
105 
106 theorem mem_belowInF {i : ℕ} {x : φ.F i} : x ∈ φ.belowInF i ↔ (x : A) ∈ φ.below i :=
107   Iff.rfl
108 
109 /-- The `i`-th graded piece `F i / F (i - 1)` of the associated graded
110 module, with `F (-1) = ⊥` at `i = 0`. -/
111 abbrev GradedQuotient (i : ℕ) : Type _ := φ.F i ⧸ φ.belowInF i
112 
113 /-- `below i * F j ≤ below (i + j)`: multiplying a below-degree-`i` element
114 by a degree-`j` element lands below degree `i + j`. -/
115 theorem below_mul_le (i j : ℕ) : φ.below i * φ.F j ≤ φ.below (i + j) := by
116   cases i with
117   | zero => simp
118   | succ n =>
119       have heq : n + 1 + j = n + j + 1 := by omega
120       rw [below_succ, heq, below_succ]
121       exact φ.mul_le n j
122 
123 /-- `F i * below j ≤ below (i + j)`: multiplying a degree-`i` element by a
124 below-degree-`j` element lands below degree `i + j`. -/
125 theorem mul_below_le (i j : ℕ) : φ.F i * φ.below j ≤ φ.below (i + j) := by
126   cases j with
127   | zero => simp
128   | succ n =>
129       have heq : i + (n + 1) = i + n + 1 := by omega
130       rw [below_succ, heq, below_succ]
131       exact φ.mul_le i n
132 
133 /-- Multiplication of representatives, landing directly in the graded
134 quotient of the sum degree. This is *not yet* shown to be well defined on
135 the graded pieces; see `gradedMul` for the induced, well-defined product. -/
136 def repMul (i j : ℕ) : φ.F i →ₗ[R] φ.F j →ₗ[R] φ.GradedQuotient (i + j) :=
137   LinearMap.mk₂' R R
138     (fun a b => Submodule.Quotient.mk
139       ⟨(a : A) * (b : A), φ.mul_mem_add a.2 b.2⟩)
140     (fun a₁ a₂ b => by
141       rw [← Submodule.Quotient.mk_add]
142       congr 1
143       exact Subtype.ext (by simp [add_mul]))
144     (fun c a b => by
145       rw [← Submodule.Quotient.mk_smul]
146       congr 1
147       exact Subtype.ext (by simp))
148     (fun a b₁ b₂ => by
149       rw [← Submodule.Quotient.mk_add]
150       congr 1
151       exact Subtype.ext (by simp [mul_add]))
152     (fun c a b => by
153       rw [← Submodule.Quotient.mk_smul]
154       congr 1
155       exact Subtype.ext (by simp))
156 
157 theorem repMul_apply (i j : ℕ) (a : φ.F i) (b : φ.F j) :
158     φ.repMul i j a b =
159       Submodule.Quotient.mk (p := φ.belowInF (i + j))
160         ⟨(a : A) * (b : A), φ.mul_mem_add a.2 b.2⟩ :=
161   rfl
162 
163 /-- Changing the first representative by a below-degree-`i` element does not
164 change `repMul`. -/
165 theorem belowInF_le_repMul_ker (i j : ℕ) :
166     φ.belowInF i ≤ (φ.repMul i j).ker := by
167   intro x hx
168   have hx' : (x : A) ∈ φ.below i := φ.mem_belowInF.1 hx
169   refine LinearMap.mem_ker.mpr (LinearMap.ext fun y => ?_)
170   show Submodule.Quotient.mk (p := φ.belowInF (i + j)) _ = (0 : φ.GradedQuotient (i + j))
171   rw [Submodule.Quotient.mk_eq_zero]
172   exact φ.mem_belowInF.2 (φ.below_mul_le i j (Submodule.mul_mem_mul hx' y.2))
173 
174 /-- Changing the second representative by a below-degree-`j` element does not
175 change `repMul`. -/
176 theorem belowInF_le_repMul_flip_ker (i j : ℕ) :
177     φ.belowInF j ≤ (φ.repMul i j).flip.ker := by
178   intro y hy
179   have hy' : (y : A) ∈ φ.below j := φ.mem_belowInF.1 hy
180   refine LinearMap.mem_ker.mpr (LinearMap.ext fun x => ?_)
181   show Submodule.Quotient.mk (p := φ.belowInF (i + j)) _ = (0 : φ.GradedQuotient (i + j))
182   rw [Submodule.Quotient.mk_eq_zero]
183   exact φ.mem_belowInF.2 (φ.mul_below_le i j (Submodule.mul_mem_mul x.2 hy'))
184 
185 /-- The well-defined multiplication `Graded i → Graded j → Graded (i + j)` on
186 the associated graded module, induced from `repMul` on representatives. This
187 is where the submultiplicativity hypothesis `mul_le` is genuinely used to
188 show a *well-defined* map, not merely assumed. -/
189 def gradedMul (i j : ℕ) :
190     φ.GradedQuotient i →ₗ[R] φ.GradedQuotient j →ₗ[R] φ.GradedQuotient (i + j) :=
191   LinearMap.liftQ₂ (φ.belowInF i) (φ.belowInF j) (φ.repMul i j)
192     (φ.belowInF_le_repMul_ker i j) (φ.belowInF_le_repMul_flip_ker i j)
193 
194 @[simp] theorem gradedMul_mk (i j : ℕ) (a : φ.F i) (b : φ.F j) :
195     φ.gradedMul i j (Submodule.Quotient.mk a) (Submodule.Quotient.mk b) =
196       Submodule.Quotient.mk (p := φ.belowInF (i + j))
197         ⟨(a : A) * (b : A), φ.mul_mem_add a.2 b.2⟩ :=
198   rfl

200 /-! ## The order function -/
201 
202 theorem exists_mem_of_directed (a : A) : ∃ i, a ∈ φ.F i := by
203   have ha : a ∈ (⊤ : Submodule R A) := Submodule.mem_top
204   rw [← φ.exhaustive] at ha
205   exact (Submodule.mem_iSup_of_directed φ.F φ.mono.directed_le).mp ha

209 /-- The order of a nonzero element `a`: the least filtration degree
210 containing it. Well defined by exhaustiveness of the filtration. -/
211 noncomputable def order {a : A} (_ha : a ≠ 0) : ℕ :=
212   Nat.find (φ.exists_mem_of_directed a)
213 
214 theorem mem_order {a : A} (ha : a ≠ 0) : a ∈ φ.F (φ.order ha) :=
215   Nat.find_spec (φ.exists_mem_of_directed a)
216 
217 theorem order_le {a : A} (ha : a ≠ 0) {i : ℕ} (hi : a ∈ φ.F i) : φ.order ha ≤ i :=
218   Nat.find_min' (φ.exists_mem_of_directed a) hi
219 
220 /-- The order of a product is at most the sum of the orders. -/
221 theorem order_mul_le {a b : A} (ha : a ≠ 0) (hb : b ≠ 0) (hab : a * b ≠ 0) :
222     φ.order hab ≤ φ.order ha + φ.order hb :=
223   φ.order_le hab (φ.mul_mem_add (φ.mem_order ha) (φ.mem_order hb))
224 
225 /-- The order of a sum is at most the maximum of the orders. -/
226 theorem order_add_le {a b : A} (ha : a ≠ 0) (hb : b ≠ 0) (hab : a + b ≠ 0) :
227     φ.order hab ≤ max (φ.order ha) (φ.order hb) := by
228   refine φ.order_le hab (add_mem ?_ ?_)
229   · exact φ.mono (le_max_left _ _) (φ.mem_order ha)
230   · exact φ.mono (le_max_right _ _) (φ.mem_order hb)

232 /-! ## Separated filtrations and the finiteness hypothesis -/
233 
234 /-- A filtration is separated when it does not oscillate: once two
235 consecutive pieces coincide, the filtration stays constant from there on
236 (equivalently, `F` is strictly increasing until it stabilises). -/
237 def Separated (φ : RingFiltration R A) : Prop :=
238   ∀ i, φ.F i = φ.F (i + 1) → ∀ j, i ≤ j → φ.F j = φ.F i
239 
240 /-- A submodule sup of two finite `R`-modules in an `R`-algebra is finite.
241 Generic fact about `Submodule R A`, not specific to a filtration. -/
242 theorem finite_sup {p q : Submodule R A} (hp : Module.Finite R p) (hq : Module.Finite R q) :
243     Module.Finite R (↥(p ⊔ q)) :=
244   Module.Finite.iff_fg.mpr
245     (Submodule.FG.sup (Module.Finite.iff_fg.mp hp) (Module.Finite.iff_fg.mp hq))
246 
247 /-- A submodule product of two finite `R`-modules in an `R`-algebra is
248 finite. Generic fact about `Submodule R A`, not specific to a filtration. -/
249 theorem finite_mul {p q : Submodule R A} (hp : Module.Finite R p) (hq : Module.Finite R q) :
250     Module.Finite R (p * q) :=
251   Module.Finite.iff_fg.mpr
252     (Submodule.FG.mul (Module.Finite.iff_fg.mp hp) (Module.Finite.iff_fg.mp hq))
253 
254 /-- A good filtration: each piece is a finite `R`-module. This is exactly
255 the finite-dimensionality hypothesis a later Hilbert-function / Bernstein-type
256 argument needs; it is not derivable from the ascending-exhaustive-unital-
257 submultiplicative axioms alone (e.g. `A` itself may fail to be Noetherian
258 over `R`), so it is recorded as an explicit class. -/
259 class IsGoodFiltration (φ : RingFiltration R A) : Prop where
260   /-- Every filtration piece is a finite `R`-module. -/
261   finite : ∀ i, Module.Finite R (φ.F i)
262 
263 theorem IsGoodFiltration.finite_sup [h : φ.IsGoodFiltration] (i j : ℕ) :
264     Module.Finite R (↥(φ.F i ⊔ φ.F j)) :=
265   RingFiltration.finite_sup (h.finite i) (h.finite j)
266 
267 theorem IsGoodFiltration.finite_mul [h : φ.IsGoodFiltration] (i j : ℕ) :
268     Module.Finite R (φ.F i * φ.F j) :=
269   RingFiltration.finite_mul (h.finite i) (h.finite j)
270 
271 end RingFiltration
