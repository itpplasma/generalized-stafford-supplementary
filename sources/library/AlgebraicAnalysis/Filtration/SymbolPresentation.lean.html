7 # Surjective symbol presentations
8 
9 A `RingFiltration.SymbolPresentation φ 𝒜` is a commutative graded algebra
10 `G = ⨁ 𝒜 n` together with `R`-linear maps `σₙ : 𝒜ₙ → Fₙ/Fₙ₋₁` which are
11 surjective and multiplicative: `σ_{i+j}(g h) = σᵢ(g) σⱼ(h)` in the associated
12 graded module of `A`. Unlike `GradedSymbol`, the `σₙ` need not be injective.
13 So `gr A` is a quotient of the commutative graded algebra `G`, not isomorphic
14 to it. This is the situation of a graded quotient of a symmetric algebra,
15 such as `S(𝔤) ↠ gr U(𝔤)`, where the enveloping algebra has no PBW theorem
16 available.
17 
18 `GradedSymbol.toPresentation` shows that an isomorphism-type symbol is a
19 presentation, so the results here contain those of `SymbolNoetherian`.
20 
21 ## Main results
22 
23 The argument is that of `SymbolNoetherian`, with one change. For an additive
24 subgroup `T ≤ A`, `grPiece T n` is the set of homogeneous `g ∈ 𝒜ₙ` with
25 `σₙ(g) = [u]` for some `u ∈ T ∩ Fₙ`. Surjectivity of `σₙ` makes every class
26 of `Fₙ/Fₙ₋₁` realised by some `g`, which is all that the equality step needs.
27 Commutativity of `G` and multiplicativity of `σ` give that `gr T` is an ideal
28 for one-sided ideals `T`. Hence `I ↦ gr I` is strictly monotone.
29 
30 * `SymbolPresentation.isNoetherian_right`, `isNoetherianRing`: if `G` is
31   Noetherian then `A` is right and left Noetherian [McConnell–Robson 1.6.9].
32 * `SymbolPresentation.devLE_right`, `devLE_left`: if moreover `ringKrullDim G ≤ m`,
33   the ideal lattices of `A` have deviation at most `m` [MR 6.5.6, 6.4.8].
34 * `SymbolPresentation.devLEOmega_right`, `devLEOmega_left`: finite Krull
35   dimension of `G` gives deviation at most omega.
36 -/

44 /-- A surjective symbol presentation of `φ` by the commutative graded algebra
45 `G = ⨁ 𝒜 n`: surjective `R`-linear maps `σₙ : 𝒜ₙ → Fₙ/Fₙ₋₁` which multiply. -/
46 structure SymbolPresentation {R A G : Type*} [CommRing R] [Ring A] [Algebra R A]
47     [CommRing G] [Algebra R G] (φ : RingFiltration R A) (𝒜 : ℕ → Submodule R G) where
48   /-- The degree-`n` presentation map `𝒜ₙ → Fₙ/Fₙ₋₁`. -/
49   sym : ∀ n, 𝒜 n →ₗ[R] φ.GradedQuotient n
50   /-- Every class of `Fₙ/Fₙ₋₁` is the image of a homogeneous element of degree `n`. -/
51   sym_surjective : ∀ n, Function.Surjective (sym n)
52   /-- Multiplicativity: if `x = g h` in `G`, then `σ_{i+j}(x) = σᵢ(g) σⱼ(h)`. -/
53   sym_mul : ∀ (i j : ℕ) (g : 𝒜 i) (h : 𝒜 j) (x : 𝒜 (i + j)),
54     (x : G) = (g : G) * (h : G) → sym (i + j) x = φ.gradedMul i j (sym i g) (sym j h)

61 /-- The degree-`n` symbols of `T ∩ Fₙ`: homogeneous `g ∈ 𝒜ₙ` whose presentation
62 class is the class of some `u ∈ T ∩ Fₙ`. -/
63 def grPiece (T : AddSubgroup A) (n : ℕ) : AddSubgroup G where
64   carrier := {g | ∃ hg : g ∈ 𝒜 n, ∃ u ∈ T, ∃ hu : u ∈ φ.F n,
65     P.sym n ⟨g, hg⟩ = Submodule.Quotient.mk ⟨u, hu⟩}
66   add_mem' := by
67     rintro a b ⟨ha, u, hu, hu', he⟩ ⟨hb, v, hv, hv', hf⟩
68     refine ⟨Submodule.add_mem _ ha hb, u + v, T.add_mem hu hv, Submodule.add_mem _ hu' hv', ?_⟩
69     rw [show (⟨a + b, Submodule.add_mem _ ha hb⟩ : 𝒜 n) = ⟨a, ha⟩ + ⟨b, hb⟩ from rfl, map_add,
70       he, hf, ← Submodule.Quotient.mk_add]
71     rfl
72   zero_mem' := ⟨Submodule.zero_mem _, 0, T.zero_mem, Submodule.zero_mem _, by
73     rw [show (⟨0, Submodule.zero_mem _⟩ : 𝒜 n) = 0 from rfl, map_zero]
74     exact (Submodule.Quotient.mk_zero _).symm⟩
75   neg_mem' := by
76     rintro a ⟨ha, u, hu, hu', he⟩
77     refine ⟨Submodule.neg_mem _ ha, -u, T.neg_mem hu, Submodule.neg_mem _ hu', ?_⟩
78     rw [show (⟨-a, Submodule.neg_mem _ ha⟩ : 𝒜 n) = -⟨a, ha⟩ from rfl, map_neg, he,
79       ← Submodule.Quotient.mk_neg]
80     rfl

89 /-- The symbol span `gr T ⊆ G`: elements all of whose homogeneous components
90 lie in the degree-wise symbol pieces of `T`. -/
91 def grSpan (T : AddSubgroup A) : AddSubgroup G where
92   carrier := {g | ∀ n, (decompose 𝒜 g n : G) ∈ P.grPiece T n}
93   add_mem' := by
94     intro a b ha hb n
95     rw [decompose_add, add_apply, Submodule.coe_add]
96     exact AddSubgroup.add_mem _ (ha n) (hb n)
97   zero_mem' := by
98     intro n
99     rw [decompose_zero, zero_apply, Submodule.coe_zero]
100     exact AddSubgroup.zero_mem _
101   neg_mem' := by
102     intro a ha n
103     rw [decompose_neg]
104     erw [DFinsupp.neg_apply]
105     rw [Submodule.coe_neg]
106     exact AddSubgroup.neg_mem _ (ha n)

116 /-- A homogeneous element of degree `n` lying in the degree-`n` piece lies in `gr T`. -/
117 theorem mem_grSpan_of_mem {T : AddSubgroup A} {n : ℕ} {g : G} (hg : g ∈ 𝒜 n)
118     (hgP : g ∈ P.grPiece T n) : g ∈ P.grSpan T := by
119   classical
120   intro m
121   by_cases hmn : m = n
122   · subst hmn
123     rw [decompose_of_mem_same 𝒜 hg]
124     exact hgP
125   · rw [decompose_of_mem_ne 𝒜 hg (Ne.symm hmn)]
126     exact AddSubgroup.zero_mem _

128 /-- A homogeneous element of `gr T` of degree `n` lies in the degree-`n` piece. -/
129 theorem mem_grPiece_of_mem_grSpan {T : AddSubgroup A} {n : ℕ} {g : G}

134 /-- **Equal symbols force equality.** If `T ≤ T'` and every symbol of `T'` is a
135 symbol of `T`, then `T = T'`. Surjectivity of the presentation realises each
136 class of `Fₙ/Fₙ₋₁` by a homogeneous element; exhaustiveness and `F (-1) = 0`
137 finish the induction. -/
138 theorem eq_of_le_of_grSpan_le {T T' : AddSubgroup A} (hle : T ≤ T')
139     (hgr : P.grSpan T' ≤ P.grSpan T) : T = T' := by
140   have key : ∀ (n : ℕ) (u : A), u ∈ T' → u ∈ φ.F n →
141       ∃ v ∈ T, v ∈ φ.F n ∧ u - v ∈ φ.below n := by
142     intro n u hu hn
143     obtain ⟨⟨g, hg'⟩, hg⟩ := P.sym_surjective n (Submodule.Quotient.mk ⟨u, hn⟩)
144     have hmem : g ∈ P.grPiece T' n := ⟨hg', u, hu, hn, hg⟩
145     obtain ⟨_, v, hv, hvn, hgv⟩ :=
146       P.mem_grPiece_of_mem_grSpan (hgr (P.mem_grSpan_of_mem hg' hmem)) hg'
147     refine ⟨v, hv, hvn, ?_⟩
148     have := (Submodule.Quotient.eq (φ.belowInF n)).mp (hg.symm.trans hgv)
149     exact φ.mem_belowInF.mp this
150   refine le_antisymm hle fun u hu => ?_
151   obtain ⟨n, hn⟩ := φ.exists_mem_of_directed u
152   induction n generalizing u with
153   | zero =>
154     obtain ⟨v, hv, _, hd⟩ := key 0 u hu hn
155     rw [below_zero, Submodule.mem_bot, sub_eq_zero] at hd
156     rw [hd]
157     exact hv
158   | succ n ih =>
159     obtain ⟨v, hv, _, hd⟩ := key (n + 1) u hu hn
160     rw [below_succ] at hd
161     have := ih (u - v) (T'.sub_mem hu (hle hv)) hd
162     simpa using T.add_mem this hv
163 
164 /-- `T ↦ gr T` is strictly monotone on additive subgroups. -/
165 theorem strictMono_grSpan : StrictMono P.grSpan := by
166   intro T T' hlt
167   refine lt_of_le_of_ne (P.grSpan_mono hlt.le) fun he => hlt.ne ?_
168   exact P.eq_of_le_of_grSpan_le hlt.le he.ge

170 /-- If homogeneous multiples of degree-wise symbols of `T` are again such
171 symbols, then `gr T` is closed under multiplication by `G`. -/
172 theorem mul_mem_grSpan {T : AddSubgroup A}
173     (hT : ∀ i m (r g : G), r ∈ 𝒜 i → g ∈ P.grPiece T m → r * g ∈ P.grPiece T (i + m))
174     (r : G) {g : G} (hg : g ∈ P.grSpan T) : r * g ∈ P.grSpan T := by
175   classical
176   intro n
177   rw [show r * g = ∑ i ∈ (decompose 𝒜 r).support, (decompose 𝒜 r i : G) * g by
178     rw [← Finset.sum_mul, sum_support_decompose]]
179   rw [decompose_sum, DFinsupp.finsetSum_apply, Submodule.coe_sum]
180   refine AddSubgroup.sum_mem _ fun i _ => ?_
181   rw [coe_decompose_mul_of_left_mem 𝒜 n (decompose 𝒜 r i).2]
182   split_ifs with hin
183   · have := hT i (n - i) _ _ (decompose 𝒜 r i).2 (hg (n - i))
184     rwa [Nat.add_sub_of_le hin] at this
185   · exact AddSubgroup.zero_mem _

187 /-- The degree-wise pieces of a right ideal absorb homogeneous elements on the right.
188 The symbol of `r g` equals `σ(g) σ(r)` by commutativity of `G`. -/
189 theorem mul_mem_grPiece_right (I : Submodule Aᵐᵒᵖ A) {i m : ℕ} {r g : G} (hr : r ∈ 𝒜 i)
190     (hg : g ∈ P.grPiece I.toAddSubgroup m) : r * g ∈ P.grPiece I.toAddSubgroup (m + i) := by
191   obtain ⟨hg', u, hu, hu', he⟩ := hg
192   obtain ⟨⟨v, hv⟩, hvr⟩ := Submodule.Quotient.mk_surjective (φ.belowInF i) (P.sym i ⟨r, hr⟩)
193   have hrg : r * g ∈ 𝒜 (m + i) := by
194     rw [mul_comm]
195     exact SetLike.mul_mem_graded hg' hr
196   refine ⟨hrg, u * v, ?_, φ.mul_mem_add hu' hv, ?_⟩
197   · have := I.smul_mem (op v) hu
198     rwa [op_smul_eq_mul] at this
199   · rw [P.sym_mul m i ⟨g, hg'⟩ ⟨r, hr⟩ ⟨r * g, hrg⟩ (mul_comm r g), he, ← hvr,
200       RingFiltration.gradedMul_mk]
201 
202 /-- The degree-wise pieces of a left ideal absorb homogeneous elements on the left. -/
203 theorem mul_mem_grPiece_left (I : Submodule A A) {i m : ℕ} {r g : G} (hr : r ∈ 𝒜 i)
204     (hg : g ∈ P.grPiece I.toAddSubgroup m) : r * g ∈ P.grPiece I.toAddSubgroup (i + m) := by
205   obtain ⟨hg', u, hu, hu', he⟩ := hg
206   obtain ⟨⟨v, hv⟩, hvr⟩ := Submodule.Quotient.mk_surjective (φ.belowInF i) (P.sym i ⟨r, hr⟩)
207   have hrg : r * g ∈ 𝒜 (i + m) := SetLike.mul_mem_graded hr hg'
208   refine ⟨hrg, v * u, ?_, φ.mul_mem_add hv hu', ?_⟩
209   · have := I.smul_mem v hu
210     rwa [smul_eq_mul] at this
211   · rw [P.sym_mul i m ⟨r, hr⟩ ⟨g, hg'⟩ ⟨r * g, hrg⟩ rfl, ← hvr, he,

214 /-- The symbol ideal `gr I ⊆ G` of a right ideal `I` of `A`. -/
215 def grIdealRight (I : Submodule Aᵐᵒᵖ A) : Ideal G where
216   carrier := P.grSpan I.toAddSubgroup
217   add_mem' := AddSubgroup.add_mem _
218   zero_mem' := AddSubgroup.zero_mem _
219   smul_mem' r _ hg :=
220     P.mul_mem_grSpan
221       (fun _ _ _ _ hr hg => by
222         rw [add_comm]
223         exact P.mul_mem_grPiece_right I hr hg) r hg

225 /-- The symbol ideal `gr I ⊆ G` of a left ideal `I` of `A`. -/
226 def grIdealLeft (I : Submodule A A) : Ideal G where
227   carrier := P.grSpan I.toAddSubgroup
228   add_mem' := AddSubgroup.add_mem _
229   zero_mem' := AddSubgroup.zero_mem _

300 /-- An isomorphism-type symbol is a surjective presentation: `σₙ` is the inverse
301 of the isomorphism `Fₙ/Fₙ₋₁ ≅ 𝒜ₙ` of `GradedSymbol.gradedQuotientEquiv`. -/
302 noncomputable def toPresentation (S : GradedSymbol φ 𝒜) : SymbolPresentation φ 𝒜 where
303   sym n := (S.gradedQuotientEquiv n).symm.toLinearMap
304   sym_surjective n := (S.gradedQuotientEquiv n).symm.surjective
305   sym_mul i j g h x hx := by
306     obtain ⟨a, ha⟩ := S.gr_surjective i g
307     obtain ⟨b, hb⟩ := S.gr_surjective j h
308     obtain ⟨c, hc⟩ := S.gr_surjective (i + j) x
309     have hab : (a : A) * (b : A) ∈ φ.F (i + j) := φ.mul_mem_add a.2 b.2
310     have hsymm : ∀ (n : ℕ) (y : φ.F n) (z : 𝒜 n), S.gr n y = z →
311         (S.gradedQuotientEquiv n).symm z = Submodule.Quotient.mk y := by
312       intro n y z hyz
313       rw [LinearEquiv.symm_apply_eq, ← hyz, gradedQuotientEquiv_mk]
314     have key : S.gr (i + j) (c - ⟨(a : A) * (b : A), hab⟩) = 0 := by
315       rw [map_sub, sub_eq_zero]
316       apply Subtype.ext
317       rw [hc, S.gr_mul i j a b, ha, hb]
318       exact hx
319     show (S.gradedQuotientEquiv (i + j)).symm x = φ.gradedMul i j
320       ((S.gradedQuotientEquiv i).symm g) ((S.gradedQuotientEquiv j).symm h)
321     rw [hsymm i a g ha, hsymm j b h hb, hsymm (i + j) c x hc, RingFiltration.gradedMul_mk]
322     exact (Submodule.Quotient.eq _).mpr (φ.mem_belowInF.mpr ((S.gr_eq_zero_iff _ _).mp key))
323 
324 end RingFiltration.GradedSymbol

331 /-- **An injective presentation is a symbol isomorphism.** If every `σₙ` is
332 injective, then `gr A ≅ G`: the symbol of `u ∈ Fₙ` is `σₙ⁻¹ [u]`. -/
333 noncomputable def toGradedSymbol [GradedAlgebra 𝒜] (P : SymbolPresentation φ 𝒜)
334     (hinj : ∀ n, Function.Injective (P.sym n)) : GradedSymbol φ 𝒜 where
335   gr n := (LinearEquiv.ofBijective (P.sym n) ⟨hinj n, P.sym_surjective n⟩).symm.toLinearMap ∘ₗ
336     (φ.belowInF n).mkQ
337   gr_surjective n := by
338     intro g
339     obtain ⟨z, hz⟩ := Submodule.Quotient.mk_surjective _ (P.sym n g)
340     refine ⟨z, ?_⟩
341     simp only [LinearMap.coe_comp, LinearEquiv.coe_coe, Function.comp_apply, Submodule.mkQ_apply,
342       hz]
343     exact (LinearEquiv.ofBijective (P.sym n) ⟨hinj n, P.sym_surjective n⟩).symm_apply_apply g
344   gr_eq_zero_iff n x := by
345     simp only [LinearMap.coe_comp, LinearEquiv.coe_coe, Function.comp_apply,
346       LinearEquiv.map_eq_zero_iff, Submodule.mkQ_apply, Submodule.Quotient.mk_eq_zero,
347       mem_belowInF]
348   gr_mul i j x y := by
349     set e := fun n => LinearEquiv.ofBijective (P.sym n) ⟨hinj n, P.sym_surjective n⟩
350     have key : ∀ n (z : φ.F n), P.sym n ((e n).symm (Submodule.Quotient.mk z)) =
351         Submodule.Quotient.mk z := fun n z => (e n).apply_symm_apply _
352     let g : 𝒜 (i + j) := ⟨_, SetLike.GradedMul.mul_mem ((e i).symm (Submodule.Quotient.mk x)).2
353       ((e j).symm (Submodule.Quotient.mk y)).2⟩
354     have hg : (e (i + j)).symm (Submodule.Quotient.mk ⟨x * y, φ.mul_mem_add x.2 y.2⟩) = g := by
355       rw [LinearEquiv.symm_apply_eq]
356       change _ = P.sym (i + j) g
357       rw [P.sym_mul i j _ _ g rfl, key, key, gradedMul_mk]
358     exact congrArg Subtype.val hg
