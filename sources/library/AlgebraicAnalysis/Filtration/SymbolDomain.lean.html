8 # Filtered rings whose associated graded ring is a domain
9 
10 Let `φ` be an ascending, exhaustive, submultiplicative filtration of the
11 `R`-algebra `A`, and let `S : φ.GradedSymbol 𝒜` identify `gr A` with a
12 commutative graded algebra `G = ⨁ 𝒜 n`. If `G` has no zero divisors, then
13 neither has `A`; if moreover `G` is nontrivial and `1 ∈ 𝒜 0`, then `A` is a
14 domain (the standard argument for filtered rings, cf.
15 [McConnell–Robson, *Noncommutative Noetherian Rings*, §1.6]).
16 
17 ## Proof
18 
19 A nonzero `a ∈ A` has an order `i = φ.order a`, the least `i` with
20 `a ∈ F i`. Its symbol in degree `i` is nonzero, since the kernel of the
21 degree-`i` symbol is `F (i - 1)` (`grAt_order_ne_zero`). For nonzero `a`, `b`
22 of orders `i`, `j` the symbol of `a b` in degree `i + j` is the product of
23 the two nonzero symbols, which is nonzero in `G`, so `a b ≠ 0`.
24 -/

35 /-- The symbol of a nonzero element in the degree of its order is nonzero. -/
36 theorem grAt_order_ne_zero {a : A} (ha : a ≠ 0) : S.grAt (φ.order ha) a ≠ 0 := by
37   rw [Ne, S.grAt_eq_zero_iff (φ.mem_order ha)]
38   intro h
39   rcases hi : φ.order ha with _ | i
40   · rw [hi, below_zero, Submodule.mem_bot] at h
41     exact ha h
42   · rw [hi, below_succ] at h
43     have := φ.order_le ha h
44     omega

47 /-- **A filtered ring whose associated graded ring has no zero divisors has
48 no zero divisors.** -/
49 theorem noZeroDivisors [NoZeroDivisors G] : NoZeroDivisors A where
50   eq_zero_or_eq_zero_of_mul_eq_zero {a b} hab := by
51     by_contra h
52     push Not at h
53     obtain ⟨ha, hb⟩ := h
54     have hmul := S.grAt_mul (φ.mem_order ha) (φ.mem_order hb)
55     have h0 : S.grAt (φ.order ha + φ.order hb) (0 : A) = 0 := by
56       rw [S.grAt_of_mem (zero_mem _)]
57       exact congrArg Subtype.val (map_zero (S.gr _))
58     rw [hab, h0] at hmul
59     exact mul_ne_zero (S.grAt_order_ne_zero ha) (S.grAt_order_ne_zero hb) hmul.symm

62 /-- A filtered ring is nontrivial when its associated graded ring is
63 nontrivial with `1` of degree `0`: an element of `F 0` with symbol `1` is
64 nonzero. -/
65 theorem nontrivial [Nontrivial G] (h1 : (1 : G) ∈ 𝒜 0) : Nontrivial A := by
66   obtain ⟨u, hu, hsymb⟩ := S.exists_grAt_eq h1
67   refine ⟨⟨u, 0, fun hu0 => one_ne_zero (α := G) ?_⟩⟩
68   rw [← hsymb, S.grAt_eq_zero_iff hu, below_zero, Submodule.mem_bot]
69   exact hu0

72 /-- **A filtered ring whose associated graded ring is a domain, with `1` of
73 degree `0`, is a domain.** -/
74 theorem isDomain [IsDomain G] (h1 : (1 : G) ∈ 𝒜 0) : IsDomain A :=
75   haveI := S.noZeroDivisors
76   haveI := S.nontrivial h1
77   NoZeroDivisors.to_isDomain A
