9 /-!
10 # Filtered rings with commutative Noetherian associated graded ring
11 
12 Let `φ` be an ascending, exhaustive filtration of the `R`-algebra `A` with
13 `1 ∈ F 0` and `F i F j ⊆ F (i+j)`, and let `S : φ.GradedSymbol 𝒜` identify
14 `gr A` with a commutative graded algebra `G = ⨁ 𝒜 n`. This file proves:
15 
16 * `RingFiltration.GradedSymbol.isNoetherian_right` and `isNoetherianRing`:
17   if `G` is Noetherian then `A` is right and left Noetherian
18   [McConnell–Robson, *Noncommutative Noetherian Rings*, 1.6.9];
19 * `RingFiltration.GradedSymbol.devLE_right` and `devLE_left`: if moreover
20   `ringKrullDim G ≤ m`, then the right- and left-ideal lattices of `A` have
21   deviation at most `m` [MR 6.5.6 with 6.4.8], hence deviation at most omega.
22 
23 ## Proof
24 
25 For an additive subgroup `T ≤ A`, `grSpan T ≤ G` consists of the elements
26 whose degree-`n` component is the degree-`n` symbol of an element of
27 `T ∩ F n`, for every `n`. If `T ≤ T'` and `grSpan T' ≤ grSpan T` then
28 `T = T'`: an element `u ∈ T' ∩ F n` has the same symbol as some `v ∈ T ∩ F n`,
29 so `u - v ∈ T' ∩ F (n-1)` and induction on `n` applies; exhaustiveness
30 finishes (`eq_of_le_of_grSpan_le`). For a one-sided ideal `T`, commutativity
31 of `G` makes `grSpan T` an ideal of `G`. Hence `I ↦ gr I` is a strictly
32 monotone map from right (or left) ideals of `A` to ideals of `G`. Strictly
33 monotone maps pull back the ascending chain condition and deviation bounds
34 (`Order.DevLE.of_strictMono`), and `Module.devLE_of_ringKrullDim_le` bounds
35 the deviation of `G` by its Krull dimension. Both sides use the same symbol
36 data; no opposite filtration has to be constructed because the definition of
37 `grSpan` is side-free.
38 -/

51 /-! ## The symbol span of an additive subgroup -/

53 /-- The degree-`n` symbols of the elements of `T ∩ F n`. -/
54 def grPiece (T : AddSubgroup A) (n : ℕ) : AddSubgroup G where
55   carrier := {g | ∃ u ∈ T, u ∈ φ.F n ∧ S.grAt n u = g}
56   add_mem' := by
57     rintro _ _ ⟨u, hu, hu', rfl⟩ ⟨v, hv, hv', rfl⟩
58     refine ⟨u + v, T.add_mem hu hv, Submodule.add_mem _ hu' hv', ?_⟩
59     rw [← sub_neg_eq_add, S.grAt_sub hu' (Submodule.neg_mem _ hv'),
60       ← zero_sub v, S.grAt_sub (Submodule.zero_mem _) hv',
61       show S.grAt n 0 = 0 from (S.grAt_eq_zero_iff (Submodule.zero_mem _)).mpr
62         (Submodule.zero_mem _)]
63     abel
64   zero_mem' := ⟨0, T.zero_mem, Submodule.zero_mem _,
65     (S.grAt_eq_zero_iff (Submodule.zero_mem _)).mpr (Submodule.zero_mem _)⟩
66   neg_mem' := by
67     rintro _ ⟨u, hu, hu', rfl⟩
68     refine ⟨-u, T.neg_mem hu, Submodule.neg_mem _ hu', ?_⟩
69     rw [← zero_sub, S.grAt_sub (Submodule.zero_mem _) hu',
70       show S.grAt n 0 = 0 from (S.grAt_eq_zero_iff (Submodule.zero_mem _)).mpr
71         (Submodule.zero_mem _), zero_sub]

73 theorem mem_grPiece {T : AddSubgroup A} {n : ℕ} {g : G} :
74     g ∈ S.grPiece T n ↔ ∃ u ∈ T, u ∈ φ.F n ∧ S.grAt n u = g := Iff.rfl

78 /-- The symbol span `gr T`: elements of `G` all of whose homogeneous
79 components are symbols of elements of `T` in the corresponding degree. -/
80 def grSpan (T : AddSubgroup A) : AddSubgroup G where
81   carrier := {g | ∀ n, (decompose 𝒜 g n : G) ∈ S.grPiece T n}
82   add_mem' := by
83     intro a b ha hb n
84     rw [decompose_add, add_apply, Submodule.coe_add]
85     exact AddSubgroup.add_mem _ (ha n) (hb n)
86   zero_mem' := by
87     intro n
88     rw [decompose_zero, zero_apply, Submodule.coe_zero]
89     exact AddSubgroup.zero_mem _
90   neg_mem' := by
91     intro a ha n
92     rw [decompose_neg]
93     erw [DFinsupp.neg_apply]
94     rw [Submodule.coe_neg]
95     exact AddSubgroup.neg_mem _ (ha n)

97 theorem mem_grSpan {T : AddSubgroup A} {g : G} :
98     g ∈ S.grSpan T ↔ ∀ n, (decompose 𝒜 g n : G) ∈ S.grPiece T n := Iff.rfl

100 theorem grSpan_mono {T T' : AddSubgroup A} (h : T ≤ T') : S.grSpan T ≤ S.grSpan T' := by
101   intro g hg n
102   obtain ⟨u, hu, hu', he⟩ := hg n
103   exact ⟨u, h hu, hu', he⟩

105 /-- The degree-`n` symbol of an element of `T ∩ F n` lies in `gr T`. -/
106 theorem grAt_mem_grSpan {T : AddSubgroup A} {n : ℕ} {u : A} (hu : u ∈ T)
107     (hn : u ∈ φ.F n) : S.grAt n u ∈ S.grSpan T := by
108   classical
109   intro m
110   by_cases hmn : m = n
111   · subst hmn
112     rw [decompose_of_mem_same 𝒜 (S.grAt_mem m u)]
113     exact ⟨u, hu, hn, rfl⟩
114   · rw [decompose_of_mem_ne 𝒜 (S.grAt_mem n u) (Ne.symm hmn)]
115     exact AddSubgroup.zero_mem _
116 
117 /-- A homogeneous element of degree `n` in `gr T` is the degree-`n` symbol of
118 an element of `T ∩ F n`. -/
119 theorem exists_of_mem_grSpan {T : AddSubgroup A} {n : ℕ} {g : G} (hg : g ∈ S.grSpan T)
120     (hgn : g ∈ 𝒜 n) : ∃ u ∈ T, u ∈ φ.F n ∧ S.grAt n u = g := by
121   have := hg n
122   rwa [decompose_of_mem_same 𝒜 hgn] at this

124 /-- **Equal symbols force equality.** If `T ≤ T'` and every symbol of `T'`
125 is a symbol of `T`, then `T = T'`. This is where exhaustiveness and
126 `F (-1) = 0` of the filtration are used. -/
127 theorem eq_of_le_of_grSpan_le {T T' : AddSubgroup A} (hle : T ≤ T')
128     (hgr : S.grSpan T' ≤ S.grSpan T) : T = T' := by
129   refine le_antisymm hle fun u hu => ?_
130   obtain ⟨n, hn⟩ := φ.exists_mem_of_directed u
131   induction n generalizing u with
132   | zero =>
133     obtain ⟨v, hv, hv0, he⟩ :=
134       S.exists_of_mem_grSpan (hgr (S.grAt_mem_grSpan hu hn)) (S.grAt_mem 0 u)
135     have hd := (S.grAt_eq_grAt_iff hv0 hn).mp he
136     rw [below_zero, Submodule.mem_bot, sub_eq_zero] at hd
137     exact hd ▸ hv
138   | succ n ih =>
139     obtain ⟨v, hv, hvn, he⟩ :=
140       S.exists_of_mem_grSpan (hgr (S.grAt_mem_grSpan hu hn)) (S.grAt_mem (n + 1) u)
141     have hd := (S.grAt_eq_grAt_iff hn hvn).mp he.symm
142     rw [below_succ] at hd
143     have := ih (u - v) (T'.sub_mem hu (hle hv)) hd
144     simpa using T.add_mem this hv
145 
146 /-- `T ↦ gr T` is strictly monotone on additive subgroups. -/
147 theorem strictMono_grSpan : StrictMono S.grSpan := by
148   intro T T' hlt
149   refine lt_of_le_of_ne (S.grSpan_mono hlt.le) fun he => hlt.ne ?_
150   exact S.eq_of_le_of_grSpan_le hlt.le he.ge

154 /-- If homogeneous multiples of symbols of `T` are again symbols of `T`, then
155 `gr T` is closed under multiplication by `G`. -/
156 theorem mul_mem_grSpan {T : AddSubgroup A}
157     (hT : ∀ i m (r g : G), r ∈ 𝒜 i → g ∈ S.grPiece T m → r * g ∈ S.grPiece T (m + i))
158     (r : G) {g : G} (hg : g ∈ S.grSpan T) : r * g ∈ S.grSpan T := by
159   classical
160   intro n
161   rw [show r * g = ∑ i ∈ (decompose 𝒜 r).support, (decompose 𝒜 r i : G) * g by
162     rw [← Finset.sum_mul, sum_support_decompose]]
163   rw [decompose_sum, DFinsupp.finsetSum_apply, Submodule.coe_sum]
164   refine AddSubgroup.sum_mem _ fun i _ => ?_
165   rw [coe_decompose_mul_of_left_mem 𝒜 n (decompose 𝒜 r i).2]
166   split_ifs with hin
167   · have := hT i (n - i) _ _ (decompose 𝒜 r i).2 (hg (n - i))
168     rwa [Nat.sub_add_cancel hin] at this
169   · exact AddSubgroup.zero_mem _

172 /-- The symbol pieces of a right ideal absorb homogeneous elements. -/
173 theorem mul_mem_grPiece_right (I : Submodule Aᵐᵒᵖ A) {i m : ℕ} {r g : G}
174     (hr : r ∈ 𝒜 i) (hg : g ∈ S.grPiece I.toAddSubgroup m) :
175     r * g ∈ S.grPiece I.toAddSubgroup (m + i) := by
176   obtain ⟨u, hu, hu', rfl⟩ := hg
177   obtain ⟨v, hv, rfl⟩ := S.exists_grAt_eq hr
178   refine ⟨u * v, ?_, φ.mul_mem_add hu' hv, ?_⟩
179   · have := I.smul_mem (op v) hu
180     rwa [op_smul_eq_mul] at this
181   · rw [S.grAt_mul hu' hv, mul_comm]

184 /-- The symbol pieces of a left ideal absorb homogeneous elements. -/
185 theorem mul_mem_grPiece_left (I : Submodule A A) {i m : ℕ} {r g : G}
186     (hr : r ∈ 𝒜 i) (hg : g ∈ S.grPiece I.toAddSubgroup m) :
187     r * g ∈ S.grPiece I.toAddSubgroup (m + i) := by
188   obtain ⟨u, hu, hu', rfl⟩ := hg
189   obtain ⟨v, hv, rfl⟩ := S.exists_grAt_eq hr
190   refine ⟨v * u, ?_, add_comm i m ▸ φ.mul_mem_add hv hu', ?_⟩

192     rwa [smul_eq_mul] at this
193   · rw [S.grAt_mul' (add_comm i m) hv hu']

195 /-- The symbol ideal `gr I ⊆ G` of a right ideal `I` of `A`. -/
196 def grIdealRight (I : Submodule Aᵐᵒᵖ A) : Ideal G where
197   carrier := S.grSpan I.toAddSubgroup
198   add_mem' := AddSubgroup.add_mem _
199   zero_mem' := AddSubgroup.zero_mem _
200   smul_mem' r _ hg :=
201     S.mul_mem_grSpan (fun _ _ _ _ hr hg => S.mul_mem_grPiece_right I hr hg) r hg
