9 # Injective polynomial presentations from dual coordinates
10 
11 Let `φ` be a filtration of the `k`-algebra `D` with polynomial generators
12 `ι : A →ₐ[k] D`, `y₁, …, y_r` (`RingFiltration.PolynomialGenerators`), so that
13 `A[ξ₁, …, ξ_r] ↠ gr D`, `a ξ^α ↦ [ι(a) y^α]`. Suppose there are dual
14 coordinates `x₁, …, x_r ∈ A` with `y_i ι(xⱼ) - ι(xⱼ) y_i = δᵢⱼ`, and `ι` is
15 injective. If `k` is a field of characteristic zero, the presentation is
16 injective (`PolynomialGenerators.presentation_sym_injective`), so
17 `gr D ≅ A[ξ]` (`PolynomialGenerators.gradedSymbol`), and `D` is a domain when
18 `A` is (`PolynomialGenerators.isDomain`).
19 
20 Examples: rings of differential operators `D_k(R)` with a coordinate frame
21 (`DifferentialOperators.IsCoordinateFrame`), and the Weyl algebra.
22 
23 ## Proof
24 
25 Commutation with `ι(xⱼ)` acts on symbols as `∂/∂ξⱼ`: for `g` homogeneous of
26 degree `n + 1`, `[ev g, ι(xⱼ)] ≡ ev (∂g/∂ξⱼ)` below degree `n`
27 (`PolynomialGenerators.commutator_polynomialEval_sub_mem_below`). On an
28 ordered monomial `y_{i₁} ⋯ y_{i_m}` the commutator deletes one factor `yⱼ` at
29 a time, and the order of the remaining factors does not matter modulo lower
30 degree because `gr D` is commutative (`list_prod_commutator_sub_mem_below`).
31 
32 If `ev g` lies below degree `n + 1`, then so does `[ev g, ι(xⱼ)]` below
33 degree `n` (`ι(xⱼ)` has degree `0`), hence `∂g/∂ξⱼ = 0` by induction, and
34 Euler's identity `(n + 1) g = ∑ⱼ ξⱼ ∂g/∂ξⱼ` gives `g = 0`. In degree zero,
35 `ev a = ι(a)`, which vanishes only for `a = 0`.
36 
37 The argument is AA-native; compare [McConnell–Robson, *Noncommutative
38 Noetherian Rings*, 15.1.20] for `D_k(R)` with étale coordinates.
39 -/

50 private theorem mul_mem_below_pred {y e : A} (hy : y ∈ φ.F 1) {m : ℕ} (he : e ∈ φ.below (m - 1)) :
51     y * e ∈ φ.below m := by
52   cases m with
53   | zero =>
54       rw [below_zero, Submodule.mem_bot] at he ⊢
55       rw [he, mul_zero]
56   | succ m => simpa [add_comm] using φ.mul_mem_below_right hy he
57 
58 /-- **Commutator with an ordered monomial.** Let `y_i ∈ F 1` and let `c`
59 satisfy `y_i c - c y_i = δᵢⱼ`. If the associated graded module is
60 commutative, then for every list `l`, the commutator of `c` with
61 `y_{l₁} ⋯ y_{l_m}` is `(count j l)` times the product with one `y_j`
62 deleted, modulo `F (m - 2)`. -/
63 theorem list_prod_commutator_sub_mem_below (hφ : φ.GrCommutative) {y : σ → A}
64     (hy : ∀ i, y i ∈ φ.F 1) {c : A} {j : σ}
65     (hc : ∀ i, y i * c - c * y i = if i = j then 1 else 0) (l : List σ) :
66     (l.map y).prod * c - c * (l.map y).prod - l.count j • ((l.erase j).map y).prod ∈
67       φ.below (l.length - 1) := by
68   induction l with
69   | nil => simp
70   | cons i l ih =>
71       set E := (l.map y).prod * c - c * (l.map y).prod - l.count j • ((l.erase j).map y).prod
72       have hE : y i * E ∈ φ.below l.length := φ.mul_mem_below_pred (hy i) ih
73       simp only [List.map_cons, List.prod_cons, List.length_cons, Nat.add_sub_cancel]
74       by_cases hij : i = j
75       · subst hij
76         rw [List.count_cons_self, List.erase_cons_head]
77         have key : y i * (l.map y).prod * c - c * (y i * (l.map y).prod) -
78               (l.count i + 1) • (l.map y).prod =
79             y i * E + l.count i • (y i * ((l.erase i).map y).prod - (l.map y).prod) +
80               (y i * c - c * y i - 1) * (l.map y).prod := by
81           simp only [E, mul_sub, mul_smul_comm, smul_sub, add_smul, one_smul, sub_mul, one_mul]
82           noncomm_ring
83         rw [key, hc]
84         simp only [↓reduceIte, sub_self, zero_mul, add_zero]
85         refine add_mem hE ?_
86         by_cases hmem : i ∈ l
87         · refine Submodule.smul_of_tower_mem _ _ ?_
88           have hperm := φ.list_prod_sub_mem_below_of_perm hφ y (fun _ => 1) hy
89             (List.perm_cons_erase hmem).symm
90           have hlen : 1 + (l.length - 1) = l.length := by
91             have := List.length_pos_of_mem hmem
92             omega
93           simpa only [List.map_cons, List.prod_cons, List.map_const', List.sum_cons,
94             List.sum_replicate, smul_eq_mul, mul_one, List.length_erase_of_mem hmem,
95             hlen] using hperm
96         · rw [List.count_eq_zero_of_not_mem hmem, zero_smul]
97           exact zero_mem _
98       · rw [List.count_cons_of_ne hij, List.erase_cons_tail (by simpa using hij)]
99         simp only [List.map_cons, List.prod_cons]
100         have key : y i * (l.map y).prod * c - c * (y i * (l.map y).prod) -
101               l.count j • (y i * ((l.erase j).map y).prod) =
102             y i * E + (y i * c - c * y i) * (l.map y).prod := by
103           simp only [E, mul_sub, mul_smul_comm]
104           noncomm_ring
105         rw [key, hc]
106         simp only [hij, ↓reduceIte, zero_mul, add_zero]
107         exact hE
108 
109 end List

144 /-- On an ordered monomial `y^α`, commutation with `c` (where
145 `y_i c - c y_i = δᵢⱼ`) is `αⱼ y^(α - eⱼ)` below degree `|α| - 1`. -/
146 theorem monomialProd_commutator_sub_mem_below (hφ : φ.GrCommutative) {y : σ → D}
147     (hy : ∀ i, y i ∈ φ.F 1) {c : D} {j : σ}
148     (hc : ∀ i, y i * c - c * y i = if i = j then 1 else 0) (α : σ →₀ ℕ) :
149     monomialProd y α * c - c * monomialProd y α - α j • monomialProd y (α - Finsupp.single j 1) ∈
150       φ.below (Finsupp.weight 1 α - 1) := by
151   have hl := φ.list_prod_commutator_sub_mem_below hφ hy hc (monomialList α)
152   rw [count_monomialList, length_monomialList] at hl
153   by_cases hj : α j = 0
154   · simpa [hj, monomialProd] using hl
155   have hperm := φ.list_prod_sub_mem_below_of_perm hφ _ (fun _ => 1) hy
156     (monomialList_erase_perm hj)
157   simp only [List.map_const', List.sum_replicate, smul_eq_mul, mul_one, List.length_erase,
158     length_monomialList] at hperm
159   have hmem : j ∈ monomialList α := List.count_pos_iff.mp (by rw [count_monomialList]; omega)
160   simp only [hmem, ↓reduceIte] at hperm
161   have h := add_mem hl (Submodule.smul_of_tower_mem _ (α j) hperm)
162   convert h using 1
163   simp only [monomialProd, smul_sub]
164   abel

172 /-- **Commutation with a dual coordinate differentiates the symbol.** For `g`
173 homogeneous of degree `n + 1`, `[ev g, ι(xⱼ)] - ev (∂g/∂ξⱼ)` lies below
174 degree `n`, where `ev (a ξ^α) = ι(a) y^α`. -/
175 theorem commutator_polynomialEval_sub_mem_below {n : ℕ} {g : MvPolynomial σ A}
176     (hg : g.IsHomogeneous (n + 1)) (j : σ) :
177     polynomialEval ι y g * ι (x j) - ι (x j) * polynomialEval ι y g -
178       polynomialEval ι y (pderiv j g) ∈ φ.below n := by
179   rw [← support_sum_monomial_coeff g, map_sum, map_sum, map_sum, Finset.sum_mul, Finset.mul_sum,
180     ← Finset.sum_sub_distrib, ← Finset.sum_sub_distrib]
181   refine Submodule.sum_mem _ fun α hα => ?_
182   have hdeg : Finsupp.weight 1 α - 1 = n := by
183     rw [hg (mem_support_iff.mp hα)]
184     rfl
185   have hm := monomialProd_commutator_sub_mem_below H.grCommutative H.gen_mem
186     (fun i => hxy i j) α
187   rw [hdeg] at hm
188   have h := φ.mul_mem_below_right (H.map_mem (g.coeff α)) hm
189   rw [zero_add] at h
190   convert h using 1
191   have hc : ι (g.coeff α) * ι (x j) = ι (x j) * ι (g.coeff α) := by
192     rw [← map_mul, mul_comm, map_mul]
193   rw [pderiv_monomial, polynomialEval_monomial, polynomialEval_monomial, map_mul, map_natCast,
194     mul_sub, mul_sub, nsmul_eq_mul, ← mul_assoc, ← mul_assoc, ← mul_assoc, ← hc, mul_assoc,
195     mul_assoc]
196   congr 1
197   rw [mul_assoc, Nat.cast_comm]
198

215 /-- **Injectivity of the symbol.** If `g` is homogeneous of degree `n` and
216 `ev g` lies below degree `n`, then `g = 0`. -/
217 theorem eq_zero_of_polynomialEval_mem_below :
218     ∀ n, ∀ g : MvPolynomial σ A, g.IsHomogeneous n → polynomialEval ι y g ∈ φ.below n → g = 0 := by
219   intro n
220   induction n with
221   | zero =>
222       intro g hg hmem
223       by_contra hg0
224       obtain ⟨a, rfl⟩ : ∃ a, g = C a :=
225         ⟨_, totalDegree_eq_zero_iff_eq_C.mp (hg.totalDegree hg0)⟩
226       rw [C_apply, polynomialEval_monomial, monomialProd_zero, mul_one, below_zero,
227         Submodule.mem_bot, ← map_zero ι] at hmem
228       exact hg0 (by rw [hι hmem, C_0])
229   | succ n ih =>
230       intro g hg hmem
231       have hpd : ∀ j, pderiv j g = 0 := by
232         intro j
233         refine ih _ (by simpa using hg.pderiv (i := j)) ?_
234         have hcomm : polynomialEval ι y g * ι (x j) - ι (x j) * polynomialEval ι y g ∈
235             φ.below n := by
236           simpa using H.grCommutative hmem (H.map_mem (x j))
237         simpa using sub_mem hcomm (H.commutator_polynomialEval_sub_mem_below hxy hg j)
238       have heuler := hg.sum_X_mul_pderiv
239       simp only [hpd, mul_zero, Finset.sum_const_zero] at heuler
240       have hne : ((n + 1 : ℕ) : k) ≠ 0 := Nat.cast_ne_zero.mpr (Nat.succ_ne_zero n)
241       rw [← Nat.cast_smul_eq_nsmul k] at heuler
242       rw [← inv_smul_smul₀ hne g, ← heuler, smul_zero]
243 
244 /-- The polynomial presentation `A[ξ] ↠ gr D` is injective in every degree. -/
245 theorem presentation_sym_injective (n : ℕ) : Function.Injective (H.presentation.sym n) := by
246   refine (injective_iff_map_eq_zero _).mpr fun g hg => ?_
247   rw [presentation_sym, Submodule.Quotient.mk_eq_zero, mem_belowInF] at hg
248   exact Subtype.ext (H.eq_zero_of_polynomialEval_mem_below hxy hι n g.1
249     (mem_polynomialGrading.mp g.2) hg)

251 /-- **`gr D ≅ A[ξ₁, …, ξ_r]`**, graded by total degree. -/
252 noncomputable def gradedSymbol : φ.GradedSymbol (polynomialGrading σ k A) :=
