10 # Principal symbols with values in a commutative graded algebra
11 
12 For a ring filtration `φ : RingFiltration R A` of a (possibly noncommutative)
13 `R`-algebra `A`, a `RingFiltration.GradedSymbol φ 𝒜` identifies the
14 associated graded ring `gr A = ⨁ F n / F (n-1)` with a commutative
15 `ℕ`-graded `R`-algebra `G = ⨁ 𝒜 n`. It consists of `R`-linear maps
16 `gr n : F n → 𝒜 n` which are surjective, have kernel `F (n-1)` (with
17 `F (-1) = 0`), and are multiplicative. These conditions say precisely that
18 each `gr n` induces `F n / F (n-1) ≃ 𝒜 n`
19 (`RingFiltration.GradedSymbol.gradedQuotientEquiv`) and that these
20 isomorphisms carry the graded multiplication of `gr A` to that of `G`.
21 Hypotheses such as "`gr A` is a commutative Noetherian ring" are stated
22 through such an identification, as in [McConnell–Robson, *Noncommutative
23 Noetherian Rings*, 1.6.9 and 6.5.6]; the ring structure of `gr A` itself is
24 never constructed.
25 
26 The interface adapts the `FilteredAlgebraData` symbol interface of
27 `itpplasma/generalized-stafford-formal` (archived `GeneralizedStafford/Setting.lean`)
28 to the shared `RingFiltration` structure.
29 
30 ## Main definitions
31 
32 * `RingFiltration.GradedSymbol φ 𝒜`: the symbol data.
33 * `RingFiltration.GradedSymbol.grAt n u`: the degree-`n` symbol of `u`,
34   extended by zero outside `F n`.
35 * `RingFiltration.ofGradedAlgebra 𝒜` and `RingFiltration.GradedSymbol.ofGradedAlgebra 𝒜`:
36   a graded ring filtered by degree, with associated graded ring itself.
37 -/

47 /-- Principal-symbol data identifying the associated graded ring of `φ` with
48 the commutative graded algebra `G = ⨁ 𝒜 n`: surjective `R`-linear maps
49 `gr n : F n → 𝒜 n` with kernel `F (n - 1)` that multiply symbols. -/
50 structure GradedSymbol (φ : RingFiltration R A) (𝒜 : ℕ → Submodule R G) where
51   /-- The degree-`n` symbol map `F n → F n / F (n-1) ≃ 𝒜 n`. -/
52   gr : ∀ n, φ.F n →ₗ[R] 𝒜 n
53   /-- Every homogeneous element of `G` is a symbol. -/
54   gr_surjective : ∀ n, Function.Surjective (gr n)
55   /-- The kernel of `gr n` is the previous filtration piece. -/
56   gr_eq_zero_iff : ∀ n (x : φ.F n), gr n x = 0 ↔ (x : A) ∈ φ.below n
57   /-- Symbols multiply: `gr (i+j) (x y) = gr i x * gr j y`. -/
58   gr_mul : ∀ i j (x : φ.F i) (y : φ.F j),
59     ((gr (i + j) ⟨x * y, φ.mul_mem_add x.2 y.2⟩ : 𝒜 (i + j)) : G) =
60       (gr i x : G) * (gr j y : G)

66 /-- The symbol map induces `F n / F (n-1) ≃ 𝒜 n`. -/
67 noncomputable def gradedQuotientEquiv (n : ℕ) : φ.GradedQuotient n ≃ₗ[R] 𝒜 n :=
68   (Submodule.quotEquivOfEq (φ.belowInF n) (LinearMap.ker (S.gr n)) (by
69     ext x
70     rw [LinearMap.mem_ker, S.gr_eq_zero_iff]
71     rfl)).trans ((S.gr n).quotKerEquivOfSurjective (S.gr_surjective n))
72 
73 theorem gradedQuotientEquiv_mk (n : ℕ) (x : φ.F n) :
74     S.gradedQuotientEquiv n (Submodule.Quotient.mk x) = S.gr n x :=
75   rfl

77 /-- The degree-`n` symbol of `u`, extended by zero outside `F n`. -/
78 noncomputable def grAt (n : ℕ) (u : A) : G := by
79   classical exact if h : u ∈ φ.F n then ((S.gr n ⟨u, h⟩ : 𝒜 n) : G) else 0
80 
81 theorem grAt_of_mem {n : ℕ} {u : A} (h : u ∈ φ.F n) :
82     S.grAt n u = ((S.gr n ⟨u, h⟩ : 𝒜 n) : G) := by
83   simp [grAt, h]
84 
85 theorem grAt_mem (n : ℕ) (u : A) : S.grAt n u ∈ 𝒜 n := by
86   unfold grAt
87   split_ifs
88   · exact (S.gr n _).2
89   · exact Submodule.zero_mem _
90 
91 theorem grAt_sub {n : ℕ} {u v : A} (hu : u ∈ φ.F n) (hv : v ∈ φ.F n) :
92     S.grAt n (u - v) = S.grAt n u - S.grAt n v := by
93   rw [S.grAt_of_mem hu, S.grAt_of_mem hv, S.grAt_of_mem (Submodule.sub_mem _ hu hv)]
94   have : (⟨u - v, Submodule.sub_mem _ hu hv⟩ : φ.F n) = ⟨u, hu⟩ - ⟨v, hv⟩ := rfl
95   rw [this, map_sub]
96   rfl
97 
98 theorem grAt_eq_zero_iff {n : ℕ} {u : A} (hu : u ∈ φ.F n) :
99     S.grAt n u = 0 ↔ u ∈ φ.below n := by
100   rw [S.grAt_of_mem hu, Submodule.coe_eq_zero, S.gr_eq_zero_iff]
101 
102 /-- Two elements of `F n` have the same degree-`n` symbol iff their
103 difference lies in `F (n - 1)`. -/
104 theorem grAt_eq_grAt_iff {n : ℕ} {u v : A} (hu : u ∈ φ.F n) (hv : v ∈ φ.F n) :
105     S.grAt n u = S.grAt n v ↔ u - v ∈ φ.below n := by
106   rw [← S.grAt_eq_zero_iff (Submodule.sub_mem _ hu hv), S.grAt_sub hu hv, sub_eq_zero]
107 
108 theorem grAt_mul {i j : ℕ} {u v : A} (hu : u ∈ φ.F i) (hv : v ∈ φ.F j) :
109     S.grAt (i + j) (u * v) = S.grAt i u * S.grAt j v := by
110   rw [S.grAt_of_mem hu, S.grAt_of_mem hv, S.grAt_of_mem (φ.mul_mem_add hu hv)]
111   exact S.gr_mul i j ⟨u, hu⟩ ⟨v, hv⟩
112 
113 theorem grAt_mul' {i j n : ℕ} (h : i + j = n) {u v : A} (hu : u ∈ φ.F i)
114     (hv : v ∈ φ.F j) : S.grAt n (u * v) = S.grAt i u * S.grAt j v := by
115   subst h
116   exact S.grAt_mul hu hv
117 
118 theorem exists_grAt_eq {n : ℕ} {g : G} (hg : g ∈ 𝒜 n) :
119     ∃ u ∈ φ.F n, S.grAt n u = g := by
120   obtain ⟨x, hx⟩ := S.gr_surjective n ⟨g, hg⟩
121   exact ⟨x, x.2, by rw [S.grAt_of_mem x.2]; simp [hx]⟩

131 /-- The degree filtration of a graded algebra: `F n` consists of the elements
132 whose homogeneous components above degree `n` vanish. -/
133 def degreeLE (n : ℕ) : Submodule R G where
134   carrier := {g | ∀ i, n < i → decompose 𝒜 g i = 0}
135   add_mem' := by
136     intro a b ha hb i hi
137     rw [decompose_add, add_apply, ha i hi, hb i hi, add_zero]
138   zero_mem' := by
139     intro i _
140     rw [decompose_zero, zero_apply]
141   smul_mem' := by
142     intro c a ha i hi
143     rw [decompose_smul, smul_apply, ha i hi, smul_zero]
144 
145 variable {𝒜}
146 
147 theorem mem_degreeLE {n : ℕ} {g : G} :
148     g ∈ degreeLE 𝒜 n ↔ ∀ i, n < i → decompose 𝒜 g i = 0 := Iff.rfl
149 
150 theorem mem_degreeLE_of_mem {n : ℕ} {g : G} (hg : g ∈ 𝒜 n) : g ∈ degreeLE 𝒜 n :=
151   fun _ hi => Subtype.ext (decompose_of_mem_ne 𝒜 hg hi.ne)
152 
153 /-- The product formula for homogeneous components of a product. -/
154 theorem coe_decompose_mul_eq_sum (x y : G) (k : ℕ) :
155     (decompose 𝒜 (x * y) k : G) =
156       ∑ ij ∈ Finset.antidiagonal k, (decompose 𝒜 x ij.1 : G) * decompose 𝒜 y ij.2 := by
157   rw [decompose_mul, coe_mul_apply_eq_sum_antidiagonal]
158 
159 /-- The degree filtration is submultiplicative. -/
160 theorem degreeLE_mul_le (i j : ℕ) :
161     degreeLE 𝒜 i * degreeLE 𝒜 j ≤ degreeLE 𝒜 (i + j) := by
162   rw [Submodule.mul_le]
163   intro x hx y hy k hk
164   apply Subtype.ext
165   rw [coe_decompose_mul_eq_sum, ZeroMemClass.coe_zero]
166   refine Finset.sum_eq_zero fun ab hab => ?_
167   rw [Finset.mem_antidiagonal] at hab
168   rcases lt_or_ge i ab.1 with h | h
169   · rw [hx _ h, ZeroMemClass.coe_zero, zero_mul]
170   · rw [hy _ (by omega), ZeroMemClass.coe_zero, mul_zero]

173 /-- The degree filtration of a graded algebra. -/
174 def ofGradedAlgebra : RingFiltration R G where
175   F := degreeLE 𝒜
176   mono := fun _ _ hab _ hx i hi => hx i (lt_of_le_of_lt hab hi)
177   one_mem := mem_degreeLE_of_mem SetLike.GradedOne.one_mem
178   mul_le := degreeLE_mul_le
179   exhaustive := by
180     classical
181     refine eq_top_iff.mpr fun g _ => ?_
182     let n := (decompose 𝒜 g).support.sup id
183     refine Submodule.mem_iSup_of_mem n fun i hi => ?_
184     by_contra hne
185     have : i ∈ (decompose 𝒜 g).support := DFinsupp.mem_support_iff.mpr hne
186     exact absurd (Finset.le_sup (f := id) this) (not_le.mpr hi)
187 
188 @[simp] theorem ofGradedAlgebra_F (n : ℕ) : (ofGradedAlgebra 𝒜).F n = degreeLE 𝒜 n := rfl

190 variable (𝒜) in
191 /-- The symbol of the degree filtration of a graded algebra is the top
192 homogeneous component; its associated graded ring is the algebra itself. -/
193 def GradedSymbol.ofGradedAlgebra : GradedSymbol (ofGradedAlgebra 𝒜) 𝒜 where
194   gr n :=
195     { toFun := fun x => decompose 𝒜 (x : G) n
196       map_add' := fun x y => by
197         change decompose 𝒜 ((x : G) + y) n = _
198         rw [decompose_add]
199         rfl
200       map_smul' := fun c x => by
201         change decompose 𝒜 (c • (x : G)) n = _
202         rw [decompose_smul]
203         rfl }
204   gr_surjective n := fun g =>
205     ⟨⟨g, mem_degreeLE_of_mem g.2⟩, Subtype.ext (decompose_of_mem_same 𝒜 g.2)⟩
206   gr_eq_zero_iff n x := by
207     obtain ⟨x, hx⟩ := x
208     simp only [LinearMap.coe_mk, AddHom.coe_mk]
209     cases n with
210     | zero =>
211       rw [below_zero, Submodule.mem_bot]
212       constructor
213       · intro h0
214         apply (decompose 𝒜).injective
215         ext i : 1
216         rw [decompose_zero, zero_apply]
217         rcases Nat.eq_zero_or_pos i with rfl | hi
218         · exact h0
219         · exact hx i hi
220       · rintro rfl
221         rw [decompose_zero, zero_apply]
222     | succ n =>
223       rw [below_succ]
224       constructor
225       · intro h0 i hi
226         rcases Nat.lt_or_ge (n + 1) i with h | h
227         · exact hx i h
228         · obtain rfl : i = n + 1 := by omega
229           exact h0
230       · intro h
231         exact h (n + 1) (Nat.lt_succ_self n)
232   gr_mul i j x y := by
233     obtain ⟨x, hx⟩ := x
234     obtain ⟨y, hy⟩ := y
235     simp only [LinearMap.coe_mk, AddHom.coe_mk]
236     rw [coe_decompose_mul_eq_sum, Finset.sum_eq_single (i, j)]
237     · rintro ⟨a, b⟩ hab hne
238       rw [Finset.mem_antidiagonal] at hab
239       rcases lt_or_ge i a with h | h
240       · rw [hx _ h, ZeroMemClass.coe_zero, zero_mul]
241       · have hb : j < b := by
242           by_contra hb
243           exact hne (Prod.ext (by dsimp only; omega) (by dsimp only; omega))
244         rw [hy _ hb, ZeroMemClass.coe_zero, mul_zero]
245     · intro h
246       exact absurd (Finset.mem_antidiagonal.mpr (rfl : i + j = i + j)) h
247 
248 end OfGradedAlgebra
