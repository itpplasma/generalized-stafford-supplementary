10 # Polynomial presentations of a commutative associated graded ring
11 
12 Let `φ` be an ascending filtration of a `k`-algebra `D` whose associated graded
13 module is commutative (`RingFiltration.GrCommutative`: `x y - y x` has degree
14 below `i + j` for `x ∈ F i`, `y ∈ F j`). Suppose a commutative `k`-algebra `A`
15 maps onto `F 0` by `ι : A →ₐ[k] D`, and finitely many elements `y₁, …, y_r` of
16 `F 1` generate the filtration in degree one:
17 `F (n + 1) ⊆ F n + F n y₁ + ⋯ + F n y_r`. Then the polynomial ring
18 `G = A[X₁, …, X_r]`, graded by total degree, maps onto `gr D` by
19 `a X^α ↦ [ι(a) y^α]`. This is a surjective symbol presentation
20 (`RingFiltration.SymbolPresentation`), so `D` is right and left Noetherian
21 when `A` is Noetherian, and the ideal lattices of `D` have deviation at most
22 `dim A + r` when `A` has finite Krull dimension.
23 
24 The monomial `y^α` is the product of the `y_i` in the order of the list
25 `(Finsupp.toMultiset α).toList`. The order does not matter modulo lower
26 degree, by commutativity of `gr D` (`list_prod_sub_mem_below_of_perm`).
27 
28 This is the generic layer for rings of differential operators `D_k(A)` (with
29 `y_i` coordinate derivations), and for enveloping-type algebras (with `A = k`).
30 Injectivity of the presentation is neither claimed nor needed.
31 
32 ## Main definitions and results
33 
34 * `RingFiltration.GrCommutative`: the associated graded module is commutative.
35 * `RingFiltration.list_prod_sub_mem_below_of_perm`: in a filtration with
36   commutative associated graded, permuting a product changes it only in lower
37   degree.
38 * `RingFiltration.polynomialGrading`: total-degree grading of `A[X_σ]` by
39   `k`-submodules, with its `GradedAlgebra` instance.
40 * `RingFiltration.PolynomialGenerators`: the hypotheses above.
41 * `RingFiltration.PolynomialGenerators.presentation`: the surjective symbol
42   presentation `A[X_σ] ↠ gr D`.
43 * `RingFiltration.PolynomialGenerators.adjoin_eq_top`: `D` is generated by
44   `ι(A)` and the `y_i`.
45 * `PolynomialGenerators.isNoetherian_right`, `isNoetherianRing`,
46   `devLE_right`, `devLE_left`, `devLEOmega_right`, `devLEOmega_left`.

64 /-- The associated graded module of `φ` is commutative: for `x ∈ F i` and
65 `y ∈ F j`, the commutator `x y - y x` lies below degree `i + j`. -/
66 def GrCommutative : Prop :=
67   ∀ ⦃i j : ℕ⦄ ⦃x y : D⦄, x ∈ φ.F i → y ∈ φ.F j → x * y - y * x ∈ φ.below (i + j)
68 
69 /-- The predecessor piece lies in the piece itself. -/
70 theorem below_le (n : ℕ) : φ.below n ≤ φ.F n := by
71   cases n with
72   | zero => exact bot_le
73   | succ n => exact φ.mono (Nat.le_succ n)
74 
75 /-- Left multiplication of a below-degree-`i` element by a degree-`j` element. -/
76 theorem mul_mem_below_left {x y : D} {i j : ℕ} (hx : x ∈ φ.below i) (hy : y ∈ φ.F j) :
77     x * y ∈ φ.below (i + j) :=
78   φ.below_mul_le i j (Submodule.mul_mem_mul hx hy)
79 
80 /-- Right multiplication of a degree-`i` element by a below-degree-`j` element. -/
81 theorem mul_mem_below_right {x y : D} {i j : ℕ} (hx : x ∈ φ.F i) (hy : y ∈ φ.below j) :
82     x * y ∈ φ.below (i + j) :=
83   φ.mul_below_le i j (Submodule.mul_mem_mul hx hy)
84 
85 /-- A product of elements `f i ∈ F (d i)` along a list has degree at most the
86 sum of the degrees. -/
87 theorem list_prod_mem {ι : Type*} (f : ι → D) (d : ι → ℕ) (hf : ∀ i, f i ∈ φ.F (d i))
88     (l : List ι) : (l.map f).prod ∈ φ.F (l.map d).sum := by
89   induction l with
90   | nil => simpa using φ.one_mem
91   | cons a l ih => simpa using φ.mul_mem_add (hf a) ih
92 
93 /-- **Products commute modulo lower degree.** If the associated graded module
94 is commutative, two products of the same factors `f i ∈ F (d i)` in permuted
95 orders differ by an element below the total degree. -/
96 theorem list_prod_sub_mem_below_of_perm (hφ : φ.GrCommutative) {ι : Type*} (f : ι → D)
97     (d : ι → ℕ) (hf : ∀ i, f i ∈ φ.F (d i)) {l l' : List ι} (h : l.Perm l') :
98     (l.map f).prod - (l'.map f).prod ∈ φ.below (l.map d).sum := by
99   induction h with
100   | nil => simp
101   | cons a _ ih =>
102       simp only [List.map_cons, List.prod_cons, List.sum_cons]
103       rw [← mul_sub]
104       exact φ.mul_mem_below_right (hf a) ih
105   | swap a b l =>
106       simp only [List.map_cons, List.prod_cons, List.sum_cons]
107       rw [← mul_assoc, ← mul_assoc, ← sub_mul, ← add_assoc]
108       exact φ.mul_mem_below_left (hφ (hf b) (hf a)) (φ.list_prod_mem f d hf l)
109   | trans h₁ _ ih₁ ih₂ =>
110       rw [(h₁.map d).sum_eq] at ih₁ ⊢
111       simpa using add_mem ih₁ ih₂
112 
113 end Commutative

119 /-- The total-degree grading of the polynomial ring `A[X_σ]`, by `k`-submodules:
120 the degree-`n` piece is the `A`-submodule of homogeneous polynomials of degree
121 `n`, with scalars restricted to `k`. The coefficient ring `A` lies in degree `0`. -/
122 noncomputable def polynomialGrading (σ k A : Type*) [CommRing k] [CommRing A] [Algebra k A] (n : ℕ) :
123     Submodule k (MvPolynomial σ A) :=
124   (homogeneousSubmodule σ A n).restrictScalars k

126 theorem mem_polynomialGrading {n : ℕ} {f : MvPolynomial σ A} :
127     f ∈ polynomialGrading σ k A n ↔ f.IsHomogeneous n :=
128   mem_homogeneousSubmodule n f

140 /-- The variables of the monomial `X^α`, listed with multiplicity in a fixed

142 noncomputable def monomialList (α : σ →₀ ℕ) : List σ :=

145 theorem length_monomialList (α : σ →₀ ℕ) : (monomialList α).length = Finsupp.weight 1 α := by

148 theorem monomialList_add_perm (α β : σ →₀ ℕ) :

153 /-- The ordered monomial `y^α = ∏ y_i` over `monomialList α`. -/
154 noncomputable def monomialProd (y : σ → D) (α : σ →₀ ℕ) : D :=
155   ((monomialList α).map y).prod

157 @[simp] theorem monomialProd_zero (y : σ → D) : monomialProd y 0 = 1 := by
158   simp [monomialProd, monomialList]

160 @[simp] theorem monomialProd_single (y : σ → D) (i : σ) :
161     monomialProd y (Finsupp.single i 1) = y i := by
162   simp [monomialProd, monomialList]

164 /-- The evaluation `A[X_σ] → D`, `a X^α ↦ ι(a) y^α`, as a `k`-linear map. -/
165 noncomputable def polynomialEval (ι : A →ₐ[k] D) (y : σ → D) : MvPolynomial σ A →ₗ[k] D :=
166   (Finsupp.lsum ℕ fun α => (LinearMap.mulRight k (monomialProd y α)).comp ι.toLinearMap) ∘ₗ
167     (AddMonoidAlgebra.coeffLinearEquiv k).toLinearMap

175 /-- An ordered monomial `y^α` in elements of degree one has degree `|α|`. -/
176 theorem monomialProd_mem {y : σ → D} (hy : ∀ i, y i ∈ φ.F 1) (α : σ →₀ ℕ) :
177     monomialProd y α ∈ φ.F (Finsupp.weight 1 α) := by
178   have h := φ.list_prod_mem y (fun _ => 1) hy (monomialList α)
179   rwa [List.map_const', List.sum_replicate, smul_eq_mul, mul_one, length_monomialList] at h

181 /-- The image of a homogeneous polynomial of degree `n` has degree `n`. -/
182 theorem polynomialEval_mem {ι : A →ₐ[k] D} (hι : ∀ a, ι a ∈ φ.F 0) {y : σ → D}
183     (hy : ∀ i, y i ∈ φ.F 1) {n : ℕ} {f : MvPolynomial σ A} (hf : f.IsHomogeneous n) :
184     polynomialEval ι y f ∈ φ.F n := by
185   rw [← support_sum_monomial_coeff f, map_sum]
186   refine Submodule.sum_mem _ fun α hα => ?_
187   rw [polynomialEval_monomial]
188   have h := φ.mul_mem_add (hι (f.coeff α)) (φ.monomialProd_mem hy α)
189   rwa [zero_add, hf (mem_support_iff.mp hα)] at h

191 /-- **Monomials multiply modulo lower degree.**
192 `ι(ab) y^(α+β) ≡ ι(a) y^α ι(b) y^β` below degree `|α| + |β|`. -/
193 theorem monomial_mul_sub_mem_below (hφ : φ.GrCommutative) {ι : A →ₐ[k] D}
194     (hι : ∀ a, ι a ∈ φ.F 0) {y : σ → D} (hy : ∀ i, y i ∈ φ.F 1) (α β : σ →₀ ℕ) (a b : A) :
195     ι (a * b) * monomialProd y (α + β) - ι a * monomialProd y α * (ι b * monomialProd y β) ∈
196       φ.below (Finsupp.weight 1 α + Finsupp.weight 1 β) := by
197   have hperm := φ.list_prod_sub_mem_below_of_perm hφ y (fun _ => 1) hy
198     (monomialList_add_perm α β).symm
199   rw [List.map_const', List.sum_replicate, smul_eq_mul, mul_one, length_monomialList,
200     map_add] at hperm
201   have h₁ := φ.mul_mem_below_right (φ.mul_mem_add (hι a) (hι b)) hperm
202   have h₂ := φ.mul_mem_below_left (φ.mul_mem_below_right (hι a)
203     (hφ (hι b) (φ.monomialProd_mem hy α))) (φ.monomialProd_mem hy β)
204   simp only [zero_add] at h₁ h₂
205   have key : ι (a * b) * monomialProd y (α + β) -
206       ι a * monomialProd y α * (ι b * monomialProd y β) =
207       ι a * ι b * (monomialProd y (α + β) - ((monomialList α ++ monomialList β).map y).prod) +
208         ι a * (ι b * monomialProd y α - monomialProd y α * ι b) * monomialProd y β := by
209     simp only [List.map_append, List.prod_append, map_mul, monomialProd]
210     noncomm_ring
211   rw [key]
212   exact add_mem h₁ h₂
213 
214 /-- **The evaluation is multiplicative modulo lower degree** on homogeneous
215 polynomials: `ev(g h) ≡ ev(g) ev(h)` below degree `i + j`. -/
216 theorem polynomialEval_mul_sub_mem_below (hφ : φ.GrCommutative) {ι : A →ₐ[k] D}
217     (hι : ∀ a, ι a ∈ φ.F 0) {y : σ → D} (hy : ∀ i, y i ∈ φ.F 1) {i j : ℕ}
218     {g h : MvPolynomial σ A} (hg : g.IsHomogeneous i) (hh : h.IsHomogeneous j) :
219     polynomialEval ι y (g * h) - polynomialEval ι y g * polynomialEval ι y h ∈
220       φ.below (i + j) := by
221   rw [← support_sum_monomial_coeff g, ← support_sum_monomial_coeff h, Finset.sum_mul_sum]
222   simp only [map_sum, Finset.sum_mul_sum, ← Finset.sum_sub_distrib]
223   refine Submodule.sum_mem _ fun α hα => Submodule.sum_mem _ fun β hβ => ?_
224   rw [monomial_mul_monomial, polynomialEval_monomial, polynomialEval_monomial,
225     polynomialEval_monomial]
226   have hm := φ.monomial_mul_sub_mem_below hφ hι hy α β (g.coeff α) (h.coeff β)
227   rwa [hg (mem_support_iff.mp hα), hh (mem_support_iff.mp hβ)] at hm

229 /-- The hypotheses for a polynomial presentation of `gr D`: the associated
230 graded module is commutative, `ι : A →ₐ[k] D` maps onto `F 0`, the elements
231 `y_i` have degree one, and they generate the filtration in degree one,
232 `F (n + 1) ⊆ F n + ∑ᵢ F n · y_i`. -/
233 structure PolynomialGenerators [Fintype σ] (ι : A →ₐ[k] D) (y : σ → D) : Prop where
234   /-- The associated graded module is commutative. -/
235   grCommutative : φ.GrCommutative
236   /-- `ι(A)` lies in degree `0`. -/
237   map_mem : ∀ a, ι a ∈ φ.F 0
238   /-- Every element of degree `0` is in the image of `ι`. -/
239   exists_map_eq : ∀ x ∈ φ.F 0, ∃ a, ι a = x
240   /-- The generators have degree one. -/
241   gen_mem : ∀ i, y i ∈ φ.F 1
242   /-- Generation in degree one: `F (n + 1) ⊆ F n + ∑ᵢ F n · y_i`. -/
243   exists_sum_mul : ∀ n, ∀ x ∈ φ.F (n + 1),
244     ∃ z : σ → D, (∀ i, z i ∈ φ.F n) ∧ x - ∑ i, z i * y i ∈ φ.F n

251 /-- Every element of degree `n` agrees, below degree `n`, with the image of a
252 homogeneous polynomial of degree `n`. -/
253 theorem exists_polynomialEval_sub_mem_below :
254     ∀ n, ∀ x ∈ φ.F n, ∃ g : MvPolynomial σ A, g.IsHomogeneous n ∧
255       x - polynomialEval ι y g ∈ φ.below n := by
256   intro n
257   induction n with
258   | zero =>
259       intro x hx
260       obtain ⟨a, rfl⟩ := H.exists_map_eq x hx
261       refine ⟨C a, isHomogeneous_C σ a, ?_⟩
262       rw [C_apply, polynomialEval_monomial, monomialProd_zero, mul_one, sub_self]
263       exact zero_mem _
264   | succ n ih =>
265       intro x hx
266       obtain ⟨z, hz, hxz⟩ := H.exists_sum_mul n x hx
267       choose g hg hgz using fun i => ih (z i) (hz i)
268       refine ⟨∑ i, g i * X i,
269         IsHomogeneous.sum _ _ _ fun i _ => (hg i).mul (isHomogeneous_X A i), ?_⟩
270       have hX : ∀ i, polynomialEval ι y (X i) = y i := by
271         intro i
272         rw [X, polynomialEval_monomial, map_one, one_mul, monomialProd_single]
273       have key : x - polynomialEval ι y (∑ i, g i * X i) =
274           (x - ∑ i, z i * y i) + ∑ i, (z i - polynomialEval ι y (g i)) * y i -
275             ∑ i, (polynomialEval ι y (g i * X i) - polynomialEval ι y (g i) * y i) := by
276         simp only [map_sum, sub_mul, Finset.sum_sub_distrib]
277         abel
278       rw [key]
279       refine sub_mem (add_mem hxz (Submodule.sum_mem _ fun i _ => ?_))
280         (Submodule.sum_mem _ fun i _ => ?_)
281       · exact φ.mul_mem_below_left (hgz i) (H.gen_mem i)
282       · have h := φ.polynomialEval_mul_sub_mem_below H.grCommutative H.map_mem H.gen_mem
283           (hg i) (isHomogeneous_X A i)
284         rwa [hX] at h

305 /-- **The polynomial presentation of `gr D`.** The symbol of a homogeneous
306 polynomial `g` of degree `n` is the class of `ev(g)` in `Fₙ/Fₙ₋₁`. -/
307 noncomputable def presentation : SymbolPresentation φ (polynomialGrading σ k A) where
308   sym n := (φ.belowInF n).mkQ ∘ₗ H.evalPiece n
309   sym_surjective n := by
310     intro q
311     obtain ⟨x, rfl⟩ := Submodule.Quotient.mk_surjective _ q
312     obtain ⟨g, hg, hd⟩ := H.exists_polynomialEval_sub_mem_below n x x.2
313     refine ⟨⟨g, mem_polynomialGrading.mpr hg⟩, (Submodule.Quotient.eq _).mpr ?_⟩
314     rw [mem_belowInF]
315     simpa [evalPiece] using neg_mem hd
316   sym_mul i j g h x hx := by
317     show Submodule.Quotient.mk _ = φ.gradedMul i j (Submodule.Quotient.mk _)
318       (Submodule.Quotient.mk _)
319     rw [gradedMul_mk]
320     refine (Submodule.Quotient.eq _).mpr ?_
321     rw [mem_belowInF]
322     simpa [evalPiece, hx] using φ.polynomialEval_mul_sub_mem_below H.grCommutative H.map_mem
323       H.gen_mem (mem_polynomialGrading.mp g.2) (mem_polynomialGrading.mp h.2)
