9 # Opposite filtrations and opposite symbols
10 
11 For a ring filtration `φ` of an `R`-algebra `A`, `φ.opposite` is the filtration
12 of `Aᵐᵒᵖ` whose piece `n` is the image of `φ.F n` under `a ↦ op a`. A symbol
13 `S : φ.GradedSymbol 𝒜` induces `S.opposite : φ.opposite.GradedSymbol 𝒜` with the
14 same graded algebra `G`. This uses only that `G` is commutative: the product
15 `op y * op x = op (x * y)` of opposite elements has symbol `gr j(y) · gr i(x)`,
16 which is the product `gr i(x) · gr j(y)` in `G`.
17 
18 Right ideals of `Aᵐᵒᵖ` are left ideals of `A`, and left ideals of `Aᵐᵒᵖ` are
19 right ideals of `A` (`rightIdealOpEquiv`, `leftIdealOpEquiv`). Deviation and
20 Noetherianity are order-theoretic, so each right-side statement about `Aᵐᵒᵖ`
21 is equivalent to the left-side statement about `A`, and conversely
22 (`isNoetherian_right_op_iff`, `devLE_right_op_iff`, `isNoetherianRing_op_iff`,
23 `devLE_left_op_iff`). Applied to the symbol `S.opposite` they give the
24 side-exchanged conclusions of `SymbolNoetherian` (`isNoetherianRing_of_opposite_right`
25 and the three companions at the end of this file).
26 
27 ## Main definitions and results
28 
29 * `RingFiltration.opposite φ`: the opposite filtration of `Aᵐᵒᵖ`.
30 * `RingFiltration.GradedSymbol.opposite S`: the opposite symbol, with the same `G`.
31 * `RingFiltration.GradedSymbol.coe_gr_opposite`: the opposite symbol of `op u` is
32   the symbol of `u`.
33 * `rightIdealOpEquiv`, `leftIdealOpEquiv`: the order isomorphisms of ideal lattices.
34 -/

44 /-- Right ideals of `Aᵐᵒᵖ` correspond to left ideals of `A`: `x` lies in the
45 left ideal iff `op x` lies in the right ideal. -/
46 def rightIdealOpEquiv : Submodule Aᵐᵒᵖᵐᵒᵖ Aᵐᵒᵖ ≃o Submodule A A where
47   toFun p :=
48     { carrier := {x | op x ∈ p}
49       add_mem' := fun {x y} hx hy => by
50         simpa [op_add] using p.add_mem hx hy
51       zero_mem' := by simp
52       smul_mem' := fun c {x} hx => by
53         change op (c * x) ∈ p
54         rw [op_mul, ← op_smul_eq_mul]
55         exact p.smul_mem _ hx }
56   invFun q :=
57     { carrier := {y | unop y ∈ q}
58       add_mem' := fun {y z} hy hz => by
59         simpa [unop_add] using q.add_mem hy hz
60       zero_mem' := by simp
61       smul_mem' := fun c {y} hy => by
62         obtain ⟨z⟩ := c
63         obtain ⟨a⟩ := z
64         change unop (op (op a) • y) ∈ q
65         rw [op_smul_eq_mul, unop_mul, unop_op]
66         exact q.smul_mem a hy }
67   left_inv p := by
68     ext y
69     simp
70   right_inv q := by
71     ext x
72     simp
73   map_rel_iff' := by
74     intro p q
75     constructor
76     · intro h y hy
77       have h' : unop y ∈ ({x | op x ∈ p} : Set A) := by simpa using hy
78       simpa using h h'
79     · intro h x hx
80       exact h hx
81 
82 /-- Left ideals of `Aᵐᵒᵖ` correspond to right ideals of `A`: `x` lies in the
83 right ideal iff `op x` lies in the left ideal. -/
84 def leftIdealOpEquiv : Submodule Aᵐᵒᵖ Aᵐᵒᵖ ≃o Submodule Aᵐᵒᵖ A where
85   toFun p :=
86     { carrier := {x | op x ∈ p}
87       add_mem' := fun {x y} hx hy => by
88         simpa [op_add] using p.add_mem hx hy
89       zero_mem' := by simp
90       smul_mem' := fun c {x} hx => by
91         obtain ⟨a⟩ := c
92         change op (x * a) ∈ p
93         rw [op_mul, ← smul_eq_mul]
94         exact p.smul_mem _ hx }
95   invFun q :=
96     { carrier := {y | unop y ∈ q}
97       add_mem' := fun {y z} hy hz => by
98         simpa [unop_add] using q.add_mem hy hz
99       zero_mem' := by simp
100       smul_mem' := fun c {y} hy => by
101         change unop (c * y) ∈ q
102         rw [unop_mul]
103         exact q.smul_mem c hy }
104   left_inv p := by
105     ext y
106     simp
107   right_inv q := by
108     ext x
109     simp
110   map_rel_iff' := by
111     intro p q
112     constructor
113     · intro h y hy
114       have h' : unop y ∈ ({x | op x ∈ p} : Set A) := by simpa using hy
115       simpa using h h'
116     · intro h x hx
117       exact h hx

125 /-- The opposite map `A →ₗ[R] Aᵐᵒᵖ`, `a ↦ op a`. -/
126 abbrev opLin (R A : Type*) [CommRing R] [Ring A] [Algebra R A] : A →ₗ[R] Aᵐᵒᵖ :=
127   (MulOpposite.opLinearEquiv R : A ≃ₗ[R] Aᵐᵒᵖ).toLinearMap
128 
129 /-- The opposite filtration of `Aᵐᵒᵖ`: piece `n` is the image of `φ.F n` under
130 `a ↦ op a`. -/
131 def opposite (φ : RingFiltration R A) : RingFiltration R Aᵐᵒᵖ where
132   F n := (φ.F n).map (opLin R A)
133   mono _ _ hij := Submodule.map_mono (φ.mono hij)
134   one_mem := Submodule.mem_map.mpr ⟨1, φ.one_mem, by simp [opLin]⟩
135   mul_le i j := by
136     show (φ.F i).map (opLin R A) * (φ.F j).map (opLin R A) ≤ (φ.F (i + j)).map (opLin R A)
137     rw [← Submodule.map_op_mul, add_comm i j]
138     exact Submodule.map_mono (φ.mul_le j i)
139   exhaustive := by
140     rw [← Submodule.map_iSup, φ.exhaustive, Submodule.map_top]
141     exact LinearMap.range_eq_top.mpr (MulOpposite.opLinearEquiv R).surjective
142 
143 theorem mem_opposite_op_iff {φ : RingFiltration R A} {n : ℕ} {a : A} :
144     op a ∈ (φ.opposite).F n ↔ a ∈ φ.F n := by
145   constructor
146   · intro h
147     obtain ⟨b, hb, hba⟩ := Submodule.mem_map.mp h
148     have : b = a := by simpa [opLin] using hba
149     exact this ▸ hb
150   · intro h
151     exact Submodule.mem_map_of_mem (f := opLin R A) h
152 
153 theorem mem_opposite_iff {φ : RingFiltration R A} {n : ℕ} {x : Aᵐᵒᵖ} :
154     x ∈ (φ.opposite).F n ↔ unop x ∈ φ.F n := by
155   rw [← mem_opposite_op_iff (a := unop x), op_unop]
156 
157 theorem mem_opposite_below_iff {φ : RingFiltration R A} (n : ℕ) (x : Aᵐᵒᵖ) :
158     x ∈ (φ.opposite).below n ↔ unop x ∈ φ.below n := by
159   cases n with
160   | zero =>
161     rw [below_zero, below_zero, Submodule.mem_bot, Submodule.mem_bot, unop_eq_zero_iff]
162   | succ n =>
163     rw [below_succ, below_succ]
164     exact mem_opposite_iff
165 
166 end RingFiltration

173 /-- The restriction of `unop` to the `n`-th piece of the opposite filtration. -/
174 def unopPiece (φ : RingFiltration R A) (n : ℕ) : (φ.opposite).F n →ₗ[R] φ.F n :=
175   LinearMap.restrict ((MulOpposite.opLinearEquiv R : A ≃ₗ[R] Aᵐᵒᵖ).symm :
176       Aᵐᵒᵖ →ₗ[R] A) fun _ hx => mem_opposite_iff.mp hx
177 
178 theorem unopPiece_apply (n : ℕ) (x : (φ.opposite).F n) :
179     ((unopPiece φ n x : φ.F n) : A) = unop (x : Aᵐᵒᵖ) := rfl
180 
181 /-- The opposite symbol: the symbol of `unop x` in the original filtration. -/
182 def opposite : GradedSymbol φ.opposite 𝒜 where
183   gr n := S.gr n ∘ₗ unopPiece φ n
184   gr_surjective n g := by
185     obtain ⟨y, hy⟩ := S.gr_surjective n g
186     have hx : op (y : A) ∈ (φ.opposite).F n := mem_opposite_op_iff.mpr y.2
187     refine ⟨⟨op (y : A), hx⟩, ?_⟩
188     have hy' : unopPiece φ n ⟨op (y : A), hx⟩ = y := Subtype.ext (unop_op _)
189     change S.gr n (unopPiece φ n ⟨op (y : A), hx⟩) = g
190     rw [hy', hy]
191   gr_eq_zero_iff n x := by
192     change S.gr n (unopPiece φ n x) = 0 ↔ (x : Aᵐᵒᵖ) ∈ (φ.opposite).below n
193     rw [S.gr_eq_zero_iff]
194     exact (mem_opposite_below_iff n (x : Aᵐᵒᵖ)).symm
195   gr_mul i j x y := by
196     have key : ∀ {n : ℕ} (z : (φ.opposite).F n),
197         (((S.gr n ∘ₗ unopPiece φ n) z : 𝒜 n) : G) = S.grAt n (unop (z : Aᵐᵒᵖ)) := by
198       intro n z
199       rw [S.grAt_of_mem (mem_opposite_iff.mp z.2)]
200       rfl
201     rw [key, key, key]
202     show S.grAt (i + j) (unop ((x : Aᵐᵒᵖ) * (y : Aᵐᵒᵖ))) = _
203     rw [unop_mul]
204     exact (S.grAt_mul' (add_comm j i) (mem_opposite_iff.mp y.2)
205       (mem_opposite_iff.mp x.2)).trans (mul_comm _ _)
206 
207 /-- The degree-`n` symbol of the opposite is the degree-`n` symbol of `unop x`. -/
208 theorem coe_gr_opposite {n : ℕ} (x : (φ.opposite).F n) :
209     ((S.opposite.gr n x : 𝒜 n) : G) = S.grAt n (unop (x : Aᵐᵒᵖ)) := by
210   rw [S.grAt_of_mem (mem_opposite_iff.mp x.2)]
211   rfl
212 
213 /-- The symbol of the opposite element `op u` is the symbol of `u`. -/
214 theorem grAt_opposite (n : ℕ) (u : A) : S.opposite.grAt n (op u) = S.grAt n u := by
215   by_cases h : u ∈ φ.F n
216   · rw [S.opposite.grAt_of_mem (mem_opposite_op_iff.mpr h), S.grAt_of_mem h]
217     rfl
218   · have h' : op u ∉ (φ.opposite).F n := fun h'' => h (mem_opposite_op_iff.mp h'')
219     simp [grAt, h, h']
220 
221 end RingFiltration.GradedSymbol

228 /-- Right Noetherianity of `Aᵐᵒᵖ` is left Noetherianity of `A`. -/
229 theorem isNoetherian_right_op_iff : IsNoetherian Aᵐᵒᵖᵐᵒᵖ Aᵐᵒᵖ ↔ IsNoetherianRing A := by
230   rw [isNoetherian_iff, IsNoetherianRing, isNoetherian_iff]
231   exact ⟨fun _ => (rightIdealOpEquiv A).symm.strictMono.wellFoundedGT,
232     fun _ => (rightIdealOpEquiv A).strictMono.wellFoundedGT⟩
233 
234 /-- Left Noetherianity of `Aᵐᵒᵖ` is right Noetherianity of `A`. -/
235 theorem isNoetherianRing_op_iff : IsNoetherianRing Aᵐᵒᵖ ↔ IsNoetherian Aᵐᵒᵖ A := by
236   rw [IsNoetherianRing, isNoetherian_iff, isNoetherian_iff]
237   exact ⟨fun _ => (leftIdealOpEquiv A).symm.strictMono.wellFoundedGT,
238     fun _ => (leftIdealOpEquiv A).strictMono.wellFoundedGT⟩
239 
240 /-- Right deviation of `Aᵐᵒᵖ` is left deviation of `A`. -/
241 theorem devLE_right_op_iff (m : ℕ) : Module.DevLE Aᵐᵒᵖᵐᵒᵖ m Aᵐᵒᵖ ↔ Module.DevLE A m A :=
242   Order.devLE_orderIso_iff m (rightIdealOpEquiv A)
243 
244 /-- Left deviation of `Aᵐᵒᵖ` is right deviation of `A`. -/
245 theorem devLE_left_op_iff (m : ℕ) : Module.DevLE Aᵐᵒᵖ m Aᵐᵒᵖ ↔ Module.DevLE Aᵐᵒᵖ m A :=
246   Order.devLE_orderIso_iff m (leftIdealOpEquiv A)
