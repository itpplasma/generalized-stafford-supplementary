10 # Corollary C.3: generator bounds by the normalized Goldie rank
11 
12 The upper bound is proved by induction on `q`, without transferring the module
13 to a matrix ring.
14 
15 Notation. `R` satisfies (S) and (N); `Q` is its classical right quotient ring
16 (Goldie), `M_Q = M ⊗_R Q` the module of fractions of a finitely generated right
17 module `M`, `l = length_Q(M_Q)` its Goldie rank (`AlgebraicAnalysis.Goldie.goldieRank`)
18 and `s = length_Q(Q_Q) = goldieRank R R`, so `ρ(M) = l / s`.
19 
20 **Lower bound** `⌈ρ(M)⌉ ≤ μ(M)` (`le_mul_of_span_eq_top`). If `m` elements generate
21 `M`, then `M_Q` is a quotient of `Q^m`, so `l ≤ m s`; by Goldie, `l = udim(M/t(M))`
22 and `s = udim(R_R)` (AlgebraicAnalysis `Goldie.le_mul_of_uniformDim`).
23 
24 **Upper bound** `μ(M) ≤ ⌊ρ(M)⌋ + 1` (`exists_span_eq_top_of_lt_mul`), assuming also
25 (Z) and (H_all): if `l < q s` then `M` is generated by `q` elements, by induction
26 on `q` (`spanRank_le_of_goldieRank_lt`). *Base case `l < s`*
27 (`cyclic_of_goldieRank_lt`): with idempotents `E_i` of `Q` and
28 common denominators `f_i = E_i c`, `r_i = c⁻¹ E_i t_i` in `R`
29 (`Goldie.exists_idempotent_denominators`), choose `m_i` successively so that the
30 fractions of `m_i e_i`, `e_i = f_i r_i`, span `M_Q` (simplicity of `R` makes each
31 choice possible); then `M / Σ m_i e_i R` is Goldie torsion, and C.1 with the
32 unused index `l` and the tail `e_l` produces one generator `y = p + h f_l`.
33 *Step `l ≥ s`*: since `Q` is simple Artinian, `Q_Q` embeds in `M_Q`, which gives
34 `p ∈ M` with `pR ≅ R_R`; the rank is additive, so `M/pR` has rank `l - s < (q-1) s`,
35 and `p` together with lifts of `q - 1` generators of `M/pR` generates `M`.
36 
37 The application-independent module theory (the free cyclic submodule, the
38 splitting of the rank and the successive choice of spanning fractions) is in
39 AlgebraicAnalysis (`AlgebraicAnalysis.RingTheory.Goldie.Splitting`).
40 -/

59 /-- **Base case `l < s`.** Assume (S), (N), (Z) and
60 (H_all). A finitely generated right module `M` with `rk(M) < rk(R_R)` is cyclic.
61 With `m_j` from the successive choice, `L = Σ m_j e_j R` has `L_Q = M_Q`, so
62 `M/L` is Goldie torsion; C.1 in `M/L` with tail `e_l` gives `h` with
63 `M/L = (h e_l + L)R`, and `y = p + h f_l` with `p = Σ m_j f_j` satisfies
64 `y r_j = m_j e_j` (`j ≠ l`) and `y r_l = h e_l`, hence `M = yR`.
65 
66 Indexing: the idempotents are indexed by `Fin n` from `0`, so the `l` chosen
67 spans use the indices `0, …, l - 1` and the free index is `l : Fin n` itself
68 (`l < n` because `rk(M) < rk(R_R)`). In one-based indexing the free index is
69 `l + 1`; the proof here uses the offset `0` in place of `p r_{l+1}`, since `m_l = 0` and hence `p r_l = 0` here. -/
70 theorem cyclic_of_goldieRank_lt (hall : PrincipalQuotientsFinite R) [Module.Finite Rᵐᵒᵖ M]
71     (hM : goldieRank R M < goldieRank R R) : ∃ y : M, Submodule.span Rᵐᵒᵖ {y} = ⊤ := by
72   classical
73   obtain ⟨n, E, c, t, f, r, hn, -, -, -, -, -, -, hfr, -, hne⟩ :=
74     exists_idempotent_denominators (R := R)
75   obtain ⟨l, hl⟩ := ENat.ne_top_iff_exists.mp (goldieRank_ne_top (R := R) (M := M))
76   have hln : l < n := by
77     rw [← hn, ← hl] at hM
78     exact_mod_cast hM
79   let e : Fin n → R := fun j => f j * r j
80   obtain ⟨m, hm0, hspan⟩ := exists_span_fraction_eq_top e hne hln hl.symm
81   let L : Submodule Rᵐᵒᵖ M := Submodule.span Rᵐᵒᵖ (Set.range fun j => op (e j) • m j)
82   have htor : IsGoldieTorsion R (M ⧸ L) := by
83     rw [← goldieRank_eq_zero_iff]
84     have hsurj : Function.Surjective
85         (OreModule.localizationMap (oppositeSubmonoid R⁰) L.subtype) := by
86       rw [← LinearMap.range_eq_top, eq_top_iff, ← hspan, Submodule.span_le]
87       rintro _ ⟨j, rfl⟩
88       exact ⟨OreModule.toLocalization _ L ⟨_, Submodule.subset_span ⟨j, rfl⟩⟩, rfl⟩
89     have hinj := OreModule.localizationMap_injective (S := oppositeSubmonoid R⁰)
90       L.injective_subtype
91     have hLM : goldieRank R L = goldieRank R M :=
92       (LinearEquiv.ofBijective _ ⟨hinj, hsurj⟩).length_eq
93     have hadd := goldieRank_eq_add_of_exact (R := R) L.injective_subtype L.mkQ_surjective
94       (LinearMap.exact_subtype_mkQ L)
95     rw [hLM, ← hl] at hadd
96     generalize goldieRank R (M ⧸ L) = x at hadd ⊢
97     induction x using ENat.recTopCoe with
98     | top => simp at hadd
99     | coe x => norm_cast at hadd ⊢; omega
100   let i : Fin n := ⟨l, hln⟩
101   obtain ⟨hbar, hh⟩ :=
102     prescribed_cyclic_of_hAll (k := k) (M := M ⧸ L) hall htor 0 (hne i)
103   obtain ⟨h, rfl⟩ := L.mkQ_surjective hbar
104   let p : M := ∑ j, op (f j) • m j
105   have hp : ∀ j, op (r j) • p = op (e j) • m j := by
106     intro j
107     simp only [p, Finset.smul_sum, smul_smul, ← op_mul]
108     rw [Finset.sum_eq_single j (fun b _ hb => by rw [hfr b j hb, op_zero, zero_smul])
109       (by simp)]
110   let y : M := p + op (f i) • h
111   have hyi : op (r i) • y = op (e i) • h := by
112     rw [smul_add, hp, hm0 i le_rfl, smul_zero, zero_add, smul_smul, ← op_mul]
113   have hLy : L ≤ Submodule.span Rᵐᵒᵖ {y} := by
114     rw [Submodule.span_le]
115     rintro _ ⟨j, rfl⟩
116     by_cases hj : j = i
117     · rw [hj]
118       dsimp only
119       rw [hm0 i le_rfl, smul_zero]
120       exact zero_mem _
121     · have : op (r j) • y = op (e j) • m j := by
122         rw [smul_add, hp, smul_smul, ← op_mul, hfr i j (Ne.symm hj), op_zero, zero_smul,
123           add_zero]
124       dsimp only
125       rw [← this]
126       exact Submodule.smul_mem _ _ (Submodule.mem_span_singleton_self y)
127   refine ⟨y, ?_⟩
128   have htop : L ⊔ Submodule.span Rᵐᵒᵖ {y} = ⊤ := by
129     rw [← Submodule.map_mkQ_eq_top, eq_top_iff, ← hh, Submodule.span_le,
130       Set.singleton_subset_iff, zero_add, ← map_smul, ← hyi]
131     exact Submodule.mem_map_of_mem
132       (Submodule.smul_mem _ _ (Submodule.mem_span_singleton_self y))
133   rwa [sup_eq_right.mpr hLy] at htop

136 /-- **Upper bound, rank form.** Assume (S), (N), (Z), (H_all). For
137 every `q`, a finitely generated right module `M` with `rk(M) < q · rk(R_R)` is
138 generated by `q` elements. Induction on `q`: if `rk(M) < rk(R_R)` then `M` is
139 cyclic; otherwise split off `pR ≅ R_R` and apply the induction hypothesis to
140 `M/pR`. -/
141 theorem spanRank_le_of_goldieRank_lt (hall : PrincipalQuotientsFinite R) (q : ℕ) :
142     ∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M],
143       goldieRank R M < q * goldieRank R R →
144         ∃ g : Finset M, g.card ≤ q ∧ Submodule.span Rᵐᵒᵖ (g : Set M) = ⊤ := by
145   classical
146   induction q with
147   | zero => intro M _ _ _ h; simp at h
148   | succ q ih =>
149     intro M _ _ _ hM
150     by_cases hlt : goldieRank R M < goldieRank R R
151     · -- Base case `l < s`: `M` is cyclic.
152       obtain ⟨y, hy⟩ := cyclic_of_goldieRank_lt (k := k) hall hlt
153       exact ⟨{y}, by simp, by simpa using hy⟩
154     -- Step: split off a copy `pR ≅ R_R`; the rank drops by `s`.
155     obtain ⟨p, hadd⟩ := exists_goldieRank_eq_add_quotient (not_lt.mp hlt)
156     let P := Submodule.span Rᵐᵒᵖ {p}
157     have hN : goldieRank R (M ⧸ P) < q * goldieRank R R := by
158       obtain ⟨s, hs⟩ := ENat.ne_top_iff_exists.mp (goldieRank_self_ne_top (R := R))
159       obtain ⟨x, hx⟩ := ENat.ne_top_iff_exists.mp (goldieRank_ne_top (R := R) (M := M ⧸ P))
160       rw [hadd, ← hs, ← hx] at hM
161       rw [← hs, ← hx]
162       norm_cast at hM ⊢
163       nlinarith
164     -- Induction: `M/pR` is generated by `q` elements; lift them and add `p`.
165     obtain ⟨g, hg, hspan⟩ := ih (M ⧸ P) hN
166     let lift : M ⧸ P → M := Function.surjInv P.mkQ_surjective
167     refine ⟨insert p (g.image lift), ?_, ?_⟩
168     · calc (insert p (g.image lift)).card ≤ (g.image lift).card + 1 :=
169           Finset.card_insert_le _ _
170         _ ≤ g.card + 1 := by gcongr; exact Finset.card_image_le
171         _ ≤ q + 1 := by omega
172     · rw [Finset.coe_insert, Submodule.span_insert, ← Submodule.map_mkQ_eq_top,
173         Submodule.map_span, Finset.coe_image, ← Set.image_comp,
174         show ⇑P.mkQ ∘ lift = id from funext (Function.surjInv_eq P.mkQ_surjective),
175         Set.image_id, hspan]
176 
177 end UpperBound

187 /-- **C.3, lower bound** `⌈ρ(M)⌉ ≤ μ(M)`. Assume (S),
188 (N). If `M` is generated by a set of at most `m` elements, `M/t(M)` contains a
189 direct sum of `ℓ` nonzero submodules and `R_R` contains no direct sum of `s + 1`
190 nonzero right ideals, then `ℓ ≤ m s`.
191 
192 This is a thin wrapper: the proof is AlgebraicAnalysis
193 `Goldie.le_mul_of_uniformDim` after supplying the
194 Ore set of regular elements from Goldie's theorem. -/
195 theorem le_mul_of_span_eq_top {m ℓ s : ℕ} (g : Finset M) (hg : g.card ≤ m)
196     (hspan : Submodule.span Rᵐᵒᵖ (g : Set M) = ⊤)
197     (hℓ : Module.UniformDimGE Rᵐᵒᵖ (M ⧸ goldieTorsionPart R M) ℓ)
198     (hs : Module.UniformDimLE Rᵐᵒᵖ R s) : ℓ ≤ m * s := by
199   obtain ⟨_⟩ := nonempty_oreSet_oppositeSubmonoid (R := R)
200   exact le_mul_of_uniformDim g hg hspan hℓ hs

203 /-- **C.3, upper bound** `μ(M) ≤ ⌊ρ(M)⌋ + 1`. Assume
204 (S), (N), (Z) and (H_all), and let `M` be finitely generated. If `M/t(M)` contains
205 no direct sum of `ℓ + 1` nonzero submodules, `R_R` contains a direct sum of `s`
206 nonzero right ideals and `ℓ < q s`, then `M` is generated by `q` elements:
207 `rk(M) ≤ ℓ < q s ≤ q · rk(R_R)`. -/
208 theorem exists_span_eq_top_of_lt_mul (hall : PrincipalQuotientsFinite R)
209     [Module.Finite Rᵐᵒᵖ M] {q ℓ s : ℕ} (hℓ : Module.UniformDimLE Rᵐᵒᵖ (M ⧸ goldieTorsionPart R M) ℓ)
210     (hs : Module.UniformDimGE Rᵐᵒᵖ R s) (hq : ℓ < q * s) :
211     ∃ g : Finset M, g.card ≤ q ∧ Submodule.span Rᵐᵒᵖ (g : Set M) = ⊤ := by
212   obtain ⟨_⟩ := nonempty_oreSet_oppositeSubmonoid (R := R)
213   refine spanRank_le_of_goldieRank_lt (k := k) hall q M ?_
214   calc goldieRank R M ≤ ℓ := uniformDimLE_quotient_iff.mp hℓ
215     _ < ((q * s : ℕ) : ℕ∞) := by exact_mod_cast hq
216     _ = q * s := by push_cast; rfl
217     _ ≤ q * goldieRank R R := by gcongr; exact uniformDimGE_self_iff.mp hs
218 
219 end CorollaryC3
