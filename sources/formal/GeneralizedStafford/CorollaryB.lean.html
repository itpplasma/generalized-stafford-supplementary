8 # Corollary B: rings of deviation at most `ω`
9 
10 **Statement.** If `R` is right Noetherian with `Kdim R ≤ ω`, then
11 `δ(d) = Kdim(R/dR)` is finite for every regular `d`, and `Kdim(R/I)` is finite
12 for every right ideal `I` containing a regular element; hence Theorems A and
13 A' hold for all such `d` and `I` under (S), (N), (Z).
14 
15 **Proof idea.** Left multiplication by a regular `d` is an injective
16 endomorphism of `R_R` with image `dR`. In the descending chain
17 `R ⊇ dR ⊇ d²R ⊇ ⋯` every factor is isomorphic to `R/dR`, so by the definition
18 of `Kdim R ≤ ω` the factor `R/dR` has finite deviation
19 (`Module.DevLEOmega.quotient_range_of_injective`). A right ideal containing a
20 regular `s` contains `sR`, so `R/I` is a quotient of `R/sR`.
21 -/

31 /-- **Principal quotients.** If `Kdim R ≤ ω`, then `R/dR` has
32 finite deviation for every regular `d`. -/
33 theorem principal_quotient_finite_of_omega (homega : Module.DevLEOmega Rᵐᵒᵖ R)
34     {d : R} (hd : d ∈ R⁰) : ∃ N, Module.DevLE Rᵐᵒᵖ N (R ⧸ principalRightIdeal d) :=
35   homega.quotient_range_of_injective (leftMulLinear d) fun x y hxy => by
36     have h : d * (x - y) = 0 := by
37       rw [mul_sub]
38       exact sub_eq_zero.mpr hxy
39     exact sub_eq_zero.mp (hd.1 _ h)
40 
41 /-- **Hypothesis (H_all).** If `Kdim R ≤ ω`, then (H_all) holds. -/
42 theorem principalQuotientsFinite_of_devLE_omega (homega : Module.DevLEOmega Rᵐᵒᵖ R) :
43     PrincipalQuotientsFinite R :=
44   fun _ hd => principal_quotient_finite_of_omega homega hd

46 /-- **Right ideals.** If `Kdim R ≤ ω`, then `R/I` has finite deviation for
47 every right ideal `I` containing a regular element. -/
48 theorem quotient_finite_of_omega (homega : Module.DevLEOmega Rᵐᵒᵖ R)
49     {I : Submodule Rᵐᵒᵖ R} (hreg : ∃ s ∈ I, s ∈ R⁰) :
50     ∃ N, Module.DevLE Rᵐᵒᵖ N (R ⧸ I) := by
51   obtain ⟨s, hsI, hs⟩ := hreg
52   obtain ⟨N, hN⟩ := principal_quotient_finite_of_omega homega hs
53   exact ⟨N, devLE_quotient_of_le hN (principalRightIdeal_le hsI)⟩

59 /-- **Corollary B for Theorem A'.** If `Kdim R ≤ ω` and `I` contains a regular
60 element, then for every `x` and `t ≠ 0` there is `H` with
61 `I + (x + H t)R = R`. -/
62 theorem fixed_ideal_of_devLE_omega (homega : Module.DevLEOmega Rᵐᵒᵖ R)
63     {I : Submodule Rᵐᵒᵖ R} (hreg : ∃ s ∈ I, s ∈ R⁰) (x : R) {t : R} (ht : t ≠ 0) :
64     ∃ H : R, I ⊔ principalRightIdeal (x + H * t) = ⊤ := by
65   obtain ⟨N, hfin⟩ := quotient_finite_of_omega homega hreg

69 /-- **Corollary B for Theorem A.** If `Kdim R ≤ ω`, then for every regular `d`,
70 every `a ≠ 0` and every `B` there are `H, U, V` with `1 = d U + (B + H a) d V`. -/
71 theorem fixed_anchor_of_devLE_omega (homega : Module.DevLEOmega Rᵐᵒᵖ R)
72     {d : R} (hd : d ∈ R⁰) {a : R} (ha : a ≠ 0) (B : R) :
73     ∃ H U V : R, 1 = d * U + (B + H * a) * d * V := by
74   obtain ⟨N, hfin⟩ := principal_quotient_finite_of_omega homega hd
