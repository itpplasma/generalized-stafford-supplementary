7 /-!
8 # Corollary C: module consequences (C.1, C.1 stable form, C.2)
9 
10 Assume (S), (N), (Z); zero divisors are allowed.
11 
12 **C.1, stable form** (`span_pair_eq_span_affine`). If `c, u` are killed by regular
13 elements and `Kdim(uR)` is finite, then for every `t ≠ 0` there is `H` with
14 `cR + uR = (c + u H t)R`. Proof: choose a regular `b` with `c b = 0`. Since
15 `uR ≅ R/ann(u)` and `ann(u)` contains a regular element, Theorem A' with offset
16 `0` and tail `t b ≠ 0` gives `1 = i + H t b r` with `u i = 0`. For
17 `z = c + u H t` this gives `z (b r) = u H t b r = u`, so `u ∈ zR`, and then
18 `c = z - u H t ∈ zR`.
19 
20 **C.1** (`prescribed_cyclic`). A finitely generated Goldie-torsion module `M`
21 with `Kdim M` finite satisfies `M = (x + h t)R` for every `x` and `t ≠ 0`.
22 Proof: induction on a finite generating set (`prescribed_cyclic_aux`); a new
23 generator `u` is absorbed into the current affine generator `c = x + y t` by the
24 stable form, with `h = y + u H`. Under (H_all) every cyclic submodule `uR ≅ R/ann(u)`
25 is a quotient of `R/sR` for a regular `s` killing `u`, so the dimension hypothesis
26 is automatic (`prescribed_cyclic_of_hAll`, and `prescribed_cyclic_of_devLE_omega`
27 under `Kdim R ≤ ω` by Corollary B).
28 
29 **C.2** (`two_generated_prescribed`, `two_generated`). Under (H_all), if `a ∈ I`
30 has full rank (every `y ∈ I` has a regular `s` with `y s ∈ aR`, the elementwise
31 form of `aQ = IQ`), then `I/aR` is a finitely generated Goldie-torsion module, and
32 C.1 applied to `y + aR` gives `h ∈ I` with `I = aR + (y + h t)R`. Such an `a`
33 exists in every right ideal by Goldie's theorem
34 (`AlgebraicAnalysis.Goldie.exists_mem_forall_exists_mul_mem`), so every right ideal
35 is generated by two elements.
36 -/

55 /-- **C.1, stable cyclicity.** If `c, u ∈ M` are
56 killed by regular elements of `R` and `Kdim(uR)` is finite, then for every
57 `t ≠ 0` there is `H` with `cR + uR = (c + u H t)R`. -/
58 theorem span_pair_eq_span_affine (c u : M)
59     (hc : ∃ s ∈ R⁰, op s • c = 0) (hu : ∃ s ∈ R⁰, op s • u = 0)
60     (hfin : ∃ N, Module.DevLE Rᵐᵒᵖ N (Submodule.span Rᵐᵒᵖ {u}))
61     {t : R} (ht : t ≠ 0) :
62     ∃ H : R, Submodule.span Rᵐᵒᵖ {c, u} =
63       Submodule.span Rᵐᵒᵖ {c + op (H * t) • u} := by
64   classical
65   obtain ⟨b, hb, hcb⟩ := hc
66   obtain ⟨N, hN⟩ := hfin
67   obtain ⟨s₀, hs₀, hs₀u⟩ := hu
68   -- One call of Theorem A' for `ann(u)`, with offset `0` and tail `t b ≠ 0`.
69   obtain ⟨H, hH⟩ := fixed_ideal (k := k) (I := LinearMap.ker (cyclicMap (R := R) u))
70     ⟨s₀, hs₀u, hs₀⟩ ((devLE_quotient_ker_cyclicMap_iff N u).mpr hN) 0 (t := t * b)
71     (fun h0 => ht (hb.2 t h0))
72   rw [zero_add] at hH
73   have h1 : (1 : R) ∈ LinearMap.ker (cyclicMap (R := R) u) ⊔
74       principalRightIdeal (H * (t * b)) := by
75     rw [hH]; exact Submodule.mem_top
76   obtain ⟨i, hi, w, ⟨r, rfl⟩, hiw⟩ := Submodule.mem_sup.mp h1
77   -- The element `z = c + u H t` generates `u`: `u = z (b r)`.
78   set z := c + op (H * t) • u with hz
79   have hu_eq : op (b * r) • z = u := by
80     have h := congrArg (cyclicMap (R := R) u) hiw
81     change op (i + H * (t * b) * r : R) • u = op (1 : R) • u at h
82     rw [op_one, one_smul, op_add, add_smul] at h
83     change cyclicMap (R := R) u i + _ = u at h
84     rw [LinearMap.mem_ker.mp hi, zero_add] at h
85     have e1 : op (b * r) • c = 0 := by rw [op_mul, mul_smul, hcb, smul_zero]
86     have e2 : op (b * r) • (op (H * t) • u) = op (H * (t * b) * r) • u := by
87       simp only [op_mul, mul_smul, mul_assoc]
88     rw [hz, smul_add, e1, e2, zero_add, h]
89   have huz : u ∈ Submodule.span Rᵐᵒᵖ {z} := by
90     rw [← hu_eq]
91     exact Submodule.smul_mem _ _ (Submodule.mem_span_singleton_self z)
92   -- Hence also `c = z - u H t ∈ zR`.
93   have hcz : c ∈ Submodule.span Rᵐᵒᵖ {z} := by
94     have hc' : c = z - op (H * t) • u := by rw [hz, add_sub_cancel_right]
95     rw [hc']
96     exact Submodule.sub_mem _ (Submodule.mem_span_singleton_self z)
97       (Submodule.smul_mem _ _ huz)
98   refine ⟨H, le_antisymm ?_ ?_⟩
99   · exact Submodule.span_le.mpr (Set.insert_subset_iff.mpr
100       ⟨hcz, Set.singleton_subset_iff.mpr huz⟩)
101   · refine Submodule.span_le.mpr (Set.singleton_subset_iff.mpr ?_)
102     exact Submodule.add_mem _ (Submodule.subset_span (by simp))
103       (Submodule.smul_mem _ _ (Submodule.subset_span (by simp)))
104 
105 end Stable

114 /-- The induction over a finite set `s` of elements, relative to
115 a submodule `W`: if every cyclic submodule of the Goldie-torsion module `M`
116 has finite deviation and `sR + W = M`, then for every `x` and `t ≠ 0` there is
117 `h` with `(x + h t)R + W = M`. -/
118 theorem prescribed_cyclic_aux
119     (hcyc : ∀ u : M, ∃ N, Module.DevLE Rᵐᵒᵖ N (Submodule.span Rᵐᵒᵖ {u}))
120     (htor : IsGoldieTorsion R M) {t : R} (ht : t ≠ 0) (s : Finset M) :
121     ∀ W : Submodule Rᵐᵒᵖ M, Submodule.span Rᵐᵒᵖ (s : Set M) ⊔ W = ⊤ →
122       ∀ x : M, ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} ⊔ W = ⊤ := by
123   classical
124   induction s using Finset.induction_on with
125   | empty =>
126     intro W hW x
127     refine ⟨0, ?_⟩
128     rw [Finset.coe_empty, Submodule.span_empty, bot_sup_eq] at hW
129     rw [hW, sup_top_eq]
130   | insert u s _ ih =>
131     intro W hW x
132     have hW' : Submodule.span Rᵐᵒᵖ (s : Set M) ⊔ (W ⊔ Submodule.span Rᵐᵒᵖ {u}) = ⊤ := by
133       rw [Finset.coe_insert, Submodule.span_insert] at hW
134       rw [← hW]
135       ac_rfl
136     -- Induction: `c = x + y t` generates `M` modulo `W + uR`.
137     obtain ⟨y, hy⟩ := ih (W ⊔ Submodule.span Rᵐᵒᵖ {u}) hW' x
138     -- Stable form: `cR + uR = (c + u H t)R`.
139     obtain ⟨H, hH⟩ := span_pair_eq_span_affine (k := k) (x + op t • y) u
140       (htor _) (htor u) (hcyc u) ht
141     refine ⟨y + op H • u, ?_⟩
142     have hz : x + op t • (y + op H • u) = (x + op t • y) + op (H * t) • u := by
143       simp only [smul_add, op_mul, mul_smul, add_assoc]
144     rw [hz, ← hH, eq_top_iff, ← hy, Submodule.span_insert]
145     exact le_of_eq (by ac_rfl)
146

148 /-- C.1 when every cyclic submodule `uR` has finite deviation: a finitely
149 generated Goldie-torsion module `M` satisfies `M = (x + h t)R` for some `h`,
150 for every `x` and `t ≠ 0`. -/
151 theorem prescribed_cyclic_of_cyclic [Module.Finite Rᵐᵒᵖ M]
152     (hcyc : ∀ u : M, ∃ N, Module.DevLE Rᵐᵒᵖ N (Submodule.span Rᵐᵒᵖ {u}))
153     (htor : IsGoldieTorsion R M) (x : M) {t : R} (ht : t ≠ 0) :
154     ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} = ⊤ := by
155   classical
156   obtain ⟨s, hs⟩ := Module.Finite.fg_top (R := Rᵐᵒᵖ) (M := M)
157   obtain ⟨h, hh⟩ := prescribed_cyclic_aux (k := k) hcyc htor ht s ⊥
158     (by rw [hs, top_sup_eq]) x
159   exact ⟨h, by rwa [sup_bot_eq] at hh⟩

162 /-- **C.1.** Let `M` be a finitely
163 generated Goldie-torsion right module with `Kdim M ≤ N` finite. For every
164 `x ∈ M` and every `t ≠ 0` there is `h ∈ M` with `M = (x + h t)R`. -/
165 theorem prescribed_cyclic [Module.Finite Rᵐᵒᵖ M]
166     (htor : IsGoldieTorsion R M) {N : ℕ} (hfin : Module.DevLE Rᵐᵒᵖ N M)
167     (x : M) {t : R} (ht : t ≠ 0) :
168     ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} = ⊤ :=
169   prescribed_cyclic_of_cyclic (k := k)
170     (fun u => ⟨N, hfin.of_injective (Submodule.span Rᵐᵒᵖ {u}).subtype Subtype.val_injective⟩)
171     htor x ht

174 /-- **C.1 under (H_all).** If `R/dR` has finite deviation for every regular
175 `d`, then every finitely generated Goldie-torsion module `M` satisfies
176 `M = (x + h t)R` for some `h`, for every `x` and `t ≠ 0`. -/
177 theorem prescribed_cyclic_of_hAll [Module.Finite Rᵐᵒᵖ M]
178     (hall : PrincipalQuotientsFinite R)
179     (htor : IsGoldieTorsion R M) (x : M) {t : R} (ht : t ≠ 0) :
180     ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} = ⊤ := by
181   refine prescribed_cyclic_of_cyclic (k := k) (fun u => ?_) htor x ht
182   obtain ⟨s, hs, hsu⟩ := htor u
183   obtain ⟨N, hN⟩ := hall s hs
184   exact ⟨N, (devLE_quotient_ker_cyclicMap_iff N u).mp
185     (devLE_quotient_of_le hN (principalRightIdeal_le (X := LinearMap.ker (cyclicMap u)) hsu))⟩

202 include k in
203 /-- **C.2, prescribed form.** Assume (H_all).
204 If `a ∈ I` and every `y ∈ I` has a regular `s` with `y s ∈ aR` (that is,
205 `aQ = IQ`), then for every `y ∈ I` and every `t ≠ 0` there is `h ∈ I` with
206 `I = aR + (y + h t)R`. -/
207 theorem two_generated_prescribed (hall : PrincipalQuotientsFinite R)
208     {I : Submodule Rᵐᵒᵖ R} {a : R} (haI : a ∈ I)
209     (hIa : ∀ y ∈ I, ∃ s ∈ R⁰, y * s ∈ principalRightIdeal a)
210     {y : R} (hyI : y ∈ I) {t : R} (ht : t ≠ 0) :
211     ∃ h ∈ I, I = principalRightIdeal a ⊔ principalRightIdeal (y + h * t) := by
212   classical
213   -- C.1 in the finitely generated Goldie-torsion module `I/aR`.
214   set K := (principalRightIdeal a).comap I.subtype
215   obtain ⟨hbar, hh⟩ := prescribed_cyclic_of_hAll (k := k) (M := ↥I ⧸ K) hall
216     (isGoldieTorsion_quotient_of_forall_exists_mul_mem hIa) (K.mkQ ⟨y, hyI⟩) ht
217   obtain ⟨⟨h, hhI⟩, rfl⟩ := Submodule.mkQ_surjective K hbar
218   refine ⟨h, hhI, le_antisymm ?_ ?_⟩
219   · intro m hm
220     have hmem : K.mkQ ⟨m, hm⟩ ∈ Submodule.span Rᵐᵒᵖ
221         {K.mkQ ⟨y, hyI⟩ + op t • K.mkQ ⟨h, hhI⟩} := by rw [hh]; exact Submodule.mem_top
222     obtain ⟨c, hc⟩ := Submodule.mem_span_singleton.mp hmem
223     rw [← map_smul, ← map_add, ← map_smul] at hc
224     have hK : c • ((⟨y, hyI⟩ : ↥I) + op t • ⟨h, hhI⟩) - ⟨m, hm⟩ ∈ K :=
225       (Submodule.Quotient.eq K).mp hc
226     obtain ⟨r, hr⟩ := hK
227     have hr' : a * r = (y + h * t) * unop c - m := hr
228     have : m = a * (-r) + (y + h * t) * unop c := by
229       rw [mul_neg, hr']; abel
230     rw [this]
231     exact Submodule.add_mem_sup ⟨-r, rfl⟩ ⟨unop c, rfl⟩
232   · refine sup_le (principalRightIdeal_le haI) (principalRightIdeal_le ?_)
233     exact I.add_mem hyI (I.smul_mem (op t) hhI)
234 
235 include k in
236 /-- **C.2, two generators.** Under (H_all) every right
237 ideal is generated by two elements: `I = aR + bR`. Take `a` of full rank
238 (Goldie) and apply the prescribed form with `y = 0`, `t = 1`. -/
239 theorem two_generated (hall : PrincipalQuotientsFinite R) (I : Submodule Rᵐᵒᵖ R) :
240     ∃ a b : R, I = principalRightIdeal a ⊔ principalRightIdeal b := by
241   obtain ⟨a, haI, hIa⟩ := exists_mem_forall_exists_mul_mem I
242   obtain ⟨h, -, hI⟩ := two_generated_prescribed (k := k) hall haI hIa I.zero_mem
243     one_ne_zero
244   exact ⟨a, 0 + h * 1, hI⟩
245 
246 end TwoGenerated
