9 # Conventions and the deviation facts used throughout
10 
11 * Modules are **right** modules: a right `R`-module `M` is `[Module Rᵐᵒᵖ M]`,
12   and `m r` is written `op r • m`. A right ideal is a `Submodule Rᵐᵒᵖ R`;
13   `principalRightIdeal x` is `xR` and `I ⊔ J` is `I + J`.
14 * (S) is `[IsSimpleRing R]`, (N) is `[IsNoetherian Rᵐᵒᵖ R]`, and (Z) is
15   supplied as an infinite field `k` with `[Algebra k R]`, i.e. `k` is central.
16 * A regular element is a two-sided non-zero-divisor, `d ∈ R⁰`.
17 * `Kdim M ≤ n` is the deviation bound `Module.DevLE Rᵐᵒᵖ n M`, and the
18   strict stratum `Kdim M < e` is `Module.Small Rᵐᵒᵖ e M`; so `Small (n + 1)` is
19   `DevLE n` and `Small 0` means `M = 0`.
20 * Goldie torsion is `AlgebraicAnalysis.Goldie.IsGoldieTorsion`; Goldie's
21   theorem is proved in `AlgebraicAnalysis.RingTheory.Goldie`.
22 
23 This file states the hypothesis (H_all) and collects the elementary deviation
24 facts used by several sections: quotients of `R/dR` inherit its bound, a cyclic
25 module `uR` is `R/ann(u)`, and `R/J` of deviation `< 0` means `J = R`.
26 -/

36 /-- Hypothesis (H_all): `δ(d) = Kdim(R/dR)` is finite for every regular `d`.
37 
38 The name `HAll` is taken by the challenge's own copy of this hypothesis,
39 `GeneralizedStaffordChallenge.HAll`, which the transport file restates while
40 opening this namespace; the library predicate keeps a descriptive name to
41 avoid that clash, and `GeneralizedStaffordChallenge.hAll_iff` identifies the
42 two. -/
43 def PrincipalQuotientsFinite (R : Type u) [Ring R] : Prop :=
44   ∀ d : R, d ∈ nonZeroDivisors R → ∃ N, Module.DevLE Rᵐᵒᵖ N (R ⧸ principalRightIdeal d)
45 
46 /-- A right ideal `X ⊇ dR` inherits the deviation bound of `R/dR`: `R/X` is a
47 quotient of `R/dR`. -/
48 theorem devLE_quotient_of_le {n : ℕ} {d : R}
49     (hd : Module.DevLE Rᵐᵒᵖ n (R ⧸ principalRightIdeal d))
50     {X : Submodule Rᵐᵒᵖ R} (hX : principalRightIdeal d ≤ X) :
51     Module.DevLE Rᵐᵒᵖ n (R ⧸ X) :=
52   hd.of_surjective (Submodule.factor hX) (Submodule.factor_surjective hX)
53 
54 /-- A right ideal containing `s` contains `sR`. -/
55 theorem principalRightIdeal_le {X : Submodule Rᵐᵒᵖ R} {s : R} (hs : s ∈ X) :
56     principalRightIdeal s ≤ X := by
57   rintro y ⟨r, rfl⟩
58   exact X.smul_mem (op r) hs
59 
60 /-- If `R/J` has deviation `< 0`, that is `R/J = 0`, then `J = R`. -/
61 theorem eq_top_of_small_zero {J : Submodule Rᵐᵒᵖ R}
62     (h : Module.Small Rᵐᵒᵖ 0 (R ⧸ J)) : J = ⊤ := by
63   have h1 : (1 : R) ∈ J := (Submodule.Quotient.mk_eq_zero J).mp (h.elim _ _)
64   refine eq_top_iff.mpr fun y _ => ?_
65   simpa using J.smul_mem (op y) h1

71 /-- The cyclic map `r ↦ u r` of a right module; its range is `uR` and its
72 kernel is the right annihilator `ann(u)`, so `uR ≅ R/ann(u)`. -/
73 def cyclicMap (u : M) : R →ₗ[Rᵐᵒᵖ] M where
74   toFun r := op r • u
75   map_add' r s := by rw [op_add, add_smul]
76   map_smul' a r := by
77     change op (r * unop a) • u = a • (op r • u)
78     rw [op_mul, op_unop, mul_smul]
79 
80 /-- The range of `r ↦ u r` is the cyclic submodule `uR`. -/
81 theorem range_cyclicMap (u : M) :
82     LinearMap.range (cyclicMap (R := R) u) = Submodule.span Rᵐᵒᵖ {u} := by
83   ext m
84   rw [Submodule.mem_span_singleton]
85   constructor
86   · rintro ⟨r, rfl⟩; exact ⟨op r, rfl⟩
87   · rintro ⟨a, rfl⟩; exact ⟨unop a, by change op (unop a) • u = a • u; rw [op_unop]⟩
88 
89 /-- `Kdim(R/ann(u)) = Kdim(uR)`: the two modules are isomorphic. -/
90 theorem devLE_quotient_ker_cyclicMap_iff (n : ℕ) (u : M) :
91     Module.DevLE Rᵐᵒᵖ n (R ⧸ LinearMap.ker (cyclicMap (R := R) u)) ↔
92       Module.DevLE Rᵐᵒᵖ n (Submodule.span Rᵐᵒᵖ {u}) :=
93   (Module.devLE_linearEquiv_iff n (cyclicMap (R := R) u).quotKerEquivRange).trans
94     (Module.devLE_linearEquiv_iff n (LinearEquiv.ofEq _ _ (range_cyclicMap (R := R) u)))
95 
96 end Cyclic
