33 /-- The conclusion of Theorem A for every regular anchor (Corollary B): for
34 every regular `d`, every nonzero `a` and every `B` there are `H, U, V` with
35 `1 = d U + (B + H a) d V`. -/
36 def FixedAnchorProperty : Prop :=
37   ∀ d a B : R, IsRegularElement d → a ≠ 0 →
38     ∃ H U V : R, 1 = d * U + (B + H * a) * d * V
39 
40 /-- The conclusion of Theorem A' for every right ideal containing a regular
41 element (Corollary B): `I + (x + H t)R = R` for some `H`, for every `x` and
42 every nonzero `t`. -/
43 def FixedIdealProperty : Prop :=
44   ∀ I : Submodule Rᵐᵒᵖ R, (∃ s ∈ I, IsRegularElement s) →
45     ∀ x t : R, t ≠ 0 → ∃ H : R, I ⊔ Submodule.span Rᵐᵒᵖ {x + H * t} = ⊤
46 
47 /-- The dimension part of Corollary B: `R` has (H_all), and `Kdim(R/I)` is a
48 finite integer for every right ideal `I` containing a regular element. -/
49 def CorollaryBProperty : Prop :=
50   HAll R ∧ ∀ I : Submodule Rᵐᵒᵖ R, (∃ s ∈ I, IsRegularElement s) →
51     HasFiniteDevDim R (R ⧸ I)
52 
53 /-- (C.1), prescribed generator: every finitely generated Goldie-torsion right
54 module `M` satisfies `M = (x + h t)R` for some `h`, for every `x ∈ M` and every
55 nonzero `t ∈ R`. -/
56 def PrescribedCyclicProperty : Prop :=
57   ∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M],
58     IsGoldieTorsion R M →
59     ∀ (x : M) (t : R), t ≠ 0 → ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} = ⊤
60 
61 /-- (C.1), stable cyclicity: for Goldie-torsion elements `c, u` of a right
62 module and every nonzero `t` there is `H` with `cR + uR = (c + u H t)R`. -/
63 def StablyCyclicProperty : Prop :=
64   ∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] (c u : M),
65     (∃ s : R, IsRegularElement s ∧ op s • c = 0) →
66     (∃ s : R, IsRegularElement s ∧ op s • u = 0) →
67     ∀ t : R, t ≠ 0 →
68       ∃ H : R, Submodule.span Rᵐᵒᵖ {c, u} = Submodule.span Rᵐᵒᵖ {c + op (H * t) • u}
69 
70 /-- (C.2): every right ideal `I` is `aR + (y + h t)R` for every `a ∈ I` with
71 `aQ = IQ` (every `y' ∈ I` has a regular `s` with `y' s ∈ aR`), every `y ∈ I`
72 and every nonzero `t`, with some `h ∈ I`; and every right ideal is generated
73 by two elements. -/
74 def TwoGeneratedProperty : Prop :=
75   (∀ (I : Submodule Rᵐᵒᵖ R) (a : R), a ∈ I →
76     (∀ y ∈ I, ∃ s : R, IsRegularElement s ∧ y * s ∈ Submodule.span Rᵐᵒᵖ {a}) →
77     ∀ y ∈ I, ∀ t : R, t ≠ 0 →
78       ∃ h ∈ I, I = Submodule.span Rᵐᵒᵖ {a} ⊔ Submodule.span Rᵐᵒᵖ {y + h * t}) ∧
79   ∀ I : Submodule Rᵐᵒᵖ R, ∃ a b : R, I = Submodule.span Rᵐᵒᵖ {a, b}
80 
81 /-- (C.3): `⌈ρ(M)⌉ ≤ μ(M) ≤ ⌊ρ(M)⌋ + 1` for finitely generated right modules,
82 in the division-free form of `GeneralizedStaffordChallenge.lean`
83 (`ρ(M) = udim(M/t(M)) / udim(R_R)`). -/
84 def GeneratorBoundsProperty : Prop :=
85   ∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M],
86     (∀ m ℓ s : ℕ, SpanRankLE R M m → UDimGE R (M ⧸ goldieTorsionPart R M) ℓ →
87       UDimLE R R s → ℓ ≤ m * s) ∧
88     (∀ q ℓ s : ℕ, UDimLE R (M ⧸ goldieTorsionPart R M) ℓ → UDimGE R R s →
89       ℓ < q * s → SpanRankLE R M q)
90 
91 end Conclusions

93 /-- An exhaustive increasing filtration `F₀ ⊆ F₁ ⊆ ⋯` of the `k`-algebra `R`
94 by `k`-subspaces with `1 ∈ F₀` and `Fᵢ Fⱼ ⊆ Fᵢ₊ⱼ`. -/
95 structure RingFiltration (k R : Type u) [CommRing k] [Ring R] [Algebra k R] where
96   /-- The filtration pieces `Fₙ`. -/
97   F : ℕ → Submodule k R
98   /-- `Fᵢ ⊆ Fⱼ` for `i ≤ j`. -/
99   mono : Monotone F
100   /-- `1 ∈ F₀`. -/
101   one_mem : (1 : R) ∈ F 0
102   /-- `Fᵢ Fⱼ ⊆ Fᵢ₊ⱼ`. -/
103   mul_mem : ∀ {i j : ℕ} {x y : R}, x ∈ F i → y ∈ F j → x * y ∈ F (i + j)
104   /-- The filtration is exhaustive. -/
105   exhaustive : ∀ x : R, ∃ n, x ∈ F n
106 
107 /-- The previous filtration piece `Fₙ₋₁`, with `F₋₁ = 0`. -/
108 def RingFiltration.below {k R : Type u} [CommRing k] [Ring R] [Algebra k R]
109     (φ : RingFiltration k R) : ℕ → Submodule k R
110   | 0 => ⊥
111   | n + 1 => φ.F n
112 
113 /-- Principal-symbol data identifying `gr R = ⨁ₙ Fₙ/Fₙ₋₁` with the commutative
114 graded algebra `G = ⨁ₙ 𝒜ₙ`: surjective `k`-linear maps `grₙ : Fₙ → 𝒜ₙ` with
115 kernel `Fₙ₋₁` that multiply symbols. -/
116 structure GradedSymbol {k R G : Type u} [CommRing k] [Ring R] [Algebra k R] [CommRing G]
117     [Algebra k G] (φ : RingFiltration k R) (𝒜 : ℕ → Submodule k G) where
118   /-- The degree-`n` symbol map `Fₙ → 𝒜ₙ`. -/
119   gr : ∀ n, φ.F n →ₗ[k] 𝒜 n
120   /-- Every homogeneous element of degree `n` is a symbol. -/
121   gr_surjective : ∀ n, Function.Surjective (gr n)
122   /-- The kernel of `grₙ` is `Fₙ₋₁`. -/
123   gr_eq_zero_iff : ∀ n (x : φ.F n), gr n x = 0 ↔ (x : R) ∈ φ.below n
124   /-- Symbols multiply: `gr_{i+j}(x y) = grᵢ(x) grⱼ(y)`. -/
125   gr_mul : ∀ i j (x : φ.F i) (y : φ.F j),
126     ((gr (i + j) ⟨x * y, φ.mul_mem x.2 y.2⟩ : 𝒜 (i + j)) : G) =
127       (gr i x : G) * (gr j y : G)
128 
129 /-- `R` has an exhaustive filtration whose associated graded ring is a
130 commutative Noetherian ring of Krull dimension at most `m`. -/
131 def HasCommNoetherianGr (k R : Type u) [CommRing k] [Ring R] [Algebra k R] (m : ℕ) :
132     Prop :=
133   ∃ (φ : RingFiltration k R) (G : Type u) (_ : CommRing G) (_ : Algebra k G)
134     (𝒜 : ℕ → Submodule k G) (_ : GradedAlgebra 𝒜),
135     Nonempty (GradedSymbol φ 𝒜) ∧ IsNoetherianRing G ∧ ringKrullDim G ≤ m
136 
137 /-- `R` has an exhaustive filtration whose associated graded ring is a
138 commutative Noetherian **domain** of finite Krull dimension. -/
139 def HasCommNoetherianDomainGr (k R : Type u) [CommRing k] [Ring R] [Algebra k R] :
140     Prop :=
141   ∃ (φ : RingFiltration k R) (G : Type u) (_ : CommRing G) (_ : Algebra k G)
142     (𝒜 : ℕ → Submodule k G) (_ : GradedAlgebra 𝒜),
143     Nonempty (GradedSymbol φ 𝒜) ∧ IsNoetherianRing G ∧ IsDomain G ∧ ringKrullDim G < ⊤
144 
145 /-- **Theorem A, both forms (filtered rings).** Let `k` be an infinite field and
146 `R` a simple `k`-algebra with an exhaustive filtration such that `gr R` is
147 commutative Noetherian of Krull dimension at most `m`. Then Theorem A holds for
148 `R` (right form `1 = d U + (B + H a) d V`) and for `Rᵐᵒᵖ` (left form
149 `1 = U d + V d (B + a H)` of `R`). -/
150 def FilteredTheoremAStatement : Prop :=
151   ∀ (k : Type u) [Field k] [Infinite k] (R : Type u) [Ring R] [Algebra k R]
152     [IsSimpleRing R] (m : ℕ), HasCommNoetherianGr k R m →
153     FixedAnchorProperty R ∧ FixedAnchorProperty Rᵐᵒᵖ
154 
155 /-- **Theorem A', both sides (filtered rings).** Under the hypotheses of
156 `FilteredTheoremAStatement`, Theorem A' holds for `R` and for `Rᵐᵒᵖ`. -/
157 def FilteredFixedIdealStatement : Prop :=
158   ∀ (k : Type u) [Field k] [Infinite k] (R : Type u) [Ring R] [Algebra k R]
159     [IsSimpleRing R] (m : ℕ), HasCommNoetherianGr k R m →
160     FixedIdealProperty R ∧ FixedIdealProperty Rᵐᵒᵖ
161 
162 /-- **Corollary B, both sides (filtered rings).** Under the hypotheses of
163 `FilteredTheoremAStatement`, the dimension part of Corollary B holds for `R`
164 and for `Rᵐᵒᵖ`. -/
165 def FilteredCorollaryBStatement : Prop :=
166   ∀ (k : Type u) [Field k] [Infinite k] (R : Type u) [Ring R] [Algebra k R]
167     [IsSimpleRing R] (m : ℕ), HasCommNoetherianGr k R m →
168     CorollaryBProperty R ∧ CorollaryBProperty Rᵐᵒᵖ
169 
170 /-- **Corollary C.1, both sides (filtered rings).** Under the hypotheses of
171 `FilteredTheoremAStatement`, the prescribed-generator and stable forms of C.1
172 hold for `R` and for `Rᵐᵒᵖ`. -/
173 def FilteredPrescribedCyclicStatement : Prop :=
174   ∀ (k : Type u) [Field k] [Infinite k] (R : Type u) [Ring R] [Algebra k R]
175     [IsSimpleRing R] (m : ℕ), HasCommNoetherianGr k R m →
176     (PrescribedCyclicProperty.{u, v} R ∧ StablyCyclicProperty.{u, v} R) ∧
177       (PrescribedCyclicProperty.{u, v} Rᵐᵒᵖ ∧ StablyCyclicProperty.{u, v} Rᵐᵒᵖ)
178 
179 /-- **Corollary C.2, both sides (filtered rings).** Under the hypotheses of
180 `FilteredTheoremAStatement`, the two-generation statements of C.2 hold for `R`
181 and for `Rᵐᵒᵖ`. -/
182 def FilteredTwoGeneratedStatement : Prop :=
183   ∀ (k : Type u) [Field k] [Infinite k] (R : Type u) [Ring R] [Algebra k R]
184     [IsSimpleRing R] (m : ℕ), HasCommNoetherianGr k R m →
185     TwoGeneratedProperty R ∧ TwoGeneratedProperty Rᵐᵒᵖ
186 
187 /-- **Corollary C.3, both sides (filtered rings).** Under the hypotheses of
188 `FilteredTheoremAStatement`, the generator bounds of C.3 hold for `R` and for
189 `Rᵐᵒᵖ`. -/
190 def FilteredGeneratorBoundsStatement : Prop :=
191   ∀ (k : Type u) [Field k] [Infinite k] (R : Type u) [Ring R] [Algebra k R]
192     [IsSimpleRing R] (m : ℕ), HasCommNoetherianGr k R m →
193     GeneratorBoundsProperty.{u, v} R ∧ GeneratorBoundsProperty.{u, v} Rᵐᵒᵖ

195 /-- **Proposition (filtered rings), hypotheses.** Let `k` be an infinite field
196 and `R` a simple `k`-algebra with an exhaustive filtration such that `gr R` is
197 commutative Noetherian of Krull dimension at most `m`. Then `R` and `Rᵐᵒᵖ`
198 satisfy (N), (Z) and `Kdim ≤ m`. (`Rᵐᵒᵖ` is simple by Mathlib's instance.) -/
199 def FilteredHypothesesStatement : Prop :=
200   ∀ (k : Type u) [Field k] [Infinite k] (R : Type u) [Ring R] [Algebra k R]
201     [IsSimpleRing R] (m : ℕ), HasCommNoetherianGr k R m →
202     (IsNoetherian Rᵐᵒᵖ R ∧ Infinite (Subring.center R) ∧ DevDimLE R R m) ∧
203     (IsNoetherian Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ ∧ Infinite (Subring.center Rᵐᵒᵖ) ∧ DevDimLE Rᵐᵒᵖ Rᵐᵒᵖ m)
204 
205 /-- **Simple quantizations.** If `R` is a simple algebra over an infinite field
206 `k` with an exhaustive filtration whose associated graded ring is a commutative
207 Noetherian domain of finite Krull dimension (for instance `gr R ≅ ℂ[X]` for an
208 irreducible affine variety `X`), then for every nonzero `d` and `a` and every
209 `B` there are `H, U, V` with `1 = d U + (B + H a) d V`, and the same holds in
210 `Rᵐᵒᵖ`, i.e. `1 = U d + V d (B + a H)` in `R`. -/
211 def FilteredDomainStatement : Prop :=
212   ∀ (k : Type u) [Field k] [Infinite k] (R : Type u) [Ring R] [Algebra k R]
213     [IsSimpleRing R], HasCommNoetherianDomainGr k R →
214     (∀ d a B : R, d ≠ 0 → a ≠ 0 → ∃ H U V : R, 1 = d * U + (B + H * a) * d * V) ∧
215     (∀ d a B : R, d ≠ 0 → a ≠ 0 → ∃ H U V : R, 1 = U * d + V * d * (B + a * H))

219 /-- The challenge filtration as an AlgebraicAnalysis filtration. -/
220 def toAA {k R : Type u} [CommRing k] [Ring R] [Algebra k R] (φ : RingFiltration k R) :
221     AlgebraicAnalysis.RingFiltration k R where
222   F := φ.F
223   mono := φ.mono
224   one_mem := φ.one_mem
225   mul_le _ _ := Submodule.mul_le.mpr fun _ hx _ hy => φ.mul_mem hx hy
226   exhaustive := eq_top_iff.mpr fun x _ =>
227     let ⟨n, hn⟩ := φ.exhaustive x; Submodule.mem_iSup_of_mem n hn
228 
229 theorem below_toAA {k R : Type u} [CommRing k] [Ring R] [Algebra k R] (φ : RingFiltration k R)
230     (n : ℕ) : (toAA φ).below n = φ.below n := by
231   cases n <;> rfl
232 
233 /-- The challenge symbol as an AlgebraicAnalysis symbol of `toAA φ`. -/
234 def symbolToAA {k R G : Type u} [CommRing k] [Ring R] [Algebra k R] [CommRing G] [Algebra k G]
235     {φ : RingFiltration k R} {𝒜 : ℕ → Submodule k G} (S : GradedSymbol φ 𝒜) :
236     AlgebraicAnalysis.RingFiltration.GradedSymbol (toAA φ) 𝒜 where
237   gr := S.gr
238   gr_surjective := S.gr_surjective
239   gr_eq_zero_iff n x := by rw [below_toAA]; exact S.gr_eq_zero_iff n x
240   gr_mul := S.gr_mul

242 /-- Both sides of a challenge filtered ring satisfy (N), (Z), the deviation
243 bound `Kdim ≤ m` and (H_omega), from `GeneralizedStafford.filtered_hypotheses`. -/
244 theorem filtered_side {k R : Type u} [Field k] [Infinite k] [Ring R] [Algebra k R]
245     [IsSimpleRing R] {m : ℕ} (h : HasCommNoetherianGr k R m) :
246     (IsNoetherian Rᵐᵒᵖ R ∧ Infinite (Subring.center R) ∧ DevDimLE R R m ∧ HOmega R) ∧
247       (IsNoetherian Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ ∧ Infinite (Subring.center Rᵐᵒᵖ) ∧
248         DevDimLE Rᵐᵒᵖ Rᵐᵒᵖ m ∧ HOmega Rᵐᵒᵖ) := by
249   obtain ⟨φ, G, _, _, 𝒜, _, ⟨S⟩, hN, hdim⟩ := h
250   have := hN
251   obtain ⟨⟨hN1, hZ1, hD1, hO1⟩, ⟨hN2, hZ2, hD2, hO2⟩⟩ :=
252     GeneralizedStafford.filtered_hypotheses (symbolToAA S) hdim
253   exact ⟨⟨hN1, hZ1, (devDimLE_iff m).mpr hD1, hOmega_iff.mpr hO1⟩,
254     ⟨hN2, hZ2, (devDimLE_iff m).mpr hD2, hOmega_iff.mpr hO2⟩⟩

280 /-- The stable form of C.1 from (H_omega): under (H_all) a cyclic submodule `uR`
281 killed by a regular element has finite deviation, and the library lemma applies. -/
282 theorem stablyCyclic_of_omega {R : Type u} [Ring R] [IsSimpleRing R]
283     [IsNoetherian Rᵐᵒᵖ R] [Infinite (Subring.center R)] (homega : HOmega R) :
284     StablyCyclicProperty.{u, v} R := by
285   intro M _ _ c u hc hu t ht
286   have hall : GeneralizedStafford.PrincipalQuotientsFinite R :=
287     hAll_iff.mp (hAll_of_omega homega)
288   have hc' : ∃ s ∈ nonZeroDivisors R, op s • c = 0 := by
289     obtain ⟨s, hs, h⟩ := hc
290     exact ⟨s, (isRegular_iff s).mp hs, h⟩
291   have hu' : ∃ s ∈ nonZeroDivisors R, op s • u = 0 := by
292     obtain ⟨s, hs, h⟩ := hu
293     exact ⟨s, (isRegular_iff s).mp hs, h⟩
294   obtain ⟨N, hN⟩ := GeneralizedStafford.devLE_span_of_torsion hall hu'
295   refine withCentre R _ fun k _ _ _ => ?_
296   exact GeneralizedStafford.span_pair_eq_span_affine (k := k) c u hc' hu' ⟨N, hN⟩ ht

395 /-- A Krull dimension below `⊤` is bounded by a natural number. -/

397     ∃ m : ℕ, ringKrullDim G ≤ m := by

403 /-- Filtered rings with a domain as associated graded ring: Theorem A for every
404 nonzero anchor, on both sides. The domain structure makes `R` a domain
405 (AlgebraicAnalysis `GradedSymbol.noZeroDivisors`), so nonzero elements are regular. -/
406 theorem filteredDomain_proof : FilteredDomainStatement.{u} := by
407   intro k _ _ R _ _ _ h
408   obtain ⟨φ, G, cG, aG, 𝒜, gA, ⟨S⟩, hN, hD, hlt⟩ := h
409   obtain ⟨m, hm⟩ := exists_nat_le_of_lt_top hlt
410   have hA : HasCommNoetherianGr k R m := ⟨φ, G, cG, aG, 𝒜, gA, ⟨S⟩, hN, hm⟩
411   obtain ⟨⟨hN1, hZ1, -, hO1⟩, ⟨hN2, hZ2, -, hO2⟩⟩ := filtered_side hA
412   have := hN1
413   have := hZ1
414   have := hN2
415   have := hZ2
416   have : NoZeroDivisors R := (symbolToAA S).noZeroDivisors
417   have hright := fixedAnchorProperty_of_omega hO1
418   have hleft := fixedAnchorProperty_of_omega hO2
419   refine ⟨fun d a B hd ha => hright d a B (isRegular_of_ne_zero' hd) ha, fun d a B hd ha => ?_⟩
420   obtain ⟨H, U, V, hUV⟩ := hleft (op d) (op a) (op B)
421     (isRegular_of_ne_zero' ((op_ne_zero_iff d).mpr hd)) ((op_ne_zero_iff a).mpr ha)
