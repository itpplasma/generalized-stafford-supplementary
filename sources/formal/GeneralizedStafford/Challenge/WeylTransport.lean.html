11 copy `Challenge.lean`) are restated verbatim below, in the same namespaces and under

14 restates it under its challenge name.

17 registered challenge. Its proofs are the AlgebraicAnalysis Weyl facts of

19 the same presentation, with the same relation body, so they agree definitionally.

24 /-! ## Weyl algebra: two-generated one-sided ideals
25 
26 The Weyl algebra `A_n(k)` is the presented algebra of the registered Stafford 3.8
27 challenge. The definitions `Stafford.freeWeylRelation`, `Stafford.FreeWeyl`,
28 `Stafford38Challenge.PhaseVar`, `Stafford38Challenge.WeylAlg`,
29 `Stafford.Reduction.Stafford38`, `Stafford38Challenge.UniversalStatement` and
30 `Stafford38.TorsionCyclicity.IsRightTorsion` are copied verbatim from
31 `stafford38-formal/Stafford38/ChallengeDefinitions.lean`; only the `import`s differ.
32 
33 The left summand of `PhaseVar n` indexes `x_i`, the right summand `∂_i`; the
34 relations `ι i ι j − ι j ι i = J i j` with `Matrix.J = [[0, −1], [1, 0]]` give
35 `[∂_i, x_j] = δ_ij`, all other generators commuting. -/
36

39 variable {k ι : Type*} [Field k]
40 
41 /-- The commutator relations presenting a Weyl-type algebra. -/
42 def freeWeylRelation (omega : Matrix ι ι k)
43     (a b : FreeAlgebra k ι) : Prop :=
44   ∃ i j,
45     a = FreeAlgebra.ι k i * FreeAlgebra.ι k j -
46       FreeAlgebra.ι k j * FreeAlgebra.ι k i ∧
47     b = algebraMap k (FreeAlgebra k ι) (omega i j)
48 
49 /-- The free algebra modulo the commutator relations encoded by `omega`. -/
50 abbrev FreeWeyl (k : Type*) [Field k] (ι : Type*)
51     (omega : Matrix ι ι k) := RingQuot (freeWeylRelation omega)
52 
53 end Stafford

88 /-- Every element of a right module is killed by some nonzero right scalar. -/
89 def IsRightTorsion {A : Type u} {M : Type v} [Ring A]
90     [AddCommGroup M] [Module Aᵐᵒᵖ M] : Prop :=
91   ∀ m : M, ∃ d : A, d ≠ 0 ∧ (MulOpposite.op d : Aᵐᵒᵖ) • m = 0

93 end Stafford38.TorsionCyclicity

101 /-- Every element of a left module is killed by some nonzero left scalar. -/
102 def IsLeftTorsion {A : Type u} {M : Type v} [Ring A]
103     [AddCommGroup M] [Module A M] : Prop :=
104   ∀ m : M, ∃ d : A, d ≠ 0 ∧ d • m = 0

106 /-! ### Stafford's Conjecture 3.8 and the affine form (Theorem A, Corollary B) -/
107 
108 /-- **Stafford's Conjecture 3.8** (Stafford 1978, p. 438): for `d ≠ 0` in
109 `Aₙ(k)` there are `F, R, S` with `1 = d R + F d S`. This is literally the
110 registered `Stafford38Challenge.UniversalStatement`; here it is the case
111 `a = 1`, `B = 0` of the affine form (`F = H`). -/
112 def Conjecture38Statement : Prop := UniversalStatement.{u}
113 
114 /-- **Affine form (Theorem A for `Aₙ(k)`).** For every nonzero `d`, every
115 nonzero `a` and every `B` in `Aₙ(k)` there are `H, U, V` with
116 `1 = d U + (B + H a) d V`. -/
117 def AffineFormStatement : Prop :=
118   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ) (d a B : WeylAlg k n),
119     d ≠ 0 → a ≠ 0 → ∃ H U V : WeylAlg k n, 1 = d * U + (B + H * a) * d * V
120 
121 /-- **Fixed-ideal form (Theorem A' for `Aₙ(k)`).** For every nonzero right
122 ideal `I` of `Aₙ(k)`, every `x` and every nonzero `t` there is `H` with
123 `I + (x + H t) Aₙ = Aₙ`. -/
124 def FixedIdealStatement : Prop :=
125   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
126     (I : Submodule (WeylAlg k n)ᵐᵒᵖ (WeylAlg k n)), I ≠ ⊥ →
127     ∀ x t : WeylAlg k n, t ≠ 0 →
128       ∃ H : WeylAlg k n, I ⊔ Submodule.span (WeylAlg k n)ᵐᵒᵖ {x + H * t} = ⊤
129 
130 /-! ### Torsion modules are cyclic (Corollary C.1), right and left -/
131 
132 /-- **Torsion cyclicity, right modules.** Every finitely generated torsion
133 right `Aₙ(k)`-module is cyclic. Same proposition as the registered
134 `Stafford38CorollaryChallenge.torsionCyclicStatement` of `stafford38-formal`. -/
135 def TorsionCyclicRightStatement : Prop :=
136   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
137     (M : Type v) [AddCommGroup M] [Module (WeylAlg k n)ᵐᵒᵖ M]
138     [Module.Finite (WeylAlg k n)ᵐᵒᵖ M],
139     Stafford38.TorsionCyclicity.IsRightTorsion (A := WeylAlg k n) (M := M) →
140       ∃ z : M, Submodule.span (WeylAlg k n)ᵐᵒᵖ ({z} : Set M) = ⊤
141 
142 /-- **Torsion cyclicity, left modules.** Every finitely generated torsion
143 left `Aₙ(k)`-module is cyclic. -/
144 def TorsionCyclicLeftStatement : Prop :=
145   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
146     (M : Type v) [AddCommGroup M] [Module (WeylAlg k n) M]
147     [Module.Finite (WeylAlg k n) M],
148     IsLeftTorsion (A := WeylAlg k n) (M := M) →
149       ∃ z : M, Submodule.span (WeylAlg k n) ({z} : Set M) = ⊤
150 
151 /-- **Prescribed cyclic generator, right modules (C.1).** In a finitely
152 generated torsion right `Aₙ(k)`-module `M`, for every `x ∈ M` and every
153 nonzero `t` there is `h ∈ M` with `M = (x + h t) Aₙ`. -/
154 def PrescribedCyclicRightStatement : Prop :=
155   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
156     (M : Type v) [AddCommGroup M] [Module (WeylAlg k n)ᵐᵒᵖ M]
157     [Module.Finite (WeylAlg k n)ᵐᵒᵖ M],
158     Stafford38.TorsionCyclicity.IsRightTorsion (A := WeylAlg k n) (M := M) →
159     ∀ (x : M) (t : WeylAlg k n), t ≠ 0 →
160       ∃ h : M, Submodule.span (WeylAlg k n)ᵐᵒᵖ ({x + op t • h} : Set M) = ⊤
161 
162 /-- **Prescribed cyclic generator, left modules (C.1 via `Aₙ ≅ Aₙᵒᵖ`).** In a
163 finitely generated torsion left `Aₙ(k)`-module `M`, for every `x ∈ M` and
164 every nonzero `t` there is `h ∈ M` with `M = Aₙ (x + t h)`. -/
165 def PrescribedCyclicLeftStatement : Prop :=
166   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
167     (M : Type v) [AddCommGroup M] [Module (WeylAlg k n) M]
168     [Module.Finite (WeylAlg k n) M],
169     IsLeftTorsion (A := WeylAlg k n) (M := M) →
170     ∀ (x : M) (t : WeylAlg k n), t ≠ 0 →
171       ∃ h : M, Submodule.span (WeylAlg k n) ({x + t • h} : Set M) = ⊤
172 
173 /-! ### One-sided ideals are two-generated (Corollary C.2), right and left -/
174 
175 /-- **Prescribed first generator, right ideals (C.2).** If `a` is a nonzero
176 element of a right ideal `I` of `Aₙ(k)`, then for every `y ∈ I` and every
177 nonzero `t` there is `h ∈ I` with `I = a Aₙ + (y + h t) Aₙ`. (Since `Aₙ(k)` is a
178 domain, any nonzero `a ∈ I` may serve as the first generator.) -/
179 def PrescribedTwoGeneratedRightStatement : Prop :=
180   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
181     (I : Submodule (WeylAlg k n)ᵐᵒᵖ (WeylAlg k n)) (a : WeylAlg k n),
182     a ∈ I → a ≠ 0 → ∀ y ∈ I, ∀ t : WeylAlg k n, t ≠ 0 →
183       ∃ h ∈ I, I = Submodule.span (WeylAlg k n)ᵐᵒᵖ {a} ⊔
184         Submodule.span (WeylAlg k n)ᵐᵒᵖ {y + h * t}
185 
186 /-- **Prescribed first generator, left ideals.** If `a` is a nonzero element
187 of a left ideal `I` of `Aₙ(k)`, then for every `y ∈ I` and every nonzero `t`
188 there is `h ∈ I` with `I = Aₙ a + Aₙ (y + t h)`. -/
189 def PrescribedTwoGeneratedLeftStatement : Prop :=
190   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
191     (I : Submodule (WeylAlg k n) (WeylAlg k n)) (a : WeylAlg k n),
192     a ∈ I → a ≠ 0 → ∀ y ∈ I, ∀ t : WeylAlg k n, t ≠ 0 →
193       ∃ h ∈ I, I = Submodule.span (WeylAlg k n) {a} ⊔
194         Submodule.span (WeylAlg k n) {y + t * h}
195 
196 end GeneralizedStaffordWeylChallenge

200 /-- **Two generators, right ideals of the Weyl algebra.** For every field `k` of
201 characteristic zero and every `n`, every right ideal `I` of `Aₙ(k)` is generated
202 by two elements: `I = a Aₙ(k) + b Aₙ(k)`. -/
203 def WeylTwoGeneratedRightStatement : Prop :=
204   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
205     (I : Submodule (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ (Stafford38Challenge.WeylAlg k n)),
206     ∃ a b : Stafford38Challenge.WeylAlg k n,
207       I = Submodule.span (Stafford38Challenge.WeylAlg k n)ᵐᵒᵖ {a, b}
208 
209 /-- **Two generators, left ideals of the Weyl algebra.** For every field `k` of
210 characteristic zero and every `n`, every left ideal `I` of `Aₙ(k)` is generated
211 by two elements: `I = Aₙ(k) a + Aₙ(k) b`. -/
212 def WeylTwoGeneratedLeftStatement : Prop :=
213   ∀ (k : Type u) [Field k] [CharZero k] (n : ℕ)
214     (I : Submodule (Stafford38Challenge.WeylAlg k n) (Stafford38Challenge.WeylAlg k n)),
215     ∃ a b : Stafford38Challenge.WeylAlg k n,
216       I = Submodule.span (Stafford38Challenge.WeylAlg k n) {a, b}
