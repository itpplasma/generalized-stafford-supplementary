13 
14 The definitions and statement definitions of `GeneralizedStaffordChallenge.lean`
15 (identical copy `Challenge.lean`) are restated verbatim below, in the same
16 namespace and under the same names, so that the solution never imports the
17 challenge. The compared `theorem`s of the challenge are not restated: each
18 challenge theorem `<name>` is proved here as `<name>_proof`, and
19 `GeneralizedStaffordSolution.lean` restates it under its challenge name.
20 
21 The bridge lemmas translate the challenge conventions into those of the
22 library: `Submodule.span Rᵐᵒᵖ {d}` is `principalRightIdeal d`
23 (`span_singleton_eq`), `IsRegularElement d` is `d ∈ R⁰` (`isRegular_iff`), the

38 /-! ## Regular elements, deviation and Goldie torsion -/
39 
40 section Definitions
41 
42 variable (R : Type u) [Ring R]
43 
44 /-- An element `d` of `R` is **regular** if it is a two-sided non-zero-divisor:
45 `d x = 0` implies `x = 0`, and `x d = 0` implies `x = 0` .
46 This is definitionally Mathlib's `d ∈ R⁰`. -/
47 def IsRegularElement {R : Type u} [Ring R] (d : R) : Prop :=
48   (∀ x : R, d * x = 0 → x = 0) ∧ (∀ x : R, x * d = 0 → x = 0)
49 
50 /-- Finite Gabriel--Rentschler deviation of a partial order. `PosetDevLE 0 α`
51 says that every descending chain in `α` stabilizes; `PosetDevLE (n + 1) α`
52 says that for every descending chain `c₀ ≥ c₁ ≥ ⋯` all but finitely many
53 closed intervals `[cᵢ₊₁, cᵢ]` satisfy `PosetDevLE n`, i.e. all but finitely
54 many factors have deviation `< n + 1` . -/
55 def PosetDevLE : (n : ℕ) → (α : Type v) → [PartialOrder α] → Prop
56   | 0, α, _ => ∀ c : ℕ → α, Antitone c → ∃ N, ∀ i, N ≤ i → c i = c N
57   | n + 1, α, _ => ∀ c : ℕ → α, Antitone c →
58       ∃ N, ∀ i, N ≤ i → PosetDevLE n (Set.Icc (c (i + 1)) (c i))
59 
60 variable (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M]
61 
62 /-- `DevDimLE R M n`: `Kdim M ≤ n` (written `Kdim`;
63 Gabriel--Rentschler deviation, not classical Krull dimension). The right
64 module `M` has right deviation dimension at most `n` , defined
65 on the lattice of right submodules of `M`. -/
66 def DevDimLE (n : ℕ) : Prop :=
67   PosetDevLE n (Submodule Rᵐᵒᵖ M)
68 
69 /-- `Kdim M` is a finite integer (possibly `-1`, which happens exactly when
70 `M = 0`). -/
71 def HasFiniteDevDim : Prop :=
72   ∃ n : ℕ, DevDimLE R M n
73 
74 /-- `DevDimLEOmega R M`: `Kdim M ≤ ω`, that is, for every descending chain of
75 right submodules `M₀ ≥ M₁ ≥ ⋯` all but finitely many factors `Mᵢ/Mᵢ₊₁`
76 (the intervals `[Mᵢ₊₁, Mᵢ]`) have finite deviation. -/
77 def DevDimLEOmega : Prop :=
78   ∀ c : ℕ → Submodule Rᵐᵒᵖ M, Antitone c → ∃ N, ∀ i, N ≤ i →
79     ∃ n : ℕ, PosetDevLE n (Set.Icc (c (i + 1)) (c i))
80 
81 /-- `M` is **Goldie-torsion**: every `m ∈ M` satisfies
82 `m s = 0` for some regular `s ∈ R`. -/
83 def IsGoldieTorsion : Prop :=
84   ∀ m : M, ∃ s : R, IsRegularElement s ∧ op s • m = 0
85 
86 /-- The Goldie-torsion part `t(M)` of `M`: the right submodule spanned by the
87 elements `m` with `m s = 0` for some regular `s`. Under (S) and (N) these
88 elements already form a submodule (Goldie's theorem); the span is used so
89 that no closure proof enters the statement. -/
90 def goldieTorsionPart : Submodule Rᵐᵒᵖ M :=
91   Submodule.span Rᵐᵒᵖ {m : M | ∃ s : R, IsRegularElement s ∧ op s • m = 0}
92 
93 /-- `UDimGE R M n`: the uniform (Goldie) dimension of `M` is at least `n`, that
94 is, `M` contains `n` nonzero right submodules whose sum is direct. -/
95 def UDimGE (n : ℕ) : Prop :=
96   ∃ N : Fin n → Submodule Rᵐᵒᵖ M, iSupIndep N ∧ ∀ i, N i ≠ ⊥
97 
98 /-- `UDimLE R M n`: the uniform (Goldie) dimension of `M` is at most `n`, that
99 is, `M` contains no direct sum of `n + 1` nonzero right submodules. -/
100 def UDimLE (n : ℕ) : Prop :=
101   ¬ UDimGE R M (n + 1)
102 
103 /-- `SpanRankLE R M m`: the right module `M` is generated by at most `m`
104 elements, i.e. `μ_R(M) ≤ m`. -/
105 def SpanRankLE (m : ℕ) : Prop :=
106   ∃ s : Finset M, s.card ≤ m ∧ Submodule.span Rᵐᵒᵖ (s : Set M) = ⊤
107 
108 /-- Hypothesis (H_all): `δ(d) = Kdim(R/dR)` is a finite integer for every
109 regular `d`. -/
110 def HAll : Prop :=
111   ∀ d : R, IsRegularElement d → HasFiniteDevDim R (R ⧸ Submodule.span Rᵐᵒᵖ {d})
112 
113 /-- Hypothesis (H_omega): `Kdim R ≤ ω` for the right regular module `R_R`. -/
114 def HOmega : Prop :=
115   DevDimLEOmega R R
116 
117 end Definitions

119 /-! ## Theorem A' and Theorem A -/
120 
121 /-- **Theorem A' (fixed ideal).** Let `R` be a simple (S), right Noetherian (N)
122 ring with infinite centre (Z). Let `I` be a right ideal containing a regular
123 element such that `Kdim(R/I)` is a finite integer. Then for
124 every `x ∈ R` and every nonzero `t ∈ R` there is `H ∈ R` with
125 `I + (x + H t)R = R`. -/
126 def FixedIdealStatement : Prop :=
127   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
128     [Infinite (Subring.center R)] (I : Submodule Rᵐᵒᵖ R),
129     (∃ s ∈ I, IsRegularElement s) → HasFiniteDevDim R (R ⧸ I) →
130     ∀ x t : R, t ≠ 0 →
131       ∃ H : R, I ⊔ Submodule.span Rᵐᵒᵖ {x + H * t} = ⊤
132 
133 /-- **Theorem A (fixed anchor).** Let `R` satisfy (S), (N), (Z) and let `d` be a
134 regular element with `δ(d) = Kdim(R/dR)` a finite integer. For every nonzero
135 `a ∈ R` (possibly a zero-divisor) and every `B ∈ R` there are `H, U, V ∈ R`
136 with `1 = d U + (B + H a) d V`. -/
137 def FixedAnchorStatement : Prop :=
138   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
139     [Infinite (Subring.center R)] (d : R),
140     IsRegularElement d → HasFiniteDevDim R (R ⧸ Submodule.span Rᵐᵒᵖ {d}) →
141     ∀ a B : R, a ≠ 0 →
142       ∃ H U V : R, 1 = d * U + (B + H * a) * d * V
143 
144 /-! ## Corollary B -/
145 
146 /-- **Corollary B, dimension part.** Let `R` be right Noetherian
147 with `Kdim R ≤ ω` (H_omega). Then `δ(d) = Kdim(R/dR)` is a finite integer for
148 every regular `d` (so (H_all) holds), and `Kdim(R/I)` is a finite integer for
149 every right ideal `I` containing a regular element. The ordinal refinement
150 `δ(d) < Kdim R` is deliberately not stated. -/
151 def CorollaryBDimensionStatement : Prop :=
152   ∀ (R : Type u) [Ring R] [IsNoetherian Rᵐᵒᵖ R], HOmega R →
153     HAll R ∧ ∀ I : Submodule Rᵐᵒᵖ R, (∃ s ∈ I, IsRegularElement s) →
154       HasFiniteDevDim R (R ⧸ I)
155 
156 /-- **Corollary B, fixed anchor.** Let `R` satisfy (S), (N), (Z) and
157 `Kdim R ≤ ω`. For every regular `d`, every nonzero `a` and every `B` there are
158 `H, U, V` with `1 = d U + (B + H a) d V`. -/
159 def CorollaryBFixedAnchorStatement : Prop :=
160   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
161     [Infinite (Subring.center R)], HOmega R →
162     ∀ d a B : R, IsRegularElement d → a ≠ 0 →
163       ∃ H U V : R, 1 = d * U + (B + H * a) * d * V
164 
165 /-- **Corollary B, fixed ideal.** Let `R` satisfy (S), (N), (Z) and
166 `Kdim R ≤ ω`. For every right ideal `I` containing a regular element, every
167 `x` and every nonzero `t` there is `H` with `I + (x + H t)R = R`. -/
168 def CorollaryBFixedIdealStatement : Prop :=
169   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
170     [Infinite (Subring.center R)], HOmega R →
171     ∀ I : Submodule Rᵐᵒᵖ R, (∃ s ∈ I, IsRegularElement s) →
172     ∀ x t : R, t ≠ 0 →
173       ∃ H : R, I ⊔ Submodule.span Rᵐᵒᵖ {x + H * t} = ⊤

211 /-- **Corollary C.1, stable cyclicity.** Let `R` satisfy (S), (N), (Z), let `c, u`
212 be Goldie-torsion elements of a right module `M` (each is killed by a regular
213 element) with `Kdim(uR)` a finite integer.
214 For every nonzero `t ∈ R` there is `H ∈ R` with `cR + uR = (c + u H t)R`. -/
215 def CorollaryC1StableStatement : Prop :=
216   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]

225 /-! ## Corollary C.2
226 
227 The condition `aQ = IQ` (with `Q` the classical right quotient ring)
228 is stated elementwise: `a ∈ I` and every `y ∈ I` has a regular `s` with
229 `y s ∈ aR`. Under (S), (N) these are equivalent: `aQ ⊆ IQ` because `a ∈ I`,
230 and `y ∈ aQ` iff `y = a r s⁻¹` iff `y s = a r` for some `r` and regular `s`. -/
231 
232 /-- **Existence of a full-rank element** (first step of Corollary C.2). Let `R`
233 satisfy (S) and (N). Every right ideal `I` contains an `a` with `aQ = IQ`, that
234 is, every `y ∈ I` has a regular `s` with `y s ∈ aR`. Corollary C.2 is stated
235 under (S), (N), (Z), (H_all); this step needs only (S) and (N) (Goldie's
236 theorem), so the statement here has fewer hypotheses and is still true. -/
237 def CorollaryC2FullRankStatement : Prop :=
238   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
239     (I : Submodule Rᵐᵒᵖ R),
240     ∃ a ∈ I, ∀ y ∈ I,
241       ∃ s : R, IsRegularElement s ∧ y * s ∈ Submodule.span Rᵐᵒᵖ {a}
242 
243 /-- **Corollary C.2, prescribed form.** Let `R` satisfy (S), (N), (Z) and
244 (H_all), and let `I` be a right ideal. If `a ∈ I` satisfies `aQ = IQ` (every
245 `y ∈ I` has a regular `s` with `y s ∈ aR`), then for every `y ∈ I` and every
246 nonzero `t ∈ R` there is `h ∈ I` with `I = aR + (y + h t)R`. -/
247 def CorollaryC2PrescribedStatement : Prop :=
248   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
249     [Infinite (Subring.center R)], HAll R →
250     ∀ (I : Submodule Rᵐᵒᵖ R) (a : R), a ∈ I →
251     (∀ y ∈ I, ∃ s : R, IsRegularElement s ∧ y * s ∈ Submodule.span Rᵐᵒᵖ {a}) →
252     ∀ y ∈ I, ∀ t : R, t ≠ 0 →
253       ∃ h ∈ I, I = Submodule.span Rᵐᵒᵖ {a} ⊔ Submodule.span Rᵐᵒᵖ {y + h * t}
254 
255 /-- **Corollary C.2 (two generators).** Let `R` satisfy (S), (N), (Z) and
256 (H_all). Every right ideal `I` of `R` is generated by two elements:
257 `I = aR + bR` for some `a, b ∈ R`. -/
258 def CorollaryC2Statement : Prop :=
259   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
260     [Infinite (Subring.center R)], HAll R →
261     ∀ I : Submodule Rᵐᵒᵖ R, ∃ a b : R, I = Submodule.span Rᵐᵒᵖ {a, b}

263 /-! ## Corollary C.3
264 
265 The classical right quotient ring is `Q = M_s(D)`, and the normalized Goldie rank is
266 `ρ(M) = length_Q(M ⊗_R Q) / s` for finitely generated `M`. Under (S) and (N)
267 (Goldie's theorem) the two numbers are uniform dimensions:
268 
269 * `s = udim(R_R)`, since `R_R` is essential in `Q_R` and `Q_Q` is a direct
270   sum of `s` simple right ideals;
271 * `length_Q(M ⊗_R Q) = udim(M / t(M))`, where `t(M)` is the Goldie-torsion
272   part: `M ⊗_R Q ≅ (M/t(M)) ⊗_R Q` by flatness of `Q`, the torsion-free module
273   `N = M/t(M)` is an essential `R`-submodule of `N ⊗_R Q`, and for a
274   `Q`-module the uniform dimension over `R` equals the `Q`-length.
275 
276 Hence `ρ(M) = udim(M/t(M)) / udim(R_R)`. Writing `ℓ = udim(M/t(M))` and
277 `s = udim(R_R)`, the two bounds are stated without division or rounding:
278 
279 * `⌈ρ(M)⌉ ≤ μ(M)` iff `ℓ ≤ μ(M) s`: if `M` is generated by `m` elements,
280   `udim(M/t(M)) ≥ ℓ` and `udim(R_R) ≤ s`, then `ℓ ≤ m s`;
281 * `μ(M) ≤ ⌊ρ(M)⌋ + 1` iff `M` is generated by `q` elements for every `q` with
282   `ℓ < q s` (the least such `q` is `⌊ℓ/s⌋ + 1`): if `udim(M/t(M)) ≤ ℓ`,
283   `udim(R_R) ≥ s` and `ℓ < q s`, then `M` is generated by `q` elements. -/
284 
285 /-- **Corollary C.3, lower bound** `⌈ρ(M)⌉ ≤ μ(M)`. Let `R` satisfy (S) and (N)
286 and let `M` be a finitely generated right module. If `M` is generated by `m`
287 elements, `M/t(M)` contains a direct sum of `ℓ` nonzero submodules and `R_R`
288 contains no direct sum of `s + 1` nonzero right ideals, then `ℓ ≤ m s`.
289 This is equivalent to `⌈ρ(M)⌉ ≤ μ(M)` given the classical identities
290 `s = udim R_R` and `length_Q(M ⊗ Q) = udim(M/t(M))` (Goldie; McConnell--Robson, 2.2--2.3); it
291 is not a literal formalization of `ρ`. -/
292 def CorollaryC3LowerStatement : Prop :=
293   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
294     (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M]
295     (m ℓ s : ℕ),
296     SpanRankLE R M m → UDimGE R (M ⧸ goldieTorsionPart R M) ℓ → UDimLE R R s →
297       ℓ ≤ m * s
298 
299 /-- **Corollary C.3, upper bound** `μ(M) ≤ ⌊ρ(M)⌋ + 1`. Let `R` satisfy (S), (N),
300 (Z) and (H_all) and let `M` be a finitely generated right module. If `M/t(M)`
301 contains no direct sum of `ℓ + 1` nonzero submodules, `R_R` contains a direct
302 sum of `s` nonzero right ideals, and `ℓ < q s`, then `M` is generated by `q`
303 elements. This is equivalent to `μ(M) ≤ ⌊ρ(M)⌋ + 1` given the classical
304 identities `s = udim R_R` and `length_Q(M ⊗ Q) = udim(M/t(M))` (Goldie; McConnell--Robson, 2.2--2.3); it is not a literal
305 formalization of `ρ`. -/
306 def CorollaryC3UpperStatement : Prop :=
307   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
308     [Infinite (Subring.center R)], HAll R →
309     ∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M]
310       (q ℓ s : ℕ),
311       UDimLE R (M ⧸ goldieTorsionPart R M) ℓ → UDimGE R R s → ℓ < q * s →
312         SpanRankLE R M q

379 /-- The centre of a simple ring is a field, so (Z) provides an infinite
380 central field `k` with `[Algebra k R]`, as the library assumes. -/
381 theorem withCentre (R : Type u) [Ring R] [IsSimpleRing R] [Infinite (Subring.center R)]
382     (P : Prop)
383     (h : ∀ (k : Type u) [Field k] [Infinite k] [Algebra k R], P) : P := by
384   let : Field (Subring.center R) := (IsSimpleRing.isField_center R).toField
385   let : Algebra (Subring.center R) R :=
386     RingHom.toAlgebra' (Subring.center R).subtype fun c x =>
387       (Subring.mem_center_iff.mp c.2 x).symm
388   exact h (Subring.center R)
