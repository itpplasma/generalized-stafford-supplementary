11 # Challenge transport: the challenge definitions and their proofs
12 
13 The definitions and statement definitions of `GeneralizedStaffordChallenge.lean`
14 (identical copy `Challenge.lean`) are restated verbatim below, in the same
15 namespace and under the same names, so that the solution never imports the
16 challenge. The compared `theorem`s of the challenge are not restated: each
17 challenge theorem `<name>` is proved here as `<name>_proof`, and
18 `GeneralizedStaffordSolution.lean` restates it under its challenge name.
19 
20 The bridge lemmas translate the challenge conventions into those of the
21 library: `Submodule.span Rᵐᵒᵖ {d}` is `principalRightIdeal d`
22 (`span_singleton_eq`), `IsRegularElement d` is `d ∈ R⁰` (`isRegular_iff`), the
23 deviation bounds are AlgebraicAnalysis' `Module.DevLE` and `Module.DevLEOmega`
24 (`devDimLE_iff`, `hOmega_iff`), (H_all) is `PrincipalQuotientsFinite`
25 (`hAll_iff`), the infinite centre becomes an infinite central field
26 (`withCentre`).

35 /-! ## Regular elements, deviation and Goldie torsion -/
36 
37 section Definitions
38 
39 variable (R : Type u) [Ring R]
40 
41 /-- An element `d` of `R` is **regular** if it is a two-sided non-zero-divisor:
42 `d x = 0` implies `x = 0`, and `x d = 0` implies `x = 0` .
43 This is definitionally Mathlib's `d ∈ R⁰`. -/
44 def IsRegularElement {R : Type u} [Ring R] (d : R) : Prop :=
45   (∀ x : R, d * x = 0 → x = 0) ∧ (∀ x : R, x * d = 0 → x = 0)
46 
47 /-- Finite Gabriel--Rentschler deviation of a partial order. `PosetDevLE 0 α`
48 says that every descending chain in `α` stabilizes; `PosetDevLE (n + 1) α`
49 says that for every descending chain `c₀ ≥ c₁ ≥ ⋯` all but finitely many
50 closed intervals `[cᵢ₊₁, cᵢ]` satisfy `PosetDevLE n`, i.e. all but finitely
51 many factors have deviation `< n + 1` . -/
52 def PosetDevLE : (n : ℕ) → (α : Type v) → [PartialOrder α] → Prop
53   | 0, α, _ => ∀ c : ℕ → α, Antitone c → ∃ N, ∀ i, N ≤ i → c i = c N
54   | n + 1, α, _ => ∀ c : ℕ → α, Antitone c →
55       ∃ N, ∀ i, N ≤ i → PosetDevLE n (Set.Icc (c (i + 1)) (c i))
56 
57 variable (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M]
58 
59 /-- `DevDimLE R M n`: `Kdim M ≤ n` (written `Kdim`;
60 Gabriel--Rentschler deviation, not classical Krull dimension). The right
61 module `M` has right deviation dimension at most `n` , defined
62 on the lattice of right submodules of `M`. -/
63 def DevDimLE (n : ℕ) : Prop :=
64   PosetDevLE n (Submodule Rᵐᵒᵖ M)
65 
66 /-- `Kdim M` is a finite integer (possibly `-1`, which happens exactly when
67 `M = 0`). -/
68 def HasFiniteDevDim : Prop :=
69   ∃ n : ℕ, DevDimLE R M n
70 
71 /-- `DevDimLEOmega R M`: `Kdim M ≤ ω`, that is, for every descending chain of
72 right submodules `M₀ ≥ M₁ ≥ ⋯` all but finitely many factors `Mᵢ/Mᵢ₊₁`
73 (the intervals `[Mᵢ₊₁, Mᵢ]`) have finite deviation. -/
74 def DevDimLEOmega : Prop :=
75   ∀ c : ℕ → Submodule Rᵐᵒᵖ M, Antitone c → ∃ N, ∀ i, N ≤ i →
76     ∃ n : ℕ, PosetDevLE n (Set.Icc (c (i + 1)) (c i))
77 
78 /-- `M` is **Goldie-torsion**: every `m ∈ M` satisfies
79 `m s = 0` for some regular `s ∈ R`. -/
80 def IsGoldieTorsion : Prop :=
81   ∀ m : M, ∃ s : R, IsRegularElement s ∧ op s • m = 0
82 
83 /-- Hypothesis (H_all): `δ(d) = Kdim(R/dR)` is a finite integer for every
84 regular `d`. -/
85 def HAll : Prop :=
86   ∀ d : R, IsRegularElement d → HasFiniteDevDim R (R ⧸ Submodule.span Rᵐᵒᵖ {d})
87 
88 /-- Hypothesis (H_omega): `Kdim R ≤ ω` for the right regular module `R_R`. -/
89 def HOmega : Prop :=
90   DevDimLEOmega R R
91 
92 end Definitions
93 
94 /-! ## Theorem A' and Theorem A -/
95 
96 /-- **Theorem A' (fixed ideal).** Let `R` be a simple (S), right Noetherian (N)
97 ring with infinite centre (Z). Let `I` be a right ideal containing a regular
98 element such that `Kdim(R/I)` is a finite integer. Then for
99 every `x ∈ R` and every nonzero `t ∈ R` there is `H ∈ R` with
100 `I + (x + H t)R = R`. -/
101 def FixedIdealStatement : Prop :=
102   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
103     [Infinite (Subring.center R)] (I : Submodule Rᵐᵒᵖ R),
104     (∃ s ∈ I, IsRegularElement s) → HasFiniteDevDim R (R ⧸ I) →
105     ∀ x t : R, t ≠ 0 →
106       ∃ H : R, I ⊔ Submodule.span Rᵐᵒᵖ {x + H * t} = ⊤
107 
108 /-- **Theorem A (fixed anchor).** Let `R` satisfy (S), (N), (Z) and let `d` be a
109 regular element with `δ(d) = Kdim(R/dR)` a finite integer. For every nonzero
110 `a ∈ R` (possibly a zero-divisor) and every `B ∈ R` there are `H, U, V ∈ R`
111 with `1 = d U + (B + H a) d V`. -/
112 def FixedAnchorStatement : Prop :=
113   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
114     [Infinite (Subring.center R)] (d : R),
115     IsRegularElement d → HasFiniteDevDim R (R ⧸ Submodule.span Rᵐᵒᵖ {d}) →
116     ∀ a B : R, a ≠ 0 →
117       ∃ H U V : R, 1 = d * U + (B + H * a) * d * V
118 
119 /-! ## Corollary B -/
120 
121 /-- **Known input to Corollary B** (McConnell--Robson 2001, 6.3.9). Let `R` be
122 right Noetherian with `Kdim R ≤ ω` (H_omega). Then `δ(d) = Kdim(R/dR)` is a
123 finite integer for every regular `d`, that is, (H_all) holds. The paper cites
124 this classical result; it is stated here because Corollary B uses it. -/
125 def KnownDimensionStatement : Prop :=
126   ∀ (R : Type u) [Ring R] [IsNoetherian Rᵐᵒᵖ R], HOmega R → HAll R
127 
128 /-- **Corollary B, fixed anchor.** Let `R` satisfy (S), (N), (Z) and
129 `Kdim R ≤ ω`. For every regular `d`, every nonzero `a` and every `B` there are
130 `H, U, V` with `1 = d U + (B + H a) d V`. -/
131 def CorollaryBFixedAnchorStatement : Prop :=
132   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
133     [Infinite (Subring.center R)], HOmega R →
134     ∀ d a B : R, IsRegularElement d → a ≠ 0 →
135       ∃ H U V : R, 1 = d * U + (B + H * a) * d * V
136 
137 /-- **Corollary B, fixed ideal.** Let `R` satisfy (S), (N), (Z) and
138 `Kdim R ≤ ω`. For every right ideal `I` containing a regular element, every
139 `x` and every nonzero `t` there is `H` with `I + (x + H t)R = R`. -/
140 def CorollaryBFixedIdealStatement : Prop :=
141   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
142     [Infinite (Subring.center R)], HOmega R →
143     ∀ I : Submodule Rᵐᵒᵖ R, (∃ s ∈ I, IsRegularElement s) →
144     ∀ x t : R, t ≠ 0 →
145       ∃ H : R, I ⊔ Submodule.span Rᵐᵒᵖ {x + H * t} = ⊤
146 
147 /-! ## Corollary C.1 -/
148 
149 /-- **Corollary C.1 (prescribed cyclic generator).** Let `R` satisfy (S), (N),
150 (Z) and let `M` be a finitely generated Goldie-torsion right module with
151 `Kdim M` a finite integer. For every `x ∈ M` and every nonzero `t ∈ R` there
152 is `h ∈ M` with `M = (x + h t)R`. -/
153 def CorollaryC1Statement : Prop :=
154   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
155     [Infinite (Subring.center R)]
156     (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M],
157     IsGoldieTorsion R M → HasFiniteDevDim R M →
158     ∀ (x : M) (t : R), t ≠ 0 →
159       ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} = ⊤
160 
161 /-- **Corollary C.1 under (H_all).** Let `R` satisfy (S), (N), (Z) and (H_all).
162 Every finitely generated Goldie-torsion right module `M` satisfies: for every
163 `x ∈ M` and nonzero `t ∈ R` there is `h ∈ M` with `M = (x + h t)R`. -/
164 def CorollaryC1HAllStatement : Prop :=
165   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
166     [Infinite (Subring.center R)], HAll R →
167     ∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M],
168     IsGoldieTorsion R M →
169     ∀ (x : M) (t : R), t ≠ 0 →
170       ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} = ⊤
171 
172 /-- **Corollary C.1 under `Kdim R ≤ ω`.** Let `R` satisfy (S), (N), (Z) and
173 (H_omega). Every finitely generated Goldie-torsion right module `M` satisfies:
174 for every `x ∈ M` and nonzero `t ∈ R` there is `h ∈ M` with `M = (x + h t)R`. -/
175 def CorollaryC1OmegaStatement : Prop :=
176   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
177     [Infinite (Subring.center R)], HOmega R →
178     ∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M],
179     IsGoldieTorsion R M →
180     ∀ (x : M) (t : R), t ≠ 0 →
181       ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} = ⊤
182 
183 /-- **Corollary C.1, stable cyclicity.** Let `R` satisfy (S), (N), (Z), let `c, u`
184 be Goldie-torsion elements of a right module `M` (each is killed by a regular
185 element) with `Kdim(uR)` a finite integer.
186 For every nonzero `t ∈ R` there is `H ∈ R` with `cR + uR = (c + u H t)R`. -/
187 def CorollaryC1StableStatement : Prop :=
188   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
189     [Infinite (Subring.center R)]
190     (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] (c u : M),
191     (∃ s : R, IsRegularElement s ∧ op s • c = 0) →
192     (∃ s : R, IsRegularElement s ∧ op s • u = 0) →
193     HasFiniteDevDim R (Submodule.span Rᵐᵒᵖ {u}) →
194     ∀ t : R, t ≠ 0 →
195       ∃ H : R, Submodule.span Rᵐᵒᵖ {c, u} = Submodule.span Rᵐᵒᵖ {c + op (H * t) • u}
196 
197 /-! ## Corollary C.2
198 
199 The condition `aQ = IQ` (with `Q` the classical right quotient ring)
200 is stated elementwise: `a ∈ I` and every `y ∈ I` has a regular `s` with
201 `y s ∈ aR`. Under (S), (N) these are equivalent: `aQ ⊆ IQ` because `a ∈ I`,
202 and `y ∈ aQ` iff `y = a r s⁻¹` iff `y s = a r` for some `r` and regular `s`. -/
203 
204 /-- **Existence of a full-rank element** (first step of Corollary C.2). Let `R`
205 satisfy (S) and (N). Every right ideal `I` contains an `a` with `aQ = IQ`, that
206 is, every `y ∈ I` has a regular `s` with `y s ∈ aR`. Corollary C.2 is stated
207 under (S), (N), (Z), (H_all); this step needs only (S) and (N) (Goldie's
208 theorem), so the statement here has fewer hypotheses and is still true. -/
209 def CorollaryC2FullRankStatement : Prop :=
210   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
211     (I : Submodule Rᵐᵒᵖ R),
212     ∃ a ∈ I, ∀ y ∈ I,
213       ∃ s : R, IsRegularElement s ∧ y * s ∈ Submodule.span Rᵐᵒᵖ {a}
214 
215 /-- **Corollary C.2, prescribed form.** Let `R` satisfy (S), (N), (Z) and
216 (H_all), and let `I` be a right ideal. If `a ∈ I` satisfies `aQ = IQ` (every
217 `y ∈ I` has a regular `s` with `y s ∈ aR`), then for every `y ∈ I` and every
218 nonzero `t ∈ R` there is `h ∈ I` with `I = aR + (y + h t)R`. -/
219 def CorollaryC2PrescribedStatement : Prop :=
220   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
221     [Infinite (Subring.center R)], HAll R →
222     ∀ (I : Submodule Rᵐᵒᵖ R) (a : R), a ∈ I →
223     (∀ y ∈ I, ∃ s : R, IsRegularElement s ∧ y * s ∈ Submodule.span Rᵐᵒᵖ {a}) →
224     ∀ y ∈ I, ∀ t : R, t ≠ 0 →
225       ∃ h ∈ I, I = Submodule.span Rᵐᵒᵖ {a} ⊔ Submodule.span Rᵐᵒᵖ {y + h * t}
226 
227 /-- **Corollary C.2 (two generators).** Let `R` satisfy (S), (N), (Z) and
228 (H_all). Every right ideal `I` of `R` is generated by two elements:
229 `I = aR + bR` for some `a, b ∈ R`. -/
230 def CorollaryC2Statement : Prop :=
231   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
232     [Infinite (Subring.center R)], HAll R →
233     ∀ I : Submodule Rᵐᵒᵖ R, ∃ a b : R, I = Submodule.span Rᵐᵒᵖ {a, b}
234 
235 /-- **Corollary C.2 under `Kdim R ≤ ω`.** Let `R` satisfy (S), (N), (Z) and
236 (H_omega). Every right ideal `I` of `R` is generated by two elements:
237 `I = aR + bR` for some `a, b ∈ R`. -/
238 def CorollaryC2OmegaStatement : Prop :=
239   ∀ (R : Type u) [Ring R] [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
240     [Infinite (Subring.center R)], HOmega R →
241     ∀ I : Submodule Rᵐᵒᵖ R, ∃ a b : R, I = Submodule.span Rᵐᵒᵖ {a, b}

248 /-- The challenge's deviation bound on a poset is AlgebraicAnalysis' `Order.DevLE`. -/
249 theorem posetDevLE_iff : ∀ (n : ℕ) (α : Type v) [PartialOrder α],
250     PosetDevLE n α ↔ Order.DevLE n α
251   | 0, α, _ => Iff.rfl
252   | n + 1, α, _ => by
253     simp only [PosetDevLE, Order.devLE_succ_iff, posetDevLE_iff n]
254 
255 /-- `DevDimLE R M n` is `Module.DevLE Rᵐᵒᵖ n M`. -/
256 theorem devDimLE_iff {R : Type u} [Ring R] {M : Type v} [AddCommGroup M]
257     [Module Rᵐᵒᵖ M] (n : ℕ) : DevDimLE R M n ↔ Module.DevLE Rᵐᵒᵖ n M :=
258   posetDevLE_iff n _
259 
260 /-- (H_omega) is `Module.DevLEOmega Rᵐᵒᵖ R`. -/
261 theorem hOmega_iff {R : Type u} [Ring R] : HOmega R ↔ Module.DevLEOmega Rᵐᵒᵖ R := by
262   simp only [HOmega, DevDimLEOmega, Module.DevLEOmega, posetDevLE_iff]
263 
264 /-- `IsRegularElement d` is `d ∈ R⁰`, by definition. -/
265 theorem isRegular_iff {R : Type u} [Ring R] (d : R) : IsRegularElement d ↔ d ∈ R⁰ :=
266   Iff.rfl
267 
268 /-- The span of `{d}` is the principal right ideal `dR`. -/
269 theorem span_singleton_eq {R : Type u} [Ring R] (d : R) :
270     Submodule.span Rᵐᵒᵖ {d} = principalRightIdeal d := by
271   ext y
272   rw [Submodule.mem_span_singleton]
273   constructor
274   · rintro ⟨a, rfl⟩; exact ⟨unop a, rfl⟩
275   · rintro ⟨r, rfl⟩; exact ⟨op r, rfl⟩
276 
277 /-- The challenge's Goldie torsion is AlgebraicAnalysis' Goldie torsion. -/
278 theorem isGoldieTorsion_iff {R : Type u} [Ring R] {M : Type v} [AddCommGroup M]
279     [Module Rᵐᵒᵖ M] :
280     GeneralizedStaffordChallenge.IsGoldieTorsion R M ↔
281       AlgebraicAnalysis.Goldie.IsGoldieTorsion R M := by
282   simp only [GeneralizedStaffordChallenge.IsGoldieTorsion,
283     AlgebraicAnalysis.Goldie.IsGoldieTorsion, isRegular_iff]
284 
285 /-- (H_all) of the challenge is the library's `PrincipalQuotientsFinite`. -/
286 theorem hAll_iff {R : Type u} [Ring R] : HAll R ↔ PrincipalQuotientsFinite R := by
287   simp only [HAll, HasFiniteDevDim, devDimLE_iff, isRegular_iff, PrincipalQuotientsFinite]
288   refine forall₂_congr fun d _ => ?_
289   rw [span_singleton_eq]

291 /-- The centre of a simple ring is a field, so (Z) provides an infinite
292 central field `k` with `[Algebra k R]`, as the library assumes. -/
293 theorem withCentre (R : Type u) [Ring R] [IsSimpleRing R] [Infinite (Subring.center R)]
294     (P : Prop)
295     (h : ∀ (k : Type u) [Field k] [Infinite k] [Algebra k R], P) : P := by
296   let : Field (Subring.center R) := (IsSimpleRing.isField_center R).toField
297   let : Algebra (Subring.center R) R :=
298     RingHom.toAlgebra' (Subring.center R).subtype fun c x =>
299       (Subring.mem_center_iff.mp c.2 x).symm
300   exact h (Subring.center R)
