6 # Challenge transport: differential operators on a smooth affine variety
7 
8 The definitions of the differential-operator section of `GeneralizedStaffordChallenge.lean`
9 (identical copy `Challenge.lean`) and the `GlobalStaffordChallenge` definitions of the Global Stafford statement are
10 restated verbatim below, in the same namespaces and under the same names, so that the solution
11 never imports the challenge. Each compared theorem `<name>` is proved here as `<name>_proof`;
12 `GeneralizedStaffordSolution.lean` restates it under its challenge name.
13 
14 The bridge: `IsDifferentialOperator P` is membership of `P` in AlgebraicAnalysis' subalgebra
15 `DifferentialOperators.algebra` of finite-order operators (`isDifferentialOperator_iff`, by
16 induction on the order), and `diffOps k A` equals that subalgebra (`diffOps_eq_algebra`). The
17 ring-level statements of `GeneralizedStafford.DiffOp` are transported along this equality by
18 substitution, and the element-level statements by the membership equivalence.
19 -/

27 /-- Multiplication by `a`, as a `k`-linear endomorphism of `A`. -/
28 def multiplication (a : A) : Module.End k A := LinearMap.mulLeft k a
29 
30 /-- The commutator `[P, a] = P ∘ a - a ∘ P` of an endomorphism with a
31 multiplication operator. -/
32 def commutator (P : Module.End k A) (a : A) : Module.End k A :=
33   P * multiplication (k := k) a - multiplication (k := k) a * P
34 
35 /-- `IsOrderLE n P` says that `P` is a differential operator of order at most
36 `n`: order `0` operators commute with every multiplication, and `P` has order
37 at most `n + 1` when every commutator `[P, a]` has order at most `n`. -/
38 def IsOrderLE : ℕ → Module.End k A → Prop
39   | 0, P => ∀ a : A, commutator P a = 0
40   | n + 1, P => ∀ a : A, IsOrderLE n (commutator P a)
41 
42 /-- A finite-order `k`-linear differential operator on `A`. -/
43 def IsDifferentialOperator (P : Module.End k A) : Prop :=
44   ∃ n : ℕ, IsOrderLE n P
45 
46 end GlobalStaffordChallenge

48 /-! ## Differential operators on a smooth affine variety
49 
50 Let `k` be a field of characteristic zero and `A` a smooth domain of finite type over `k`.
51 Let `D_k(A)` be the ring of `k`-linear finite-order differential operators on `A`, as defined by
52 `GlobalStaffordChallenge.IsDifferentialOperator`. Products are compositions. The closure proofs
53 of `diffOps` below are complete and elementary (`[PQ, a] = P [Q, a] + [P, a] Q`); they are not
54 compared theorems. The statements that need only elements (Theorem A, the Global Stafford
55 identity) are written in `Module.End k A` with explicit `IsDifferentialOperator` side
56 conditions. -/

66 theorem endCommutator_zero (a : A) :
67     GlobalStaffordChallenge.commutator (0 : Module.End k A) a = 0 := by
68   simp [GlobalStaffordChallenge.commutator]
69 
70 theorem endCommutator_add (P Q : Module.End k A) (a : A) :
71     GlobalStaffordChallenge.commutator (P + Q) a =
72       GlobalStaffordChallenge.commutator P a + GlobalStaffordChallenge.commutator Q a := by
73   simp only [GlobalStaffordChallenge.commutator, add_mul, mul_add]; abel
74 
75 theorem endCommutator_mul (P Q : Module.End k A) (a : A) :
76     GlobalStaffordChallenge.commutator (P * Q) a =
77       P * GlobalStaffordChallenge.commutator Q a + GlobalStaffordChallenge.commutator P a * Q := by
78   simp only [GlobalStaffordChallenge.commutator, mul_sub, sub_mul, mul_assoc]; abel
79 
80 theorem isOrderLE_zero : ∀ n, IsOrderLE n (0 : Module.End k A)
81   | 0 => fun a => endCommutator_zero a
82   | n + 1 => fun a => by rw [endCommutator_zero]; exact isOrderLE_zero n
83 
84 theorem isOrderLE_add : ∀ {n} {P Q : Module.End k A},
85     IsOrderLE n P → IsOrderLE n Q → IsOrderLE n (P + Q)
86   | 0, _, _, hP, hQ => fun a => by rw [endCommutator_add, hP a, hQ a, add_zero]
87   | _ + 1, _, _, hP, hQ => fun a => by
88       rw [endCommutator_add]; exact isOrderLE_add (hP a) (hQ a)
89 
90 theorem isOrderLE_succ : ∀ {n} {P : Module.End k A}, IsOrderLE n P → IsOrderLE (n + 1) P
91   | 0, _, hP => fun a => by rw [hP a]; exact isOrderLE_zero 0
92   | _ + 1, _, hP => fun a => isOrderLE_succ (hP a)
93 
94 theorem isOrderLE_mono {m n : ℕ} {P : Module.End k A} (h : m ≤ n) (hP : IsOrderLE m P) :
95     IsOrderLE n P := by
96   induction n, h using Nat.le_induction with
97   | base => exact hP
98   | succ n _ ih => exact isOrderLE_succ ih
99 
100 /-- Orders add under composition. -/
101 theorem isOrderLE_mul : ∀ (N : ℕ) {m n : ℕ} {P Q : Module.End k A}, m + n = N →
102     IsOrderLE m P → IsOrderLE n Q → IsOrderLE N (P * Q)
103   | 0, m, n, P, Q, h, hP, hQ => by
104       obtain ⟨rfl, rfl⟩ : m = 0 ∧ n = 0 := by omega
105       intro a
106       rw [endCommutator_mul, hP a, hQ a, mul_zero, zero_mul, add_zero]
107   | N + 1, m, n, P, Q, h, hP, hQ => fun a => by
108       rw [endCommutator_mul]
109       refine isOrderLE_add ?_ ?_
110       · match n, hQ with
111         | 0, hQ => rw [hQ a, mul_zero]; exact isOrderLE_zero N
112         | n + 1, hQ => exact isOrderLE_mul N (by omega) hP (hQ a)
113       · match m, hP with
114         | 0, hP => rw [hP a, zero_mul]; exact isOrderLE_zero N
115         | m + 1, hP => exact isOrderLE_mul N (by omega) (hP a) hQ
116 
117 end DiffOps

119 /-- The `k`-subalgebra `D_k(A)` of `Module.End k A`: the finite-order differential operators,
120 with composition as product. -/
121 def diffOps (k A : Type u) [CommRing k] [CommRing A] [Algebra k A] :
122     Subalgebra k (Module.End k A) where
123   carrier := {P | IsDifferentialOperator P}
124   add_mem' := fun ⟨m, hP⟩ ⟨n, hQ⟩ =>
125     ⟨max m n, isOrderLE_add (isOrderLE_mono (le_max_left m n) hP)
126       (isOrderLE_mono (le_max_right m n) hQ)⟩
127   mul_mem' := fun ⟨m, hP⟩ ⟨n, hQ⟩ => ⟨m + n, isOrderLE_mul _ rfl hP hQ⟩
128   algebraMap_mem' c := ⟨0, fun _ => sub_eq_zero.mpr (Algebra.commutes c _)⟩
129 
130 end GeneralizedStaffordChallenge

136 /-- Two generators for right ideals of `D_k(A)`. For a field `k` of characteristic zero and a
137 smooth domain `A` of finite type over `k`, every right ideal `I` of `D_k(A)` is generated by two
138 elements: `I = a D_k(A) + b D_k(A)`. -/
139 def DiffOpTwoGeneratedRightStatement : Prop :=
140   ∀ (k : Type u) [Field k] [CharZero k]
141     (A : Type u) [CommRing A] [IsDomain A] [Algebra k A] [Algebra.Smooth k A]
142     (I : Submodule (diffOps k A)ᵐᵒᵖ (diffOps k A)),
143     ∃ a b : diffOps k A, I = Submodule.span (diffOps k A)ᵐᵒᵖ {a, b}
144 
145 /-- Two generators for left ideals of `D_k(A)`. With `k` and `A` as above, every left ideal `I`
146 of `D_k(A)` is generated by two elements: `I = D_k(A) a + D_k(A) b`. -/
147 def DiffOpTwoGeneratedLeftStatement : Prop :=
148   ∀ (k : Type u) [Field k] [CharZero k]
149     (A : Type u) [CommRing A] [IsDomain A] [Algebra k A] [Algebra.Smooth k A]
150     (I : Submodule (diffOps k A) (diffOps k A)),
151     ∃ a b : diffOps k A, I = Submodule.span (diffOps k A) {a, b}
152 
153 /-- Theorem A for `D_k(A)`. With `k` and `A` as above, for all differential operators `d, a, B`
154 with `d ≠ 0` and `a ≠ 0` there are differential operators `H, U, V` with
155 `1 = d U + (B + H a) d V` (right form), and differential operators `H, U, V` with
156 `1 = U d + V d (B + a H)` (left form). Products are compositions. -/
157 def DiffOpFixedAnchorStatement : Prop :=
158   ∀ (k : Type u) [Field k] [CharZero k]
159     (A : Type u) [CommRing A] [IsDomain A] [Algebra k A] [Algebra.Smooth k A]
160     (d a B : Module.End k A),
161     IsDifferentialOperator d → IsDifferentialOperator a → IsDifferentialOperator B →
162     d ≠ 0 → a ≠ 0 →
163       (∃ H U V : Module.End k A,
164         IsDifferentialOperator H ∧ IsDifferentialOperator U ∧ IsDifferentialOperator V ∧
165         (1 : Module.End k A) = d * U + (B + H * a) * d * V) ∧
166       (∃ H U V : Module.End k A,
167         IsDifferentialOperator H ∧ IsDifferentialOperator U ∧ IsDifferentialOperator V ∧
168         (1 : Module.End k A) = U * d + V * d * (B + a * H))
169 
170 /-- The Global Stafford identity for `D_k(A)`. With `k` and `A` as above, for every nonzero
171 differential operator `d` there are differential operators `F, R, S` with `1 = d R + F d S`,
172 the products being compositions. -/
173 def DiffOpGlobalStaffordStatement : Prop :=
174   ∀ (k : Type u) [Field k] [CharZero k]
175     (A : Type u) [CommRing A] [IsDomain A] [Algebra k A] [Algebra.Smooth k A]
176     (d : Module.End k A),
177     IsDifferentialOperator d → d ≠ 0 →
178       ∃ F R S : Module.End k A,
179         IsDifferentialOperator F ∧ IsDifferentialOperator R ∧
180         IsDifferentialOperator S ∧
181         (1 : Module.End k A) = d * R + F * d * S
182 
183 /-- Cyclicity of torsion right `D_k(A)`-modules. With `k` and `A` as above, let `M` be a finitely
184 generated right `D_k(A)`-module in which every element is killed by a nonzero operator. For every
185 `x ∈ M` and every nonzero `t ∈ D_k(A)` there is `h ∈ M` with `M = (x + h t) D_k(A)`; in
186 particular `M` is cyclic. -/
187 def DiffOpCyclicRightStatement : Prop :=
188   ∀ (k : Type u) [Field k] [CharZero k]
189     (A : Type u) [CommRing A] [IsDomain A] [Algebra k A] [Algebra.Smooth k A]
190     (M : Type v) [AddCommGroup M] [Module (diffOps k A)ᵐᵒᵖ M]
191     [Module.Finite (diffOps k A)ᵐᵒᵖ M],
192     (∀ m : M, ∃ s : diffOps k A, s ≠ 0 ∧ op s • m = 0) →
193     ∀ (x : M) (t : diffOps k A), t ≠ 0 →
194       ∃ h : M, Submodule.span (diffOps k A)ᵐᵒᵖ {x + op t • h} = ⊤
195 
196 /-- Cyclicity of torsion left `D_k(A)`-modules. With `k` and `A` as above, let `M` be a finitely
197 generated left `D_k(A)`-module in which every element is killed by a nonzero operator. For every
198 `x ∈ M` and every nonzero `t ∈ D_k(A)` there is `h ∈ M` with `M = D_k(A) (x + t h)`; in
199 particular `M` is cyclic. -/
200 def DiffOpCyclicLeftStatement : Prop :=
201   ∀ (k : Type u) [Field k] [CharZero k]
202     (A : Type u) [CommRing A] [IsDomain A] [Algebra k A] [Algebra.Smooth k A]
203     (M : Type v) [AddCommGroup M] [Module (diffOps k A) M]
204     [Module.Finite (diffOps k A) M],
205     (∀ m : M, ∃ s : diffOps k A, s ≠ 0 ∧ s • m = 0) →
206     ∀ (x : M) (t : diffOps k A), t ≠ 0 →
207       ∃ h : M, Submodule.span (diffOps k A) {x + t • h} = ⊤

213 /-- `IsOrderLE n P` is membership of `P` in the order-`n` filtration of
214 AlgebraicAnalysis (induction on `n`). -/
215 theorem isOrderLE_iff_mem_order {k A : Type u} [CommRing k] [CommRing A] [Algebra k A]
216     (n : ℕ) (P : Module.End k A) :
217     GlobalStaffordChallenge.IsOrderLE n P ↔
218       P ∈ AlgebraicAnalysis.DifferentialOperators.order (k := k) (R := A) n := by
219   induction n generalizing P with
220   | zero => exact (AlgebraicAnalysis.DifferentialOperators.mem_order_zero_iff P).symm
221   | succ n ih =>
222       rw [AlgebraicAnalysis.DifferentialOperators.mem_order_succ_iff]
223       exact forall_congr' fun a => ih _
224 
225 /-- `IsDifferentialOperator P` is membership of `P` in AlgebraicAnalysis' subalgebra
226 `DifferentialOperators.algebra` of finite-order operators. -/
227 theorem isDifferentialOperator_iff {k A : Type u} [CommRing k] [CommRing A] [Algebra k A]
228     (P : Module.End k A) :
229     GlobalStaffordChallenge.IsDifferentialOperator P ↔
230       P ∈ AlgebraicAnalysis.DifferentialOperators.algebra (k := k) (R := A) := by
231   simp only [GlobalStaffordChallenge.IsDifferentialOperator,
232     AlgebraicAnalysis.DifferentialOperators.mem_algebra_iff]
233   exact exists_congr fun n => isOrderLE_iff_mem_order n P
234 
235 /-- The challenge subalgebra `diffOps k A` is AlgebraicAnalysis' `DifferentialOperators.algebra`. -/
236 theorem diffOps_eq_algebra (k A : Type u) [Field k] [CommRing A] [Algebra k A] :
237     diffOps k A = AlgebraicAnalysis.DifferentialOperators.algebra (k := k) (R := A) := by
238   ext P
239   exact isDifferentialOperator_iff P
240 
241 end Bridge

304 /-- Theorem A for `D_k(A)`, right and left forms. -/
305 theorem diffOpFixedAnchor_proof : DiffOpFixedAnchorStatement.{u} := by
306   intro k _ _ A _ _ _ _ d a B hd ha hB hd0 ha0
307   obtain ⟨H, U, V, hH, hU, hV, h⟩ := GeneralizedStafford.DiffOp.affineForm_right k A d a B
308     ((isDifferentialOperator_iff d).mp hd) hd0 ((isDifferentialOperator_iff a).mp ha) ha0
309     ((isDifferentialOperator_iff B).mp hB)
310   obtain ⟨H', U', V', hH', hU', hV', h'⟩ := GeneralizedStafford.DiffOp.affineForm_left k A d a B
311     ((isDifferentialOperator_iff d).mp hd) hd0 ((isDifferentialOperator_iff a).mp ha) ha0
312     ((isDifferentialOperator_iff B).mp hB)
313   exact ⟨⟨H, U, V, (isDifferentialOperator_iff H).mpr hH, (isDifferentialOperator_iff U).mpr hU,
314       (isDifferentialOperator_iff V).mpr hV, h⟩,
315     ⟨H', U', V', (isDifferentialOperator_iff H').mpr hH', (isDifferentialOperator_iff U').mpr hU',
316       (isDifferentialOperator_iff V').mpr hV', h'⟩⟩
317 
318 /-- The Global Stafford identity for `D_k(A)`. -/
319 theorem diffOpGlobalStafford_proof : DiffOpGlobalStaffordStatement.{u} := by
320   intro k _ _ A _ _ _ _ d hd hd0
321   obtain ⟨F, R, S, hF, hR, hS, h⟩ :=
322     GeneralizedStafford.DiffOp.globalStafford k A d ((isDifferentialOperator_iff d).mp hd) hd0
323   exact ⟨F, R, S, (isDifferentialOperator_iff F).mpr hF, (isDifferentialOperator_iff R).mpr hR,
324     (isDifferentialOperator_iff S).mpr hS, h⟩
