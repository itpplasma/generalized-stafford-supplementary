6 /-!
7 # Rings of differential operators `D_k(A)`
8 
9 Let `k` be a field of characteristic zero and `A` a smooth domain of finite type over `k`.
10 The ring `D = D_k(A)` of finite-order `k`-linear differential operators on `A` satisfies (S),
11 (N), (Z) and (H_omega) on both sides, by AlgebraicAnalysis (`SmoothOperators`): `D` is simple
12 and a domain, it is right and left Noetherian, its right and left deviations are at most
13 `2r` for `r` generators of `A`, and its centre is infinite. The conclusions of Theorem A,
14 Theorem A', Corollary B and Corollaries C.1 to C.3 then follow for `D` and for `Dᵐᵒᵖ` from
15 the abstract theorems of `GeneralizedStafford.Challenge.FilteredTransport`.
16 
17 This file proves:
18 
19 * `GeneralizedStafford.DiffOp.conclusions_right`, `conclusions_left`: all conclusions for `D`
20   (right ideals and modules) and for `Dᵐᵒᵖ` (left ideals and modules).
21 * `affineForm_right`, `affineForm_left`: Theorem A, right form `1 = d U + (B + H a) d V` and
22   left form `1 = U d + V d (B + a H)`, for nonzero `d` and `a`, as statements about elements of
23   `Module.End k A` lying in `D_k(A)`.
24 * `globalStafford`: `1 = d R + F d S` for nonzero `d`.
25 * `twoGenerated_left_ideal`, `prescribedCyclicLeft`: the left-ideal and left-module forms, with
26   the same proofs transported through the opposite ring.
27 -/

141 /-- `D_k(A)` as a subalgebra of `Module.End k A`: the operators of finite order. -/
142 abbrev Dsub (k A : Type u) [Field k] [CommRing A] [Algebra k A] :
143     Subalgebra k (Module.End k A) :=
144   AlgebraicAnalysis.DifferentialOperators.algebra (k := k) (R := A)
145 
146 /-- **Theorem A, right form.** For differential operators `d, a, B` on `A` with `d ≠ 0` and
147 `a ≠ 0` there are differential operators `H, U, V` with `1 = d U + (B + H a) d V`. -/
148 theorem affineForm_right (d a B : Module.End k A) (hd : d ∈ Dsub k A) (hd0 : d ≠ 0)
149     (ha : a ∈ Dsub k A) (ha0 : a ≠ 0) (hB : B ∈ Dsub k A) :
150     ∃ H U V : Module.End k A, H ∈ Dsub k A ∧ U ∈ Dsub k A ∧ V ∈ Dsub k A ∧
151       (1 : Module.End k A) = d * U + (B + H * a) * d * V := by
152   have := isDomain k A
153   have hS : IsSimpleRing (DiffOp k A) := isSimpleRing k A
154   have hN : IsNoetherian (DiffOp k A)ᵐᵒᵖ (DiffOp k A) := noetherianRight k A
155   have hZ : Infinite (Subring.center (DiffOp k A)) := infiniteCenterRight k A
156   have hd' : (⟨d, hd⟩ : DiffOp k A) ≠ 0 := fun h => hd0 (by simpa using congrArg Subtype.val h)
157   have hreg : IsRegularElement (⟨d, hd⟩ : DiffOp k A) := Weyl.isRegular_of_ne_zero hd'
158   have ha' : (⟨a, ha⟩ : DiffOp k A) ≠ 0 := fun h => ha0 (by simpa using congrArg Subtype.val h)
159   obtain ⟨H, U, V, h⟩ :=
160     (fixedAnchorProperty_of_omega (R := DiffOp k A) (homegaRight k A)
161       : FixedAnchorProperty (DiffOp k A)) ⟨d, hd⟩ ⟨a, ha⟩ ⟨B, hB⟩ hreg ha'
162   refine ⟨H, U, V, H.2, U.2, V.2, ?_⟩
163   have := congrArg Subtype.val h
164   simpa using this
165 
166 /-- **Theorem A, left form.** For differential operators `d, a, B` on `A` with `d ≠ 0` and
167 `a ≠ 0` there are differential operators `H, U, V` with `1 = U d + V d (B + a H)`. -/
168 theorem affineForm_left (d a B : Module.End k A) (hd : d ∈ Dsub k A) (hd0 : d ≠ 0)
169     (ha : a ∈ Dsub k A) (ha0 : a ≠ 0) (hB : B ∈ Dsub k A) :
170     ∃ H U V : Module.End k A, H ∈ Dsub k A ∧ U ∈ Dsub k A ∧ V ∈ Dsub k A ∧
171       (1 : Module.End k A) = U * d + V * d * (B + a * H) := by
172   have := isDomain k A
173   have hS : IsSimpleRing (DiffOp k A) := isSimpleRing k A
174   have hN : IsNoetherian (DiffOp k A)ᵐᵒᵖᵐᵒᵖ (DiffOp k A)ᵐᵒᵖ := noetherianLeft k A
175   have hZ : Infinite (Subring.center (DiffOp k A)ᵐᵒᵖ) := infiniteCenterLeft k A
176   have hS' : IsSimpleRing (DiffOp k A)ᵐᵒᵖ := inferInstance
177   have hd' : (⟨d, hd⟩ : DiffOp k A) ≠ 0 := fun h => hd0 (by simpa using congrArg Subtype.val h)
178   have ha' : (⟨a, ha⟩ : DiffOp k A) ≠ 0 := fun h => ha0 (by simpa using congrArg Subtype.val h)
179   have hreg : IsRegularElement (op (⟨d, hd⟩ : DiffOp k A)) :=
180     Weyl.isRegular_of_ne_zero (fun h => hd' (by simpa using h))
181   have ha'' : op (⟨a, ha⟩ : DiffOp k A) ≠ 0 := fun h => ha' (by simpa using h)
182   obtain ⟨H, U, V, h⟩ :=
183     (fixedAnchorProperty_of_omega (R := (DiffOp k A)ᵐᵒᵖ) (homegaLeft k A)
184       : FixedAnchorProperty (DiffOp k A)ᵐᵒᵖ)
185       (op ⟨d, hd⟩) (op ⟨a, ha⟩) (op ⟨B, hB⟩) hreg ha''
186   have h' := congrArg (fun z : (DiffOp k A)ᵐᵒᵖ => ((unop z : DiffOp k A) : Module.End k A)) h
187   refine ⟨((unop H : DiffOp k A) : Module.End k A), ((unop U : DiffOp k A) : Module.End k A),
188     ((unop V : DiffOp k A) : Module.End k A), (unop H).2, (unop U).2, (unop V).2, ?_⟩
189   simpa [mul_assoc] using h'
190 
191 /-- **Global Stafford identity.** For a nonzero differential operator `d` on `A` there are
192 differential operators `F, R, S` with `1 = d R + F d S`. This is Theorem A with `a = 1`
193 and `B = 0`. -/
194 theorem globalStafford (d : Module.End k A) (hd : d ∈ Dsub k A) (hd0 : d ≠ 0) :
195     ∃ F R S : Module.End k A, F ∈ Dsub k A ∧ R ∈ Dsub k A ∧ S ∈ Dsub k A ∧
196       (1 : Module.End k A) = d * R + F * d * S := by
197   obtain ⟨H, U, V, hH, hU, hV, h⟩ :=
198     affineForm_right k A d 1 0 hd hd0 (Dsub k A).one_mem one_ne_zero (Dsub k A).zero_mem
199   exact ⟨H, U, V, hH, hU, hV, by simpa using h⟩
200 
201 end Elements

209 /-- The ring homomorphism `Rᵐᵒᵖᵐᵒᵖ → R`, `op (op r) ↦ r`. -/
210 def opOpHom (R : Type u) [Ring R] : Rᵐᵒᵖᵐᵒᵖ →+* R where
211   toFun x := unop (unop x)
212   map_one' := rfl
213   map_mul' _ _ := rfl
214   map_zero' := rfl
215   map_add' _ _ := rfl
216 
217 /-- A left `R`-module as a module over `Rᵐᵒᵖᵐᵒᵖ`, with `op (op r) • m = r • m`. A left module
218 is a right module over the opposite ring. This is a local construction, not an instance. -/
219 abbrev leftAsRight (M : Type v) [AddCommGroup M] [Module R M] : Module Rᵐᵒᵖᵐᵒᵖ M :=
220   Module.compHom M (opOpHom R)
221 
222 /-- The span over `Rᵐᵒᵖᵐᵒᵖ` of a set is its span over `R`. -/
223 theorem span_leftAsRight (M : Type v) [AddCommGroup M] [Module R M] (S : Set M) :
224     (@Submodule.span Rᵐᵒᵖᵐᵒᵖ M _ _ (leftAsRight M) S : Set M) = Submodule.span R S := by
225   letI := leftAsRight (R := R) M
226   apply le_antisymm
227   · let N : Submodule Rᵐᵒᵖᵐᵒᵖ M :=
228       { carrier := (Submodule.span R S : Set M)
229         add_mem' := fun hx hy => Submodule.add_mem _ hx hy
230         zero_mem' := Submodule.zero_mem _
231         smul_mem' := fun c _ hm => Submodule.smul_mem _ (opOpHom R c) hm }
232     have hle : Submodule.span Rᵐᵒᵖᵐᵒᵖ S ≤ N :=
233       Submodule.span_le.mpr fun s hs => Submodule.subset_span hs
234     exact fun m hm => hle hm
235   · let N : Submodule R M :=
236       { carrier := (Submodule.span Rᵐᵒᵖᵐᵒᵖ S : Set M)
237         add_mem' := fun hx hy => Submodule.add_mem _ hx hy
238         zero_mem' := Submodule.zero_mem _
239         smul_mem' := fun r _ hm => Submodule.smul_mem _ (op (op r)) hm }
240     have hle : Submodule.span R S ≤ N :=
241       Submodule.span_le.mpr fun s hs => Submodule.subset_span hs
242     exact fun m hm => hle hm
243 
244 /-- Left ideals of `R` are two-generated, from the two-generation of right ideals of `Rᵐᵒᵖ`:
245 a left ideal `I` is the right ideal `{op x | x ∈ I}` of `Rᵐᵒᵖ`. -/
246 theorem twoGenerated_left_of_op
247     (hR : ∀ J : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ, ∃ a b : Rᵐᵒᵖ, J = Submodule.span Rᵐᵒᵖᵐᵒᵖ {a, b})
248     (I : Submodule R R) : ∃ a b : R, I = Submodule.span R {a, b} := by
249   obtain ⟨a, b, hab⟩ := hR ((AlgebraicAnalysis.rightIdealOpEquiv R).symm I)
250   refine ⟨unop a, unop b, ?_⟩
251   ext x
252   have hx : x ∈ I ↔ op x ∈ (AlgebraicAnalysis.rightIdealOpEquiv R).symm I := Iff.rfl
253   rw [hx, hab, Submodule.mem_span_pair, Submodule.mem_span_pair]
254   constructor
255   · rintro ⟨c, d, hcd⟩
256     refine ⟨unop (unop c), unop (unop d), ?_⟩
257     have h1 := congrArg unop hcd
258     simpa [MulOpposite.smul_eq_mul_unop, smul_eq_mul] using h1
259   · rintro ⟨c, d, hcd⟩
260     refine ⟨op (op c), op (op d), ?_⟩
261     apply unop_injective
262     simpa [MulOpposite.smul_eq_mul_unop, smul_eq_mul] using hcd
263 
264 /-- Prescribed cyclic generator for left modules over `R`, from the right statement for `Rᵐᵒᵖ`:
265 a finitely generated left module killed by nonzero elements `d` (each `d • m = 0`) satisfies
266 `M = R (x + t h)` for every `x` and nonzero `t`. -/
267 theorem prescribedCyclicLeft_of_property [IsDomain R]
268     (hP : PrescribedCyclicProperty.{u, v} Rᵐᵒᵖ)
269     (M : Type v) [AddCommGroup M] [Module R M] [Module.Finite R M]
270     (hM : ∀ m : M, ∃ d : R, d ≠ 0 ∧ d • m = 0) (x : M) (t : R) (ht : t ≠ 0) :
271     ∃ h : M, Submodule.span R {x + t • h} = ⊤ := by
272   letI := leftAsRight (R := R) M
273   have hfin : Module.Finite Rᵐᵒᵖᵐᵒᵖ M := by
274     obtain ⟨S, hS⟩ := Module.finite_def.mp ‹Module.Finite R M›
275     refine Module.finite_def.mpr ⟨S, ?_⟩
276     have key := span_leftAsRight (R := R) M (S : Set M)
277     rw [hS] at key
278     apply Submodule.eq_top_iff'.mpr
279     intro m
280     have hm : m ∈ (Submodule.span Rᵐᵒᵖᵐᵒᵖ (S : Set M) : Set M) := by
281       rw [show (Submodule.span Rᵐᵒᵖᵐᵒᵖ (S : Set M) : Set M) = ↑(⊤ : Submodule R M) from key]
282       exact Submodule.mem_top
283     exact hm
284   have hG : IsGoldieTorsion Rᵐᵒᵖ M := fun m => by
285     obtain ⟨d, hd, hdm⟩ := hM m
286     exact ⟨op d, Weyl.isRegular_of_ne_zero (fun h => hd (by simpa using h)), hdm⟩
287   obtain ⟨h, hh⟩ := hP M hG x (op t) (fun h => ht (by simpa using h))
288   refine ⟨h, ?_⟩
289   apply Submodule.eq_top_iff'.mpr
290   intro m
291   have key := span_leftAsRight (R := R) M ({x + t • h} : Set M)
292   have hh' : Submodule.span Rᵐᵒᵖᵐᵒᵖ ({x + t • h} : Set M) = ⊤ := hh
293   have hm : m ∈ (Submodule.span Rᵐᵒᵖᵐᵒᵖ ({x + t • h} : Set M) : Set M) := by
294     rw [hh']
295     exact Submodule.mem_top
296   rw [key] at hm
297   exact hm
298 
299 end Opposite
