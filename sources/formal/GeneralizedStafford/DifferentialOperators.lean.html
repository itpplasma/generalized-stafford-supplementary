6 /-!
7 # Rings of differential operators `D_k(A)`
8 
9 Let `k` be a field of characteristic zero and `A` a smooth domain of finite type over `k`.
10 The ring `D = D_k(A)` of finite-order `k`-linear differential operators on `A` satisfies (S),
11 (N), (Z) and (H_omega) on both sides, by AlgebraicAnalysis (`SmoothOperators`): `D` is simple
12 and a domain, it is right and left Noetherian, its right and left deviations are at most
13 `2r` for `r` generators of `A`, and its centre is infinite. The conclusions of Theorem A,
14 Theorem A', Corollary B and Corollaries C.1 and C.2 then follow for `D` and for `Dᵐᵒᵖ` from
15 the abstract theorems of `GeneralizedStafford.Challenge.FilteredTransport`.
16 
17 This file proves:
18 
19 * `GeneralizedStafford.DiffOp.conclusions_right`, `conclusions_left`: all conclusions for `D`
20   (right ideals and modules) and for `Dᵐᵒᵖ` (left ideals and modules).
21 * `affineForm_right`, `affineForm_left`: Theorem A, right form `1 = d U + (B + H a) d V` and
22   left form `1 = U d + V d (B + a H)`, for nonzero `d` and `a`, as statements about elements of
23   `Module.End k A` lying in `D_k(A)`.
24 * `globalStafford`: `1 = d R + F d S` for nonzero `d`.
25 * `twoGenerated_left_ideal`, `prescribedCyclicLeft`: the left-ideal and left-module forms, with
26   the same proofs transported through the opposite ring.
27 -/

133 section Elements

135 variable (k : Type u) (A : Type u) [Field k] [CharZero k] [CommRing A] [IsDomain A]

138 /-- `D_k(A)` as a subalgebra of `Module.End k A`: the operators of finite order. -/
139 abbrev Dsub (k A : Type u) [Field k] [CommRing A] [Algebra k A] :
140     Subalgebra k (Module.End k A) :=
141   AlgebraicAnalysis.DifferentialOperators.algebra (k := k) (R := A)
142 
143 /-- **Theorem A, right form.** For differential operators `d, a, B` on `A` with `d ≠ 0` and
144 `a ≠ 0` there are differential operators `H, U, V` with `1 = d U + (B + H a) d V`. -/
145 theorem affineForm_right (d a B : Module.End k A) (hd : d ∈ Dsub k A) (hd0 : d ≠ 0)
146     (ha : a ∈ Dsub k A) (ha0 : a ≠ 0) (hB : B ∈ Dsub k A) :
147     ∃ H U V : Module.End k A, H ∈ Dsub k A ∧ U ∈ Dsub k A ∧ V ∈ Dsub k A ∧
148       (1 : Module.End k A) = d * U + (B + H * a) * d * V := by
149   have := isDomain k A
150   have hS : IsSimpleRing (DiffOp k A) := isSimpleRing k A
151   have hN : IsNoetherian (DiffOp k A)ᵐᵒᵖ (DiffOp k A) := noetherianRight k A
152   have hZ : Infinite (Subring.center (DiffOp k A)) := infiniteCenterRight k A
153   have hd' : (⟨d, hd⟩ : DiffOp k A) ≠ 0 := fun h => hd0 (by simpa using congrArg Subtype.val h)
154   have hreg : IsRegularElement (⟨d, hd⟩ : DiffOp k A) := Weyl.isRegular_of_ne_zero hd'
155   have ha' : (⟨a, ha⟩ : DiffOp k A) ≠ 0 := fun h => ha0 (by simpa using congrArg Subtype.val h)
156   obtain ⟨H, U, V, h⟩ :=
157     (fixedAnchorProperty_of_omega (R := DiffOp k A) (homegaRight k A)
158       : FixedAnchorProperty (DiffOp k A)) ⟨d, hd⟩ ⟨a, ha⟩ ⟨B, hB⟩ hreg ha'
159   refine ⟨H, U, V, H.2, U.2, V.2, ?_⟩
160   have := congrArg Subtype.val h
161   simpa using this
162 
163 /-- **Theorem A, left form.** For differential operators `d, a, B` on `A` with `d ≠ 0` and
164 `a ≠ 0` there are differential operators `H, U, V` with `1 = U d + V d (B + a H)`. -/
165 theorem affineForm_left (d a B : Module.End k A) (hd : d ∈ Dsub k A) (hd0 : d ≠ 0)
166     (ha : a ∈ Dsub k A) (ha0 : a ≠ 0) (hB : B ∈ Dsub k A) :
167     ∃ H U V : Module.End k A, H ∈ Dsub k A ∧ U ∈ Dsub k A ∧ V ∈ Dsub k A ∧
168       (1 : Module.End k A) = U * d + V * d * (B + a * H) := by
169   have := isDomain k A
170   have hS : IsSimpleRing (DiffOp k A) := isSimpleRing k A
171   have hN : IsNoetherian (DiffOp k A)ᵐᵒᵖᵐᵒᵖ (DiffOp k A)ᵐᵒᵖ := noetherianLeft k A
172   have hZ : Infinite (Subring.center (DiffOp k A)ᵐᵒᵖ) := infiniteCenterLeft k A
173   have hS' : IsSimpleRing (DiffOp k A)ᵐᵒᵖ := inferInstance
174   have hd' : (⟨d, hd⟩ : DiffOp k A) ≠ 0 := fun h => hd0 (by simpa using congrArg Subtype.val h)
175   have ha' : (⟨a, ha⟩ : DiffOp k A) ≠ 0 := fun h => ha0 (by simpa using congrArg Subtype.val h)
176   have hreg : IsRegularElement (op (⟨d, hd⟩ : DiffOp k A)) :=
177     Weyl.isRegular_of_ne_zero (fun h => hd' (by simpa using h))
178   have ha'' : op (⟨a, ha⟩ : DiffOp k A) ≠ 0 := fun h => ha' (by simpa using h)
179   obtain ⟨H, U, V, h⟩ :=
180     (fixedAnchorProperty_of_omega (R := (DiffOp k A)ᵐᵒᵖ) (homegaLeft k A)
181       : FixedAnchorProperty (DiffOp k A)ᵐᵒᵖ)
182       (op ⟨d, hd⟩) (op ⟨a, ha⟩) (op ⟨B, hB⟩) hreg ha''
183   have h' := congrArg (fun z : (DiffOp k A)ᵐᵒᵖ => ((unop z : DiffOp k A) : Module.End k A)) h
184   refine ⟨((unop H : DiffOp k A) : Module.End k A), ((unop U : DiffOp k A) : Module.End k A),
185     ((unop V : DiffOp k A) : Module.End k A), (unop H).2, (unop U).2, (unop V).2, ?_⟩
186   simpa [mul_assoc] using h'
187 
188 /-- **Global Stafford identity.** For a nonzero differential operator `d` on `A` there are
189 differential operators `F, R, S` with `1 = d R + F d S`. This is Theorem A with `a = 1`
190 and `B = 0`. -/
191 theorem globalStafford (d : Module.End k A) (hd : d ∈ Dsub k A) (hd0 : d ≠ 0) :
192     ∃ F R S : Module.End k A, F ∈ Dsub k A ∧ R ∈ Dsub k A ∧ S ∈ Dsub k A ∧
193       (1 : Module.End k A) = d * R + F * d * S := by
194   obtain ⟨H, U, V, hH, hU, hV, h⟩ :=
195     affineForm_right k A d 1 0 hd hd0 (Dsub k A).one_mem one_ne_zero (Dsub k A).zero_mem
196   exact ⟨H, U, V, hH, hU, hV, by simpa using h⟩
197 
198 end Elements

204 variable {R : Type u} [Ring R]

206 /-- The ring homomorphism `Rᵐᵒᵖᵐᵒᵖ → R`, `op (op r) ↦ r`. -/
207 def opOpHom (R : Type u) [Ring R] : Rᵐᵒᵖᵐᵒᵖ →+* R where
208   toFun x := unop (unop x)
209   map_one' := rfl
210   map_mul' _ _ := rfl
211   map_zero' := rfl
212   map_add' _ _ := rfl
213 
214 /-- A left `R`-module as a module over `Rᵐᵒᵖᵐᵒᵖ`, with `op (op r) • m = r • m`. A left module
215 is a right module over the opposite ring. This is a local construction, not an instance. -/
216 abbrev leftAsRight (M : Type v) [AddCommGroup M] [Module R M] : Module Rᵐᵒᵖᵐᵒᵖ M :=
217   Module.compHom M (opOpHom R)
218 
219 /-- The span over `Rᵐᵒᵖᵐᵒᵖ` of a set is its span over `R`. -/
220 theorem span_leftAsRight (M : Type v) [AddCommGroup M] [Module R M] (S : Set M) :
221     (@Submodule.span Rᵐᵒᵖᵐᵒᵖ M _ _ (leftAsRight M) S : Set M) = Submodule.span R S := by
222   letI := leftAsRight (R := R) M
223   apply le_antisymm
224   · let N : Submodule Rᵐᵒᵖᵐᵒᵖ M :=
225       { carrier := (Submodule.span R S : Set M)
226         add_mem' := fun hx hy => Submodule.add_mem _ hx hy
227         zero_mem' := Submodule.zero_mem _
228         smul_mem' := fun c _ hm => Submodule.smul_mem _ (opOpHom R c) hm }
229     have hle : Submodule.span Rᵐᵒᵖᵐᵒᵖ S ≤ N :=
230       Submodule.span_le.mpr fun s hs => Submodule.subset_span hs
231     exact fun m hm => hle hm
232   · let N : Submodule R M :=
233       { carrier := (Submodule.span Rᵐᵒᵖᵐᵒᵖ S : Set M)
234         add_mem' := fun hx hy => Submodule.add_mem _ hx hy
235         zero_mem' := Submodule.zero_mem _
236         smul_mem' := fun r _ hm => Submodule.smul_mem _ (op (op r)) hm }
237     have hle : Submodule.span R S ≤ N :=
238       Submodule.span_le.mpr fun s hs => Submodule.subset_span hs
239     exact fun m hm => hle hm
240 
241 /-- Left ideals of `R` are two-generated, from the two-generation of right ideals of `Rᵐᵒᵖ`:
242 a left ideal `I` is the right ideal `{op x | x ∈ I}` of `Rᵐᵒᵖ`. -/
243 theorem twoGenerated_left_of_op
244     (hR : ∀ J : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ, ∃ a b : Rᵐᵒᵖ, J = Submodule.span Rᵐᵒᵖᵐᵒᵖ {a, b})
245     (I : Submodule R R) : ∃ a b : R, I = Submodule.span R {a, b} := by
246   obtain ⟨a, b, hab⟩ := hR ((AlgebraicAnalysis.rightIdealOpEquiv R).symm I)
247   refine ⟨unop a, unop b, ?_⟩
248   ext x
249   have hx : x ∈ I ↔ op x ∈ (AlgebraicAnalysis.rightIdealOpEquiv R).symm I := Iff.rfl
250   rw [hx, hab, Submodule.mem_span_pair, Submodule.mem_span_pair]
251   constructor
252   · rintro ⟨c, d, hcd⟩
253     refine ⟨unop (unop c), unop (unop d), ?_⟩
254     have h1 := congrArg unop hcd
255     simpa [MulOpposite.smul_eq_mul_unop, smul_eq_mul] using h1
256   · rintro ⟨c, d, hcd⟩
257     refine ⟨op (op c), op (op d), ?_⟩
258     apply unop_injective
259     simpa [MulOpposite.smul_eq_mul_unop, smul_eq_mul] using hcd
260 
261 /-- Prescribed cyclic generator for left modules over `R`, from the right statement for `Rᵐᵒᵖ`:
262 a finitely generated left module killed by nonzero elements `d` (each `d • m = 0`) satisfies
263 `M = R (x + t h)` for every `x` and nonzero `t`. -/
264 theorem prescribedCyclicLeft_of_property [IsDomain R]
265     (hP : PrescribedCyclicProperty.{u, v} Rᵐᵒᵖ)
266     (M : Type v) [AddCommGroup M] [Module R M] [Module.Finite R M]
267     (hM : ∀ m : M, ∃ d : R, d ≠ 0 ∧ d • m = 0) (x : M) (t : R) (ht : t ≠ 0) :
268     ∃ h : M, Submodule.span R {x + t • h} = ⊤ := by
269   letI := leftAsRight (R := R) M
270   have hfin : Module.Finite Rᵐᵒᵖᵐᵒᵖ M := by
271     obtain ⟨S, hS⟩ := Module.finite_def.mp ‹Module.Finite R M›
272     refine Module.finite_def.mpr ⟨S, ?_⟩
273     have key := span_leftAsRight (R := R) M (S : Set M)
274     rw [hS] at key
275     apply Submodule.eq_top_iff'.mpr
276     intro m
277     have hm : m ∈ (Submodule.span Rᵐᵒᵖᵐᵒᵖ (S : Set M) : Set M) := by
278       rw [show (Submodule.span Rᵐᵒᵖᵐᵒᵖ (S : Set M) : Set M) = ↑(⊤ : Submodule R M) from key]
279       exact Submodule.mem_top
280     exact hm
281   have hG : IsGoldieTorsion Rᵐᵒᵖ M := fun m => by
282     obtain ⟨d, hd, hdm⟩ := hM m
283     exact ⟨op d, Weyl.isRegular_of_ne_zero (fun h => hd (by simpa using h)), hdm⟩
284   obtain ⟨h, hh⟩ := hP M hG x (op t) (fun h => ht (by simpa using h))
285   refine ⟨h, ?_⟩
286   apply Submodule.eq_top_iff'.mpr
287   intro m
288   have key := span_leftAsRight (R := R) M ({x + t • h} : Set M)
289   have hh' : Submodule.span Rᵐᵒᵖᵐᵒᵖ ({x + t • h} : Set M) = ⊤ := hh
290   have hm : m ∈ (Submodule.span Rᵐᵒᵖᵐᵒᵖ ({x + t • h} : Set M) : Set M) := by
291     rw [hh']
292     exact Submodule.mem_top
293   rw [key] at hm
294   exact hm
295 
296 end Opposite
