10 # Germs and source selection
11 
12 The Gabriel quotient `F_e` and its germs are represented concretely by the
13 germ calculus of `AlgebraicAnalysis.Module.Germ`: a germ of level `e` of a
14 module `X` is an element of `Section R e X = Hom Rᵐᵒᵖ e R X`, i.e. a partial
15 map from a submodule of `R` with small cokernel, modulo maps that agree on a
16 dense submodule; `Hom.Epic w` is "`w` is epic modulo `Kdim < e`".
17 
18 * **Central scalars.** A central `λ ∈ k` acts on sections of a critical
19   module through the centre of its Schur division ring
20   (`section_central_scalar`).
21 * **Unit germs.** `q_J r` is epic iff `Kdim R/(J + rR) < e`
22   (`smul_mkQ_epic_iff`).
23 * **Epic criterion, pencils and pencil stability.** An epic
24   germ `v` stays epic along `v + u λ` for all but finitely many `λ`
25   (`germ_pencil_finite`).
26 * **Escape and maximal rank.** These are
27   `AlgebraicAnalysis.Goldie.IsGoldieTorsion.exists_smul_mul_not_mem` and
28   `IsGoldieTorsion.exists_injective_right_mul`; together with the pencil lemma
29   they give finite affine source selection in Goldie-torsion bimodules
30   (`exists_affine_source_of_goldie`).
31 * **Source selection.** A *test* (`GermTest`) is
32   `(e, V, q, z, c)` with `q` an epic germ and `c` regular; `f` satisfies it if
33   `z + q f c` is epic. Every finite family of tests is satisfied by some
34   `f = x + H t` (`GermTest.exists_affine_satisfied`), and a satisfied test stays
35   satisfied along `f + λ g` for almost all `λ` (`GermTest.pencil_finite`).
36 -/

52 /-- A central field `k` of `R`, viewed in the centre of `Rᵐᵒᵖ`. -/
53 def oppositeCentralField (k : Type*) [Field k] [Algebra k R] : k →+* Subring.center Rᵐᵒᵖ :=
54   (algebraMap k Rᵐᵒᵖ).codRestrict _ (fun c =>
55     Subring.mem_center_iff.mpr fun r => (Algebra.commutes c r).symm)
56 
57 /-- Right multiplication of a section by a central scalar `λ` is the left
58 action of the Schur endomorphism `φ_λ` , also for germs defined only on a dense submodule. -/
59 theorem section_central_scalar (a : Subring.center Rᵐᵒᵖ) (w : Section R e C) :
60     a.val • w = End.centralMap (R := Rᵐᵒᵖ) (e := e) (C := C) a • w := by
61   have he : Hom.reducedMap (e := e) (End.centralScalar (C := C) a) =
62       End.centralScalar (C := C ⧸ smallRadical Rᵐᵒᵖ e C) a := by
63     apply LinearMap.ext
64     intro x
65     obtain ⟨x, rfl⟩ := Submodule.mkQ_surjective (smallRadical Rᵐᵒᵖ e C) x
66     rfl
67   have hs : Hom.reducedMap (e := e) (Section.leftMul (unop a.val)) =
68       End.centralScalar (C := R ⧸ smallRadical Rᵐᵒᵖ e R) a := by
69     apply LinearMap.ext
70     intro x
71     obtain ⟨x, rfl⟩ := Submodule.mkQ_surjective (smallRadical Rᵐᵒᵖ e R) x
72     change (smallRadical Rᵐᵒᵖ e R).mkQ (unop a.val * x) =
73       (smallRadical Rᵐᵒᵖ e R).mkQ (x * unop a.val)
74     congr 1
75     exact congrArg unop (Subring.mem_center_iff.mp a.property (op x))
76   change PartialHom.comp w (PartialHom.ofLinearMap _) =
77     PartialHom.comp (PartialHom.ofLinearMap _) w
78   rw [he, hs]
79   induction w using Quotient.inductionOn with | _ w =>
80   apply Quotient.sound
81   apply Representative.rel_of_eq
82   intro x hx hy
83   have hm := w.val.toFun.map_smul a.val (⟨x, w.pullback_le _ hy⟩ : w.val.domain)
84   exact hm
85 
86 end CentralScalars

94 /-- The germ of `R → R/J, r ↦ x r + J` is epic iff `Kdim R/(J + xR) < e`. -/
95 theorem quotient_section_epic_iff (J : Submodule Rᵐᵒᵖ R) (x : R) :
96     Hom.Epic (Hom.ofLinearMap (e := e) (J.mkQ.comp (leftMulLinear x))) ↔
97       Module.Small Rᵐᵒᵖ e (R ⧸ (J ⊔ principalRightIdeal x)) := by
98   rw [Hom.epic_ofLinearMap_iff]
99   have hr : (J.mkQ.comp (leftMulLinear x)).range =
100       (principalRightIdeal x).map J.mkQ := by
101     rw [LinearMap.range_comp]
102     rfl
103   rw [hr]
104   exact Module.small_linearEquiv_iff e
105     (Submodule.quotientQuotientEquivQuotientSup J (principalRightIdeal x))
106 
107 /-- **Unit germs.** For the unit germ `q_J` of `R/J` and `x ∈ R`,
108 the germ `q_J x` is epic iff `Kdim R/(J + xR) < e`. -/
109 theorem smul_mkQ_epic_iff (J : Submodule Rᵐᵒᵖ R) (x : R) :
110     Hom.Epic (op x • Hom.ofLinearMap (e := e) J.mkQ) ↔
111       Module.Small Rᵐᵒᵖ e (R ⧸ (J ⊔ principalRightIdeal x)) := by
112   rw [← quotient_section_epic_iff]
113   have he : op x • Hom.ofLinearMap (e := e) J.mkQ =
114       Hom.ofLinearMap (e := e) (J.mkQ.comp (leftMulLinear x)) := by
115     change Hom.comp (Hom.ofLinearMap J.mkQ) (Hom.ofLinearMap (Section.leftMul x)) = _
116     rw [← Hom.ofLinearMap_comp]
117     rfl
118   rw [he]
119 
120 end UnitGerms

129 /-- **Pencil stability.** If `v` is an epic germ of a module `M` with
130 `Kdim M ≤ e` and `u` is any germ, then `v + u λ` is epic for all but finitely
131 many `λ ∈ k`. Proof: by the epic criterion it suffices to test the
132 finitely many critical top factors `C` of a critical filtration; on each,
133 `h ↦ h(v + u λ) = P_v h + φ_λ P_u h` is a central pencil with `P_v` injective,
134 so it is injective for almost all `λ`. -/
135 theorem germ_pencil_finite (hM : Module.DevLE Rᵐᵒᵖ e M)
136     (F : Module.CriticalFiltration Rᵐᵒᵖ e M) (v u : Section R e M) (hv : Hom.Epic v) :
137     {c : k | ¬ Hom.Epic (v + op (algebraMap k R c) • u)}.Finite := by
138   classical
139   have hfin : (⋃ i : F.TopIndex,
140       {c : k | ¬ Function.Injective (fun h : Hom Rᵐᵒᵖ e M (F.TopFactor i) =>
141         Hom.comp h (v + op (algebraMap k R c) • u))}).Finite := by
142     refine Set.finite_iUnion fun i => ?_
143     let := End.divisionRing (F.topFactor_critical i)
144     let := Hom.finite_of_devLE hM (F.topFactor_critical i)
145     let κ : k →+* End Rᵐᵒᵖ e (F.TopFactor i) :=
146       End.centralMap.comp (oppositeCentralField (R := R) k)
147     have hκ : ∀ c x, κ c * x = x * κ c := fun c x =>
148       (End.centralMap_commute (oppositeCentralField (R := R) k c) x).eq
149     refine (centralPencil_injective_finite κ hκ (Hom.precompLinear v)
150       (Hom.precompLinear u) (fun h h' he => hv.right_cancel h h' he)).subset ?_
151     intro c hc hinj
152     apply hc
153     have heq : (fun h : Hom Rᵐᵒᵖ e M (F.TopFactor i) =>
154         Hom.comp h (v + op (algebraMap k R c) • u)) =
155         centralPencil κ hκ (Hom.precompLinear v) (Hom.precompLinear u) c := by
156       funext h
157       rw [Hom.comp_add]
158       change Hom.comp h v + Hom.comp h (Hom.comp u (Hom.ofLinearMap
159           (Section.leftMul (algebraMap k R c)))) = Hom.comp h v + κ c • Hom.comp h u
160       rw [← Hom.comp_assoc]
161       exact congrArg (Hom.comp h v + ·)
162         (section_central_scalar (oppositeCentralField (R := R) k c) (Hom.comp h u))
163     rw [heq]
164     exact hinj
165   refine hfin.subset fun c hc => ?_
166   by_contra hn
167   apply hc
168   refine (Hom.epic_iff_injective_comp_topFactors hM F _).mpr fun i => ?_
169   by_contra hi
170   exact hn (Set.mem_iUnion.mpr ⟨i, hi⟩)
171 
172 end Stability

187 /-- Finite affine selection in Goldie-torsion bimodules, direction form
188 (Steps 1 and 2 below). Let `Pᵢ` be injective maps from finite-dimensional
189 spaces into Goldie-torsion bimodules `Wᵢ`, `Zᵢ` arbitrary, `a ≠ 0` and `bᵢ`
190 regular. There is a direction `H` such that `d ↦ Zᵢ d + Pᵢ(d) (B + σ(c) H a) bᵢ`
191 is injective for all `i` and all but finitely many central scalars `c`.
192 Step 1 combines maximal rank for each `i` by ordinary pencils
193 (AlgebraicAnalysis `exists_simultaneously_injective`); Step 2 is a reversed
194 pencil (AlgebraicAnalysis `reversed_centralPencil_injective_finite`). -/
195 theorem exists_affine_direction_of_goldie
196     (hW : ∀ i, IsGoldieTorsion R (Wᵢ i))
197     (σ : k →+* R) (κ : ∀ i, k →+* Eᵢ i)
198     (hκ : ∀ i c a, κ i c * a = a * κ i c)
199     (hscalar : ∀ i c (w : Wᵢ i), op (σ c) • w = κ i c • w)
200     (P Z : ∀ i, Dᵢ i →ₗ[Eᵢ i] Wᵢ i) (hP : ∀ i, Function.Injective (P i))
201     (s : Finset ι) (b : ι → R) (hb : ∀ i ∈ s, b i ∈ R⁰)
202     {a : R} (ha : a ≠ 0) (B : R) :
203     ∃ H : R, {c : k | ∃ i ∈ s, ¬ Function.Injective
204       (fun d => Z i d + op ((B + σ c * H * a) * b i) • P i d)}.Finite := by
205   classical
206   let F (i : ι) (r : R) : Dᵢ i →ₗ[Eᵢ i] Wᵢ i :=
207     (rightMulLinear (E := Eᵢ i) (r * (a * b i))).comp (P i)
208   have hF (i : ι) (r t : R) (c : k) (d : Dᵢ i) :
209       F i (r + σ c * t) d = F i r d + κ i c • F i t d := by
210     change op ((r + σ c * t) * (a * b i)) • P i d =
211       op (r * (a * b i)) • P i d + κ i c • (op (t * (a * b i)) • P i d)
212     rw [add_mul, op_add, add_smul, mul_assoc]
213     congr 1
214     rw [op_mul, mul_smul, hscalar, smul_comm]
215   obtain ⟨H, hH⟩ := exists_simultaneously_injective σ κ hκ F hF s (fun i hi =>
216     (hW i).exists_injective_right_mul σ (κ i) (hκ i) (hscalar i)
217       (P i) (hP i) (fun h0 => ha ((hb i hi).2 a h0)))
218   let A (i : ι) : Dᵢ i →ₗ[Eᵢ i] Wᵢ i :=
219     Z i + (rightMulLinear (E := Eᵢ i) (B * b i)).comp (P i)
220   have hfinite : (⋃ i ∈ s,
221       {c | ¬ Function.Injective (centralPencil (κ i) (hκ i) (A i) (F i H) c)}).Finite :=
222     s.finite_toSet.biUnion fun i hi =>
223       reversed_centralPencil_injective_finite (κ i) (hκ i) _ _ (hH i hi)
224   refine ⟨H, hfinite.subset ?_⟩
225   rintro c ⟨i, hi, hbad⟩
226   refine Set.mem_iUnion.mpr ⟨i, Set.mem_iUnion.mpr ⟨hi, ?_⟩⟩
227   have heq : (fun d => Z i d + op ((B + σ c * H * a) * b i) • P i d) =
228       centralPencil (κ i) (hκ i) (A i) (F i H) c := by
229     funext d
230     change Z i d + op ((B + σ c * H * a) * b i) • P i d =
231       (Z i d + op (B * b i) • P i d) + κ i c • (op (H * (a * b i)) • P i d)
232     have hscaled : op (σ c * (H * (a * b i))) • P i d =
233         κ i c • (op (H * (a * b i)) • P i d) := by
234       rw [op_mul, mul_smul, hscalar, smul_comm]
235     rw [add_mul, op_add, add_smul, mul_assoc, mul_assoc, hscaled]
236     abel
237   change ¬ Function.Injective (centralPencil (κ i) (hκ i) (A i) (F i H) c)
238   rwa [← heq]
239 
240 /-- Finite affine selection in Goldie-torsion bimodules: some `H` makes every
241 `d ↦ Zᵢ d + Pᵢ(d) (B + H a) bᵢ` injective (take `H := σ(c) H₀` for a good
242 central `c`). -/
243 theorem exists_affine_source_of_goldie
244     (hW : ∀ i, IsGoldieTorsion R (Wᵢ i))
245     (σ : k →+* R) (κ : ∀ i, k →+* Eᵢ i)
246     (hκ : ∀ i c a, κ i c * a = a * κ i c)
247     (hscalar : ∀ i c (w : Wᵢ i), op (σ c) • w = κ i c • w)
248     (P Z : ∀ i, Dᵢ i →ₗ[Eᵢ i] Wᵢ i) (hP : ∀ i, Function.Injective (P i))
249     (s : Finset ι) (b : ι → R) (hb : ∀ i ∈ s, b i ∈ R⁰)
250     {a : R} (ha : a ≠ 0) (B : R) :
251     ∃ H : R, ∀ i ∈ s, Function.Injective
252       (fun d => Z i d + op ((B + H * a) * b i) • P i d) := by
253   obtain ⟨H, hH⟩ := exists_affine_direction_of_goldie hW σ κ hκ hscalar P Z hP s b hb ha B
254   obtain ⟨c, hc⟩ := hH.exists_notMem
255   refine ⟨σ c * H, ?_⟩
256   intro i hi
257   by_contra hbad
258   exact hc ⟨i, hi, hbad⟩
259 
260 end FiniteSelection

264 /-- A **test** `(e, V, q, z, c)`: a Noetherian
265 Goldie-torsion module `V` (the target) with `Kdim V ≤ e`, an epic germ `q`, an
266 arbitrary germ `z` (the offset) and a regular tail `c`. -/
267 structure GermTest (R : Type u) [Ring R] where
268   /-- The level `e`. -/
269   level : ℕ
270   /-- The target module `V`. -/
271   Target : Type u
272   [addCommGroup : AddCommGroup Target]
273   [module : Module Rᵐᵒᵖ Target]
274   [noetherian : IsNoetherian Rᵐᵒᵖ Target]
275   /-- `Kdim V ≤ e`. -/
276   bounded : Module.DevLE Rᵐᵒᵖ level Target
277   /-- `V` is Goldie-torsion. -/
278   torsion : IsGoldieTorsion R Target
279   /-- The epic germ `q`. -/
280   q : Hom Rᵐᵒᵖ level R Target
281   /-- The offset germ `z`. -/
282   offset : Hom Rᵐᵒᵖ level R Target
283   /-- `q` is epic. -/
284   epic : Hom.Epic q
285   /-- The tail `c`. -/
286   tail : R
287   /-- The tail `c` is regular. -/
288   tail_reg : tail ∈ R⁰
289 
290 attribute [instance] GermTest.addCommGroup GermTest.module GermTest.noetherian

296 /-- An element `f` **satisfies** the test `(e, V, q, z, c)` if `z + q f c` is
297 epic. -/
298 def GermTest.Satisfied (τ : GermTest R) (f : R) : Prop :=
299   Hom.Epic (τ.offset + op (f * τ.tail) • τ.q)

311 /-- Source selection for finitely many critical germ tests: given epic germs
312 `qᵢ`, offsets `zᵢ`, regular tails `bᵢ`, `a ≠ 0` and `B`, some `H` makes every
313 `h ↦ h (zᵢ + qᵢ (B + H a) bᵢ)` injective on the Schur space of the
314 Goldie-torsion critical module `Cᵢ`. -/
315 theorem exists_affine_germ_tests
316     {ι : Type*} [Fintype ι] (e : ι → ℕ) (M C : ι → Type u)
317     [∀ i, AddCommGroup (M i)] [∀ i, Module Rᵐᵒᵖ (M i)]
318     [∀ i, IsNoetherian Rᵐᵒᵖ (M i)]
319     [∀ i, AddCommGroup (C i)] [∀ i, Module Rᵐᵒᵖ (C i)]
320     [∀ i, IsNoetherian Rᵐᵒᵖ (C i)]
321     (hM : ∀ i, Module.DevLE Rᵐᵒᵖ (e i) (M i))
322     (hC : ∀ i, Module.Critical Rᵐᵒᵖ (e i) (C i))
323     (ht : ∀ i, IsGoldieTorsion R (C i))
324     (q z : ∀ i, Hom Rᵐᵒᵖ (e i) R (M i)) (hq : ∀ i, Hom.Epic (q i))
325     (b : ι → R) (hb : ∀ i, b i ∈ R⁰) {a : R} (ha : a ≠ 0) (B : R) :
326     ∃ H : R, ∀ i, Function.Injective
327       (fun h : Hom Rᵐᵒᵖ (e i) (M i) (C i) =>
328         Hom.comp h (z i + op ((B + H * a) * b i) • q i)) := by
329   classical
330   let : ∀ i, DivisionRing (End Rᵐᵒᵖ (e i) (C i)) := fun i => End.divisionRing (hC i)
331   let : ∀ i, Module.Finite (End Rᵐᵒᵖ (e i) (C i))
332       (Hom Rᵐᵒᵖ (e i) (M i) (C i)) := fun i => Hom.finite_of_devLE (hM i) (hC i)
333   let κ (i : ι) : k →+* End Rᵐᵒᵖ (e i) (C i) :=
334     End.centralMap.comp (oppositeCentralField (R := R) k)
335   let P (i : ι) := Hom.precompLinear (C := C i) (q i)
336   let Z (i : ι) := Hom.precompLinear (C := C i) (z i)
337   obtain ⟨H, hH⟩ := exists_affine_source_of_goldie
338     (fun i => isGoldieTorsion_section (hC i) (ht i))
339     (algebraMap k R) κ
340     (fun i c x => (End.centralMap_commute (oppositeCentralField (R := R) k c) x).eq)
341     (fun i c w => section_central_scalar (oppositeCentralField (R := R) k c) w)
342     P Z (fun i => (hq i).right_cancel)
343     Finset.univ b (fun i _ => hb i) ha B
344   refine ⟨H, fun i => ?_⟩
345   have hi := hH i (Finset.mem_univ i)
346   convert hi using 1
347   funext h
348   exact Hom.comp_add h (z i) (op ((B + H * a) * b i) • q i) |>.trans
349     (congrArg (Hom.comp h (z i) + ·)
350       (Hom.comp_assoc h (q i) (Hom.ofLinearMap (Section.leftMul _))).symm)

353 /-- **Source selection, epic form.** For finitely many
354 Goldie-torsion modules `Mᵢ` with `Kdim Mᵢ ≤ eᵢ`, epic germs `qᵢ`, offsets `zᵢ`
355 and regular tails `bᵢ`, and for `a ≠ 0` and `B`, some `H` makes every germ
356 `zᵢ + qᵢ (B + H a) bᵢ` epic. By the epic criterion it suffices to
357 satisfy the critical tests of the finitely many top factors. -/
358 theorem exists_affine_epic_tests
359     {ι : Type*} [Fintype ι] (e : ι → ℕ) (M : ι → Type u)
360     [∀ i, AddCommGroup (M i)] [∀ i, Module Rᵐᵒᵖ (M i)]
361     [∀ i, IsNoetherian Rᵐᵒᵖ (M i)]
362     (hM : ∀ i, Module.DevLE Rᵐᵒᵖ (e i) (M i))
363     (ht : ∀ i, IsGoldieTorsion R (M i))
364     (q z : ∀ i, Hom Rᵐᵒᵖ (e i) R (M i)) (hq : ∀ i, Hom.Epic (q i))
365     (b : ι → R) (hb : ∀ i, b i ∈ R⁰) {a : R} (ha : a ≠ 0) (B : R) :
366     ∃ H : R, ∀ i, Hom.Epic (z i + op ((B + H * a) * b i) • q i) := by
367   classical
368   let F (i : ι) : Module.CriticalFiltration Rᵐᵒᵖ (e i) (M i) :=
369     Classical.choice (hM i).nonempty_criticalFiltration
370   let J := (i : ι) × (F i).TopIndex
371   let C (j : J) := (F j.1).TopFactor j.2
372   have hC (j : J) : Module.Critical Rᵐᵒᵖ (e j.1) (C j) :=
373     (F j.1).topFactor_critical j.2
374   have htC (j : J) : IsGoldieTorsion R (C j) := by
375     let U := (F j.1).toRelSeries j.2.val.succ
376     let V := ((F j.1).toRelSeries j.2.val.castSucc).comap U.subtype
377     exact IsGoldieTorsion.of_surjective V.mkQ V.mkQ_surjective
378       (IsGoldieTorsion.of_injective U.subtype Subtype.val_injective (ht j.1))
379   obtain ⟨H, hH⟩ := exists_affine_germ_tests (k := k)
380     (fun j : J => e j.1) (fun j => M j.1) C
381     (fun j => hM j.1) hC htC (fun j => q j.1) (fun j => z j.1)
382     (fun j => hq j.1) (fun j => b j.1) (fun j => hb j.1) ha B
383   exact ⟨H, fun i => (Hom.epic_iff_injective_comp_topFactors (hM i) (F i) _).mpr
384     (fun j => hH ⟨i, j⟩)⟩

394 /-- For a central `λ`: `z + q (f + λ g) c = (z + q f c) + (q g c) λ`. -/
395 theorem test_pencil_identity {M : Type u} [AddCommGroup M] [Module Rᵐᵒᵖ M]
396     {e : ℕ} (q z : Section R e M) (f g b lam : R)
397     (hlam : ∀ x : R, lam * x = x * lam) :
398     z + op ((f + lam * g) * b) • q =
399       (z + op (f * b) • q) + op lam • (op (g * b) • q) := by
400   have he : (f + lam * g) * b = f * b + (g * b) * lam := by
401     rw [add_mul, mul_assoc, hlam]
402   rw [he, op_add, add_smul, ← add_assoc, Section.right_action_assoc]
403 
404 /-- **Stability for tests.** If `f` satisfies a test, then so does
405 `f + λ g` for all but finitely many central `λ ∈ k`. -/
406 theorem GermTest.pencil_finite {k : Type*} [Field k] [Algebra k R]
407     (τ : GermTest R) {f : R} (hf : τ.Satisfied f) (g : R) :
408     {c : k | ¬ τ.Satisfied (f + algebraMap k R c * g)}.Finite := by
409   classical
410   let F := Classical.choice τ.bounded.nonempty_criticalFiltration
411   have hfin := germ_pencil_finite (k := k) τ.bounded F
412     (τ.offset + op (f * τ.tail) • τ.q) (op (g * τ.tail) • τ.q) hf
413   simpa only [GermTest.Satisfied,
414     test_pencil_identity _ _ _ _ _ _ (fun x => (Algebra.commutes _ x))] using hfin

416 /-- **Selection for tests.** Assume (S), (N), (Z). For every `x`,
417 every `t ≠ 0` and every finite family of tests, some `f = x + H t` satisfies
418 all of them. -/
419 theorem GermTest.exists_affine_satisfied [IsSimpleRing R] [IsNoetherian Rᵐᵒᵖ R]
420     (k : Type*) [Field k] [Infinite k] [Algebra k R]
421     {ι : Type*} [Fintype ι] (tests : ι → GermTest R) (x : R) {t : R} (ht : t ≠ 0) :
422     ∃ H : R, ∀ i, (tests i).Satisfied (x + H * t) :=
423   exists_affine_epic_tests (k := k)
424     (fun i => (tests i).level) (fun i => (tests i).Target)
425     (fun i => (tests i).bounded) (fun i => (tests i).torsion)
426     (fun i => (tests i).q) (fun i => (tests i).offset) (fun i => (tests i).epic)
427     (fun i => (tests i).tail) (fun i => (tests i).tail_reg) ht x
428 
429 /-- The **unit test** `(e, R/J, q_J, 0, 1)` of a Goldie-torsion quotient `R/J`
430 with `Kdim R/J ≤ e`. -/
431 def unitTest [IsNoetherian Rᵐᵒᵖ R] (J : Submodule Rᵐᵒᵖ R) (e : ℕ)
432     (hJ : Module.DevLE Rᵐᵒᵖ e (R ⧸ J)) (hT : IsGoldieTorsion R (R ⧸ J)) :
433     GermTest R where
434   level := e
435   Target := R ⧸ J
436   bounded := hJ
437   torsion := hT
438   q := Hom.ofLinearMap J.mkQ
439   offset := 0
440   epic := Hom.epic_ofLinearMap_of_surjective _ (Submodule.mkQ_surjective _)
441   tail := 1
442   tail_reg := Submonoid.one_mem _
443 
444 /-- `f` satisfies the unit test of `R/J` iff `Kdim R/(J + fR) < e`. -/
445 theorem unitTest_satisfied_iff [IsNoetherian Rᵐᵒᵖ R] (J : Submodule Rᵐᵒᵖ R) (e : ℕ)
446     (hJ : Module.DevLE Rᵐᵒᵖ e (R ⧸ J)) (hT : IsGoldieTorsion R (R ⧸ J)) (f : R) :
447     (unitTest J e hJ hT).Satisfied f ↔
448       Module.Small Rᵐᵒᵖ e (R ⧸ (J ⊔ principalRightIdeal f)) := by
449   change Hom.Epic (0 + op (f * 1) • Hom.ofLinearMap (e := e) J.mkQ) ↔ _
450   rw [zero_add, mul_one]
451   exact smul_mkQ_epic_iff J f
