8 # Filtered rings with commutative Noetherian associated graded ring
9 
10 Proposition `prop:filtered` of the manuscript. Let `k` be an infinite field and
11 `R` a simple `k`-algebra with an exhaustive filtration `φ` and a symbol
12 `S : φ.GradedSymbol 𝒜` identifying `gr R` with a commutative graded algebra
13 `G = ⨁ 𝒜ₙ` of Krull dimension at most `m`, with `G` Noetherian. Then `R` and
14 `Rᵐᵒᵖ` satisfy (S), (N), (Z) and (H_ω), and hence Theorem A, Theorem A',
15 Corollary B, Corollary C.1 and C.2 hold for `R` and for `Rᵐᵒᵖ`, that is
16 on both sides.
17 
18 * `GeneralizedStafford.right_side`: right Noetherianity and right deviation
19   bounds from the symbol (AlgebraicAnalysis' `GradedSymbol` lemmas).
20 * `GeneralizedStafford.filtered_hypotheses`: (N), (Z) and `Kdim ≤ m` on both
21   sides, using the opposite filtration and the opposite symbol for `Rᵐᵒᵖ`.
22 * `GeneralizedStafford.filtered_theoremA` and the following theorems: the
23   conclusions for each side, reusing the abstract theorems of
24   `TheoremA`, `CorollaryB` and `Modules`.
25 -/

43 /-- **Right side of a symbol.** Let `S` identify `gr R` with a commutative
44 Noetherian graded algebra `G` of Krull dimension at most `m`. Then `R` is right
45 Noetherian, and the right ideals of `R` have deviation at most `m` and at most
46 `ω`. -/
47 theorem right_side [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
48     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
49     IsNoetherian Rᵐᵒᵖ R ∧ Module.DevLE Rᵐᵒᵖ m R ∧ Module.DevLEOmega Rᵐᵒᵖ R :=
50   ⟨S.isNoetherian_right, S.devLE_right m hm,
51     S.devLEOmega_right (ringKrullDim_ne_top_of_le hm)⟩
52 
53 /-- **Filtered rings: (N), (Z) and `Kdim ≤ m` on both sides.** Let `R` be a
54 simple `k`-algebra, `k` an infinite field, with a symbol `S` onto a commutative
55 Noetherian graded algebra of Krull dimension at most `m`. Then `R` is right
56 Noetherian with infinite centre and right deviation at most `m` and at most
57 `ω`; the same holds for `Rᵐᵒᵖ`, using the opposite filtration and the opposite
58 symbol. -/
59 theorem filtered_hypotheses [Infinite k] [IsSimpleRing R] [IsNoetherianRing G]
60     [GradedAlgebra 𝒜] {φ : RingFiltration k R} (S : φ.GradedSymbol 𝒜) {m : ℕ}
61     (hm : ringKrullDim G ≤ m) :
62     (IsNoetherian Rᵐᵒᵖ R ∧ Infinite (Subring.center R) ∧
63         Module.DevLE Rᵐᵒᵖ m R ∧ Module.DevLEOmega Rᵐᵒᵖ R) ∧
64       (IsNoetherian Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ ∧ Infinite (Subring.center Rᵐᵒᵖ) ∧
65         Module.DevLE Rᵐᵒᵖᵐᵒᵖ m Rᵐᵒᵖ ∧ Module.DevLEOmega Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ) :=
66   ⟨⟨(right_side S hm).1, infinite_subring_center k R, (right_side S hm).2⟩,
67     ⟨(right_side S.opposite hm).1, infinite_subring_center k Rᵐᵒᵖ,
68       (right_side S.opposite hm).2⟩⟩
69 
70 end Hypotheses

79 /-- Under (H_all), every cyclic submodule `uR` killed on the right by a regular
80 element has finite deviation: `uR` is a quotient of `R / sR`. -/
81 theorem devLE_span_of_torsion {M : Type v} [AddCommGroup M] [Module Rᵐᵒᵖ M]
82     (hall : PrincipalQuotientsFinite R) {u : M} (hu : ∃ s ∈ R⁰, op s • u = 0) :
83     ∃ N, Module.DevLE Rᵐᵒᵖ N (Submodule.span Rᵐᵒᵖ {u}) := by
84   obtain ⟨s, hs, hsu⟩ := hu
85   obtain ⟨N, hN⟩ := hall s hs
86   exact ⟨N, (devLE_quotient_ker_cyclicMap_iff N u).mp
87     (devLE_quotient_of_le hN (principalRightIdeal_le (X := LinearMap.ker (cyclicMap u)) hsu))⟩

89 /-- **Theorem A, both sides.** Let `R` be as in `filtered_hypotheses`. For every
90 regular `d`, every nonzero `a` and every `B` there are `H, U, V` with
91 `1 = d U + (B + H a) d V`, in `R` and in `Rᵐᵒᵖ` (the left form
92 `1 = U d + V d (B + a H)` of `R`). -/
93 theorem filtered_theoremA [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
94     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
95     (∀ d a B : R, d ∈ R⁰ → a ≠ 0 → ∃ H U V : R, 1 = d * U + (B + H * a) * d * V) ∧
96       (∀ d a B : Rᵐᵒᵖ, d ∈ Rᵐᵒᵖ⁰ → a ≠ 0 →
97         ∃ H U V : Rᵐᵒᵖ, 1 = d * U + (B + H * a) * d * V) := by
98   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := filtered_hypotheses S hm
99   have := hN
100   have := hN'
101   exact ⟨fun d a B hd ha => fixed_anchor_of_devLE_omega (k := k) hO hd ha B,
102     fun d a B hd ha => fixed_anchor_of_devLE_omega (k := k) hO' hd ha B⟩
103 
104 /-- **Theorem A', both sides.** For every right ideal `I` of `R` containing a
105 regular element, every `x` and every nonzero `t` there is `H` with
106 `I + (x + H t) R = R`; the same holds for `Rᵐᵒᵖ`. -/
107 theorem filtered_fixedIdeal [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
108     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
109     (∀ I : Submodule Rᵐᵒᵖ R, (∃ s ∈ I, s ∈ R⁰) → ∀ x t : R, t ≠ 0 →
110         ∃ H : R, I ⊔ principalRightIdeal (x + H * t) = ⊤) ∧
111       (∀ I : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ, (∃ s ∈ I, s ∈ Rᵐᵒᵖ⁰) → ∀ x t : Rᵐᵒᵖ, t ≠ 0 →
112         ∃ H : Rᵐᵒᵖ, I ⊔ principalRightIdeal (x + H * t) = ⊤) := by
113   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := filtered_hypotheses S hm
114   have := hN
115   have := hN'
116   exact ⟨fun I hreg x t ht => by
117       obtain ⟨H, hH⟩ := fixed_ideal_of_devLE_omega (k := k) hO hreg x ht
118       exact ⟨H, hH⟩,
119     fun I hreg x t ht => by
120       obtain ⟨H, hH⟩ := fixed_ideal_of_devLE_omega (k := k) hO' hreg x ht
121       exact ⟨H, hH⟩⟩

123 /-- **Corollary C.1, both sides.** Every finitely generated Goldie-torsion right
124 module `M` over `R` (and over `Rᵐᵒᵖ`) satisfies: for every `x ∈ M` and every
125 nonzero `t` there is `h ∈ M` with `M = (x + h t) R`. Also, for Goldie-torsion
126 elements `c, u` of `M` and every nonzero `t` there is `H` with
127 `cR + uR = (c + u H t) R`. -/
128 theorem filtered_prescribedCyclic [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
129     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
130     ((∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M],
131         IsGoldieTorsion R M → ∀ (x : M) (t : R), t ≠ 0 →
132           ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} = ⊤) ∧
133       (∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] (c u : M),
134         (∃ s ∈ R⁰, op s • c = 0) → (∃ s ∈ R⁰, op s • u = 0) → ∀ t : R, t ≠ 0 →
135           ∃ H : R, Submodule.span Rᵐᵒᵖ {c, u} = Submodule.span Rᵐᵒᵖ {c + op (H * t) • u})) ∧
136     ((∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖᵐᵒᵖ M],
137         IsGoldieTorsion Rᵐᵒᵖ M → ∀ (x : M) (t : Rᵐᵒᵖ), t ≠ 0 →
138           ∃ h : M, Submodule.span Rᵐᵒᵖᵐᵒᵖ {x + op t • h} = ⊤) ∧
139       (∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖᵐᵒᵖ M] (c u : M),
140         (∃ s ∈ Rᵐᵒᵖ⁰, op s • c = 0) → (∃ s ∈ Rᵐᵒᵖ⁰, op s • u = 0) → ∀ t : Rᵐᵒᵖ, t ≠ 0 →
141           ∃ H : Rᵐᵒᵖ, Submodule.span Rᵐᵒᵖᵐᵒᵖ {c, u} =
142             Submodule.span Rᵐᵒᵖᵐᵒᵖ {c + op (H * t) • u})) := by
143   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := filtered_hypotheses S hm
144   have := hN
145   have := hN'
146   refine ⟨⟨fun M _ _ _ htor x t ht => prescribed_cyclic_of_devLE_omega (k := k) hO htor x ht,
147     fun M _ _ c u hc hu t ht => ?_⟩,
148     ⟨fun M _ _ _ htor x t ht => prescribed_cyclic_of_devLE_omega (k := k) hO' htor x ht,
149     fun M _ _ c u hc hu t ht => ?_⟩⟩
150   · obtain ⟨N, hN⟩ := devLE_span_of_torsion (principalQuotientsFinite_of_devLE_omega hO) hu
151     exact span_pair_eq_span_affine (k := k) c u hc hu ⟨N, hN⟩ ht
152   · obtain ⟨N, hN⟩ := devLE_span_of_torsion (principalQuotientsFinite_of_devLE_omega hO') hu
153     exact span_pair_eq_span_affine (k := k) c u hc hu ⟨N, hN⟩ ht

155 /-- **Corollary C.2, both sides.** Let `I` be a right ideal of `R` and `a ∈ I`
156 with `aQ = IQ`, that is every `y ∈ I` has a regular `s` with `y s ∈ aR`. Then for
157 every `y ∈ I` and nonzero `t` there is `h ∈ I` with `I = aR + (y + h t) R`, and
158 every right ideal of `R` is generated by two elements; the same holds for
159 `Rᵐᵒᵖ`. -/
160 theorem filtered_twoGenerated [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
161     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
162     ((∀ (I : Submodule Rᵐᵒᵖ R) (a : R), a ∈ I →
163         (∀ y ∈ I, ∃ s ∈ R⁰, y * s ∈ principalRightIdeal a) →
164         ∀ y ∈ I, ∀ t : R, t ≠ 0 →
165           ∃ h ∈ I, I = principalRightIdeal a ⊔ principalRightIdeal (y + h * t)) ∧
166       ∀ I : Submodule Rᵐᵒᵖ R, ∃ a b : R, I = principalRightIdeal a ⊔ principalRightIdeal b) ∧
167     ((∀ (I : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ) (a : Rᵐᵒᵖ), a ∈ I →
168         (∀ y ∈ I, ∃ s ∈ Rᵐᵒᵖ⁰, y * s ∈ principalRightIdeal a) →
169         ∀ y ∈ I, ∀ t : Rᵐᵒᵖ, t ≠ 0 →
170           ∃ h ∈ I, I = principalRightIdeal a ⊔ principalRightIdeal (y + h * t)) ∧
171       ∀ I : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ, ∃ a b : Rᵐᵒᵖ,
172         I = principalRightIdeal a ⊔ principalRightIdeal b) := by
173   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := filtered_hypotheses S hm
174   have := hN
175   have := hN'
176   exact ⟨⟨fun I a haI hIa y hyI t ht => by
177         obtain ⟨h, hhI, hI⟩ := two_generated_prescribed (k := k)
178           (principalQuotientsFinite_of_devLE_omega hO) haI hIa hyI ht
179         exact ⟨h, hhI, hI⟩,
180       fun I => two_generated (k := k) (principalQuotientsFinite_of_devLE_omega hO) I⟩,
181     ⟨fun I a haI hIa y hyI t ht => by
182         obtain ⟨h, hhI, hI⟩ := two_generated_prescribed (k := k)
183           (principalQuotientsFinite_of_devLE_omega hO') haI hIa hyI ht
184         exact ⟨h, hhI, hI⟩,
185       fun I => two_generated (k := k) (principalQuotientsFinite_of_devLE_omega hO') I⟩⟩
