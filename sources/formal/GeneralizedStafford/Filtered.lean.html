9 # Filtered rings with commutative Noetherian associated graded ring
10 
11 Proposition `prop:filtered` of the manuscript. Let `k` be an infinite field and
12 `R` a simple `k`-algebra with an exhaustive filtration `φ` and a symbol
13 `S : φ.GradedSymbol 𝒜` identifying `gr R` with a commutative graded algebra
14 `G = ⨁ 𝒜ₙ` of Krull dimension at most `m`, with `G` Noetherian. Then `R` and
15 `Rᵐᵒᵖ` satisfy (S), (N), (Z) and (H_ω), and hence Theorem A, Theorem A',
16 Corollary B, Corollary C.1, C.2 and C.3 hold for `R` and for `Rᵐᵒᵖ`, that is
17 on both sides.
18 
19 * `GeneralizedStafford.right_side`: right Noetherianity and right deviation
20   bounds from the symbol (AlgebraicAnalysis' `GradedSymbol` lemmas).
21 * `GeneralizedStafford.filtered_hypotheses`: (N), (Z) and `Kdim ≤ m` on both
22   sides, using the opposite filtration and the opposite symbol for `Rᵐᵒᵖ`.
23 * `GeneralizedStafford.filtered_theoremA` and the following theorems: the
24   conclusions for each side, reusing the abstract theorems of
25   `TheoremA`, `CorollaryB`, `Modules` and `GeneratorBounds`.
26 -/

44 /-- **Right side of a symbol.** Let `S` identify `gr R` with a commutative
45 Noetherian graded algebra `G` of Krull dimension at most `m`. Then `R` is right
46 Noetherian, and the right ideals of `R` have deviation at most `m` and at most
47 `ω`. -/
48 theorem right_side [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
49     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
50     IsNoetherian Rᵐᵒᵖ R ∧ Module.DevLE Rᵐᵒᵖ m R ∧ Module.DevLEOmega Rᵐᵒᵖ R :=
51   ⟨S.isNoetherian_right, S.devLE_right m hm,
52     S.devLEOmega_right (ringKrullDim_ne_top_of_le hm)⟩
53 
54 /-- **Filtered rings: (N), (Z) and `Kdim ≤ m` on both sides.** Let `R` be a
55 simple `k`-algebra, `k` an infinite field, with a symbol `S` onto a commutative
56 Noetherian graded algebra of Krull dimension at most `m`. Then `R` is right
57 Noetherian with infinite centre and right deviation at most `m` and at most
58 `ω`; the same holds for `Rᵐᵒᵖ`, using the opposite filtration and the opposite
59 symbol. -/
60 theorem filtered_hypotheses [Infinite k] [IsSimpleRing R] [IsNoetherianRing G]
61     [GradedAlgebra 𝒜] {φ : RingFiltration k R} (S : φ.GradedSymbol 𝒜) {m : ℕ}
62     (hm : ringKrullDim G ≤ m) :
63     (IsNoetherian Rᵐᵒᵖ R ∧ Infinite (Subring.center R) ∧
64         Module.DevLE Rᵐᵒᵖ m R ∧ Module.DevLEOmega Rᵐᵒᵖ R) ∧
65       (IsNoetherian Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ ∧ Infinite (Subring.center Rᵐᵒᵖ) ∧
66         Module.DevLE Rᵐᵒᵖᵐᵒᵖ m Rᵐᵒᵖ ∧ Module.DevLEOmega Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ) :=
67   ⟨⟨(right_side S hm).1, infinite_subring_center k R, (right_side S hm).2⟩,
68     ⟨(right_side S.opposite hm).1, infinite_subring_center k Rᵐᵒᵖ,
69       (right_side S.opposite hm).2⟩⟩
70 
71 end Hypotheses

80 /-- Under (H_all), every cyclic submodule `uR` killed on the right by a regular
81 element has finite deviation: `uR` is a quotient of `R / sR`. -/
82 theorem devLE_span_of_torsion {M : Type v} [AddCommGroup M] [Module Rᵐᵒᵖ M]
83     (hall : PrincipalQuotientsFinite R) {u : M} (hu : ∃ s ∈ R⁰, op s • u = 0) :
84     ∃ N, Module.DevLE Rᵐᵒᵖ N (Submodule.span Rᵐᵒᵖ {u}) := by
85   obtain ⟨s, hs, hsu⟩ := hu

88     (devLE_quotient_of_le hN (principalRightIdeal_le (X := LinearMap.ker (cyclicMap u)) hsu))⟩

90 /-- **Theorem A, both sides.** Let `R` be as in `filtered_hypotheses`. For every
91 regular `d`, every nonzero `a` and every `B` there are `H, U, V` with
92 `1 = d U + (B + H a) d V`, in `R` and in `Rᵐᵒᵖ` (the left form
93 `1 = U d + V d (B + a H)` of `R`). -/
94 theorem filtered_theoremA [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
95     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
96     (∀ d a B : R, d ∈ R⁰ → a ≠ 0 → ∃ H U V : R, 1 = d * U + (B + H * a) * d * V) ∧
97       (∀ d a B : Rᵐᵒᵖ, d ∈ Rᵐᵒᵖ⁰ → a ≠ 0 →
98         ∃ H U V : Rᵐᵒᵖ, 1 = d * U + (B + H * a) * d * V) := by
99   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := filtered_hypotheses S hm
100   have := hN
101   have := hN'
102   exact ⟨fun d a B hd ha => fixed_anchor_of_devLE_omega (k := k) hO hd ha B,
103     fun d a B hd ha => fixed_anchor_of_devLE_omega (k := k) hO' hd ha B⟩

105 /-- **Theorem A', both sides.** For every right ideal `I` of `R` containing a
106 regular element, every `x` and every nonzero `t` there is `H` with
107 `I + (x + H t) R = R`; the same holds for `Rᵐᵒᵖ`. -/
108 theorem filtered_fixedIdeal [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
109     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
110     (∀ I : Submodule Rᵐᵒᵖ R, (∃ s ∈ I, s ∈ R⁰) → ∀ x t : R, t ≠ 0 →
111         ∃ H : R, I ⊔ principalRightIdeal (x + H * t) = ⊤) ∧
112       (∀ I : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ, (∃ s ∈ I, s ∈ Rᵐᵒᵖ⁰) → ∀ x t : Rᵐᵒᵖ, t ≠ 0 →
113         ∃ H : Rᵐᵒᵖ, I ⊔ principalRightIdeal (x + H * t) = ⊤) := by
114   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := filtered_hypotheses S hm
115   have := hN
116   have := hN'
117   exact ⟨fun I hreg x t ht => by
118       obtain ⟨H, hH⟩ := fixed_ideal_of_devLE_omega (k := k) hO hreg x ht
119       exact ⟨H, hH⟩,
120     fun I hreg x t ht => by
121       obtain ⟨H, hH⟩ := fixed_ideal_of_devLE_omega (k := k) hO' hreg x ht
122       exact ⟨H, hH⟩⟩

124 /-- **Corollary B, both sides.** `R` has `Kdim (R / dR)` finite for every
125 regular `d`, and `Kdim (R / I)` finite for every right ideal `I` containing a
126 regular element; the same holds for `Rᵐᵒᵖ`. -/
127 theorem filtered_corollaryB [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
128     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
129     (PrincipalQuotientsFinite R ∧
130         ∀ I : Submodule Rᵐᵒᵖ R, (∃ s ∈ I, s ∈ R⁰) → ∃ N, Module.DevLE Rᵐᵒᵖ N (R ⧸ I)) ∧
131       (PrincipalQuotientsFinite Rᵐᵒᵖ ∧
132         ∀ I : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ, (∃ s ∈ I, s ∈ Rᵐᵒᵖ⁰) →
133           ∃ N, Module.DevLE Rᵐᵒᵖᵐᵒᵖ N (Rᵐᵒᵖ ⧸ I)) := by
134   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := filtered_hypotheses S hm

136   have := hN'
137   exact ⟨⟨principalQuotientsFinite_of_devLE_omega hO, fun _ hreg => quotient_finite_of_omega hO hreg⟩,
138     ⟨principalQuotientsFinite_of_devLE_omega hO', fun _ hreg => quotient_finite_of_omega hO' hreg⟩⟩

140 /-- **Corollary C.1, both sides.** Every finitely generated Goldie-torsion right
141 module `M` over `R` (and over `Rᵐᵒᵖ`) satisfies: for every `x ∈ M` and every
142 nonzero `t` there is `h ∈ M` with `M = (x + h t) R`. Also, for Goldie-torsion
143 elements `c, u` of `M` and every nonzero `t` there is `H` with
144 `cR + uR = (c + u H t) R`. -/
145 theorem filtered_prescribedCyclic [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
146     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
147     ((∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M],
148         IsGoldieTorsion R M → ∀ (x : M) (t : R), t ≠ 0 →
149           ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} = ⊤) ∧
150       (∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] (c u : M),
151         (∃ s ∈ R⁰, op s • c = 0) → (∃ s ∈ R⁰, op s • u = 0) → ∀ t : R, t ≠ 0 →
152           ∃ H : R, Submodule.span Rᵐᵒᵖ {c, u} = Submodule.span Rᵐᵒᵖ {c + op (H * t) • u})) ∧
153     ((∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖᵐᵒᵖ M],
154         IsGoldieTorsion Rᵐᵒᵖ M → ∀ (x : M) (t : Rᵐᵒᵖ), t ≠ 0 →
155           ∃ h : M, Submodule.span Rᵐᵒᵖᵐᵒᵖ {x + op t • h} = ⊤) ∧
156       (∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖᵐᵒᵖ M] (c u : M),
157         (∃ s ∈ Rᵐᵒᵖ⁰, op s • c = 0) → (∃ s ∈ Rᵐᵒᵖ⁰, op s • u = 0) → ∀ t : Rᵐᵒᵖ, t ≠ 0 →
158           ∃ H : Rᵐᵒᵖ, Submodule.span Rᵐᵒᵖᵐᵒᵖ {c, u} =
159             Submodule.span Rᵐᵒᵖᵐᵒᵖ {c + op (H * t) • u})) := by
160   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := filtered_hypotheses S hm
161   have := hN
162   have := hN'
163   refine ⟨⟨fun M _ _ _ htor x t ht => prescribed_cyclic_of_devLE_omega (k := k) hO htor x ht,
164     fun M _ _ c u hc hu t ht => ?_⟩,
165     ⟨fun M _ _ _ htor x t ht => prescribed_cyclic_of_devLE_omega (k := k) hO' htor x ht,
166     fun M _ _ c u hc hu t ht => ?_⟩⟩
167   · obtain ⟨N, hN⟩ := devLE_span_of_torsion (principalQuotientsFinite_of_devLE_omega hO) hu
168     exact span_pair_eq_span_affine (k := k) c u hc hu ⟨N, hN⟩ ht
169   · obtain ⟨N, hN⟩ := devLE_span_of_torsion (principalQuotientsFinite_of_devLE_omega hO') hu
170     exact span_pair_eq_span_affine (k := k) c u hc hu ⟨N, hN⟩ ht

172 /-- **Corollary C.2, both sides.** Let `I` be a right ideal of `R` and `a ∈ I`
173 with `aQ = IQ`, that is every `y ∈ I` has a regular `s` with `y s ∈ aR`. Then for
174 every `y ∈ I` and nonzero `t` there is `h ∈ I` with `I = aR + (y + h t) R`, and
175 every right ideal of `R` is generated by two elements; the same holds for
176 `Rᵐᵒᵖ`. -/
177 theorem filtered_twoGenerated [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
178     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
179     ((∀ (I : Submodule Rᵐᵒᵖ R) (a : R), a ∈ I →
180         (∀ y ∈ I, ∃ s ∈ R⁰, y * s ∈ principalRightIdeal a) →
181         ∀ y ∈ I, ∀ t : R, t ≠ 0 →
182           ∃ h ∈ I, I = principalRightIdeal a ⊔ principalRightIdeal (y + h * t)) ∧
183       ∀ I : Submodule Rᵐᵒᵖ R, ∃ a b : R, I = principalRightIdeal a ⊔ principalRightIdeal b) ∧
184     ((∀ (I : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ) (a : Rᵐᵒᵖ), a ∈ I →
185         (∀ y ∈ I, ∃ s ∈ Rᵐᵒᵖ⁰, y * s ∈ principalRightIdeal a) →
186         ∀ y ∈ I, ∀ t : Rᵐᵒᵖ, t ≠ 0 →
187           ∃ h ∈ I, I = principalRightIdeal a ⊔ principalRightIdeal (y + h * t)) ∧
188       ∀ I : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ, ∃ a b : Rᵐᵒᵖ,
189         I = principalRightIdeal a ⊔ principalRightIdeal b) := by
190   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := filtered_hypotheses S hm
191   have := hN
192   have := hN'
193   exact ⟨⟨fun I a haI hIa y hyI t ht => by
194         obtain ⟨h, hhI, hI⟩ := two_generated_prescribed (k := k)
195           (principalQuotientsFinite_of_devLE_omega hO) haI hIa hyI ht
196         exact ⟨h, hhI, hI⟩,
197       fun I => two_generated (k := k) (principalQuotientsFinite_of_devLE_omega hO) I⟩,
198     ⟨fun I a haI hIa y hyI t ht => by
199         obtain ⟨h, hhI, hI⟩ := two_generated_prescribed (k := k)
200           (principalQuotientsFinite_of_devLE_omega hO') haI hIa hyI ht
201         exact ⟨h, hhI, hI⟩,
202       fun I => two_generated (k := k) (principalQuotientsFinite_of_devLE_omega hO') I⟩⟩

204 /-- **Corollary C.3, both sides.** For finitely generated right modules `M`
205 over `R` (and over `Rᵐᵒᵖ`), with `ℓ` and `s` the uniform dimensions of `M / t(M)`
206 and of `R_R`: if `M` is generated by `m` elements then `ℓ ≤ m s`, and if
207 `ℓ < q s` then `M` is generated by `q` elements. -/
208 theorem filtered_generatorBounds [IsNoetherianRing G] [GradedAlgebra 𝒜] {φ : RingFiltration k R}
209     (S : φ.GradedSymbol 𝒜) {m : ℕ} (hm : ringKrullDim G ≤ m) :
210     ((∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M]
211         (n ℓ s : ℕ), (∃ g : Finset M, g.card ≤ n ∧ Submodule.span Rᵐᵒᵖ (g : Set M) = ⊤) →
212         Module.UniformDimGE Rᵐᵒᵖ (M ⧸ goldieTorsionPart R M) ℓ →
213         Module.UniformDimLE Rᵐᵒᵖ R s → ℓ ≤ n * s) ∧
214       (∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M]
215         (q ℓ s : ℕ), Module.UniformDimLE Rᵐᵒᵖ (M ⧸ goldieTorsionPart R M) ℓ →
216         Module.UniformDimGE Rᵐᵒᵖ R s → ℓ < q * s →
217           ∃ g : Finset M, g.card ≤ q ∧ Submodule.span Rᵐᵒᵖ (g : Set M) = ⊤)) ∧
218     ((∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖᵐᵒᵖ M]
219         (n ℓ s : ℕ), (∃ g : Finset M, g.card ≤ n ∧ Submodule.span Rᵐᵒᵖᵐᵒᵖ (g : Set M) = ⊤) →
220         Module.UniformDimGE Rᵐᵒᵖᵐᵒᵖ (M ⧸ goldieTorsionPart Rᵐᵒᵖ M) ℓ →
221         Module.UniformDimLE Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ s → ℓ ≤ n * s) ∧
222       (∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖᵐᵒᵖ M]
223         (q ℓ s : ℕ), Module.UniformDimLE Rᵐᵒᵖᵐᵒᵖ (M ⧸ goldieTorsionPart Rᵐᵒᵖ M) ℓ →
224         Module.UniformDimGE Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ s → ℓ < q * s →
225           ∃ g : Finset M, g.card ≤ q ∧ Submodule.span Rᵐᵒᵖᵐᵒᵖ (g : Set M) = ⊤)) := by
226   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := filtered_hypotheses S hm
227   have := hN
228   have := hN'
229   exact ⟨⟨fun M _ _ _ n ℓ s ⟨g, hg, hspan⟩ hℓ hs => le_mul_of_span_eq_top g hg hspan hℓ hs,
230       fun M _ _ _ q ℓ s hℓ hs hq =>
231         exists_span_eq_top_of_lt_mul (k := k) (principalQuotientsFinite_of_devLE_omega hO) hℓ hs hq⟩,
232     ⟨fun M _ _ _ n ℓ s ⟨g, hg, hspan⟩ hℓ hs => le_mul_of_span_eq_top g hg hspan hℓ hs,
233       fun M _ _ _ q ℓ s hℓ hs hq =>
234         exists_span_eq_top_of_lt_mul (k := k) (principalQuotientsFinite_of_devLE_omega hO') hℓ hs hq⟩⟩
