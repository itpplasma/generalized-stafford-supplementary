5 /-!
6 # Left statements from right statements
7 
8 For a ring `A` with an anti-automorphism `τ : A ≃+* Aᵐᵒᵖ`, the map
9 `I ↦ {y | τ⁻¹(op y) ∈ I}` is a bijection from left ideals of `A` to right ideals
10 of `A`, and a left `A`-module is a right `A`-module through `τ`. Hence a
11 statement about right ideals and right modules of `A` yields the same statement
12 for left ideals and left modules, with no extra hypothesis on `A`.
13 
14 The results here are generic: they take the right-hand statement as a
15 hypothesis. The two-generator transfer is used for the Weyl algebra in
16 `GeneralizedStafford.Weyl`.
17 -/

29 /-- The right ideal `τ(I)` attached to a left ideal `I` by the anti-automorphism
30 `τ`: `y ∈ τ(I)` iff `τ⁻¹(op y) ∈ I`. -/
31 def leftToRight (τ : A ≃+* Aᵐᵒᵖ) (I : Submodule A A) : Submodule Aᵐᵒᵖ A where
32   carrier := {y | τ.symm (op y) ∈ I}
33   add_mem' {x y} hx hy := by
34     change τ.symm (op (x + y)) ∈ I
35     rw [op_add, map_add]
36     exact I.add_mem hx hy
37   zero_mem' := by
38     change τ.symm (op 0) ∈ I
39     simp
40   smul_mem' r y hy := by
41     change τ.symm (op (y * r.unop)) ∈ I
42     rw [op_mul, map_mul]
43     exact I.smul_mem _ hy

45 /-- A left ideal with two generators, from the two-generator statement for right
46 ideals. -/
47 theorem twoGeneratedLeft_of_right (τ : A ≃+* Aᵐᵒᵖ)
48     (hR : ∀ J : Submodule Aᵐᵒᵖ A, ∃ a b : A, J = Submodule.span Aᵐᵒᵖ {a, b})
49     (I : Submodule A A) : ∃ a b : A, I = Submodule.span A {a, b} := by
50   obtain ⟨a, b, hab⟩ := hR (leftToRight τ I)
51   refine ⟨τ.symm (op a), τ.symm (op b), ?_⟩
52   ext x
53   have hx : x ∈ I ↔ unop (τ x) ∈ leftToRight τ I := by
54     change x ∈ I ↔ τ.symm (op (unop (τ x))) ∈ I
55     simp
56   rw [hx, hab, Submodule.mem_span_pair, Submodule.mem_span_pair]
57   constructor
58   · rintro ⟨r, s, hrs⟩
59     refine ⟨τ.symm r, τ.symm s, ?_⟩
60     apply τ.injective
61     apply unop_injective
62     simp only [smul_eq_mul, map_add, map_mul, RingEquiv.apply_symm_apply, unop_add,
63       unop_mul, unop_op, ← hrs, MulOpposite.smul_eq_mul_unop]
64   · rintro ⟨r, s, hrs⟩
65     refine ⟨τ r, τ s, ?_⟩
66     rw [← hrs]
67     simp only [smul_eq_mul, map_add, map_mul, RingEquiv.apply_symm_apply, unop_add,
68       unop_mul, unop_op, MulOpposite.smul_eq_mul_unop]

70 theorem mem_leftToRight (τ : A ≃+* Aᵐᵒᵖ) (I : Submodule A A) (x : A) :
71     unop (τ x) ∈ leftToRight τ I ↔ x ∈ I := by
72   change τ.symm (op (unop (τ x))) ∈ I ↔ x ∈ I
73   simp

75 /-- The transfer `I ↦ τ(I)` is injective. -/
76 theorem leftToRight_injective (τ : A ≃+* Aᵐᵒᵖ) :
77     Function.Injective (leftToRight τ) := by
78   intro I I' h
79   ext x
80   rw [← mem_leftToRight τ I, ← mem_leftToRight τ I', h]

82 /-- The transfer of a left span is the right span of the transferred generators. -/
83 theorem leftToRight_span (τ : A ≃+* Aᵐᵒᵖ) (S : Set A) :
84     leftToRight τ (Submodule.span A S) =
85       Submodule.span Aᵐᵒᵖ ((fun x => unop (τ x)) '' S) := by
86   apply le_antisymm
87   · intro y hy
88     have hy' : τ.symm (op y) ∈ Submodule.span A S := hy
89     have key : ∀ z ∈ Submodule.span A S,
90         unop (τ z) ∈ Submodule.span Aᵐᵒᵖ ((fun x => unop (τ x)) '' S) := by
91       intro z hz
92       induction hz using Submodule.span_induction with
93       | mem z hz => exact Submodule.subset_span ⟨z, hz, rfl⟩
94       | zero => simp
95       | add z w _ _ hz hw => simpa using Submodule.add_mem _ hz hw
96       | smul r z _ hz =>
97         have := Submodule.smul_mem _ (τ r) hz
98         simpa [smul_eq_mul, MulOpposite.smul_eq_mul_unop] using this
99     simpa using key _ hy'
100   · rw [Submodule.span_le]
101     rintro _ ⟨z, hz, rfl⟩
102     exact (mem_leftToRight τ _ z).2 (Submodule.subset_span hz)

104 /-- Prescribed first generator for left ideals, from the right statement. -/
105 theorem prescribedTwoGeneratedLeft_of_right (τ : A ≃+* Aᵐᵒᵖ)
106     (hR : ∀ (J : Submodule Aᵐᵒᵖ A) (a : A), a ∈ J → a ≠ 0 → ∀ y ∈ J, ∀ t : A, t ≠ 0 →
107       ∃ h ∈ J, J = Submodule.span Aᵐᵒᵖ {a} ⊔ Submodule.span Aᵐᵒᵖ {y + h * t})
108     (I : Submodule A A) (a : A) (haI : a ∈ I) (ha : a ≠ 0) (y : A) (hy : y ∈ I)
109     (t : A) (ht : t ≠ 0) :
110     ∃ h ∈ I, I = Submodule.span A {a} ⊔ Submodule.span A {y + t * h} := by
111   have hne : ∀ z : A, z ≠ 0 → unop (τ z) ≠ 0 := fun z hz h0 =>
112     hz (τ.injective (by simpa using congrArg op h0))
113   obtain ⟨h', hh', hJ⟩ := hR (leftToRight τ I) (unop (τ a))
114     ((mem_leftToRight τ I a).2 haI) (hne a ha) (unop (τ y))
115     ((mem_leftToRight τ I y).2 hy) (unop (τ t)) (hne t ht)
116   refine ⟨τ.symm (op h'), hh', leftToRight_injective τ ?_⟩
117   rw [← Submodule.span_insert, leftToRight_span, hJ, ← Submodule.span_insert]
118   congr 1
119   simp [Set.image_insert_eq]

127 /-- A left `A`-module viewed as a right module through `τ`:
128 `m * r := τ⁻¹(op r) • m`. This is a local definition, not an instance. -/
129 abbrev rightOfLeft (τ : A ≃+* Aᵐᵒᵖ) : Module Aᵐᵒᵖ M :=
130   Module.compHom M (τ.symm : Aᵐᵒᵖ →+* A)
131 
132 /-- The right span of `S` in `rightOfLeft τ` is the left span of `S`. -/
133 theorem span_rightOfLeft (τ : A ≃+* Aᵐᵒᵖ) (S : Set M) :
134     ((@Submodule.span Aᵐᵒᵖ M _ _ (rightOfLeft τ) S : Set M)) = Submodule.span A S := by
135   let := rightOfLeft (M := M) τ
136   apply le_antisymm
137   · let N : Submodule Aᵐᵒᵖ M :=
138       { carrier := Submodule.span A S
139         add_mem' := fun hx hy => Submodule.add_mem _ hx hy
140         zero_mem' := Submodule.zero_mem _
141         smul_mem' := fun r m hm => Submodule.smul_mem _ (τ.symm r) hm }
142     exact (Submodule.span_le (p := N)).2 Submodule.subset_span
143   · let N : Submodule A M :=
144       { carrier := Submodule.span Aᵐᵒᵖ S
145         add_mem' := fun hx hy => Submodule.add_mem _ hx hy
146         zero_mem' := Submodule.zero_mem _
147         smul_mem' := fun r m hm => by
148           have := Submodule.smul_mem _ (τ r) hm
149           change τ.symm (τ r) • m ∈ _ at this
150           simpa using this }
151     exact (Submodule.span_le (p := N)).2 Submodule.subset_span

153 /-- Prescribed cyclic generator for left modules, from the right statement. -/
154 theorem prescribedCyclicLeft_of_right (τ : A ≃+* Aᵐᵒᵖ)
155     (hR : ∀ (M : Type v) [AddCommGroup M] [Module Aᵐᵒᵖ M] [Module.Finite Aᵐᵒᵖ M],
156       (∀ m : M, ∃ d : A, d ≠ 0 ∧ op d • m = 0) →
157       ∀ (x : M) (t : A), t ≠ 0 → ∃ h : M, Submodule.span Aᵐᵒᵖ {x + op t • h} = ⊤)
158     [Module.Finite A M] (hM : ∀ m : M, ∃ d : A, d ≠ 0 ∧ d • m = 0)
159     (x : M) (t : A) (ht : t ≠ 0) :
160     ∃ h : M, Submodule.span A {x + t • h} = ⊤ := by
161   let := rightOfLeft (M := M) τ
162   have hsmul : ∀ (r : Aᵐᵒᵖ) (m : M), r • m = τ.symm r • m := fun _ _ => rfl
163   have htop : ∀ S : Set M, Submodule.span Aᵐᵒᵖ S = ⊤ ↔ Submodule.span A S = ⊤ := by
164     intro S
165     rw [← SetLike.coe_set_eq, ← SetLike.coe_set_eq (q := (⊤ : Submodule A M)),
166       span_rightOfLeft]
167     rfl
168   have : Module.Finite Aᵐᵒᵖ M := by
169     obtain ⟨S, hS⟩ := (Module.finite_def.mp ‹Module.Finite A M›)
170     exact Module.finite_def.mpr ⟨S, (htop _).2 hS⟩
171   have hne : ∀ z : A, z ≠ 0 → unop (τ z) ≠ 0 := fun z hz h0 =>
172     hz (τ.injective (by simpa using congrArg op h0))
173   obtain ⟨h, hh⟩ := hR M (fun m => by
174       obtain ⟨d, hd, hdm⟩ := hM m
175       exact ⟨unop (τ d), hne d hd, by simpa [hsmul] using hdm⟩)
176     x (unop (τ t)) (hne t ht)
177   refine ⟨h, (htop _).1 ?_⟩
178   simpa [hsmul] using hh
