8 # Theorem A' (fixed ideal) and Theorem A (fixed anchor)
9 
10 Assume (S) `R` simple, (N) right Noetherian, and (Z) `k` an infinite central
11 field. Zero divisors are allowed: the anchor `d` and the regular element of `I`
12 are regular, while the tails `t` and `a` need only be nonzero.
13 
14 * **Theorem A'** (`fixed_ideal`) is `P(N+1)` of the two-call descent with the
15   empty family of tests: `Kdim R/(I + fR) < 0`, i.e. `I + fR = R`.
16 * **Theorem A** (`fixed_anchor`) is Theorem A' for `I = dR`, `x = B d` and
17   `t = a d` (nonzero because `d` is regular): `dR + (B d + H a d)R = R` reads
18   `1 = d U + (B + H a) d V`.
19 -/

31 /-- **Theorem A' (fixed ideal).** Let `I` be a right ideal containing a
32 regular element with `Kdim(R/I) ≤ N` finite. For every `x ∈ R`
33 and every `t ≠ 0` there is `H ∈ R` with `I + (x + H t)R = R`. -/
34 theorem fixed_ideal {I : Submodule Rᵐᵒᵖ R} (hreg : ∃ s ∈ I, s ∈ R⁰)
35     {N : ℕ} (hfin : Module.DevLE Rᵐᵒᵖ N (R ⧸ I)) (x : R) {t : R} (ht : t ≠ 0) :
36     ∃ H : R, I ⊔ principalRightIdeal (x + H * t) = ⊤ := by
37   obtain ⟨H, hsmall, -⟩ := idealBound_zero (k := k) hreg hfin x t ht
38     (fun i : PEmpty.{u + 1} => nomatch i)
39   exact ⟨H, eq_top_of_small_zero hsmall⟩

42 /-- **Theorem A (fixed anchor).** Let `d` be regular with
43 `δ(d) = Kdim(R/dR) ≤ N` finite. For every `a ≠ 0` (possibly a zero-divisor)
44 and every `B` there are `H, U, V` with `1 = d U + (B + H a) d V`. -/
45 theorem fixed_anchor {d : R} (hd : d ∈ R⁰) {N : ℕ}
46     (hfin : Module.DevLE Rᵐᵒᵖ N (R ⧸ principalRightIdeal d)) {a : R} (ha : a ≠ 0) (B : R) :
47     ∃ H U V : R, 1 = d * U + (B + H * a) * d * V := by
48   obtain ⟨H, hH⟩ := fixed_ideal (k := k) (I := principalRightIdeal d)
49     ⟨d, ⟨1, mul_one d⟩, hd⟩ hfin (B * d) (t := a * d) (fun h0 => ha (hd.2 a h0))
50   have h1 : (1 : R) ∈ principalRightIdeal d ⊔ principalRightIdeal (B * d + H * (a * d)) := by
51     rw [hH]; exact Submodule.mem_top
52   obtain ⟨y, ⟨U, rfl⟩, z, ⟨V, rfl⟩, he⟩ := Submodule.mem_sup.mp h1
53   refine ⟨H, U, V, ?_⟩
54   rw [← he]
55   change d * U + (B * d + H * (a * d)) * V = d * U + (B + H * a) * d * V
56   simp only [add_mul, mul_assoc]
