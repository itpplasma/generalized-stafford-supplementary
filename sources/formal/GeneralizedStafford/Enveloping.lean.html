6 # Simple quotients of enveloping algebras
7 
8 Let `𝔤` be a finite-dimensional Lie
9 algebra over an infinite field `k`, and let `R` be a simple `k`-algebra with a surjective
10 `k`-algebra map `π : U(𝔤) → R` (so `R ≅ U(𝔤)/J` for a two-sided ideal `J`). Then `R` and
11 `Rᵐᵒᵖ` satisfy (N), (Z) and `Kdim ≤ dim 𝔤`, and hence Theorem A, Theorem A', Corollary B
12 and Corollary C hold for `R` on both sides.
13 
14 Proof idea. The PBW symbol presentation of AlgebraicAnalysis (`pbwSymbolPresentation`)
15 presents the filtration of `R` induced by the generators of `𝔤` by the polynomial ring in
16 `dim 𝔤` variables, so `R` is Noetherian on both sides with deviation at most `dim 𝔤`. The
17 opposite statements are the same statements for left ideals, transported by
18 `isNoetherian_right_op_iff` and `devLE_right_op_iff`. The conclusions then follow from the
19 abstract theorems of `GeneralizedStafford.Filtered`, using `Module.DevLE.omega`.
20 
21 * `GeneralizedStafford.enveloping_hypotheses`: (N), (Z) and `Kdim ≤ dim 𝔤` on both sides.
22 * `GeneralizedStafford.enveloping_theoremA` and the following theorems: the conclusions
23   for each side.
24 -/

38 /-- **Enveloping algebras: (N), (Z) and `Kdim ≤ dim 𝔤` on both sides.** Let `k` be an
39 infinite field, `𝔤` a finite-dimensional Lie algebra and `R` a simple quotient of `U(𝔤)`,
40 given by a surjective algebra map `π`. Then `R` is Noetherian with infinite centre and
41 right deviation at most `dim 𝔤` and at most `ω`; the same holds for `Rᵐᵒᵖ`. -/
42 theorem enveloping_hypotheses [Infinite k] [IsSimpleRing R]
43     (π : UniversalEnvelopingAlgebra k L →ₐ[k] R) (hπ : Function.Surjective π) :
44     (IsNoetherian Rᵐᵒᵖ R ∧ Infinite (Subring.center R) ∧
45         Module.DevLE Rᵐᵒᵖ (Module.finrank k L) R ∧
46         Module.DevLEOmega Rᵐᵒᵖ R) ∧
47       (IsNoetherian Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ ∧ Infinite (Subring.center Rᵐᵒᵖ) ∧
48         Module.DevLE Rᵐᵒᵖᵐᵒᵖ (Module.finrank k L) Rᵐᵒᵖ ∧
49         Module.DevLEOmega Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ) := by
50   have hD := devLE_right_of_surjective π hπ
51   have hDL : Module.DevLE R (Module.finrank k L) R := devLE_left_of_surjective π hπ
52   have hop : IsNoetherian Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ := (isNoetherian_right_op_iff R).mpr (isNoetherianRing_of_surjective π hπ)
53   have hopD : Module.DevLE Rᵐᵒᵖᵐᵒᵖ (Module.finrank k L) Rᵐᵒᵖ := (devLE_right_op_iff R _).mpr hDL
54   exact ⟨⟨isNoetherian_right_of_surjective π hπ, infinite_subring_center k R, hD, hD.omega⟩,
55     ⟨hop, infinite_subring_center k Rᵐᵒᵖ, hopD, hopD.omega⟩⟩
56 
57 end Hypotheses

64 /-- **Theorem A, both sides (enveloping algebras).** For every regular `d`, every nonzero
65 `a` and every `B` there are `H, U, V` with `1 = d U + (B + H a) d V`, in `R` and in `Rᵐᵒᵖ`
66 (the left form `1 = U d + V d (B + a H)` of `R`). -/
67 theorem enveloping_theoremA (π : UniversalEnvelopingAlgebra k L →ₐ[k] R)
68     (hπ : Function.Surjective π) :
69     (∀ d a B : R, d ∈ R⁰ → a ≠ 0 → ∃ H U V : R, 1 = d * U + (B + H * a) * d * V) ∧
70       (∀ d a B : Rᵐᵒᵖ, d ∈ Rᵐᵒᵖ⁰ → a ≠ 0 →
71         ∃ H U V : Rᵐᵒᵖ, 1 = d * U + (B + H * a) * d * V) := by
72   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := enveloping_hypotheses π hπ
73   have := hN
74   have := hN'
75   exact ⟨fun d a B hd ha => fixed_anchor_of_devLE_omega (k := k) hO hd ha B,
76     fun d a B hd ha => fixed_anchor_of_devLE_omega (k := k) hO' hd ha B⟩
77 
78 /-- **Theorem A', both sides (enveloping algebras).** For every right ideal `I` of `R`
79 containing a regular element, every `x` and every nonzero `t` there is `H` with
80 `I + (x + H t) R = R`; the same holds for `Rᵐᵒᵖ`. -/
81 theorem enveloping_fixedIdeal (π : UniversalEnvelopingAlgebra k L →ₐ[k] R)
82     (hπ : Function.Surjective π) :
83     (∀ I : Submodule Rᵐᵒᵖ R, (∃ s ∈ I, s ∈ R⁰) → ∀ x t : R, t ≠ 0 →
84         ∃ H : R, I ⊔ principalRightIdeal (x + H * t) = ⊤) ∧
85       (∀ I : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ, (∃ s ∈ I, s ∈ Rᵐᵒᵖ⁰) → ∀ x t : Rᵐᵒᵖ, t ≠ 0 →
86         ∃ H : Rᵐᵒᵖ, I ⊔ principalRightIdeal (x + H * t) = ⊤) := by
87   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := enveloping_hypotheses π hπ
88   have := hN
89   have := hN'
90   exact ⟨fun I hreg x t ht => by
91       obtain ⟨H, hH⟩ := fixed_ideal_of_devLE_omega (k := k) hO hreg x ht
92       exact ⟨H, hH⟩,
93     fun I hreg x t ht => by
94       obtain ⟨H, hH⟩ := fixed_ideal_of_devLE_omega (k := k) hO' hreg x ht
95       exact ⟨H, hH⟩⟩

97 /-- **Corollary C.1, both sides (enveloping algebras).** Every finitely generated
98 Goldie-torsion right module `M` over `R` (and over `Rᵐᵒᵖ`) satisfies: for every `x ∈ M` and
99 every nonzero `t` there is `h ∈ M` with `M = (x + h t) R`. Also, for Goldie-torsion elements
100 `c, u` of `M` and every nonzero `t` there is `H` with `cR + uR = (c + u H t) R`. -/
101 theorem enveloping_prescribedCyclic (π : UniversalEnvelopingAlgebra k L →ₐ[k] R)
102     (hπ : Function.Surjective π) :
103     ((∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖ M],
104         IsGoldieTorsion R M → ∀ (x : M) (t : R), t ≠ 0 →
105           ∃ h : M, Submodule.span Rᵐᵒᵖ {x + op t • h} = ⊤) ∧
106       (∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖ M] (c u : M),
107         (∃ s ∈ R⁰, op s • c = 0) → (∃ s ∈ R⁰, op s • u = 0) → ∀ t : R, t ≠ 0 →
108           ∃ H : R, Submodule.span Rᵐᵒᵖ {c, u} = Submodule.span Rᵐᵒᵖ {c + op (H * t) • u})) ∧
109     ((∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖᵐᵒᵖ M] [Module.Finite Rᵐᵒᵖᵐᵒᵖ M],
110         IsGoldieTorsion Rᵐᵒᵖ M → ∀ (x : M) (t : Rᵐᵒᵖ), t ≠ 0 →
111           ∃ h : M, Submodule.span Rᵐᵒᵖᵐᵒᵖ {x + op t • h} = ⊤) ∧
112       (∀ (M : Type v) [AddCommGroup M] [Module Rᵐᵒᵖᵐᵒᵖ M] (c u : M),
113         (∃ s ∈ Rᵐᵒᵖ⁰, op s • c = 0) → (∃ s ∈ Rᵐᵒᵖ⁰, op s • u = 0) → ∀ t : Rᵐᵒᵖ, t ≠ 0 →
114           ∃ H : Rᵐᵒᵖ, Submodule.span Rᵐᵒᵖᵐᵒᵖ {c, u} =
115             Submodule.span Rᵐᵒᵖᵐᵒᵖ {c + op (H * t) • u})) := by
116   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := enveloping_hypotheses π hπ
117   have := hN
118   have := hN'
119   refine ⟨⟨fun M _ _ _ htor x t ht => prescribed_cyclic_of_devLE_omega (k := k) hO htor x ht,
120     fun M _ _ c u hc hu t ht => ?_⟩,
121     ⟨fun M _ _ _ htor x t ht => prescribed_cyclic_of_devLE_omega (k := k) hO' htor x ht,
122     fun M _ _ c u hc hu t ht => ?_⟩⟩
123   · obtain ⟨N, hN⟩ := devLE_span_of_torsion (principalQuotientsFinite_of_devLE_omega hO) hu
124     exact span_pair_eq_span_affine (k := k) c u hc hu ⟨N, hN⟩ ht
125   · obtain ⟨N, hN⟩ := devLE_span_of_torsion (principalQuotientsFinite_of_devLE_omega hO') hu
126     exact span_pair_eq_span_affine (k := k) c u hc hu ⟨N, hN⟩ ht

128 /-- **Corollary C.2, both sides (enveloping algebras).** Let `I` be a right ideal of `R`
129 and `a ∈ I` with `aQ = IQ`, that is every `y ∈ I` has a regular `s` with `y s ∈ aR`. Then
130 for every `y ∈ I` and nonzero `t` there is `h ∈ I` with `I = aR + (y + h t) R`, and every
131 right ideal of `R` is generated by two elements; the same holds for `Rᵐᵒᵖ`. -/
132 theorem enveloping_twoGenerated (π : UniversalEnvelopingAlgebra k L →ₐ[k] R)
133     (hπ : Function.Surjective π) :
134     ((∀ (I : Submodule Rᵐᵒᵖ R) (a : R), a ∈ I →
135         (∀ y ∈ I, ∃ s ∈ R⁰, y * s ∈ principalRightIdeal a) →
136         ∀ y ∈ I, ∀ t : R, t ≠ 0 →
137           ∃ h ∈ I, I = principalRightIdeal a ⊔ principalRightIdeal (y + h * t)) ∧
138       ∀ I : Submodule Rᵐᵒᵖ R, ∃ a b : R, I = principalRightIdeal a ⊔ principalRightIdeal b) ∧
139     ((∀ (I : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ) (a : Rᵐᵒᵖ), a ∈ I →
140         (∀ y ∈ I, ∃ s ∈ Rᵐᵒᵖ⁰, y * s ∈ principalRightIdeal a) →
141         ∀ y ∈ I, ∀ t : Rᵐᵒᵖ, t ≠ 0 →
142           ∃ h ∈ I, I = principalRightIdeal a ⊔ principalRightIdeal (y + h * t)) ∧
143       ∀ I : Submodule Rᵐᵒᵖᵐᵒᵖ Rᵐᵒᵖ, ∃ a b : Rᵐᵒᵖ,
144         I = principalRightIdeal a ⊔ principalRightIdeal b) := by
145   obtain ⟨⟨hN, -, -, hO⟩, ⟨hN', -, -, hO'⟩⟩ := enveloping_hypotheses π hπ
146   have := hN
147   have := hN'
148   exact ⟨⟨fun I a haI hIa y hyI t ht => by
149         obtain ⟨h, hhI, hI⟩ := two_generated_prescribed (k := k)
150           (principalQuotientsFinite_of_devLE_omega hO) haI hIa hyI ht
151         exact ⟨h, hhI, hI⟩,
152       fun I => two_generated (k := k) (principalQuotientsFinite_of_devLE_omega hO) I⟩,
153     ⟨fun I a haI hIa y hyI t ht => by
154         obtain ⟨h, hhI, hI⟩ := two_generated_prescribed (k := k)
155           (principalQuotientsFinite_of_devLE_omega hO') haI hIa hyI ht
156         exact ⟨h, hhI, hI⟩,
157       fun I => two_generated (k := k) (principalQuotientsFinite_of_devLE_omega hO') I⟩⟩
158
