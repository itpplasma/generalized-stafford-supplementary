10 # The Weyl algebra `A_n(k)`
11 
12 The hypotheses (S), (N), (Z) and (H_omega) of the generalized Stafford theorems
13 hold for the Weyl algebra `A_n(k)` over a field `k` of characteristic zero, by
14 AlgebraicAnalysis (`Weyl.Simplicity`, `Weyl.BernsteinSymbol`,
15 `Weyl.Transposition`). This file collects these facts and the consequences
16 for `A_n(k)`:
17 
18 * unconditionally, every right ideal and every left ideal of `A_n(k)` is
19   two-generated (Corollary C.2), with the left case transferred from the right
20   case by the transposition anti-automorphism;
21 * the Weyl algebra is a domain (AlgebraicAnalysis `Weyl.isDomain`), so every
22   nonzero element is regular; this is the input of the regular-element
23   consequences below (Theorem A, Theorem A', Corollary C.1 and the
24   prescribed-generator form of Corollary C.2).
25 -/

36 /-- `A_n(k)` is simple (S) for a field of characteristic zero. -/
37 theorem isSimpleRing (k : Type u) [Field k] [CharZero k] (n : ℕ) :
38     IsSimpleRing (WeylAlg k n) :=
39   AlgebraicAnalysis.Weyl.Simplicity.isSimpleRing_weylAlg k n
40 
41 /-- `A_n(k)` is right Noetherian (N), from the Bernstein symbol. -/
42 theorem isNoetherian (k : Type u) [Field k] (n : ℕ) :
43     IsNoetherian (WeylAlg k n)ᵐᵒᵖ (WeylAlg k n) :=
44   AlgebraicAnalysis.Weyl.BernsteinSymbol.isNoetherian_right k n
45 
46 /-- `Kdim A_n(k) ≤ ω` (H_omega), from the Bernstein symbol. -/
47 theorem hOmega (k : Type u) [Field k] (n : ℕ) : HOmega (WeylAlg k n) :=
48   hOmega_iff.mpr (AlgebraicAnalysis.Weyl.BernsteinSymbol.devLEOmega_right k n)
49 
50 /-- The transposition anti-automorphism `A_n(k) ≃+* A_n(k)ᵐᵒᵖ`. -/
51 def transposeRingEquiv (k : Type u) [Field k] (n : ℕ) :
52     WeylAlg k n ≃+* (WeylAlg k n)ᵐᵒᵖ :=
53   (AlgebraicAnalysis.Weyl.Transposition.transpositionEquiv k n).toRingEquiv
54 
55 /-- The centre of `A_n(k)` contains `k`, which is infinite (Z). -/
56 theorem infinite_center (k : Type u) [Field k] [CharZero k] (n : ℕ) :
57     Infinite (Subring.center (WeylAlg k n)) := by
58   have := isSimpleRing k n
59   have hinj : Function.Injective (algebraMap k (WeylAlg k n)) :=
60     (algebraMap k (WeylAlg k n)).injective
61   let f : k → Subring.center (WeylAlg k n) := fun c =>
62     ⟨algebraMap k (WeylAlg k n) c,
63       Subring.mem_center_iff.mpr fun z => (Algebra.commutes c z).symm⟩
64   exact Infinite.of_injective f fun a b hab => hinj (congrArg Subtype.val hab)

70 variable {A : Type u} [Ring A]
71 
72 /-- A nonzero element of a domain is regular. -/
73 theorem isRegular_of_ne_zero [IsDomain A] {d : A} (hd : d ≠ 0) : IsRegularElement d :=
74   ⟨fun _ hx => (mul_eq_zero.mp hx).resolve_left hd,
75    fun _ hx => (mul_eq_zero.mp hx).resolve_right hd⟩
76 
77 /-- Theorem A for a domain (the Weyl algebra, by AlgebraicAnalysis): the nonzero
78 anchor `d`, `a` and `B` satisfy `1 = d U + (B + H a) d V`. -/
79 theorem affine_of_domain [IsSimpleRing A] [IsNoetherian Aᵐᵒᵖ A]
80     [Infinite (Subring.center A)] [IsDomain A] (hO : HOmega A)
81     (d a B : A) (hd : d ≠ 0) (ha : a ≠ 0) :
82     ∃ H U V : A, 1 = d * U + (B + H * a) * d * V :=
83   corollaryB_fixedAnchor_proof A hO d a B (isRegular_of_ne_zero hd) ha
84 
85 /-- Theorem A' for a domain: for a nonzero right ideal `I`, `I + (x + H t) A = A`. -/
86 theorem fixedIdeal_of_domain [IsSimpleRing A] [IsNoetherian Aᵐᵒᵖ A]
87     [Infinite (Subring.center A)] [IsDomain A] (hO : HOmega A)
88     (I : Submodule Aᵐᵒᵖ A) (hI : I ≠ ⊥)
89     (x t : A) (ht : t ≠ 0) : ∃ H : A, I ⊔ Submodule.span Aᵐᵒᵖ {x + H * t} = ⊤ := by
90   obtain ⟨s, hs, hs0⟩ := Submodule.exists_mem_ne_zero_of_ne_bot hI
91   exact corollaryB_fixedIdeal_proof A hO I ⟨s, hs, isRegular_of_ne_zero hs0⟩ x t ht
92 
93 /-- Corollary C.1 for a domain: a finitely generated module
94 killed by nonzero scalars has a prescribed cyclic generator. -/
95 theorem prescribedCyclic_of_domain [IsSimpleRing A] [IsNoetherian Aᵐᵒᵖ A]
96     [Infinite (Subring.center A)] [IsDomain A] (hO : HOmega A)
97     (M : Type v) [AddCommGroup M] [Module Aᵐᵒᵖ M] [Module.Finite Aᵐᵒᵖ M]
98     (hM : ∀ m : M, ∃ d : A, d ≠ 0 ∧ op d • m = 0) (x : M) (t : A) (ht : t ≠ 0) :
99     ∃ h : M, Submodule.span Aᵐᵒᵖ {x + op t • h} = ⊤ :=
100   corollaryC1_omega_proof A hO M (by
101     intro m
102     obtain ⟨d, hd, hdm⟩ := hM m
103     exact ⟨d, isRegular_of_ne_zero hd, hdm⟩) x t ht
104 
105 /-- Corollary C.2 for a domain: a nonzero element `a` of a
106 right ideal `I` is the first generator of a two-generated presentation. -/
107 theorem prescribedTwoGenerated_of_domain [IsSimpleRing A] [IsNoetherian Aᵐᵒᵖ A]
108     [Infinite (Subring.center A)] [IsDomain A] (hO : HOmega A)
109     (I : Submodule Aᵐᵒᵖ A) (a : A) (haI : a ∈ I) (ha : a ≠ 0) (y : A) (hy : y ∈ I)
110     (t : A) (ht : t ≠ 0) :
111     ∃ h ∈ I, I = Submodule.span Aᵐᵒᵖ {a} ⊔ Submodule.span Aᵐᵒᵖ {y + h * t} := by
112   refine corollaryC2_prescribed_proof A (hAll_of_omega hO)
113     I a haI ?_ y hy t ht
114   intro z _
115   by_cases hz : z = 0
116   · exact ⟨1, isRegular_of_ne_zero one_ne_zero, by simp [hz]⟩
117   obtain ⟨r, s, hrs, _⟩ :=
118     AlgebraicAnalysis.OreRightIntersection.rightOreCondition_of_isNoetherian ha hz
119   have hs : s ≠ 0 := by rintro rfl; simp_all
120   refine ⟨s, isRegular_of_ne_zero hs, ?_⟩
121   rw [← hrs, Submodule.mem_span_singleton]
122   exact ⟨op r, by simp⟩
123 
124 end FromDomain

126 /-! ## Two-generated one-sided ideals of `A_n(k)` (unconditional) -/
127 
128 section TwoGenerated
129 
130 variable (k : Type u) [Field k] [CharZero k] (n : ℕ)
131 
132 /-- Every right ideal of `A_n(k)` is two-generated (Corollary C.2). -/
133 theorem twoGeneratedRight (I : Submodule (WeylAlg k n)ᵐᵒᵖ (WeylAlg k n)) :
134     ∃ a b : WeylAlg k n, I = Submodule.span (WeylAlg k n)ᵐᵒᵖ {a, b} := by
135   have := isSimpleRing k n
136   have := isNoetherian k n
137   have := infinite_center k n
138   exact corollaryC2_proof (WeylAlg k n) (hAll_of_omega (hOmega k n)) I
139 
140 /-- Every left ideal of `A_n(k)` is two-generated, transferred from the right
141 case by the transposition anti-automorphism. -/
142 theorem twoGeneratedLeft (I : Submodule (WeylAlg k n) (WeylAlg k n)) :
143     ∃ a b : WeylAlg k n, I = Submodule.span (WeylAlg k n) {a, b} :=
144   twoGeneratedLeft_of_right (transposeRingEquiv k n) (twoGeneratedRight k n) I
145 
146 end TwoGenerated

158 /-- Stafford's Conjecture 3.8 for `A_n(k)`: `1 = d R + F d S` for every nonzero `d`,
159 from the affine form with `a = 1` and `B = 0`. -/
160 theorem conjecture38 (d : WeylAlg k n) (hd : d ≠ 0) :
161     ∃ F R S : WeylAlg k n, (1 : WeylAlg k n) = d * R + F * d * S := by
162   have := isSimpleRing k n
163   have := isNoetherian k n
164   have := infinite_center k n
165   obtain ⟨H, U, V, h⟩ := affine_of_domain (hOmega k n) d 1 0 hd one_ne_zero
166   exact ⟨H, U, V, by simpa using h⟩

168 /-- Theorem A for `A_n(k)`: for `d, a ≠ 0` and any `B` there are `H, U, V`
169 with `1 = d U + (B + H a) d V`. -/
170 theorem affineForm (d a B : WeylAlg k n) (hd : d ≠ 0) (ha : a ≠ 0) :
171     ∃ H U V : WeylAlg k n, 1 = d * U + (B + H * a) * d * V := by
172   have := isSimpleRing k n
173   have := isNoetherian k n
174   have := infinite_center k n
175   exact affine_of_domain (hOmega k n) d a B hd ha

177 /-- Theorem A' for `A_n(k)`: for a nonzero right ideal `I`, `x` and `t ≠ 0`

179 theorem fixedIdealForm (I : Submodule (WeylAlg k n)ᵐᵒᵖ (WeylAlg k n)) (hI : I ≠ ⊥)
180     (x t : WeylAlg k n) (ht : t ≠ 0) :
181     ∃ H : WeylAlg k n, I ⊔ Submodule.span (WeylAlg k n)ᵐᵒᵖ {x + H * t} = ⊤ := by
182   have := isSimpleRing k n
183   have := isNoetherian k n
184   have := infinite_center k n
185   exact fixedIdeal_of_domain (hOmega k n) I hI x t ht

187 /-- Corollary C.1 for right modules over `A_n(k)`: a finitely generated torsion
188 right module has, for every `x` and `t ≠ 0`, an `h` with `M = (x + h t)A_n(k)`. -/
189 theorem prescribedCyclicRight (M : Type v) [AddCommGroup M]
190     [Module (WeylAlg k n)ᵐᵒᵖ M] [Module.Finite (WeylAlg k n)ᵐᵒᵖ M]
191     (hM : ∀ m : M, ∃ d : WeylAlg k n, d ≠ 0 ∧ op d • m = 0)
192     (x : M) (t : WeylAlg k n) (ht : t ≠ 0) :
193     ∃ h : M, Submodule.span (WeylAlg k n)ᵐᵒᵖ {x + op t • h} = ⊤ := by
194   have := isSimpleRing k n
195   have := isNoetherian k n
196   have := infinite_center k n
197   exact prescribedCyclic_of_domain (hOmega k n) M hM x t ht

199 /-- Corollary C.1 for left modules over `A_n(k)`, transferred from the right case
200 by the transposition anti-automorphism. -/
201 theorem prescribedCyclicLeft (M : Type v) [AddCommGroup M]
202     [Module (WeylAlg k n) M] [Module.Finite (WeylAlg k n) M]
203     (hM : ∀ m : M, ∃ d : WeylAlg k n, d ≠ 0 ∧ d • m = 0)
204     (x : M) (t : WeylAlg k n) (ht : t ≠ 0) :
205     ∃ h : M, Submodule.span (WeylAlg k n) {x + t • h} = ⊤ := by
206   refine prescribedCyclicLeft_of_right (M := M) (transposeRingEquiv k n) ?_ hM x t ht
207   intro M _ _ _ hM x t ht
208   exact prescribedCyclicRight k n M hM x t ht

210 /-- Torsion cyclicity for right modules: a finitely generated torsion right
211 `A_n(k)`-module is cyclic (the case `x = 0`, `t = 1` of the prescribed form). -/
212 theorem torsionCyclicRight (M : Type v) [AddCommGroup M]
213     [Module (WeylAlg k n)ᵐᵒᵖ M] [Module.Finite (WeylAlg k n)ᵐᵒᵖ M]
214     (hM : ∀ m : M, ∃ d : WeylAlg k n, d ≠ 0 ∧ op d • m = 0) :
215     ∃ z : M, Submodule.span (WeylAlg k n)ᵐᵒᵖ ({z} : Set M) = ⊤ := by
216   have := isSimpleRing k n
217   obtain ⟨h, hh⟩ := prescribedCyclicRight k n M hM 0 1 one_ne_zero
